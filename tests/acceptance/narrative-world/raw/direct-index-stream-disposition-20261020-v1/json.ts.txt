import { createNarrativeSha256V1 } from './sha256'

/** Separate from immutable v3 release canonicalization. Never change the v3 comparator. */
export class NarrativeContractErrorV1 extends Error {
  constructor(readonly code: string, message: string) {
    super(`[narrative-ir:${code}] ${message}`)
    this.name = 'NarrativeContractErrorV1'
  }
}
export function narrativeFail(code: string, message: string): never {
  throw new NarrativeContractErrorV1(code, message)
}
function isNarrativeNumber(value: unknown): value is number {
  return typeof value === 'number' && Number.isFinite(value) && !Object.is(value, -0)
    && (!Number.isInteger(value) || Number.isSafeInteger(value))
}
const SURROGATE_UNIT = /[\ud800-\udfff]/
// JSON escapes only quotes, backslashes and C0 controls once lone surrogates
// have been rejected. Ordinary text can retain its existing string storage.
const JSON_ESCAPE = /["\\\u0000-\u001f]/
export function compareUnicodeCodePointsV1(a: string, b: string): number {
  if (a === b) return 0
  // Native UTF-16 order is exactly scalar order when neither string contains
  // surrogate units. Keep the scalar comparison for astral characters.
  if (!SURROGATE_UNIT.test(a) && !SURROGATE_UNIT.test(b)) return a < b ? -1 : 1
  let left = 0, right = 0
  while (left < a.length && right < b.length) {
    const aPoint = a.codePointAt(left)!, bPoint = b.codePointAt(right)!
    const difference = aPoint - bPoint
    if (difference) return difference
    left += aPoint > 0xffff ? 2 : 1
    right += bPoint > 0xffff ? 2 : 1
  }
  // Only the sign matters to callers; code-point order is unchanged and no
  // temporary arrays proportional to long semantic IDs are allocated.
  return left < a.length ? 1 : right < b.length ? -1 : 0
}
export function assertUnicodeV1(text: string): void {
  if (!SURROGATE_UNIT.test(text)) return
  for (let index = 0; index < text.length; index++) {
    const unit = text.charCodeAt(index)
    if (unit >= 0xd800 && unit <= 0xdbff) {
      const next = text.charCodeAt(++index)
      if (!(next >= 0xdc00 && next <= 0xdfff)) narrativeFail('unicode', '孤立 surrogate 不可编码为原文证据')
    } else if (unit >= 0xdc00 && unit <= 0xdfff) narrativeFail('unicode', '孤立 surrogate 不可编码为原文证据')
  }
}
/** Exact raw UTF-8 length, without allocating a byte array. */
export function narrativeUtf8ByteLengthV1(text: string): number {
  let bytes = 0
  for (let index = 0; index < text.length; index++) {
    const unit = text.charCodeAt(index)
    if (unit >= 0xd800 && unit <= 0xdbff) {
      const next = text.charCodeAt(++index)
      if (!(next >= 0xdc00 && next <= 0xdfff)) narrativeFail('unicode', '孤立 surrogate 不可编码为原文证据')
      bytes += 4
    } else if (unit >= 0xdc00 && unit <= 0xdfff) narrativeFail('unicode', '孤立 surrogate 不可编码为原文证据')
    else bytes += unit < 0x80 ? 1 : unit < 0x800 ? 2 : 3
  }
  return bytes
}
/** Validation and encoding share one traversal. Validation alone must not
 * allocate a second, potentially very large serialized artifact. */
interface FrozenNarrativeShape {
  id: number
  sourceKeys: readonly string[]
  keys: readonly string[]
  tokens: readonly string[]
  units: number
}
// The weak receipt proves JSON shape only, never semantic validity or read
// authority. Store a numeric handle, not the shape itself: evicting the bounded
// pool also releases metadata for still-live objects with unusual key sets.
const frozenJsonObjects = new WeakMap<object, number>()
const frozenShapes = new Map<string, FrozenNarrativeShape>()
const frozenShapesById = new Map<number, FrozenNarrativeShape>()
let frozenShapeUnits = 0, nextFrozenShapeId = 1
function rememberFrozenShape(keys: string[]): number {
  if (keys.length > 64) return 0
  const keyUnits = keys.reduce((sum, key) => sum + key.length, 0)
  if (keyUnits + keys.length > 4096) return 0
  const signature = keys.join('\u0000'), cached = frozenShapes.get(signature)
  // A separator can occur inside a key. The signature is only a lookup hint;
  // reuse requires exact ordered keys, not merely an equal joined string.
  if (cached && cached.sourceKeys.length === keys.length
    && keys.every((key, index) => key === cached.sourceKeys[index])) return cached.id
  if (nextFrozenShapeId >= Number.MAX_SAFE_INTEGER) return 0
  const sorted = [...keys].sort(compareUnicodeCodePointsV1)
  const tokens = sorted.map(key => `${JSON.stringify(key)}:`)
  const units = signature.length + keyUnits * 2 + tokens.reduce((sum, token) => sum + token.length, 0)
  const remove = (key: string) => {
    const shape = frozenShapes.get(key)!
    frozenShapeUnits -= shape.units; frozenShapesById.delete(shape.id); frozenShapes.delete(key)
  }
  if (cached) remove(signature)
  while (frozenShapes.size && (frozenShapes.size >= 256 || frozenShapeUnits + units > 65_536)) {
    remove(frozenShapes.keys().next().value!)
  }
  if (units > 65_536) return 0
  const shape: FrozenNarrativeShape = { id: nextFrozenShapeId++, sourceKeys: keys, keys: sorted, tokens, units }
  frozenShapes.set(signature, shape); frozenShapesById.set(shape.id, shape); frozenShapeUnits += units
  return shape.id
}
function walkNarrativeJsonV1(value: unknown, write?: (part: string) => void, omittedKeys?: ReadonlySet<string>,
  rootProjection?: { omittedKeys: ReadonlySet<string>; write: (part: string) => void }, parsedHere = false, freezeHere = false,
  batchNumbers = false): void {
  const stack = new Set<object>()
  // Repeated schema keys and scalar identities otherwise allocate the same
  // escaped token at every occurrence. These bounded pools belong only to
  // this synchronous traversal; they retain neither objects nor authority.
  // Validation-only walks allocate neither pool. Long prose is never pooled.
  let keyTokens: Map<string, string> | undefined
  let stringTokens: Map<string, string> | undefined
  let stringTokenOrder: Array<string | undefined> | undefined
  let stringTokenHead = 0
  let stringTokenUnits = 0
  const keyToken = (key: string): string => {
    const cached = keyTokens?.get(key)
    if (cached !== undefined) return cached
    const token = `${JSON.stringify(key)}:`
    if (key.length <= 64 && (keyTokens?.size ?? 0) < 256) (keyTokens ??= new Map()).set(key, token)
    return token
  }
  const stringToken = (text: string): string => {
    const cached = stringTokens?.get(text)
    if (cached !== undefined) return cached
    // A hit refers to this exact immutable string, already checked in this
    // walk. Omitted values and validation-only walks still validate below.
    assertUnicodeV1(text)
    const token = JSON_ESCAPE.test(text) ? JSON.stringify(text) : `"${text}"`
    const units = text.length + token.length
    if (text.length <= 512 && units <= 131_072) {
      const tokens = stringTokens ??= new Map()
      const order = stringTokenOrder ??= []
      // A bounded FIFO keeps nearby repeated identities reusable throughout
      // a large artifact, instead of reserving every slot for its first rows.
      // Keys and tokens contain no object identity or validation authority.
      while (tokens.size && (tokens.size >= 1024 || stringTokenUnits + units > 131_072)) {
        const first = order[stringTokenHead]!
        stringTokenUnits -= first.length + tokens.get(first)!.length
        tokens.delete(first)
        order[stringTokenHead] = undefined
        stringTokenHead = (stringTokenHead + 1) % 1024
      }
      order[(stringTokenHead + tokens.size) % 1024] = text
      tokens.set(text, token); stringTokenUnits += units
    }
    return token
  }
  const visit = (item: unknown, emit?: (part: string) => void, atRoot = false): void => {
    if (item === null) { emit?.('null'); return }
    if (typeof item === 'string') { if (emit) emit(stringToken(item)); else assertUnicodeV1(item); return }
    if (typeof item === 'boolean') { emit?.(String(item)); return }
    if (isNarrativeNumber(item)) { emit?.(String(item)); return }
    if (typeof item !== 'object' || !item) narrativeFail('json', '仅允许精确 JSON 值；禁止 undefined/非有限数/不安全整数')
    const shapeId = frozenJsonObjects.get(item), frozen = shapeId !== undefined, plain = parsedHere || frozen
    const shape = shapeId ? frozenShapesById.get(shapeId) : undefined
    let validatedKeys: string[] | undefined
    if (frozen && !emit && !rootProjection) return
    // Freeze before inspection so validated data descriptors cannot change
    // during traversal. A receipt is minted only after all children succeed.
    if (freezeHere && !frozen) Object.freeze(item)
    // Plain traversal needs either our complete frozen-JSON receipt or a
    // JSON.parse performed here without a reviver or async/caller handoff.
    // Both exclude accessors, symbols, holes and cycles. Scalar validation
    // and exact canonical wire comparison still visit every encoded value.
    if (!plain) {
      if (stack.has(item)) narrativeFail('json-cycle', 'JSON 不允许循环对象')
      stack.add(item)
      if (Object.getOwnPropertySymbols(item).length) narrativeFail('json', '不允许 symbol 属性')
    }
    if (Array.isArray(item)) {
      if (!plain && Object.keys(item).length !== item.length) narrativeFail('json', '数组必须稠密且无额外属性')
      emit?.('[')
      let numbers: number[] | undefined, numberStart = 0
      for (let index = 0; index < item.length; index++) {
        const descriptor = plain ? undefined : Object.getOwnPropertyDescriptor(item, index)
        if (!plain && (!descriptor || !('value' in descriptor))) narrativeFail('json', '数组必须稠密且不允许 accessor')
        const child = plain ? item[index] : descriptor!.value
        // Hash-only batching reduces tiny writes for long numeric postings.
        // Inspect every original descriptor before copying a scalar into a
        // bounded, private run. Never join the caller's array or invoke its
        // getters, toJSON or valueOf; no mutable object gains a receipt.
        if (batchNumbers && emit && item.length >= 32 && isNarrativeNumber(child)) {
          if (!numbers?.length) numberStart = index
          numbers ??= []
          numbers.push(child)
          if (numbers.length === 256) {
            if (numberStart) emit(',')
            emit(numbers.join(',')); numbers.length = 0
          }
          continue
        }
        if (numbers?.length) {
          if (numberStart) emit?.(',')
          emit?.(numbers.join(',')); numbers.length = 0
        }
        if (index) emit?.(',')
        visit(child, emit)
      }
      if (numbers?.length) {
        if (numberStart) emit?.(',')
        emit?.(numbers.join(','))
      }
      emit?.(']')
    } else {
      if (!plain && Object.getPrototypeOf(item) !== Object.prototype && Object.getPrototypeOf(item) !== null) {
        narrativeFail('json', '仅允许普通 JSON 对象')
      }
      const keys = shape?.keys ?? Object.keys(item)
      if (!shape && emit) (keys as string[]).sort(compareUnicodeCodePointsV1)
      if (freezeHere && !frozen) validatedKeys = keys as string[]
      emit?.('{')
      const projection = atRoot ? rootProjection : undefined
      projection?.write('{')
      let emitted = 0, projected = 0
      for (let index = 0; index < keys.length; index++) {
        const key = keys[index]
        if (!shape) assertUnicodeV1(key)
        const descriptor = plain ? undefined : Object.getOwnPropertyDescriptor(item, key)!
        if (!plain && !('value' in descriptor!)) narrativeFail('json', '不允许 accessor')
        const child = plain ? (item as Record<string, unknown>)[key] : descriptor!.value
        const output = omittedKeys?.has(key) ? undefined : emit
        if (output) { if (emitted++) output(','); output(shape ? shape.tokens[index] : keyToken(key)) }
        // Omission affects only encoded bytes. Invalid values, accessors and
        // cycles inside omitted fields are still rejected by the same walk.
        if (projection && !projection.omittedKeys.has(key)) {
          if (projected++) projection.write(',')
          projection.write(shape ? shape.tokens[index] : keyToken(key))
          visit(child, part => { output?.(part); projection.write(part) })
        } else visit(child, output)
      }
      emit?.('}')
      projection?.write('}')
    }
    if (!plain) stack.delete(item)
    if (freezeHere && !frozen) frozenJsonObjects.set(item, validatedKeys ? rememberFrozenShape(validatedKeys) : 0)
  }
  visit(value, write, true)
}
export function assertNarrativeJsonV1(value: unknown): void { walkNarrativeJsonV1(value) }
/** For compiler-owned decoded values: deeply freeze and validate once. The
 * weak receipt proves JSON shape only, never semantic validity or authority.
 * Merely calling Object.freeze elsewhere cannot mint this receipt. */
export function freezeNarrativeJsonV1<T>(value: T): T {
  walkNarrativeJsonV1(value, undefined, undefined, undefined, false, true)
  return value
}
export function canonicalNarrativeJsonV1(value: unknown): string {
  return encodeNarrativeJsonV1(value)!
}
/** Check exact canonical wire text without allocating a second full encoding.
 * Continue the strict walk after a mismatch: a false comparison must not hide
 * an accessor, invalid scalar, cycle or other invalid value in the remainder. */
export function matchesCanonicalNarrativeJsonV1(value: unknown, serialized: string): boolean {
  if (typeof serialized !== 'string') narrativeFail('json', '规范 JSON 比较需要原始字符串')
  let offset = 0, matches = true
  walkNarrativeJsonV1(value, part => {
    if (matches && !serialized.startsWith(part, offset)) matches = false
    offset += part.length
  })
  return matches && offset === serialized.length
}
/** A disposable cache may stop encoding when its byte budget is exhausted.
 * null is a refusal, never validation of the unvisited remainder. */
export function boundedCanonicalNarrativeJsonV1(value: unknown, maxBytes: number): string | null {
  if (!Number.isSafeInteger(maxBytes) || maxBytes < 0) narrativeFail('json-budget', 'JSON 字节预算必须是安全非负整数')
  return encodeNarrativeJsonV1(value, maxBytes)
}
function encodeNarrativeJsonV1(value: unknown, maxBytes?: number, writeChunk?: (chunk: string) => void): string | null {
  // Keep complete string segments rather than forcing one flat UTF-16 copy.
  // Separating single-byte tokens also avoids widening a large ASCII identity
  // block because an adjacent label contains a non-Latin character. All tokens
  // and hashes are finished here; no serialization is deferred to a consumer.
  const segments: Array<string | undefined> = []
  let parts: string[] = [], length = 0, bytes = 0, wide = false
  const flush = () => {
    if (!parts.length) return
    let chunk = parts.join('')
    writeChunk?.(chunk)
    // Binary merging bounds the concatenation depth and releases individual
    // segment references. The result remains an ordinary, complete string.
    let level = 0
    while (segments[level] !== undefined) {
      chunk = segments[level]! + chunk; segments[level++] = undefined
    }
    segments[level] = chunk
    parts = []; length = 0
  }
  const exhausted = Symbol('cache JSON budget exhausted')
  try {
    walkNarrativeJsonV1(value, part => {
      if (maxBytes !== undefined) {
        if (part.length > maxBytes - bytes || (bytes += narrativeUtf8ByteLengthV1(part)) > maxBytes) throw exhausted
      }
      const partWide = /[\u0100-\uffff]/.test(part)
      if (partWide !== wide) { flush(); wide = partWide }
      parts.push(part); length += part.length
      if (length >= 32_768) flush()
    })
  } catch (error) { if (error === exhausted) return null; throw error }
  if (parts.length) flush()
  let result = ''
  for (let level = segments.length - 1; level >= 0; level--) if (segments[level] !== undefined) result += segments[level]
  return result
}
/** Materialize the complete canonical string and its SHA-256 together.
 * Validation/encoding and the bounded UTF-8 hash consume the same chunks;
 * the returned string is complete, not a deferred serializer or a prefix. */
export function canonicalNarrativeJsonWithHashV1(value: unknown): { json: string; hash: string } {
  const hash = incrementalNarrativeHash()
  try {
    const json = encodeNarrativeJsonV1(value, undefined, hash.updateText)!
    return { json, hash: hash.digest() }
  } finally { hash.destroy() }
}
/** Exact canonical-value equality without allocating two complete encodings.
 * Invalid JSON is still rejected, including accessors and cyclic objects. */
export function equalNarrativeJsonV1(left: unknown, right: unknown): boolean {
  assertNarrativeJsonV1(left); assertNarrativeJsonV1(right)
  const equal = (a: unknown, b: unknown): boolean => {
    if (a === b) return true
    if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') return false
    if (Array.isArray(a) !== Array.isArray(b)) return false
    const aKeys = Object.keys(a), bKeys = Object.keys(b)
    return aKeys.length === bKeys.length && aKeys.every(key => Object.prototype.hasOwnProperty.call(b, key)
      && equal((a as Record<string, unknown>)[key], (b as Record<string, unknown>)[key]))
  }
  return equal(left, right)
}
/** SHA-256 is unchanged. Feed bounded UTF-8 pieces instead of allocating and
 * transferring another complete encoding of a potentially large artifact.
 * Updates consume each piece synchronously; caller mutation after the call
 * cannot change a digest already being returned. */
// Byte scratch space only, never a JSON/semantic validation receipt or a
// content cache. Live sinks own separate buffers; excess nested sinks allocate
// their own 32 KiB and return at most four cleared buffers to this pool.
const hashByteBuffers: Uint8Array[] = []
function incrementalNarrativeHash() {
  const hash = createNarrativeSha256V1(), encoder = new TextEncoder()
  let buffer: Uint8Array | undefined, used = 0
  const flush = () => {
    if (used) { hash.update(buffer!.subarray(0, used)); used = 0 }
  }
  const updateText = (text: string) => {
    if (!text.length) return
    const bytes = buffer ??= hashByteBuffers.pop() ?? new Uint8Array(32_768)
    // Most canonical tokens are short ASCII keys, numbers or delimiters.
    // Encode these directly into the same byte buffer instead of retaining a
    // parts array, joining it, and then encoding another temporary string.
    // All callers already validate Unicode through the strict JSON/raw codec.
    if (text.length <= 128) {
      // A validated UTF-16 unit needs at most three UTF-8 bytes (a surrogate
      // pair uses four bytes for two units). Reserve this small upper bound
      // once, then keep the hot cursor local instead of checking capacity
      // and updating captured state on every byte.
      if (bytes.length - used < text.length * 3) flush()
      let offset = used
      for (let at = 0; at < text.length; at++) {
        let code = text.charCodeAt(at)
        if (code < 0x80) bytes[offset++] = code
        else if (code < 0x800) {
          bytes[offset++] = 0xc0 | (code >>> 6); bytes[offset++] = 0x80 | (code & 0x3f)
        } else if (code >= 0xd800 && code <= 0xdbff) {
          code = 0x10000 + ((code - 0xd800) << 10) + (text.charCodeAt(++at) - 0xdc00)
          bytes[offset++] = 0xf0 | (code >>> 18); bytes[offset++] = 0x80 | ((code >>> 12) & 0x3f)
          bytes[offset++] = 0x80 | ((code >>> 6) & 0x3f); bytes[offset++] = 0x80 | (code & 0x3f)
        } else {
          bytes[offset++] = 0xe0 | (code >>> 12); bytes[offset++] = 0x80 | ((code >>> 6) & 0x3f)
          bytes[offset++] = 0x80 | (code & 0x3f)
        }
      }
      used = offset
      return
    }
    for (let start = 0; start < text.length;) {
      if (bytes.length - used < 4) flush()
      let end = Math.min(start + 32_768, text.length)
      const finalUnit = text.charCodeAt(end - 1)
      if (end < text.length && finalUnit >= 0xd800 && finalUnit <= 0xdbff) end--
      // encodeInto never emits a partial scalar. Keep unconsumed characters
      // for the next flush, even when a token crosses the scratch boundary.
      const encoded = encoder.encodeInto(text.slice(start, end), used ? bytes.subarray(used) : bytes)
      if (!encoded.read) narrativeFail('encoding', 'UTF-8 分段编码未消费输入')
      used += encoded.written; start += encoded.read
    }
  }
  return { updateText, updateBytes: (bytes: Uint8Array) => { flush(); hash.update(bytes) },
    digest: () => { flush(); return hash.digestHex() },
    destroy: () => {
      hash.destroy(); used = 0
      if (buffer) {
        buffer.fill(0)
        if (hashByteBuffers.length < 4) hashByteBuffers.push(buffer)
        buffer = undefined
      }
    } }
}
export async function hashNarrativeBytesV1(value: string | Uint8Array): Promise<string> {
  if (typeof value === 'string') assertUnicodeV1(value)
  const hash = incrementalNarrativeHash()
  try {
    if (typeof value === 'string') hash.updateText(value)
    else hash.updateBytes(value)
    return hash.digest()
  } finally { hash.destroy() }
}
/** Raw UTF-8 streaming only, like hashNarrativeBytesV1: this does not validate
 * JSON, issue a receipt or establish semantic authority. Each part must end at
 * a scalar boundary. Consume synchronously and always release the byte sink,
 * including when a producer throws partway through its output. */
export async function hashNarrativeTextPartsV1(parts: Iterable<string>): Promise<string> {
  return hashNarrativeTextStreamV1(write => {
    for (const part of parts) write(part)
    return undefined
  })
}
/** Synchronous raw-text producer. The writer expires before this call returns;
 * retaining it cannot access a released or subsequently borrowed hash buffer.
 * Like the iterable form, this grants no JSON or semantic verification. */
export async function hashNarrativeTextStreamV1(produce: (write: (part: string) => void) => undefined): Promise<string> {
  const hash = incrementalNarrativeHash()
  let active = true
  try {
    const result = produce(part => {
      if (!active) narrativeFail('encoding', 'UTF-8 写入器已结束')
      if (typeof part !== 'string') narrativeFail('encoding', 'UTF-8 分段输入必须是字符串')
      assertUnicodeV1(part)
      hash.updateText(part)
    })
    if (result !== undefined) {
      // Reject asynchronous producers without leaving their rejected promise
      // unobserved. Their retained writer is already closed by finally below.
      void Promise.resolve(result).catch(() => undefined)
      narrativeFail('encoding', 'UTF-8 生产器必须同步完成且不返回结果')
    }
    return hash.digest()
  } finally { active = false; hash.destroy() }
}
function hashCanonicalNarrativeValueV1(value: unknown, omittedKeys?: ReadonlySet<string>, freezeHere = false): Promise<string> {
  const sink = canonicalNarrativeHashSink()
  try {
    // Use the same strict canonical traversal, including exact key order and
    // invalid-JSON rejection. Only the sink changes, never the wire format.
    walkNarrativeJsonV1(value, sink.write, omittedKeys, undefined, false, freezeHere, true)
    return Promise.resolve(sink.digest())
  } finally { sink.destroy() }
}
function canonicalNarrativeHashSink() {
  const hash = incrementalNarrativeHash()
  return { write: hash.updateText, digest: hash.digest, destroy: hash.destroy }
}

/** Compare the entire canonical wire and hash a root-field projection in one
 * strict traversal. Omitted root fields still participate in wire matching
 * and JSON validation; fields with the same names below the root are retained.
 * A digest accompanied by matches=false is never a verification receipt. */
export function inspectCanonicalNarrativeObjectV1(value: unknown, serialized: string, omittedRootKeys: readonly string[]): {
  matches: boolean; projectionHash: string
} {
  return inspectCanonicalObject(value, serialized, omittedRootKeys, false)
}

/** Own the parse and the complete synchronous inspection. A caller cannot
 * promote an arbitrary object to the private plain-JSON traversal. No trust
 * survives this return: subsequent use of value follows the strict codec. */
export function parseCanonicalNarrativeObjectV1(serialized: string, omittedRootKeys: readonly string[]): {
  value: Record<string, unknown>; matches: boolean; projectionHash: string
} {
  if (typeof serialized !== 'string') narrativeFail('json', '规范 JSON 解码需要原始字符串')
  const value: unknown = JSON.parse(serialized)
  const spans = canonicalRootSpans(serialized)
  if (spans) {
    for (const key of omittedRootKeys) {
      if (typeof key !== 'string') narrativeFail('json-key', '摘要排除键必须是字符串')
      assertUnicodeV1(key)
    }
    const omitted = new Set(omittedRootKeys), hash = incrementalNarrativeHash()
    try {
      hash.updateText('{')
      let emitted = false
      for (const span of spans) if (!omitted.has(span.key)) {
        if (emitted) hash.updateText(',')
        hash.updateText(serialized.slice(span.start, span.end))
        emitted = true
      }
      hash.updateText('}')
      return { value: value as Record<string, unknown>, matches: true, projectionHash: hash.digest() }
    } finally { hash.destroy() }
  }
  const proof = inspectCanonicalObject(value, serialized, omittedRootKeys, true)
  return { value: value as Record<string, unknown>, ...proof }
}

/** Recognize complete JSON grammar and exact canonical wire without decoding
 * or re-encoding the entire object. Literals, controls, scalar Unicode, number
 * spellings, separators and sorted unique keys are all checked here. Bounds limit speculative metadata/stack use, not the
 * accepted JSON domain: every miss falls back to the existing strict inspector.
 * In particular, a duplicate key may overwrite an invalid scalar; only the
 * retained parsed value determines that inspector's rejection/projection hash.
 */
function canonicalRootSpans(serialized: string): Array<{ key: string; start: number; valueStart: number; end: number }> | null {
  let offset = 0
  const roots: Array<{ key: string; start: number; valueStart: number; end: number }> = []
  const string = (): string | null => {
    if (serialized.charCodeAt(offset++) !== 34) return null
    const start = offset
    for (;;) {
      const quote = serialized.indexOf('"', offset)
      if (quote === -1) return null
      let back = quote
      while (back > start && serialized.charCodeAt(back - 1) === 92) back--
      offset = quote + 1
      if ((quote - back) % 2) continue
      const token = serialized.slice(start, quote)
      // This recognizer can now run before JSON.parse; raw control units are
      // never JSON string data, even when the remaining escapes are canonical.
      if (/[\u0000-\u001f]/.test(token)) return null
      for (let at = token.indexOf('\\'); at !== -1; at = token.indexOf('\\', at)) {
        const escape = token[++at]
        if ('"\\bfnrt'.includes(escape)) { at++; continue }
        if (escape !== 'u') return null
        const digits = token.slice(at + 1, at + 5)
        if (!/^00[01][0-9a-f]$/.test(digits) || ['0008', '0009', '000a', '000c', '000d'].includes(digits)) return null
        at += 5
      }
      try { assertUnicodeV1(token) } catch (error) {
        if (error instanceof NarrativeContractErrorV1 && error.code === 'unicode') return null
        throw error
      }
      return token
    }
  }
  const visit = (root = false, depth = 0): boolean => {
    if (depth > 256) return false
    const unit = serialized.charCodeAt(offset)
    if (unit === 34) return string() !== null
    if (unit === 123) {
      offset++
      if (serialized[offset] === '}') { offset++; return true }
      let previous: string | undefined
      for (;;) {
        const start = offset, token = string()
        if (token === null) return false
        const key: string = token.includes('\\') ? JSON.parse('"' + token + '"') : token
        if (previous !== undefined && compareUnicodeCodePointsV1(previous, key) >= 0) return false
        previous = key
        if (serialized[offset++] !== ':') return false
        const valueStart = offset
        if (!visit(false, depth + 1)) return false
        if (root) {
          if (roots.length === 256) return false
          roots.push({ key, start, valueStart, end: offset })
        }
        const next = serialized[offset++]
        if (next === '}') return true
        if (next !== ',') return false
      }
    }
    if (unit === 91) {
      offset++
      if (serialized[offset] === ']') { offset++; return true }
      for (;;) {
        if (!visit(false, depth + 1)) return false
        const next = serialized[offset++]
        if (next === ']') return true
        if (next !== ',') return false
      }
    }
    if (unit === 116 || unit === 110 || unit === 102) {
      const literal = unit === 116 ? 'true' : unit === 110 ? 'null' : 'false'
      if (!serialized.startsWith(literal, offset)) return false
      offset += literal.length
      return true
    }
    const start = offset
    while (offset < serialized.length) {
      const code = serialized.charCodeAt(offset)
      if (!(code >= 48 && code <= 57 || code === 46 || code === 69 || code === 101 || code === 43 || code === 45)) break
      offset++
    }
    if (offset === start) return false
    const text = serialized.slice(start, offset), value = Number(text)
    return Number.isFinite(value) && !Object.is(value, -0)
      && (!Number.isInteger(value) || Number.isSafeInteger(value)) && String(value) === text
  }
  return serialized[0] === '{' && visit(true) && offset === serialized.length ? roots : null
}

/** Speculatively defer one root value while retaining every canonical byte.
 * A miss is not an error or a validity decision: the caller must use the normal
 * JSON.parse/strict-codec path. A hit proves JSON shape and wire form only.
 * The deferred value still needs the caller's complete semantic comparison.
 * Bounds belong to the recognizer, so large/deep valid inputs keep the fallback.
 */
export function splitCanonicalNarrativeObjectV1(serialized: string, deferredKey: string, omittedRootKeys: readonly string[]): {
  value: Record<string, unknown>; deferredJson: string; projectionHash: string
} | null {
  if (typeof serialized !== 'string') narrativeFail('json', '规范 JSON 解码需要原始字符串')
  for (const key of [deferredKey, ...omittedRootKeys]) {
    if (typeof key !== 'string') narrativeFail('json-key', '分段键和摘要排除键必须是字符串')
    assertUnicodeV1(key)
  }
  const spans = canonicalRootSpans(serialized)
  const deferred = spans?.find(span => span.key === deferredKey)
  if (!spans || !deferred) return null
  const value: Record<string, unknown> = {}, omitted = new Set(omittedRootKeys)
  const hash = incrementalNarrativeHash()
  try {
    hash.updateText('{')
    let emitted = false
    for (const span of spans) {
      if (!omitted.has(span.key)) {
        if (emitted) hash.updateText(',')
        hash.updateText(serialized.slice(span.start, span.end)); emitted = true
      }
      if (span !== deferred) {
        // Define data properties explicitly: __proto__ is ordinary JSON data.
        // All remaining root values are owned parses without a caller reviver.
        Object.defineProperty(value, span.key, { value: JSON.parse(serialized.slice(span.valueStart, span.end)),
          enumerable: true, writable: true, configurable: true })
      }
    }
    hash.updateText('}')
    return { value, deferredJson: serialized.slice(deferred.valueStart, deferred.end), projectionHash: hash.digest() }
  } finally { hash.destroy() }
}

function inspectCanonicalObject(value: unknown, serialized: string, omittedRootKeys: readonly string[], parsedHere: boolean): {
  matches: boolean; projectionHash: string
} {
  if (typeof serialized !== 'string' || value === null || typeof value !== 'object' || Array.isArray(value)) {
    narrativeFail('json', '规范根对象摘要需要对象及完整原始字符串')
  }
  for (const key of omittedRootKeys) {
    if (typeof key !== 'string') narrativeFail('json-key', '摘要排除键必须是字符串')
    assertUnicodeV1(key)
  }
  const sink = canonicalNarrativeHashSink()
  let offset = 0, matches = true
  try {
    walkNarrativeJsonV1(value, part => {
      if (matches && !serialized.startsWith(part, offset)) matches = false
      offset += part.length
    }, undefined, { omittedKeys: new Set(omittedRootKeys), write: sink.write }, parsedHere)
    return { matches: matches && offset === serialized.length, projectionHash: sink.digest() }
  } finally { sink.destroy() }
}

export const hashNarrativeValueV1 = (value: unknown): Promise<string> => hashCanonicalNarrativeValueV1(value)

/** For compiler-owned content that is already destined to be immutable.
 * The complete strict walk validates, freezes and hashes together. There is
 * no trusted-input bypass: invalid children still fail, and only successfully
 * completed objects receive the private JSON-shape receipt. */
export function freezeAndHashNarrativeJsonV1(value: unknown, omittedKeys?: readonly string[]): Promise<string> {
  for (const key of omittedKeys ?? []) {
    if (typeof key !== 'string') narrativeFail('json-key', '摘要排除键必须是字符串')
    assertUnicodeV1(key)
  }
  // Omission changes only the digest. Every omitted child is still validated
  // and frozen before its parent can receive a private JSON-shape receipt.
  return hashCanonicalNarrativeValueV1(value, omittedKeys === undefined ? undefined : new Set(omittedKeys), true)
}

/** Hash the canonical projection without allocating a recursively copied
 * object tree. Keys are omitted at every object depth, never array elements;
 * the entire input remains subject to strict JSON validation. */
export function hashNarrativeValueWithoutKeysV1(value: unknown, omittedKeys: readonly string[]): Promise<string> {
  for (const key of omittedKeys) {
    if (typeof key !== 'string') narrativeFail('json-key', '摘要排除键必须是字符串')
    assertUnicodeV1(key)
  }
  return hashCanonicalNarrativeValueV1(value, new Set(omittedKeys))
}

export type DecoderV1<T> = (value: unknown, path: string) => T
export const textV1: DecoderV1<string> = (value, path) => {
  if (typeof value !== 'string') narrativeFail('shape', `${path} 必须是 string`)
  assertUnicodeV1(value)
  return value
}
export const keyV1: DecoderV1<string> = (value, path) => {
  const text = textV1(value, path)
  if (!text || text.trim() !== text) narrativeFail('shape', `${path} 必须是精确非空键`)
  return text
}
export const hashV1: DecoderV1<string> = (value, path) => {
  const text = textV1(value, path)
  if (!/^[a-f0-9]{64}$/.test(text)) narrativeFail('hash', `${path} 必须是 SHA-256`)
  return text
}
export const pointerV1: DecoderV1<string> = (value, path) => {
  const pointer = textV1(value, path)
  if (!/^(\/(?:[^~]|~[01])*)+$/.test(pointer)) narrativeFail('json-pointer', `${path} 必须是非空规范 JSON pointer`)
  return pointer
}
export const uintV1: DecoderV1<number> = (value, path) => {
  if (typeof value !== 'number' || !Number.isSafeInteger(value) || value < 0 || Object.is(value, -0)) {
    narrativeFail('shape', `${path} 必须是安全非负整数`)
  }
  return value
}
export const integerTextV1: DecoderV1<string> = (value, path) => {
  const text = textV1(value, path)
  if (!/^(0|-?[1-9][0-9]*)$/.test(text)) narrativeFail('tick', `${path} 必须是规范十进制整数`)
  return text
}
export const nullableV1 = <T>(decoder: DecoderV1<T>): DecoderV1<T | null> => (value, path) => value === null ? null : decoder(value, path)
export const enumV1 = <T extends string | number>(values: readonly T[]): DecoderV1<T> => (value, path) => {
  if (!values.includes(value as T)) narrativeFail('enum', `${path} 不支持 ${String(value)}`)
  return value as T
}
export const arrayV1 = <T>(decoder: DecoderV1<T>): DecoderV1<T[]> => (value, path) => {
  if (!Array.isArray(value)) narrativeFail('shape', `${path} 必须是数组`)
  return value.map((item, index) => decoder(item, `${path}[${index}]`))
}
export const uniqueKeysV1: DecoderV1<string[]> = (value, path) => {
  const keys = arrayV1(keyV1)(value, path)
  if (new Set(keys).size !== keys.length) narrativeFail('duplicate', `${path} 键重复`)
  return keys.sort(compareUnicodeCodePointsV1)
}
export function objectV1<T extends object>(shape: { [K in keyof T]-?: DecoderV1<T[K]> }): DecoderV1<T> {
  const expected = Object.keys(shape)
  return (value, path) => {
    if (!value || typeof value !== 'object' || Array.isArray(value)) narrativeFail('shape', `${path} 必须是对象`)
    if (Object.keys(value).length !== expected.length || expected.some(key => !Object.prototype.hasOwnProperty.call(value, key))) {
      narrativeFail('exact-keys', `${path} 缺少字段或含额外字段`)
    }
    const decoded: Record<string, unknown> = {}
    for (const key of expected) {
      const field = (shape[key as keyof T] as DecoderV1<unknown>)((value as Record<string, unknown>)[key], `${path}.${key}`)
      if (key === '__proto__') Object.defineProperty(decoded, key, { value: field, writable: true, enumerable: true, configurable: true })
      else decoded[key] = field
    }
    return decoded as T
  }
}
