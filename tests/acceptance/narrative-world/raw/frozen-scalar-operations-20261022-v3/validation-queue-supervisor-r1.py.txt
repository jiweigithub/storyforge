from pathlib import Path
from datetime import datetime, timezone
import fcntl, hashlib, json, os, shutil, signal, subprocess, sys, time
from process_load import snapshot
ROOT=Path('/tmp/nw-frozen-scalar-operations-20261022-v3/frozen-workspace')
BASE=Path('/tmp/nw-frozen-scalar-operations-20261022-v3');PREFIX=BASE/'validation'
STATE=BASE/'validation-queue-r1-state.json';TERMINAL=BASE/'validation-queue-r1-terminal.json';LOG=BASE/'validation-queue-r1.log'
now=lambda:datetime.now(timezone.utc).isoformat()
def sha(path):
    with Path(path).open('rb') as stream:return hashlib.file_digest(stream,'sha256').hexdigest()
def save(path,value):
    with path.open('x') as stream:json.dump(value,stream,ensure_ascii=False,indent=2);stream.write('\n')
def replace(path,value):
    stage=Path(str(path)+'.tmp');stage.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n');os.replace(stage,path)
def main():
    signal.signal(signal.SIGHUP,signal.SIG_IGN)
    with (BASE/'validation-queue-r1.lock').open('a') as lock:
        fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
        assert not STATE.exists() and not TERMINAL.exists() and not LOG.exists()
        assert not (BASE/'validation-terminal.json').exists() and not (BASE/'result.json').exists()
        pin_path=BASE/'validation-sources.json';pin=json.loads(pin_path.read_text());plan=json.loads((BASE/'plan.json').read_text())
        pin_sha=sys.argv[1];assert sha(pin_path)==pin_sha
        record={'schema':'storyforge.narrative-durable-queue-attempt','version':2,'attempt':1,
            'startedAt':now(),'supervisorPid':os.getpid(),'root':str(ROOT),'queueCwd':str(BASE),
            'runner':str(BASE/'validation.py'),'pinSha256':pin_sha,'log':str(LOG),
            'productionChanged':False,'actualModelCalls':0,'accepted':False,'queuePid':None}
        child=None;code=2
        def interrupted(signum,frame):
            if child is not None and child.poll() is None:
                try:os.killpg(child.pid,signum)
                except ProcessLookupError:pass
            raise SystemExit(128+signum)
        signal.signal(signal.SIGTERM,interrupted);signal.signal(signal.SIGINT,interrupted)
        def verify():
            assert sha(pin_path)==pin_sha and sha(BASE/'validation.py')==pin['runnerSha256']
            for path,digest in pin['files'].items():assert sha(ROOT/path)==digest,path
            for group in ['runtimeFiles','extraFiles','executionFiles']:
                for path,digest in pin[group].items():assert sha(path)==digest,path
            for directory in [ROOT,BASE]:
                assert not (directory/'reclaimed.json').exists() and not (directory.parent/'reclaimed.json').exists()
            assert shutil.disk_usage(ROOT).free>=plan['readiness']['diskReserveBytes']
        def observe():
            return {**snapshot(os.getpid()), 'at':now(),'freeBytes':shutil.disk_usage(ROOT).free,
                'supervisorPid':os.getpid(),'pinSha256':pin_sha,'queueSubmitted':False,'measurementStarted':False}
        try:
            verify();quiet=0;deadline=time.monotonic()+plan['readiness']['maximumWaitSeconds']
            while quiet<plan['readiness']['consecutiveQuietObservations']:
                observation=observe();quiet=quiet+1 if observation['quiet'] else 0
                observation['consecutiveQuietObservations']=quiet
                replace(BASE/'admission-state.json',observation)
                with (BASE/'admission-observations.jsonl').open('a') as stream:stream.write(json.dumps(observation,ensure_ascii=False)+'\n')
                if time.monotonic()>=deadline and quiet<plan['readiness']['consecutiveQuietObservations']:
                    record.update(disposition='deferred-no-low-load-window',queueSubmitted=False,measurementStarted=False,workspaceFrozen=False)
                    raise SystemExit(75)
                if quiet<plan['readiness']['consecutiveQuietObservations']:time.sleep(plan['readiness']['intervalSeconds'])
            record['admittedAt']=now();verify()
            command=[sys.executable,'/tmp/storyforge-serial-ci-20261013.py','--',sys.executable,str(BASE/'validation.py')]
            record['command']=command
            with LOG.open('x') as log:
                child=subprocess.Popen(command,cwd=BASE,stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
                record['queuePid']=child.pid;save(STATE,record);code=child.wait()
        except BaseException as error:
            record['error']=type(error).__name__+': '+str(error)
            code=error.code if isinstance(error,SystemExit) and isinstance(error.code,int) else 2
        finally:
            record.update(endedAt=now(),exitCode=code,logSha256=sha(LOG) if LOG.exists() else None,
                runnerTerminalPresent=(BASE/'validation-terminal.json').exists())
            save(TERMINAL,record)
        return code
if __name__=='__main__':sys.exit(main())
