"""Read-only admission classification; the 20% CPU criterion is unchanged."""
import re, shlex, json, os, subprocess, sys

def is_test_process(row):
    if row.get('state','').startswith('Z'):
        return False
    command=row['command'].strip()
    if re.fullmatch(r'npm run (?:ci(?::e2e)?|test:coverage)(?: --.*)?',command):
        return True
    if re.fullmatch(r'node \(vitest(?: \d+)?\)',command):
        return True
    try:args=shlex.split(command)
    except ValueError:return False
    if not args or args[0].rsplit('/',1)[-1] not in ('node','nodejs'):
        return False
    script_index=next((i for i in range(1,len(args)) if not args[i].startswith('-')),None)
    if script_index is None:return False
    script=args[script_index];following=args[script_index+1:]
    # Inspect the executable's script and verb, never quoted shell/Python source.
    if script.endswith(('/vitest.mjs','/vitest.js')):
        return 'run' in following or '--run' in following or '--coverage' in following
    if script.endswith(('/playwright/cli.js','/playwright-core/cli.js')):
        return bool(following) and following[0]=='test'
    return False

def classify(rows,excluded=(),threshold=20):
    foreign=[r for r in rows if r['pid'] not in excluded and not r.get('state','').startswith('Z')]
    tests=[r for r in foreign if is_test_process(r)]
    by_pid={r['pid']:r for r in rows};test_pids={r['pid'] for r in tests}
    roots=[]
    for test in tests:
        parent=test['ppid'];seen={test['pid']};nested=False
        while parent in by_pid and parent not in seen:
            if parent in test_pids:nested=True;break
            seen.add(parent);parent=by_pid[parent]['ppid']
        if not nested:roots.append(test)
    busy=[r for r in foreign if r['cpu']>=threshold]
    return {'busy':busy,'activeTests':tests,'activeTestRoots':roots,'quiet':not busy and not tests}

def snapshot(owner_pid):
    rows=[]
    for line in subprocess.check_output(['ps','-axo','pid=,ppid=,stat=,%cpu=,command='],text=True).splitlines():
        parts=line.strip().split(None,4)
        if len(parts)==5:rows.append({'pid':int(parts[0]),'ppid':int(parts[1]),'state':parts[2],'cpu':float(parts[3]),'command':parts[4]})
    by_pid={r['pid']:r for r in rows};owned={owner_pid}
    while True:
        children={r['pid'] for r in rows if r['ppid'] in owned}
        if children<=owned:break
        owned|=children
    ancestors={owner_pid};pid=owner_pid
    while pid in by_pid:
        pid=by_pid[pid]['ppid']
        if not pid or pid in ancestors:break
        ancestors.add(pid)
    return {**classify(rows,owned|ancestors),'ownedProcesses':[r for r in rows if r['pid'] in owned]}

if __name__=='__main__':
    print(json.dumps(snapshot(int(sys.argv[1]) if len(sys.argv)>1 else os.getpid()),ensure_ascii=False))
