"""Synthetic control-flow fixtures; no product or model acceptance evidence."""
import hashlib
import json
from pathlib import Path
import tempfile
import unittest

from baseline_promotion import promote, sha


class BaselinePromotionTest(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name)
        self.manifest = self.root / 'manifest.json'
        self.previous = self.root / 'previous.json'
        self.generated = self.root / 'generated.json'
        self.target = self.root / 'new.json'
        (self.root / 'corpus.json').write_text('{"fixtureOnly":true}')
        (self.root / 'source.txt').write_text('source A')
        self.spec = {'baseline': {'path': 'new.json'},
                     'sourceFingerprintPaths': ['manifest.json', 'source.txt'],
                     'corpus': {'path': 'corpus.json', 'sha256': sha(self.root / 'corpus.json')}}
        self.manifest.write_text(json.dumps(self.spec))
        hashes = {name: sha(self.root / name) for name in sorted(['manifest.json', 'source.txt', 'corpus.json'])}
        self.report = {'observations': [{'status': 'gap', 'fixtureOnly': True}],
                       'input': {'hashes': hashes, 'sha256': hashlib.sha256(json.dumps(hashes, separators=(',', ':')).encode()).hexdigest()},
                       'run': {'fixtureOnly': True}}
        self.previous.write_text(json.dumps({'observations': self.report['observations'], 'input': {}, 'run': {}}))
        self.generated.write_text(json.dumps(self.report))
        self.args = (self.root, self.manifest, self.previous, self.generated, self.target)

    def test_predeclared_self_fingerprinted_manifest_copies_exact_measured_bytes(self):
        result = promote(*self.args)
        self.assertTrue(result['manifestSelfFingerprintRetained'])
        self.assertEqual(self.generated.read_bytes(), self.target.read_bytes())

    def test_changing_manifest_after_measurement_does_not_rebind_or_copy(self):
        self.spec['baseline']['history'] = ['post-measurement change']
        self.manifest.write_text(json.dumps(self.spec))
        with self.assertRaisesRegex(ValueError, 'Measured input no longer matches: manifest.json'):
            promote(*self.args)
        self.assertFalse(self.target.exists())

    def test_changed_observation_is_not_silently_promoted(self):
        self.report['observations'][0]['status'] = 'pass'
        self.generated.write_text(json.dumps(self.report))
        with self.assertRaisesRegex(ValueError, 'Observed behavior changed'):
            promote(*self.args)
        self.assertFalse(self.target.exists())

    def test_existing_target_is_not_overwritten(self):
        self.target.write_text('preserved earlier result')
        with self.assertRaises(FileExistsError):
            promote(*self.args)
        self.assertEqual(self.target.read_text(), 'preserved earlier result')


if __name__ == '__main__':
    unittest.main(verbosity=2)
