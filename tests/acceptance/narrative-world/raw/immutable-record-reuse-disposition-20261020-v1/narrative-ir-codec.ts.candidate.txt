type ResourceIdentity = NonNullable<(typeof PROJECT_TABLES)[number]['resourceIdentity']>
// Compile-time coverage of the current flat identity contract. Adding a field
// requires an explicit freshness comparison here; tables still come only from
// PROJECT_TABLES, including newly registered or duplicate resource kinds.
const identityFields: { [K in keyof ResourceIdentity]-?: true } = {
  version: true, field: true, resourceKind: true, narrativeEntityType: true,
  contextKind: true, label: true, descriptorMode: true,
}
const identityKeys = Object.keys(identityFields) as Array<keyof ResourceIdentity>
type IdentityContract = Array<{ table: string; identity: ResourceIdentity }>
let lastIdentityContract: IdentityContract | undefined
function currentIdentityContract(): IdentityContract {
  let position = 0, unchanged = lastIdentityContract !== undefined
  for (const spec of PROJECT_TABLES) {
    if (!spec.resourceIdentity) continue
    const previous = lastIdentityContract?.[position++]
    if (!previous || previous.table !== spec.name) { unchanged = false; continue }
    for (const key of identityKeys) {
      if (previous.identity[key] !== spec.resourceIdentity[key]) unchanged = false
    }
  }
  if (unchanged && position === lastIdentityContract!.length) return lastIdentityContract!
  lastIdentityContract = PROJECT_TABLES.filter(spec => spec.resourceIdentity)
    .map(spec => ({ table: spec.name, identity: { ...spec.resourceIdentity! } }))
  return lastIdentityContract
}

export const semanticRefDecoderV1 = <K extends SemanticKindV1>(kinds: readonly K[]): DecoderV1<SemanticRefV1<K>> => (value, path) => {
  // The kind check below already parses and validates the complete identity;
  // avoid decoding the same URI a second time for every edge.
  const ref = objectV1<SemanticRefV1<K>>({ kind: enumV1(kinds), semanticId: keyV1, revisionHash: hashV1 })(value, path)
  if (parseNarrativeSemanticIdV1(ref.semanticId).kind !== ref.kind) narrativeFail('ref-kind', '引用 kind 与 semanticId 不一致')
  return ref
}

/** Ordered values preserve order. Semantic reference collections elsewhere are
 * sets sorted by stable identity; neither uses labels or locale ordering. */
export const typedValueDecoderV1: DecoderV1<TypedValueV1> = (value, path) => {
  switch ((value as TypedValueV1 | null)?.kind) {
    case 'boolean': return objectV1<Extract<TypedValueV1, { kind: 'boolean' }>>({ kind: enumV1(['boolean']), value: (item, at) => {
      if (typeof item !== 'boolean') narrativeFail('shape', `${at} 必须是 boolean`)
      return item
    } })(value, path)
    case 'decimal': return objectV1<Extract<TypedValueV1, { kind: 'decimal' }>>({ kind: enumV1(['decimal']), unit: nullableV1(keyV1), value: (item, at) => {
      const text = textV1(item, at)
      if (!/^-?(0|[1-9][0-9]*)(\.[0-9]*[1-9])?$/.test(text) || text === '-0') narrativeFail('decimal', 'decimal 必须是规范十进制字符串')
      return text
    } })(value, path)
    case 'text': return objectV1<Extract<TypedValueV1, { kind: 'text' }>>({ kind: enumV1(['text']), value: textV1 })(value, path)
    case 'enum': return objectV1<Extract<TypedValueV1, { kind: 'enum' }>>({ kind: enumV1(['enum']), value: keyV1 })(value, path)
    case 'entity-ref': return objectV1<Extract<TypedValueV1, { kind: 'entity-ref' }>>({ kind: enumV1(['entity-ref']), value: entityRef })(value, path)
    case 'ordered-set': {
      const set = objectV1<Extract<TypedValueV1, { kind: 'ordered-set' }>>({ kind: enumV1(['ordered-set']), values: arrayV1(typedValueDecoderV1) })(value, path)
      if (set.values.some(item => item.kind === 'ordered-set')) narrativeFail('value-depth', 'v1 ordered-set 仅包含原子值')
      if (new Set(set.values.map(item => canonicalNarrativeJsonV1(item))).size !== set.values.length) narrativeFail('duplicate', '有序集合值重复')
      return set
    }
    default: return narrativeFail('value-kind', `${path} 未知 typed value`)
  }
}
const setOf = <T>(decoder: DecoderV1<T>, identity: (item: T) => string): DecoderV1<T[]> => (value, path) => {
  const values = arrayV1(decoder)(value, path)
  if (new Set(values.map(identity)).size !== values.length) narrativeFail('duplicate', `${path} 包含重复项`)
  return values.sort((a, b) => compareUnicodeCodePointsV1(identity(a), identity(b)))
}

const recordDecoder: DecoderV1<NarrativeRecordV1> = (value, path) => {
  const kind = enumV1(NARRATIVE_KINDS_V1)((value as NarrativeRecordV1 | null)?.kind, `${path}.kind`)
  const record = objectV1({ ...envelopeShape, payload: payloadDecoders[kind] as DecoderV1<NarrativePayloadsV1[SemanticKindV1]> })(value, path) as NarrativeRecordV1
  const identity = parseNarrativeSemanticIdV1(record.semanticId)
  if (identity.kind !== kind || identity.ownerKey !== narrativeOwnerKeyV1(record.scope.owner)) narrativeFail('identity', '信封 owner/kind 与身份不一致')
  const spec = narrativeResourceIdentitySpecV1(identity.source.resourceKind)
  if (record.provenance.sourceResourceKey !== `${spec.resourceIdentity!.contextKind}:${identity.source.sourceUid}`) {
    narrativeFail('provenance', '身份与来源资源 key 不一致')
  }
  if (record.authorUpdatedAt < record.authorCreatedAt) narrativeFail('audit-time', '审计修改时间早于创建')
  if (record.visibility.class === 'restricted' ? record.visibility.audienceKeys.length === 0 : record.visibility.audienceKeys.length !== 0) {
    narrativeFail('visibility', 'restricted 必须声明受众，其他分类不能夹带受众授权')
  }
  if (record.modality === 'intent' && record.visibility.class !== 'author-only') narrativeFail('intent-visibility', 'v1 意图保持 author-only')
  if (kind === 'entity' && record.kind === 'entity') {
    if (record.payload.entityType !== spec.resourceIdentity!.narrativeEntityType
      || canonicalNarrativeJsonV1(identity.source) !== canonicalNarrativeJsonV1(record.payload.sourceIdentity)) narrativeFail('entity-kind', '实体类型/来源与注册表不一致')
  }
  if (record.kind === 'event') {
    if ((record.payload.eventType === 'observation') !== (record.payload.observerRef !== null)
      || (record.payload.eventType === 'communication') !== (record.payload.communication !== null)
      || record.payload.communication?.recipientRefs.length === 0) narrativeFail('event-variant', '观察/通信必须提供对应 typed payload')
  }
  if (record.kind === 'rule' && record.payload.strength === 'hard' && !record.payload.validatorRef) narrativeFail('hard-rule', 'hard rule 必須有版本化 validator')
  if (record.kind === 'epistemic-event' && ((record.payload.transition === 'revise') !== (record.payload.revisesRef !== null)
    || (record.payload.transition === 'deny-awareness' && record.payload.attitude !== 'unaware'))) narrativeFail('epistemic-transition', '认知修订/不知声明结构不闭合')
  record.time.disclosedAt = setOf(objectV1({ audienceKey: keyV1, sequenceKey: keyV1, unitKey: keyV1 }), canonicalNarrativeJsonV1)(record.time.disclosedAt, 'disclosedAt')
  return record
}

function parsedInput(input: unknown): unknown {
  let value = input
  if (typeof input === 'string') {
    try { value = JSON.parse(input) } catch { narrativeFail('json', '不是合法 JSON') }
  }
  assertNarrativeJsonV1(value)
  return value
}
export function parseNarrativeRecordV1(input: unknown): NarrativeRecordV1 {
  return recordDecoder(parsedInput(input), 'record')
}
/** Compiler-owned decoded copy; the caller's input remains mutable. */
export function parseImmutableNarrativeRecordV1(input: unknown): NarrativeRecordV1 {
  const contract = currentIdentityContract()
  const record = freezeNarrativeJsonV1(parseNarrativeRecordV1(input))
  immutableRecords.set(record, contract)
  return record
}
export function parseNarrativeIdentityAliasV1(input: unknown): NarrativeIdentityAliasV1 {
  return aliasDecoder(parsedInput(input), 'alias')
}
export function parseNarrativeIRBundleV1(input: unknown): NarrativeIRBundleV1 {
  return parseBundle(input, recordDecoder)
}
/** Internal immutable compilation may share only our exact frozen records.
 * The bundle shell/arrays remain private and all bundle checks still run.
 * Ordinary parsing above always returns fresh, mutable decoded records. */
export function parseNarrativeIRBundleWithImmutableRecordsV1(input: unknown): NarrativeIRBundleV1 {
  const contract = currentIdentityContract()
  return parseBundle(input, (value, path) => value !== null && typeof value === 'object'
    && immutableRecords.get(value as NarrativeRecordV1) === contract
    ? value as NarrativeRecordV1 : recordDecoder(value, path))
}
function parseBundle(input: unknown, decodeRecord: DecoderV1<NarrativeRecordV1>): NarrativeIRBundleV1 {
  const bundle = objectV1<NarrativeIRBundleV1>({ schema: enumV1(['storyforge.narrative-ir']), irVersion: enumV1([1]),
    scope: scopeDecoder, records: setOf(decodeRecord, r => r.semanticId), evidence: setOf(evidenceDecoderV1, e => e.evidenceId),
    timeContext: value => parseNarrativeTimeContextV1(value), aliases: setOf(aliasDecoder, a => canonicalNarrativeJsonV1(a.from)),
    semanticRootHash: hashV1 })(parsedInput(input), 'bundle')
  const context = bundle.timeContext
  context.timelines.sort((a, b) => compareUnicodeCodePointsV1(a.timelineKey, b.timelineKey))
  context.sequences.sort((a, b) => compareUnicodeCodePointsV1(a.sequenceKey, b.sequenceKey))
  context.anchors.sort((a, b) => compareUnicodeCodePointsV1(a.eventKey, b.eventKey))
  context.before.sort((a, b) => compareUnicodeCodePointsV1(canonicalNarrativeJsonV1(a), canonicalNarrativeJsonV1(b)))
  context.mappings.sort((a, b) => compareUnicodeCodePointsV1(a.mappingKey, b.mappingKey))
  const bundleScopeKey = canonicalNarrativeJsonV1(bundle.scope)
  for (const record of bundle.records) {
    if (canonicalNarrativeJsonV1(record.scope) !== bundleScopeKey) narrativeFail('scope', '同一 IR bundle 不允许混装 owner/branch/revision')
  }
  return bundle
}
export function encodeNarrativeIRBundleV1(input: NarrativeIRBundleV1): string {
  return canonicalNarrativeJsonV1(parseNarrativeIRBundleV1(input))
}

/** Walk only an already exact-decoded record. Ref-shaped leaves cannot be
 * supplied as arbitrary JSON payloads. Includes every nested TypedValue. */
function isReferenceLeaf(value: Record<string, unknown>): boolean {
  return Object.keys(value).length === 3 && typeof value.semanticId === 'string'
    && typeof value.revisionHash === 'string' && typeof value.kind === 'string'
}

/** The same validated leaves as mapNarrativeReferencesV1, without allocating
 * an unused copy of every non-reference object during read-only traversal. */
export function forEachNarrativeReferenceV1(record: NarrativeRecordV1, visit: (ref: SemanticRefV1) => void): void {
  const walk = (value: unknown): void => {
    if (Array.isArray(value)) { for (const child of value) walk(child); return }
    if (!value || typeof value !== 'object') return
    const obj = value as Record<string, unknown>
    if (isReferenceLeaf(obj)) { visit(ref(obj, 'ref')); return }
    for (const key of Object.keys(obj)) walk(obj[key])
  }
  walk(record)
}

export function mapNarrativeReferencesV1(record: NarrativeRecordV1, map: (ref: SemanticRefV1) => SemanticRefV1): NarrativeRecordV1 {
  const walk = (value: unknown): unknown => {
    if (Array.isArray(value)) return value.map(walk)
    if (!value || typeof value !== 'object') return value
    const obj = value as Record<string, unknown>
    if (isReferenceLeaf(obj)) return map(ref(obj, 'ref'))
    return Object.fromEntries(Object.entries(obj).map(([key, item]) => [key, walk(item)]))
  }
  return walk(record) as NarrativeRecordV1
}
