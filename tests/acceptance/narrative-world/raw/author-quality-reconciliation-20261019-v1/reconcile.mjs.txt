import { readFileSync, writeFileSync, existsSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { resolve } from 'node:path'
const root = '/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge'
const base = 'tests/acceptance/narrative-world/'
const draft = JSON.parse(readFileSync('/tmp/nw-author-quality-reconciliation-20261018-v1.json', 'utf8'))
const read = path => JSON.parse(readFileSync(resolve(root, path), 'utf8'))
const hash = value => createHash('sha256').update(value).digest('hex')
const sourceHashes = { ...draft.sourceHashes }
for (const [path, expected] of Object.entries(sourceHashes)) {
  if (hash(readFileSync(resolve(root, path))) !== expected) throw Error('Changed review source: ' + path)
}
const human = read(base + 'author-quality-human-review-20261017-v1.json')
const model = read(base + 'raw/author-quality-20261016-v1/independent-judgment.json')
const run = read(base + 'author-quality-model-run-20261017-v1.json')
const packetPath = base + 'raw/author-quality-20261016-v1/blind-packet.json'
const packet = read(packetPath)
sourceHashes[packetPath] = hash(readFileSync(resolve(root, packetPath)))
if (hash(JSON.stringify(packet)) !== draft.packetHash || human.packetHash !== draft.packetHash || model.packetHash !== draft.packetHash || run.packetHash !== draft.packetHash) throw Error('Review packet mismatch')
const comparisons = packet.items.map(item => {
  const h = human.judgments.find(row => row.label === item.label)
  const m = model.judgment.judgments.find(row => row.label === item.label)
  if (!h || !m || h.outputHash !== item.outputHash || hash(JSON.stringify(item.writing)) !== item.outputHash) throw Error('Unbound paired output')
  return { label: item.label, outputHash: item.outputHash,
    human: { consistency: h.consistency, consistencyChange: h.consistencyChange, motivation: h.motivation, acceptance: h.acceptance,
      hardContradictions: h.hardContradictions, reason: h.reason, editRecord: h.editRecord, elapsedSeconds: h.elapsedSeconds },
    independentModel: m,
    agreement: { consistencyChange: h.consistencyChange === m.consistencyChange,
      noHardContradiction: h.hardContradictions.length === 0 && m.hardContradictions.length === 0 },
    scoreDifferenceHumanMinusModel: { consistency: h.consistency - m.consistency, motivation: h.motivation - m.motivation, acceptance: h.acceptance - m.acceptance },
  }
})
const receipt = { schema: 'storyforge.narrative-author-quality-reconciliation', version: 1,
  generatedAt: new Date().toISOString(), evidenceLevel: 'Joined existing independent model and explicit human judgments; no new judgment',
  packetHash: draft.packetHash, humanReviewerId: human.reviewerId, humanCompletedAt: human.completedAt,
  provider: run.provider, writerModel: run.writerModel, independentReviewerModel: model.model,
  comparisons, actualNewModelCalls: 0, actualNewHumanReviews: 0, existingHumanReviewers: 1,
  totalsFromExistingModelRun: run.totals, knownCostUsd: run.costUsd, sourceHashes,
  accepted: false, qualityState: 'limited-reviewed-pair; actual author edit/time records missing',
  limitations: ['One familiar synthetic story and one author, not arbitrary narrative quality.',
    'Human motivation and acceptance scores are lower than the independent model scores; differences are retained.',
    'Null edit records and elapsed time remain unknown; no inference of no edits or zero time.',
    'Equal paired scores and different actual context costs do not establish IR superiority.',
    'This joined ledger does not alter historical model-run pending status, original outputs or any review.'] }
const target = '/tmp/nw-author-quality-reconciliation-20261019-v1.json'
if (existsSync(target)) throw Error('Receipt destination exists; retain immutable evidence')
writeFileSync(target, JSON.stringify(receipt, null, 2) + '\n', { flag: 'wx' })
process.stdout.write(JSON.stringify({ path: target, sha256: hash(readFileSync(target)), accepted: false, actualNewModelCalls: 0 }) + '\n')
