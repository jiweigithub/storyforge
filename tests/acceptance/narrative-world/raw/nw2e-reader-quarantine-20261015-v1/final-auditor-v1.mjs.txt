import assert from 'node:assert/strict'
import { readFile, readdir, writeFile } from 'node:fs/promises'
import { resolve, relative } from 'node:path'
import { pathToFileURL } from 'node:url'

const root = process.cwd()
const moduleAt = name => import(pathToFileURL(resolve(root, 'scripts', name)).href)
const { sha256, readerMessages, scoreReader } = await moduleAt('narrative-reader-contract.mjs')
const { readerEvaluationProtocol, readReaderQuarantine } = await moduleAt('narrative-reader-quarantine.mjs')
const { validateReaderCall, verifyReaderHistory, readerExecutionCounts } = await moduleAt('narrative-reader-continuation.mjs')
const { inspectReaderReceiptJournal } = await moduleAt('narrative-reader-storage.mjs')
const { scoreGroundedReaderV2, summarizeGroundedReaderV2 } = await moduleAt('narrative-reader-report.mjs')
const run = 'tests/acceptance/narrative-world/raw/reader-fixed-20261015-v5'
const output = process.argv[2]
assert(output && relative(root, resolve(root, output)).startsWith('tests/acceptance/narrative-world/'))
const terminal = JSON.parse(await readFile('/tmp/nw2e-reader-quarantine-remaining-v1-terminal.json', 'utf8'))
assert.equal(terminal.run, run)
assert(Number.isInteger(terminal.exitCode))
const hashes = {}, inventories = {}
async function bytes(path) {
  const value = await readFile(resolve(root, path))
  hashes[path] = sha256(value)
  return value
}
async function json(path) { return JSON.parse(await bytes(path)) }
const manifest = await json(`${run}/manifest.json`)
assert.equal(hashes[`${run}/manifest.json`], terminal.manifestSha256)
const inputs = await json(manifest.inputsPath), protocol = await json(manifest.protocolPath)
assert.equal(hashes[manifest.inputsPath], manifest.inputsSha256)
assert.equal(hashes[manifest.protocolPath], manifest.protocolSha256)
const corpusPath = 'tests/acceptance/narrative-world/corpus-v1.json', corpus = await json(corpusPath)
assert.equal(hashes[corpusPath], inputs.sourceHashes[corpusPath])
for (const [path, hash] of Object.entries(manifest.runtimeHashes)) assert.equal(sha256(await bytes(path)), hash, path)
const prepared = inputs.families.flatMap(family => family.cases).map(item => ({ item, request: readerMessages(item, readerEvaluationProtocol(protocol)) }))
const unique = new Map(prepared.map(({ request }) => [request.promptHash, request]))
assert.equal(prepared.length, 720)
assert.equal(unique.size, 461)
assert.deepEqual(prepared.map(({ item, request }) => ({ id: item.id, promptHash: request.promptHash,
  inputBytes: request.inputBytes, estimatedInputTokens: request.estimatedInputTokens, omittedDocumentIds: request.omittedDocumentIds })), manifest.cases)
const history = await readReaderQuarantine(root, manifest.recovery.run, manifest, unique, protocol)
assert.equal(sha256(history.evidence), sha256(manifest.recovery.evidence))
assert.equal(history.results.size, 70)
assert.equal(history.quarantined.size, 1)
assert.equal(sha256(await bytes(manifest.recovery.claimPath)), manifest.recovery.claimSha256)
Object.assign(hashes, history.evidence.files)
Object.assign(inventories, history.evidence.directories)
const results = new Map(history.results), provenance = new Map()
for (const [hash] of history.results) {
  const matches = Object.keys(history.evidence.files).filter(path => path.endsWith(`/calls/${hash}.result.json`))
  assert.equal(matches.length, 1)
  provenance.set(hash, matches[0])
}
const names = (await readdir(resolve(root, run, 'calls'))).sort(), files = new Set(names)
inventories[`${run}/calls`] = names
for (const name of names) {
  const match = /^([a-f0-9]{64})\.(request|started|result)\.json$/.exec(name)
  assert(match && unique.has(match[1]) && !history.results.has(match[1]) && !history.quarantined.has(match[1]), name)
}
const journals = []
for (const [hash, request] of unique) {
  if (history.results.has(hash) || history.quarantined.has(hash)) continue
  const prior = {}
  for (const kind of ['request', 'started', 'result']) if (files.has(`${hash}.${kind}.json`)) prior[kind] = await json(`${run}/calls/${hash}.${kind}.json`)
  // A new unknown is a failed audit, not a fabricated call or automatic retry.
  const value = validateReaderCall(prior, request, protocol)
  if (!value) continue
  results.set(hash, value); provenance.set(hash, `${run}/calls/${hash}.result.json`)
  const path = `${run}/receipt-journal/${hash}.jsonl`, journal = inspectReaderReceiptJournal(await bytes(path), hash)
  assert.equal(journal.corruption, null, path)
  assert.deepEqual(journal.terminalEvidence, value, path)
  if (value.status === 'completed') {
    assert.equal(journal.generationCompletedEvidence.rawText, value.rawText)
    assert.deepEqual(journal.generationCompletedEvidence.usage, value.usage)
  }
  journals.push({ promptHash: hash, path, sha256: hashes[path], status: value.status, events: journal.events.map(event => event.kind) })
}
const journalNames = (await readdir(resolve(root, run, 'receipt-journal'))).sort()
assert.deepEqual(journalNames.filter(name => name !== 'staging'), journals.map(item => `${item.promptHash}.jsonl`).sort())
inventories[`${run}/receipt-journal`] = journalNames
if (journalNames.includes('staging')) {
  const staging = await readdir(resolve(root, run, 'receipt-journal/staging'))
  assert.deepEqual(staging, []); inventories[`${run}/receipt-journal/staging`] = []
}
const runNames = (await readdir(resolve(root, run))).sort(); inventories[run] = runNames
const summaries = []
for (const name of runNames.filter(name => /^summary-.*\.json$/.test(name))) {
  const summary = await json(`${run}/${name}`)
  if (!summary.dryRun && summary.manifestSha256 === sha256(manifest) && summary.counts.recordedAttempts === results.size + history.quarantined.size) summaries.push({ name, summary })
}
summaries.sort((a, b) => b.summary.savedAt.localeCompare(a.summary.savedAt))
assert(summaries.length > 0, 'No matching terminal summary')
const { name: summaryName, summary } = summaries[0]
const values = [...results.values()]
const expectedCounts = { ...readerExecutionCounts(values, unique.size), recordedAttempts: results.size + history.quarantined.size,
  quarantinedUnknownAttempts: history.quarantined.size, unknownDispatchAttempts: history.quarantined.size,
  unpersistedAttempts: 0, unattempted: unique.size - results.size - history.quarantined.size }
assert.deepEqual(summary.counts, expectedCounts)
assert.equal(summary.executionComplete, false)
assert.equal(summary.accepted, false)
assert.equal(summary.historicalEvidenceUnchanged, true)
assert.equal(summary.evaluationEvidenceUnchanged, true)
assert.deepEqual(summary.unpersistedAttempts, [])
const rows = [], observations = []
for (const { item, request } of prepared) {
  const value = results.get(request.promptHash), original = corpus.families.find(family => family.id === item.familyId)
  const status = value ? value.status : history.quarantined.has(request.promptHash) ? 'unknown' : 'unattempted'
  const metadata = { id: item.id, questionId: item.question.id, familyId: item.familyId, split: item.split,
    operation: item.question.operation, condition: item.condition, promptHash: request.promptHash }
  observations.push({ ...metadata, status, resultPath: provenance.get(request.promptHash) ?? null,
    unknownStartPath: history.quarantined.get(request.promptHash)?.startReceipt ?? null })
  if (value?.status !== 'completed') continue
  const prior = summary.rows.find(row => row.id === item.id)
  assert(prior, item.id)
  assert.equal(prior.promptHash, request.promptHash)
  assert.deepEqual(prior.answer, value.answer)
  assert.deepEqual(prior.usage, value.usage)
  assert.equal(prior.durationMs, value.durationMs)
  assert.deepEqual(prior.score, scoreReader(item, request, value.answer, original))
  rows.push({ ...metadata, inputBytes: request.inputBytes, estimatedInputTokens: request.estimatedInputTokens,
    omittedDocuments: request.omittedDocumentIds, retrievalStatus: item.context.status,
    answer: value.answer, usage: value.usage, durationMs: value.durationMs,
    score: scoreGroundedReaderV2(item, request, value.answer, original) })
}
assert.equal(rows.length, summary.rows.length)
assert.equal(rows.length, summary.completedCases)
assert.equal(new Set(summary.rows.map(row => row.id)).size, rows.length)
const statuses = ['completed', 'failed', 'unknown', 'unattempted']
const denominators = (key) => Object.fromEntries([...new Set(observations.map(row => row[key]))].map(group => {
  const selected = observations.filter(row => row[key] === group)
  return [group, { planned: selected.length, ...Object.fromEntries(statuses.map(status => [status, selected.filter(row => row.status === status).length])) }]
}))
const total = Object.fromEntries(statuses.map(status => [status, observations.filter(row => row.status === status).length]))
assert.equal(Object.values(total).reduce((sum, count) => sum + count, 0), 720)
for (const path of ['scripts/narrative-reader-report.mjs', 'scripts/narrative-reader-report.test.mjs']) await bytes(path)
await verifyReaderHistory(root, history.evidence)
for (const [path, expected] of Object.entries(hashes)) assert.equal(sha256(await readFile(resolve(root, path))), expected, path)
for (const [path, expected] of Object.entries(inventories)) assert.deepEqual((await readdir(resolve(root, path))).sort(), expected, path)
const receipt = {
  schema: 'storyforge.narrative-reader-incomplete-diagnostic', version: 1, createdAt: new Date().toISOString(),
  run, terminal, terminalSummary: `${run}/${summaryName}`, manifestSha256: hashes[`${run}/manifest.json`],
  counts: expectedCounts, conditionObservations: { planned: 720, ...total },
  denominatorByCondition: denominators('condition'), denominatorBySplit: denominators('split'),
  denominatorByFamily: denominators('familyId'), denominatorByOperation: denominators('operation'),
  newCohort: { calls: journals.length, completed: journals.filter(row => row.status === 'completed').length,
    failed: journals.filter(row => row.status === 'failed').length, independentlyReplayedJournals: journals.length },
  immutableEvidence: { historicalFiles: Object.keys(history.evidence.files).length,
    checkedFiles: Object.keys(hashes).length, checkedDirectories: Object.keys(inventories).length, unchanged: true },
  protocol: { provider: protocol.provider, model: protocol.model, retries: protocol.retries, historicalRepeatedInputs: 0 },
  totalObservedUsage: summary.totalObservedUsage, costUsd: null,
  metricsOnSuccessfulResponsesOnly: summarizeGroundedReaderV2(rows),
  scoringBoundary: 'This separately named incomplete diagnostic replays grounded-quotation scorer v2 on actual successful responses only. Missing/failed/unknown observations remain in explicit planned denominators and are never filled in as answers or judgments. Full-report and independent-judge completeness gates are unchanged.',
  evidenceBoundary: 'Synthetic engineering scenes, fixed reader, shared identical prompts. Not an independent semantic review, fresh unseen holdout, author judgment or creative-quality acceptance.',
  summarySourceDrift: summary.sourceDrift, observations, rows, journals, hashes, inventories,
  cohortFinishedUnattemptedInputs: summary.cohort.completed, executionComplete: false,
  independentHumanReviews: 0, semanticAcceptance: false, accepted: false,
}
await writeFile(resolve(root, output), JSON.stringify(receipt, null, 2) + '\n', { flag: 'wx' })
console.log(JSON.stringify({ output, counts: expectedCounts, observations: receipt.conditionObservations,
  journalsVerified: journals.length, filesVerified: receipt.immutableEvidence.checkedFiles, accepted: false }))
