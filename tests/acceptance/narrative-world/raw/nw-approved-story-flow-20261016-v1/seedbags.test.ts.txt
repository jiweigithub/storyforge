import { afterEach, expect, it } from 'vitest'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { db } from '../../src/lib/db/schema'
import { createWorkspace } from '../../src/lib/workspace/create-workspace'
import { stampNewRecord } from '../../src/lib/workspace/scope'
import { captureNarrativeDomainV1, hashNarrativeDomainSourceV1 } from '../../src/lib/world-engine/narrative-domain-capture'
import { confirmNarrativeDomainAnnotationsV1 } from '../../src/lib/world-engine/narrative-domain-authoring'
import { prepareNarrativeCandidateV1 } from '../../src/lib/world-engine/narrative-domain-candidate'
import { generateNarrativeAnnotationCandidateV1, adoptNarrativeAnnotationCandidateV1 } from '../../src/lib/agent/run/narrative-annotation-durable'
import { deriveNarrativeNovelToWorldV4 } from '../../src/lib/world-engine/narrative-derivation'
import { verifyWorldReleaseRecordV1 } from '../../src/lib/world-engine/release-codec'
import { authorWorldQueryV1, openAuthorWorldQueryAccessV1, queryWorldReleaseV1, closeWorldQueryAccessV1 } from '../../src/lib/context-gateway/world-query-client'

const sha = (value: Uint8Array | string) => createHash('sha256').update(value).digest('hex')
const folder = 'tests/acceptance/narrative-world/'
const sourcePath = folder + 'target-state-independent-story-eval-20261015-v1.json'
const reviewPath = folder + 'author-review-status-20261016-v2.json'
const sourceBytes = readFileSync(sourcePath), reviewBytes = readFileSync(reviewPath)
const history = JSON.parse(sourceBytes.toString()), actual = history.cases[0].result
const review = JSON.parse(reviewBytes.toString())
const checkpoint = JSON.parse(readFileSync(folder + 'engineering-checkpoint-20261016-v30.json', 'utf8'))
const raw = actual.calls[0].output
const report: Record<string, any> = {
  schema: 'storyforge.author-reviewed-candidate-flow-diagnostic', version: 1,
  boundary: 'Offline replay of an unchanged real response with recorded human meaning review. Fixture transport and simulated adoption/derivation in isolated fake IndexedDB; not a new model call, native UI run, user-data adoption or complete quality acceptance.',
  source: { path: sourcePath, sha256: sha(sourceBytes), rawResponseSha256: sha(raw) },
  authorReview: { path: reviewPath, sha256: sha(reviewBytes), reviewedStory: '晒谷台上的灯' },
  sourceHashes: checkpoint.sourceHashes, actualNewModelCalls: 0, fixtureReplayDeliveries: 0,
  simulatedAdoption: true, completedStages: [], completed: false, accepted: false,
}
const stage = (name: string, data: unknown) => report.completedStages.push({ name, data })
const verifySources = () => {
  for (const [path, hash] of Object.entries(checkpoint.sourceHashes)) expect(sha(readFileSync(path)), path).toBe(hash)
  expect(sha(readFileSync(sourcePath))).toBe(sha(sourceBytes))
  expect(sha(readFileSync(reviewPath))).toBe(sha(reviewBytes))
}
afterEach(() => db.close())

it('author-reviewed raw candidate survives simulated adoption, explicit derivation and frozen queries', async () => {
  try {
    verifySources()
    expect(review.referenceHashes[sourcePath]).toBe(sha(sourceBytes))
    expect(review.assessments.find((item: any) => item.story === '晒谷台上的灯').adjudication.authorAcceptsActualCandidate).toBe(true)
    await db.delete(); await db.open()
    const created = await createWorkspace({ name: '已抽检候选贯穿诊断', genres: [], description: '隔离评测数据', status: 'drafting', targetWordCount: 2000, enableMultiWorld: false },
      { purpose: 'independent-work', kind: 'novel', novelProfile: 'long' })
    const { scope } = created, owner = { kind: 'work' as const, workCode: created.work.code }, now = Date.now()
    const add = async (table: string, fields: Record<string, unknown>, ownership: 'world' | 'work') =>
      Number(await db.table(table).add(stampNewRecord(scope, table, { ...fields, createdAt: now, updatedAt: now }, { owner: ownership })))
    const confirm = async (table: string, recordId: number, annotations: any) => {
      const capture = await captureNarrativeDomainV1({ scope, owner }), row = capture.rows.get(table)!.find(item => item.id === recordId)!
      await confirmNarrativeDomainAnnotationsV1({ scope, table, recordId,
        expectedSourceHash: await hashNarrativeDomainSourceV1(capture, table, row), annotations })
    }
    const metadata = { narrativeVisibilityV1: { class: 'unclassified', audienceKeys: [], policyVersion: 'evaluation-source-v1' } }
    const categoryId = await add('codexCategories', { domain: 'humanity', parentId: null, name: '评测物品', fieldSchema: '[]', order: 0 }, 'world')
    let targetId = 0, chapterId = 0
    for (const field of actual.baseline.originalFields) {
      const ragDocumentId = field.sourceUid
      if (field.table === 'characters') {
        const id = await add('characters', { ragDocumentId, name: field.text, shortDescription: '', roleWeight: 'main', moralAxis: 'neutral', orderAxis: 'neutral', background: '', personality: '' }, 'world')
        await confirm('characters', id, metadata)
      } else if (field.table === 'codexEntries') {
        const id = await add('codexEntries', { ragDocumentId, categoryId, name: field.text, description: '', fields: '{}', refs: '{}', origin: 'manual', sourceEvidenceQuotes: '[]', sourceContentHash: '', producerRunId: null, producerCandidateHash: null, order: 0 }, 'world')
        await confirm('codexEntries', id, metadata)
      } else if (field.table === 'chapters') {
        const outlineNodeId = await add('outlineNodes', { parentId: null, type: 'chapter', title: history.corpus.title, summary: '', order: 0 }, 'work')
        chapterId = await add('chapters', { ragDocumentId, outlineNodeId, title: history.corpus.title, content: field.text, summary: '', wordCount: [...field.text].length, notes: '', status: 'draft', order: 0 }, 'work')
      } else if (field.table === 'foreshadows') {
        targetId = await add('foreshadows', { ragDocumentId, name: history.corpus.cases[0].title, type: 'dialogue', status: history.corpus.cases[0].initialBusinessStatus,
          description: field.text, notes: '', plantChapterId: null, echoChapterIds: '[]', resolveChapterId: null }, 'work')
      } else throw Error('Unexpected original source table: ' + field.table)
    }
    expect(targetId).toBeGreaterThan(0)
    await confirm('foreshadows', targetId, actual.baseline.targetState.annotations)
    const selection = { table: 'foreshadows', sourceUid: actual.baseline.targetState.sourceUid, owner,
      entryKeys: actual.baseline.originalFields.map((field: any) => field.key) }
    const prepared = await prepareNarrativeCandidateV1({ scope, selection })
    const selected = (fields: any[]) => fields.map(({ key, text, contentHash }) => ({ key, text, contentHash }))
    expect(selected(prepared.baseline.originalFields)).toEqual(selected(actual.baseline.originalFields))
    expect(prepared.baseline.targetState?.annotations).toEqual(actual.baseline.targetState.annotations)
    expect((await db.foreshadows.get(targetId))!.status).toBe('planted')
    stage('original-input-and-prior-identities', { exactFields: prepared.baseline.originalFields.length })

    const sourceWork = await db.works.get(scope.workId), sourceChapter = await db.chapters.get(chapterId)
    const prior = await deriveNarrativeNovelToWorldV4({ sourceScope: scope, targetName: '候选采纳前的测试封存', publish: true })
    expect(prior.release).not.toBeNull()
    expect(await db.works.get(scope.workId)).toEqual(sourceWork)
    expect(await db.chapters.get(chapterId)).toEqual(sourceChapter)
    stage('prior-release', { contentHash: prior.release!.contentHash })

    const before = await db.foreshadows.get(targetId)
    const generated = await generateNarrativeAnnotationCandidateV1({ scope, selection,
      instruction: '离线复核已获作者认可的原始响应；不发起模型请求。',
      modelIdentity: { provider: 'fixture', model: 'author-reviewed-original-response-replay' },
      runAI: async () => { report.fixtureReplayDeliveries++; return { output: raw, durationMs: 0 } } })
    expect(generated.candidate.artifact.status).toBe('ready')
    expect(await db.foreshadows.get(targetId)).toEqual(before)
    await adoptNarrativeAnnotationCandidateV1({ scope, runId: generated.snapshot.run.id, candidateHash: generated.candidate.candidateHash })
    const adopted = (await db.foreshadows.get(targetId))!
    expect(adopted.status).toBe('planted')
    expect(adopted.narrativeDevelopmentsV1!.map(child => [child.semanticSubKey, child.state, child.narrativeModalityV1]))
      .toEqual(JSON.parse(raw).annotations.narrativeDevelopmentsV1.map((child: any) => [child.semanticSubKey, child.state, child.narrativeModalityV1]))
    stage('simulated-author-adoption', { candidateHash: generated.candidate.candidateHash, children: adopted.narrativeDevelopmentsV1 })

    const derived = await deriveNarrativeNovelToWorldV4({ sourceScope: scope, targetName: '已抽检候选的新测试封存', publish: true })
    expect(await db.foreshadows.get(targetId)).toEqual(adopted)
    expect(await db.works.get(scope.workId)).toEqual(sourceWork)
    expect(await db.chapters.get(chapterId)).toEqual(sourceChapter)
    expect((await db.worldReleases.get(prior.release!.id!))!.manifestJson).toBe(prior.release!.manifestJson)
    const release = derived.release!, verified = await verifyWorldReleaseRecordV1(release)
    if (verified.version !== 4) throw Error('Expected frozen v4')
    const developments = verified.manifest.narrative.bundle.records.filter(record => record.kind === 'development')
    expect(developments.map(record => [record.payload.state, record.modality]).sort()).toEqual([['blocked', 'intent'], ['fulfilled', 'established']])
    stage('explicit-work-to-world-derivation', { contentHash: release.contentHash, oldReleasePreserved: true, sourceOwnerAndTextPreserved: true, developments })

    for (const [view, state, modality] of [['development', 'fulfilled', 'established'], ['author-plan', 'blocked', 'intent']] as const) {
      const access = await openAuthorWorldQueryAccessV1({ scope: derived.targetScope, localReleaseRecordId: release.id!, view,
        perspective: { kind: 'author' }, knowledgeThrough: null, disclosedThrough: null })
      try {
        const result = await queryWorldReleaseV1(access, { ...authorWorldQueryV1(access), text: null, kinds: ['development'], evidenceDepth: 'original' })
        report['query-' + view] = result
        expect(result.items).toHaveLength(1)
        expect(result.items[0].atom).toHaveProperty('state', state)
        expect(result.items[0].modality).toBe(modality)
        expect(result.items[0].evidence.length).toBeGreaterThan(0)
        expect(result.items[0].evidence.every(span => actual.baseline.originalFields.some((field: any) => field.text.includes(span.text)))).toBe(true)
        stage('query-' + view, { state, modality, evidenceSpans: result.items[0].evidence.length })
      } finally { await closeWorldQueryAccessV1(access) }
    }
    expect(report.fixtureReplayDeliveries).toBe(1)
    verifySources(); report.completed = true
  } catch (error) {
    report.error = error instanceof Error ? { name: error.name, message: error.message } : String(error)
    throw error
  } finally {
    report.sourceFilesUnchanged = Object.entries(checkpoint.sourceHashes).every(([path, h]) => sha(readFileSync(path)) === h)
    writeFileSync(process.env.NW_APPROVED_FLOW_REPORT!, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' })
  }
}, 30000)
