import { afterEach, describe, expect, it } from 'vitest'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { db } from '../../src/lib/db/schema'
import { createWorkspace } from '../../src/lib/workspace/create-workspace'
import { stampNewRecord } from '../../src/lib/workspace/scope'
import { captureNarrativeDomainV1, hashNarrativeDomainSourceV1 } from '../../src/lib/world-engine/narrative-domain-capture'
import { confirmNarrativeDomainAnnotationsV1 } from '../../src/lib/world-engine/narrative-domain-authoring'
import { prepareNarrativeCandidateV1, compileNarrativeAnnotationProposalV1, assertNarrativeCandidateBaselineV1 } from '../../src/lib/world-engine/narrative-domain-candidate'
import { generateNarrativeAnnotationCandidateV1, adoptNarrativeAnnotationCandidateV1 } from '../../src/lib/agent/run/narrative-annotation-durable'
import { useForeshadowStore } from '../../src/stores/foreshadow'

const file = 'tests/acceptance/narrative-world/target-state-story-eval-20261015-v1.json'
const bytes = readFileSync(file), sha = (s: Uint8Array | string) => createHash('sha256').update(s).digest('hex')
const historical = JSON.parse(bytes.toString()), actual = historical.cases[0].result
const results: unknown[] = []
afterEach(() => { db.close() })

describe('Diagnostic only: unchanged real responses under explicit business admission', () => {
  for (const call of actual.calls) it('immutable response ' + call.index, async () => {
    await db.delete(); await db.open()
    const created = await createWorkspace({ name: '离线准入诊断', genres: [], description: '', status: 'drafting', targetWordCount: 2000, enableMultiWorld: false }, { purpose: 'independent-work', kind: 'novel', novelProfile: 'long' })
    const { scope } = created, owner = { kind: 'work' as const, workCode: created.work.code }, now = Date.now()
    const add = async (table: string, fields: Record<string, unknown>, ownership: 'world' | 'work') => db.table(table).add(stampNewRecord(scope, table, { ...fields, createdAt: now, updatedAt: now }, { owner: ownership }))
    const categoryId = await add('codexCategories', { domain: 'humanity', parentId: null, name: '评测物品', fieldSchema: '[]', order: 0 }, 'world')
    let targetId = 0
    for (const field of actual.baseline.originalFields) {
      const ragDocumentId = field.sourceUid
      if (field.table === 'characters') await add('characters', { ragDocumentId, name: field.text, shortDescription: '', roleWeight: 'main', moralAxis: 'neutral', orderAxis: 'neutral', background: '', personality: '' }, 'world')
      else if (field.table === 'codexEntries') await add('codexEntries', { ragDocumentId, categoryId, name: field.text, description: '', fields: '{}', refs: '{}', origin: 'manual', sourceEvidenceQuotes: '[]', sourceContentHash: '', producerRunId: null, producerCandidateHash: null, order: 0 }, 'world')
      else if (field.table === 'chapters') {
        const outlineNodeId = await add('outlineNodes', { parentId: null, type: 'chapter', title: '风里的铜铃', summary: '', order: 0 }, 'work')
        await add('chapters', { ragDocumentId, outlineNodeId, title: '风里的铜铃', content: field.text, summary: '', wordCount: [...field.text].length, notes: '', status: 'draft', order: 0 }, 'work')
      } else if (field.table === 'foreshadows') targetId = Number(await add('foreshadows', { ragDocumentId, name: '铜铃归还与第一名孩子上岸', type: 'dialogue', status: 'planned', description: field.text, notes: '', plantChapterId: null, echoChapterIds: '[]', resolveChapterId: null }, 'work'))
      else throw Error('Unexpected historical field')
    }
    expect(targetId).toBeGreaterThan(0)
    const initial = await captureNarrativeDomainV1({ scope, owner }), row = initial.rows.get('foreshadows')!.find(row => row.id === targetId)!
    await confirmNarrativeDomainAnnotationsV1({ scope, table: 'foreshadows', recordId: targetId,
      expectedSourceHash: await hashNarrativeDomainSourceV1(initial, 'foreshadows', row), annotations: actual.baseline.targetState.annotations })
    const selection = { table: 'foreshadows', sourceUid: actual.baseline.targetState.sourceUid, owner, entryKeys: actual.baseline.originalFields.map((field: { key: string }) => field.key) }
    const planned = await prepareNarrativeCandidateV1({ scope, selection })
    expect(planned.baseline.originalFields.map(field => [field.key, field.text])).toEqual(actual.baseline.originalFields.map((field: { key: string; text: string }) => [field.key, field.text]))
    await expect(compileNarrativeAnnotationProposalV1({ ...planned, raw: call.output })).rejects.toThrow('计划不能冒充已兑现')
    const plannedBefore = await db.foreshadows.get(targetId)
    expect(plannedBefore!.status).toBe('planned')
    // Existing author-facing store operation, only in this isolated fake DB.
    await useForeshadowStore.getState().loadAll(scope)
    await useForeshadowStore.getState().updateStatus(targetId, 'planted')
    await expect(assertNarrativeCandidateBaselineV1(await captureNarrativeDomainV1({ scope, owner }), planned.baseline)).rejects.toThrow('变化')
    const fresh = await prepareNarrativeCandidateV1({ scope, selection })
    const compiled = await compileNarrativeAnnotationProposalV1({ ...fresh, raw: call.output })
    const illegal = JSON.parse(call.output); illegal.annotations.status = 'resolved'
    await expect(compileNarrativeAnnotationProposalV1({ ...fresh, raw: JSON.stringify(illegal) })).rejects.toThrow()
    const before = await db.foreshadows.get(targetId)
    const generated = await generateNarrativeAnnotationCandidateV1({ scope, selection, instruction: '离线回放既有响应，不调用模型。',
      modelIdentity: { provider: 'fixture', model: 'immutable-output-replay-not-model-acceptance' },
      runAI: async () => ({ output: call.output, durationMs: 0 }) })
    expect(generated.candidate.artifact.status).toBe('ready')
    expect(await db.foreshadows.get(targetId)).toEqual(before)
    await adoptNarrativeAnnotationCandidateV1({ scope, runId: generated.snapshot.run.id, candidateHash: generated.candidate.candidateHash })
    const adopted = (await db.foreshadows.get(targetId))!
    expect(adopted.status).toBe('planted')
    expect(adopted.narrativeAuthorityV1!.mode).toBe('candidate-adoption')
    expect(adopted.narrativeDevelopmentsV1!.map(child => [child.semanticSubKey, child.state])).toEqual(JSON.parse(call.output).annotations.narrativeDevelopmentsV1.map((child: { semanticSubKey: string; state: string }) => [child.semanticSubKey, child.state]))
    expect(sha(readFileSync(file))).toBe(sha(bytes))
    results.push({ historicalCall: call.index, originalResponseHash: sha(call.output), exactOriginalFieldCount: planned.baseline.originalFields.length,
      plannedRejected: true, authorStoreStatusAfterExplicitChange: before!.status, oldBaselineRejected: true, newCandidateReady: true,
      modelCannotWriteBusinessStatus: true, sourceUnchangedBeforeAdoption: true, simulatedAdoption: true,
      finalBusinessStatus: adopted.status, finalChildren: adopted.narrativeDevelopmentsV1!.map(child => ({ semanticSubKey: child.semanticSubKey, state: child.state })),
      portableAnnotations: compiled.portableAnnotations })
    writeFileSync('/tmp/nw1c-target-admission-replay-v1.json', JSON.stringify({ schema: 'storyforge.narrative-target-admission-offline-replay', version: 1,
      evidence: { path: file, sha256: sha(bytes) }, actualNewModelCalls: 0, humanAcceptance: false, historicalCaseAccepted: false,
      boundary: 'Offline counterfactual replay of unchanged successful responses. Only isolated fixture business status is explicitly changed through the existing author store. This neither retries the model nor repairs/relabels the rejected historical candidate.', results }, null, 2) + '\n')
  }, 20000)
})
