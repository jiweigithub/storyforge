import type { NarrativeOwnerV1 } from './narrative-ir'

export interface WorldImpactRevisionVectorV1 {
  semanticRootHash: string
  revisionVectorHash: string
  timeContextHash: string
  records: Array<{ semanticId: string; revisionHash: string }>
  sources: Array<{ sourceKey: string; fieldPointer: string; sourceRevision: string; contentHash: string }>
  originalFields: Array<{ sourceKey: string; fieldPointer: string; contentHash: string }>
}
export interface WorldImpactEdgeV1 {
  /** From the changed input to the artifact which consumes it. */
  from: string
  to: string
  kind: 'evidence' | 'provenance' | 'identity' | 'dependency' | 'sequence' | 'time-context'
  strength: 'hard' | 'review'
  role: string
  evidenceIds: string[]
}
export interface WorldImpactItemV1 {
  key: string
  label: string
  kind: 'record' | 'source' | 'time-context'
  disposition: 'rebuild' | 'stale' | 'review' | 'unaffected'
  operation: 'added' | 'removed' | 'changed' | 'retained'
  oldHash: string | null
  newHash: string | null
  fieldDiffs: Array<{ pointer: string; oldHash: string | null; newHash: string | null }>
  /** One bounded witness per item; the complete direct graph is separately
   * retained, so cycles do not cause exponential path materialization. */
  path: WorldImpactEdgeV1[]
  pathRootKey: string | null
  pathComplete: boolean
  evidenceIds: string[]
  unknownEdges: string[]
  confidence: 'deterministic' | 'unknown-impact'
  locked: boolean
  authorDecisionRequired: boolean
}
export interface WorldImpactPlanV1 {
  schema: 'storyforge.world-impact-plan'
  version: 1
  algorithmVersion: 'world-impact-evidence-v1'
  operation: 'compare-draft-revision'
  operationKey: string
  scope: { owner: NarrativeOwnerV1; branchKey: string }
  baseRevisionVector: WorldImpactRevisionVectorV1
  nextRevisionVector: WorldImpactRevisionVectorV1
  changedSemanticIds: string[]
  dependencyGraphHash: string
  edges: WorldImpactEdgeV1[]
  items: WorldImpactItemV1[]
  rebuildSourceKeys: string[]
  staleSemanticIds: string[]
  reviewKeys: string[]
  unknownCoverage: string[]
  protectedSemanticIds: string[]
  writes: []
  frozenArtifactsChanged: 0
  planHash: string
}
