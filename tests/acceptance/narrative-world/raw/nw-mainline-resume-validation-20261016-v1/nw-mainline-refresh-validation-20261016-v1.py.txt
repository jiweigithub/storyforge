from pathlib import Path
import datetime, hashlib, json, os, shutil, signal, socket, subprocess, sys

root = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
prefix = Path('/tmp/nw-mainline-refresh-validation-20261016-v1')
pin_path = Path('/tmp/nw-mainline-refresh-validation-sources-20261016-v1.json')
pin = json.loads(pin_path.read_text())
terminal = Path(str(prefix) + '-terminal.json')
baseline = root / 'tests/acceptance/narrative-world/baseline-20261016-nw2e-v43.json'
profile_prefix = Path('/tmp/nw-mainline-refresh-profile-20261016-v1')
profile_report = Path(str(profile_prefix) + '.json')
assert not terminal.exists() and not baseline.exists() and not profile_report.exists()

def verify():
    assert Path.cwd() == root
    for ref, expected in [('HEAD', pin['head']), ('MERGE_HEAD', pin['mergeHead'])]:
        assert subprocess.check_output(['git', 'rev-parse', ref], text=True).strip() == expected
    bad = [name for name, digest in pin['files'].items()
           if not (root / name).is_file() or hashlib.sha256((root / name).read_bytes()).hexdigest() != digest]
    assert not bad, 'Pinned source drift: ' + json.dumps(bad)

def stop(signum, _frame):
    raise SystemExit(128 + signum)

signal.signal(signal.SIGTERM, stop)
record = {'schema': 'storyforge.nw-resumed-validation-terminal', 'version': 1,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'sourceRoot': str(root), 'head': pin['head'], 'mergeParent': pin['mergeHead'],
          'sourceFiles': len(pin['files']), 'pinSha256': hashlib.sha256(pin_path.read_bytes()).hexdigest(),
          'stages': [], 'sourceUnchanged': False, 'modelCalls': 0, 'fullCI': False,
          'formalPerformanceAcceptance': False, 'accepted': False}

def run_phase(name, args, env):
    verify()
    free = shutil.disk_usage(root).free / 1024**3
    assert free >= 5, 'Insufficient free space; stage not dispatched'
    log = Path(str(prefix) + '-' + name + '.log')
    stage = {'name': name, 'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
             'command': args, 'log': str(log), 'freeGiB': free}
    record['stages'].append(stage)
    print('Starting ' + name, flush=True)
    with log.open('x') as output:
        result = subprocess.call(args, env=env, stdout=output, stderr=subprocess.STDOUT)
    stage.update(exitCode=result, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    print('Finished ' + name + '; exit=' + str(result), flush=True)
    verify()
    if result:
        raise RuntimeError(name + ' exited ' + str(result) + '; later stages not dispatched')

code = 2
try:
    verify()
    tests = ['tests/regression/R-NW-0B-evidence.test.ts',
             'tests/regression/R-NW-1A-ir-codec.test.ts',
             'tests/regression/R-NW-1B-corpus.test.ts',
             'tests/regression/R-NW-2A-package.test.ts',
             'tests/regression/R-NW-2A-lineage.test.ts',
             'tests/regression/R-NW-2C-source-plan.test.ts',
             'tests/regression/R-AGENT1-chat-copilot-outline.test.ts',
             'tests/regression/R-HARNESS1-master-durable-orchestrator.test.ts',
             'tests/regression/R-CLIENT-native-transport.test.ts',
             'tests/regression/R-CLIENT-save-file.test.ts']
    assert all((root / test).is_file() for test in tests)
    env = dict(os.environ, NW_BASELINE_REPORT=str(baseline))
    run_phase('targeted', ['node', 'node_modules/vitest/vitest.mjs', 'run', *tests, '--reporter=verbose'], env)
    assert baseline.is_file(), 'Baseline receipt missing'
    current = json.loads(baseline.read_text())
    previous = json.loads((root / 'tests/acceptance/narrative-world/baseline-20261016-nw2e-v42.json').read_text())
    assert {k: v for k, v in current.items() if k not in ('input', 'run')} == {
        k: v for k, v in previous.items() if k not in ('input', 'run')}, 'Baseline observations changed; inspect before promotion'
    record['baseline'] = {'path': str(baseline.relative_to(root)),
                          'sha256': hashlib.sha256(baseline.read_bytes()).hexdigest(),
                          'observationsMatchV42': True, 'manifestNotPromoted': True}
    run_phase('types', ['node', 'node_modules/typescript/bin/tsc', '--noEmit'], dict(os.environ))
    with socket.socket() as sock:
        sock.bind(('127.0.0.1', 0)); port = sock.getsockname()[1]
    profile_env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1', PLAYWRIGHT_PORT=str(port),
                       PLAYWRIGHT_HTML_OUTPUT_DIR=str(profile_prefix) + '-html',
                       NW_SCALE_BROWSER_MODE='diagnostic', NW_SCALE_PROFILE='compiler-v2',
                       NW_SCALE_BROWSER_SIZES='1000000', NW_SCALE_BROWSER_REPORT=str(profile_report),
                       NW_SCALE_BROWSER_VECTORS='tests/acceptance/narrative-world/scale-vectors-20261015-v2.json',
                       NW_SCALE_BROWSER_SOURCES='/tmp/nw-mainline-refresh-profile-sources-20261016-v1.json',
                       NW_SCALE_BROWSER_PROFILE_DIR=str(profile_prefix))
    run_phase('cpu-profile', ['node', 'node_modules/@playwright/test/cli.js', 'test',
                             'tests/e2e/narrative-scale.spec.ts', '--reporter=line', '--retries=0',
                             '--output=' + str(profile_prefix) + '-test-results'], profile_env)
    result = json.loads(profile_report.read_text())
    assert result['profile'] == 'compiler-v2' and result['mode'] == 'diagnostic'
    assert result['sourceUnchanged'] and result['cpuProfiling'] and result['accepted'] is False
    record['profile'] = {'path': str(profile_report), 'sha256': hashlib.sha256(profile_report.read_bytes()).hexdigest(),
                         'failures': result['failures'], 'formalAcceptance': False}
    code = 0
except BaseException as error:
    record['error'] = type(error).__name__ + ': ' + str(error)
    code = error.code if isinstance(error, SystemExit) and isinstance(error.code, int) else 2
finally:
    try:
        verify(); record['sourceUnchanged'] = True
    except BaseException as error:
        record['sourceCheckError'] = type(error).__name__ + ': ' + str(error); code = 2
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    with terminal.open('x') as output:
        json.dump(record, output, ensure_ascii=False, indent=2); output.write('\n')
print('Resumed validation terminal exit=' + str(code), flush=True)
sys.exit(code)
