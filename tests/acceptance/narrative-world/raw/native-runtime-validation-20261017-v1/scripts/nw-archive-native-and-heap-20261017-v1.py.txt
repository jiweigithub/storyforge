"""Preserve exact terminal evidence. Run only after both owned sessions close."""
from pathlib import Path
import datetime
import gzip
import hashlib
import json
import re
import shutil
import subprocess
import sys

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
DEST = ROOT / 'tests/acceptance/narrative-world'
JOBS = {
    'heap': ('nw-retained-heap-20261017-v1', 'b5f0eedefe3b020d2f276600d8ae4327fd1eb6bd1894f92bc9c0e00f8dc1975c'),
    'native': ('nw-native-runtime-validation-20261017-v1', '02f698dbe97147de53b465fb3f8ef847b869251a656954fe069a19444f186d97'),
}


def digest(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def preserve(source, target):
    target.parent.mkdir(parents=True, exist_ok=True)
    if not target.exists():
        with source.open('rb') as original, target.open('xb') as output:
            shutil.copyfileobj(original, output)
    assert digest(target) == digest(source), 'Existing archive differs: ' + str(target)
    return {'path': str(target.relative_to(DEST)), 'bytes': target.stat().st_size, 'sha256': digest(target)}


def archive(kind, stem, expected_pin):
    prefix = Path('/tmp') / stem
    pin_path = Path(str(prefix) + '-sources.json')
    terminal_path = Path(str(prefix) + '-terminal.json')
    terminal = json.loads(terminal_path.read_text())
    assert terminal.get('endedAt') and isinstance(terminal.get('exitCode'), int)
    assert digest(pin_path) == expected_pin == terminal['pinSha256']
    pin = json.loads(pin_path.read_text())
    for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
        assert subprocess.check_output(['git', '-C', str(ROOT), 'rev-parse', ref], text=True).strip() == pin[key]
    hashes = pin.get('files', pin.get('sourceHashes'))
    assert isinstance(hashes, dict) and hashes
    changed = [name for name, sha in hashes.items() if not (ROOT / name).is_file() or digest(ROOT / name) != sha]
    assert not changed, 'Source changed since run: ' + json.dumps(changed)
    raw = DEST / 'raw' / stem.removeprefix('nw-')
    receipt = DEST / (stem.removeprefix('nw-').replace('-20261017-v1', '-result-20261017-v1') + '.json')
    if receipt.exists():
        saved = json.loads(receipt.read_text())
        assert saved['source']['pinSha256'] == expected_pin
        for item in saved['artifacts']:
            assert digest(DEST / item['path']) == item['sha256']
        print(json.dumps({'existingVerifiedReceipt': str(receipt)}))
        return

    files = []
    for path in sorted(prefix.parent.glob(prefix.name + '*')):
        if path.is_file() and path != pin_path and path.suffix in {'.json', '.log', '.cpuprofile', '.gz'}:
            files.append((path, Path(path.name)))
        elif path.is_dir() and path.name.endswith('-test-results'):
            files.extend((child, Path(path.name) / child.relative_to(path))
                         for child in sorted(path.rglob('*')) if child.is_file())
    scripts = ['/tmp/nw-archive-native-and-heap-20261017-v1.py']
    if kind == 'heap':
        scripts += ['/tmp/nw-retained-heap-runner-20261017-v1.py', '/tmp/nw-retained-heap-probe-20261017-v1.mjs',
                    '/tmp/nw-retained-heap-summary-20261017-v1.py', '/tmp/nw-cpu-profile-summary-20261017-v1.py',
                    '/tmp/nw-retained-heap-input-scope-review-20261017-v1.json',
                    '/tmp/nw-retained-heap-runner-20261017-v1-dispatched-original.py']
    else:
        scripts += ['/tmp/nw-native-runtime-validation-20261017-v1.py']
    for script in scripts:
        p = Path(script)
        assert p.is_file()
        # Text suffix keeps archived executable evidence outside build/lint discovery.
        files.append((p, Path('scripts') / (p.name + '.txt')))
    assert len({str(target) for _, target in files}) == len(files)
    assert shutil.disk_usage(ROOT).free >= sum(p.stat().st_size for p, _ in files) + 5 * 2**30
    inventory = [preserve(source, raw / relative) for source, relative in files]
    packed_pin = raw / (pin_path.name + '.gz')
    if not packed_pin.exists():
        with packed_pin.open('xb') as output:
            output.write(gzip.compress(pin_path.read_bytes(), mtime=0))
    assert hashlib.sha256(gzip.decompress(packed_pin.read_bytes())).hexdigest() == expected_pin
    inventory.append({'path': str(packed_pin.relative_to(DEST)), 'bytes': packed_pin.stat().st_size,
                      'sha256': digest(packed_pin), 'decompressedSha256': expected_pin})
    phases = []
    for item in terminal['phases']:
        log_path = Path(item['log'])
        assert digest(log_path) == item['logSha256']
        plain = re.sub(r'\x1b\[[0-9;]*m', '', log_path.read_text(errors='replace'))
        phases.append({**item, 'log': str((raw / log_path.name).relative_to(DEST)),
                       'browserSummaryLines': re.findall(r'^\s*\d+ (?:passed|failed|skipped)(?: [^\n]*)?$', plain, re.M)})
    result = {'schema': 'storyforge.narrative-' + kind + '-validation-result', 'version': 1,
              'recordedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
              'source': {'head': pin['head'], 'mergeHead': pin['mergeHead'], 'sourceFiles': len(hashes),
                         'pinSha256': expected_pin, 'currentPinnedFilesUnchanged': True},
              'terminal': str((raw / terminal_path.name).relative_to(DEST)),
              'exitCode': terminal['exitCode'], 'error': terminal.get('error'), 'phases': phases,
              'actualModelCalls': 0, 'performanceAcceptance': False, 'humanAcceptance': False,
              'accepted': False, 'artifacts': inventory}
    if kind == 'heap':
        probe_path = Path(str(prefix) + '-result.json')
        if probe_path.exists():
            probe = json.loads(probe_path.read_text())
            assert probe['diagnosticOnly'] and not probe['accepted']
            for item in probe['snapshots']:
                path = Path(item['path'])
                assert path.suffix == '.gz', 'Uncompressed snapshot requires review before archive'
                assert digest(path) == item['compressedSha256']
                h = hashlib.sha256()
                with gzip.open(path, 'rb') as stream:
                    for block in iter(lambda: stream.read(1024 * 1024), b''): h.update(block)
                assert h.hexdigest() == item['sha256']
            result['probe'] = {'path': str((raw / probe_path.name).relative_to(DEST)),
                               'sourceUnchanged': probe['sourceUnchanged'], 'error': probe.get('error'),
                               'compileMs': probe.get('compilation', {}).get('compileMs'),
                               'coldIndexMs': probe.get('cold', {}).get('coldIndexMs'),
                               'instrumentation': probe['instrumentation'],
                               'exactEvidenceQueries': [q.get('exactEvidenceFound') for q in probe.get('queries', [])]}
            for suffix, key in [('-summary.json', 'heapSummary'), ('-cpu-summary.json', 'cpuSummary')]:
                path = Path(str(prefix) + suffix)
                result[key] = str((raw / path.name).relative_to(DEST)) if path.exists() else None
        result['diagnosticOnly'] = True
        result['diagnosticInputScope'] = terminal['diagnosticInputScope']
        result['limitations'] = ['Forced GC and profiling alter measurements; timings and heap bytes are not acceptance RSS.',
                                'Reachable component self sizes are not dominator retained sizes and can overlap.']
    else:
        result['source']['runSourceUnchanged'] = terminal['sourceUnchanged']
        result['nativeTwoProductsPassed'] = terminal['nativeTwoProductsPassed']
        result['priorFullValidation'] = terminal['priorFullValidation']
        result['onlyChangedInput'] = terminal['onlyChangedInput']
        result['fullCIRerun'] = False
        result['fullE2ERerun'] = False
        result['baselineFingerprintRefreshStillRequired'] = True
    with receipt.open('x') as output:
        json.dump(result, output, ensure_ascii=False, indent=2)
        output.write('\n')
    print(json.dumps({'receipt': str(receipt), 'artifacts': len(inventory),
                      'archivedBytes': sum(item['bytes'] for item in inventory), 'accepted': False}))


def main():
    assert sys.argv[1:] == ['--sessions-closed'], 'Close both93465/92075 handles, then pass --sessions-closed'
    for stem, _ in JOBS.values():
        terminal = Path('/tmp') / (stem + '-terminal.json')
        assert terminal.is_file(), 'A pinned run is not terminal; no repository writes allowed'
    for kind, (stem, expected_pin) in JOBS.items():
        archive(kind, stem, expected_pin)


if __name__ == '__main__':
    main()
