from pathlib import Path
import datetime
import hashlib
import json
import os
import shutil
import socket
import subprocess
import sys

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
PREFIX = Path('/tmp/nw-native-runtime-validation-20261017-v1')
PIN_PATH = Path(str(PREFIX) + '-sources.json')
pin = json.loads(PIN_PATH.read_text())
assert Path.cwd() == ROOT
terminal = Path(str(PREFIX) + '-terminal.json')
assert not terminal.exists()

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

pin_sha = digest(PIN_PATH)

def verify():
    assert digest(PIN_PATH) == pin_sha
    assert digest(Path(__file__)) == pin['runnerSha256']
    for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
        assert subprocess.check_output(['git', 'rev-parse', ref], text=True).strip() == pin[key]
    changed = [name for name, sha in pin['files'].items()
               if not (ROOT / name).is_file() or digest(ROOT / name) != sha]
    assert not changed, 'Pinned source changed: ' + json.dumps(changed)

record = {'schema': 'storyforge.narrative-native-runtime-revalidation', 'version': 1,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'head': pin['head'], 'mergeHead': pin['mergeHead'], 'pinSha256': pin_sha,
          'sourceFiles': len(pin['files']), 'phases': [], 'sourceUnchanged': False,
          'priorFullValidation': 'bounded-token-validation-result-20261017-v1.json',
          'onlyChangedInput': 'tests/e2e/helpers/narrative-v4-production.ts',
          'nativeTwoProductsPassed': False, 'actualModelCalls': 0,
          'performanceAcceptance': False, 'humanAcceptance': False, 'accepted': False}
env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1')
for key in list(env):
    if key.startswith('NW_SCALE_'):
        env.pop(key, None)
with socket.socket() as sock:
    sock.bind(('127.0.0.1', 0))
    env['PLAYWRIGHT_PORT'] = str(sock.getsockname()[1])

code = 2
try:
    verify()
    assert shutil.disk_usage(ROOT).free >= 5 * 2**30, 'Less than5GiB free'
    commands = [
        ('native', ['node', 'node_modules/@playwright/test/cli.js', 'test',
                    'tests/e2e/narrative-source-plan.spec.ts', '--grep', 'selects v4 sources',
                    '--reporter=line', '--retries=0', '--output=' + str(PREFIX) + '-test-results']),
        ('lint', ['node', 'node_modules/eslint/bin/eslint.js', 'tests/e2e/helpers/narrative-v4-production.ts']),
        ('types', ['node', 'node_modules/typescript/bin/tsc', '--noEmit']),
    ]
    for name, command in commands:
        verify()
        log = Path(str(PREFIX) + '-' + name + '.log')
        item = {'name': name, 'command': command, 'log': str(log),
                'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat()}
        record['phases'].append(item)
        print(json.dumps({'starting': name, 'log': str(log)}), flush=True)
        with log.open('x') as output:
            result = subprocess.call(command, env=env, stdout=output, stderr=subprocess.STDOUT)
        item.update(exitCode=result, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat(), logSha256=digest(log))
        if name == 'native':
            record['nativeTwoProductsPassed'] = result == 0
        verify()
        print(json.dumps({'completed': name, 'exitCode': result}), flush=True)
    code = 0 if all(item['exitCode'] == 0 for item in record['phases']) else 1
except BaseException as error:
    record['error'] = type(error).__name__ + ': ' + str(error)
finally:
    try:
        verify()
        record['sourceUnchanged'] = True
    except BaseException as error:
        record['sourceCheckError'] = str(error)
        code = 2
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    with terminal.open('x') as output:
        json.dump(record, output, indent=2)
        output.write('\n')
    print(json.dumps({'terminal': str(terminal), 'exitCode': code}), flush=True)
sys.exit(code)
