import { canonicalNarrativeJsonV1, compareUnicodeCodePointsV1, hashNarrativeValueV1 } from '../world-engine/narrative-json'
import { parseWorldQueryGrantV1, parseWorldQueryV1, WORLD_QUERY_SORT_VERSION_V1, WORLD_QUERY_VERSION_V1, worldQueryFailV1 } from './world-query-contract'
import { matchWorldQueryTimeV1, validateWorldQueryTimeV1 } from './world-query-time'

interface VerifiedQueryInputV1 {
  reference: WorldReferenceV2
  projection: NarrativeDomainProjectionV1
  index: NarrativeIndexV1
}
interface QueryCursorV1 { version: 1; offset: number; queryHash: string; viewFingerprint: string; sortVersion: string }
const bytes = (text: string) => new TextEncoder().encode(text).length
const same = (a: unknown, b: unknown) => canonicalNarrativeJsonV1(a) === canonicalNarrativeJsonV1(b)
const sorted = (values: Iterable<string>) => [...new Set(values)].sort(compareUnicodeCodePointsV1)
const preview = (text: string, max: number) => Array.from(text).slice(0, max).join('')
const lexical = (text: string) => text.replace(/[A-Z]/g, char => char.toLowerCase())
const predicateLabel = (key: string) => WORLD_SEMANTIC_REGISTRY_V4.predicates.find(predicate => predicate.key === key)?.label ?? '属性'

function assertGrant(query: WorldQueryV1, grant: WorldQueryGrantV1, reference: WorldReferenceV2): void {
  if (grant.purpose === 'product-preparation' && query.evidenceDepth !== 'metadata') worldQueryFailV1('depth', '产品定向目录不能作为正式生产原文授权')
  if (query.worldReference.referenceHash !== reference.referenceHash || query.worldReference.releaseHash !== reference.releaseHash
    || grant.releaseReferenceHash !== reference.referenceHash || query.sourcePlanHash !== grant.sourcePlanHash
    || !grant.views.includes(query.view) || !same(query.perspective, grant.perspective)
    || query.branch.kind === 'canon' && query.branch.key !== grant.branchKey) {
    worldQueryFailV1('access-unavailable', '请求的版本、视角或用途不可用')
  }
  const portable = ({ localReleaseRecordId: _locator, ...body }: WorldQueryV1['worldReference']) => body
  if (!same(portable(query.worldReference), portable(reference))) worldQueryFailV1('reference', '冻结来源身份不一致')
  if (query.budget.maxTokens > grant.maxTokens || query.budget.maxRecords > grant.maxRecords
    || query.budget.maxEvidenceBytes > grant.maxEvidenceBytes) worldQueryFailV1('budget', '查询超过冻结读取授权预算')
  if (query.time?.axis === 'disclosed' && grant.perspective.kind !== 'author'
    && !grant.audienceKeys.includes(query.time.audienceKey!)) worldQueryFailV1('access-unavailable', '请求的版本、视角或用途不可用')
}

function decodeCursor(value: string | null, queryHash: string, viewFingerprint: string): number {
  if (!value) return 0
  try {
    const cursor = JSON.parse(decodeURIComponent(value)) as QueryCursorV1
    if (Object.keys(cursor).sort().join(',') !== 'offset,queryHash,sortVersion,version,viewFingerprint'
      || cursor.version !== 1 || !Number.isSafeInteger(cursor.offset) || cursor.offset < 0
      || cursor.queryHash !== queryHash || cursor.viewFingerprint !== viewFingerprint || cursor.sortVersion !== WORLD_QUERY_SORT_VERSION_V1) throw Error('cursor')
    return cursor.offset
  } catch { return worldQueryFailV1('cursor', '查询游标与版本、条件、授权或预算不一致') }
}

function recordLabel(record: NarrativeRecordV1, atom: NonNullable<WorldQueryItemV1['atom']>, records: Map<string, NarrativeRecordV1>): string {
  const label = (key: string) => { const entity = records.get(key); return entity?.kind === 'entity' ? entity.payload.canonicalLabel : '已确认对象' }
  if ('holderKey' in atom) return `${label(atom.holderKey)}的认知`
  switch (record.kind) {
    case 'entity': return record.payload.canonicalLabel
    case 'assertion': return `${label(record.payload.subjectRef.semanticId)} · ${predicateLabel(record.payload.predicateKey)}`
    case 'event': return ({ action: '行动', observation: '观察', communication: '传递信息' })[record.payload.eventType]
    case 'relation': return `${label(record.payload.fromRef.semanticId)} · ${record.payload.relationType} · ${label(record.payload.toRef.semanticId)}`
    case 'rule': return preview(record.payload.statement, 160) || '已确认规则'
    case 'development': return `${({ thread: '故事线', goal: '目标', conflict: '冲突', commitment: '承诺', foreshadow: '伏笔' })[record.payload.developmentType]} · ${({ open: '未完成', advancing: '推进中', blocked: '受阻', resolved: '已解决', fulfilled: '已兑现', abandoned: '已放弃' })[record.payload.state]}`
    case 'disclosure': return '已确认揭示'
    case 'epistemic-event': return `${label(record.payload.holderRef.semanticId)}的认知`
  }
}

function readableSummary(record: NarrativeRecordV1, atom: NonNullable<WorldQueryItemV1['atom']>, sourceText: string | undefined,
  records: Map<string, NarrativeRecordV1>): string {
  const label = (key: string) => { const entity = records.get(key); return entity?.kind === 'entity' ? entity.payload.canonicalLabel : '已确认对象' }
  const value = (item: TypedValueV1): string => item.kind === 'entity-ref' ? label(item.value.semanticId)
    : item.kind === 'ordered-set' ? item.values.map(value).join('、') : String(item.value)
  if ('holderKey' in atom) return `${label(atom.holderKey)}${atom.attitude === 'doubt' ? '怀疑' : atom.attitude === 'know' ? '知道' : '相信'}：${value(atom.assertedValue)}`
  if (record.kind === 'entity') return record.payload.aliases.length ? `别名：${record.payload.aliases.join('、')}` : record.payload.canonicalLabel
  if (record.kind === 'assertion') return `${predicateLabel(record.payload.predicateKey)}：${record.payload.polarity === 'negative' ? '并非' : ''}${value(record.payload.value)}`
  if (sourceText) return preview(sourceText, 400)
  if (record.kind === 'event') return `参与者：${record.payload.participants.map(item => label(item.entityRef.semanticId)).join('、') || '未注明'}`
  if (record.kind === 'rule') return record.payload.statement
  return recordLabel(record, atom, records)
}

/** Pure computation behind the registered WorldRelease provider. The provider
 * verifies release/index bytes and mints a trusted grant before calling this;
 * there is no database, latest-draft reader, model, or network fallback here. */
export function createWorldQueryEvaluatorV1(input: VerifiedQueryInputV1) {
  const { reference, projection, index } = input
  if (index.indexHash !== reference.indexIdentity.indexHash || index.inputRootHash !== reference.indexIdentity.inputRootHash
    || index.recipeVersion !== reference.indexIdentity.recipeVersion) worldQueryFailV1('index', '查询索引与冻结引用不一致')
  const byId = new Map(projection.bundle.records.map(record => [record.semanticId, record]))
  const records = index.indexes.recordKeys.map(key => byId.get(key) ?? worldQueryFailV1('index', '查询索引未闭合'))
  const positions = new Map(records.map((record, position) => [record.semanticId, position]))
  const evidence = new Map(projection.bundle.evidence.map(span => [span.evidenceId, span]))
  const sourceKey = (source: { sourceResourceKey: string; fieldPointer: string; sourceRevision: string }) =>
    canonicalNarrativeJsonV1([source.sourceResourceKey, source.fieldPointer, source.sourceRevision])
  const sources = new Map(projection.sources.map(source => [sourceKey(source), new TextEncoder().encode(source.text)]))
  const recordProofs = new Map<number, string[]>()
  for (let position = 0; position < records.length; position++) {
    const record = records[position], ids = [...record.evidenceIds]
    if (!ids.length) {
      const provenance = { sourceResourceKey: record.provenance.sourceResourceKey, fieldPointer: record.provenance.sourceField,
        sourceRevision: record.provenance.sourceRevision }, source = sources.get(sourceKey(provenance))
      if (source?.length) {
        const evidenceId = `original:v1:${record.semanticId}`
        evidence.set(evidenceId, { ...provenance, evidenceId, sourceContentHash: record.provenance.sourceContentHash,
          normalizationVersion: 'raw-utf8-v1', start: 0, end: source.length, spanHash: record.provenance.sourceContentHash, supportRole: 'context' })
        ids.push(evidenceId)
      }
    }
    recordProofs.set(position, ids)
  }
  const chunks = new Map<string, NarrativeIndexV1['indexes']['chunks']>()
  for (const chunk of index.indexes.chunks) {
    const key = sourceKey(chunk), list = chunks.get(key) ?? []; list.push(chunk); chunks.set(key, list)
  }
  const byEntity = new Map(index.indexes.byEntity.map(item => [item.entityIndex, new Set(item.recordIndexes)]))
  const threads = new Map(index.indexes.developments.filter(item => item.type === 'thread').map(item => [item.recordIndex, new Set([item.recordIndex, ...item.memberIndexes])]))

  return async (queryInput: WorldQueryV1, grantInput: WorldQueryGrantV1, exactReadSemanticId?: string): Promise<WorldQueryResultV1> => {
    const query = parseWorldQueryV1(queryInput), grant = parseWorldQueryGrantV1(grantInput)
    assertGrant(query, grant, reference)
    const viewFingerprint = await hashNarrativeValueV1({ version: WORLD_QUERY_VERSION_V1, grant, view: query.view,
      indexHash: index.indexHash, releaseHash: reference.releaseHash })
    const queryHash = await hashNarrativeValueV1({ ...query, cursor: null,
      worldReference: { ...query.worldReference, localReleaseRecordId: 0 }, viewFingerprint, sortVersion: WORLD_QUERY_SORT_VERSION_V1,
      exactReadSemanticId: exactReadSemanticId ?? null })
    const offset = decodeCursor(query.cursor, queryHash, viewFingerprint)
    const base = { schema: 'storyforge.world-query-result' as const, version: 1 as const, queryHash, releaseRef: reference.referenceHash,
      indexRef: index.indexHash, viewFingerprint, items: [] as WorldQueryItemV1[], edges: [] as WorldQueryResultV1['edges'],
      diagnostics: [] as WorldQueryResultV1['diagnostics'], budgetUsed: { tokens: 0, records: 0, evidenceBytes: 0 }, nextCursor: null as string | null }
    const finish = async (status: WorldQueryResultV1['status']): Promise<WorldQueryResultV1> => {
      const result = { ...base, status }
      // Include the envelope, diagnostics, cursor and receipt in the returned
      // token estimate. The receipt hash has a fixed width; only the decimal
      // token count itself may require a second pass to settle.
      for (let attempt = 0; attempt < 4; attempt++) {
        const tokens = estimateTokens(canonicalNarrativeJsonV1({ ...result, receiptRef: '0'.repeat(64) }))
        if (tokens === result.budgetUsed.tokens) break
        result.budgetUsed.tokens = tokens
      }
      if (result.budgetUsed.tokens > query.budget.maxTokens) worldQueryFailV1('budget', '完整查询回执超过预算；请增加预算或减少本页条数')
      return { ...result, receiptRef: await hashNarrativeValueV1(result) }
    }
    if (query.branch.kind === 'hypothesis' || query.view === 'hypothesis') {
      base.diagnostics.push({ code: 'unsupported', semanticIds: [], reason: 'hypothesis-overlay-not-implemented' })
      return finish('unsupported')
    }
    validateWorldQueryTimeV1(query.time, projection.bundle.timeContext)
    const { visible, required, atoms } = buildWorldQueryVisibilityV1({ records, index, query, grant, timeContext: projection.bundle.timeContext })
    if (grant.perspective.kind === 'character') {
      const holder = positions.get(grant.perspective.characterKey), record = holder === undefined ? undefined : records[holder]
      if (holder === undefined || !visible.has(holder) || record?.kind !== 'entity' || record.payload.entityType !== 'character') {
        worldQueryFailV1('access-unavailable', '请求的角色视角不可用')
      }
    }
    if (exactReadSemanticId !== undefined && (!positions.has(exactReadSemanticId) || !visible.has(positions.get(exactReadSemanticId)!))) {
      worldQueryFailV1('access-unavailable', '请求的资源身份不可用')
    }
    const resolveFacet = (keys: string[], kind: 'entity' | 'thread') => keys.map(key => {
      const position = positions.get(key)
      if (position === undefined || !visible.has(position) || (kind === 'entity' ? records[position].kind !== 'entity' : !threads.has(position))) {
        worldQueryFailV1('access-unavailable', '请求的资源身份不可用')
      }
      return position
    })
    const entities = resolveFacet(query.entityKeys, 'entity'), arcs = resolveFacet(query.storyArcKeys, 'thread')
    const permittedChunks = new Set(index.indexes.chunks.filter(chunk => queryVisibilityAllowsV1(chunk.visibility, grant)
      && chunk.recordIndexes.every(position => visible.has(position))))
    const spanText = new Map<string, string>()
    for (const position of visible) for (const id of recordProofs.get(position)!) {
      if (spanText.has(id)) continue
      const span = evidence.get(id)
      if (!span) worldQueryFailV1('evidence', '冻结证据不完整')
      const touched = (chunks.get(sourceKey(span)) ?? []).filter(chunk => chunk.start < span.end && chunk.end > span.start)
      if (!touched.length || touched.some(chunk => !permittedChunks.has(chunk))) continue
      const source = sources.get(sourceKey(span))
      if (!source || span.end > source.length) worldQueryFailV1('evidence', '冻结证据范围无效')
      spanText.set(id, new TextDecoder('utf-8', { fatal: true }).decode(source.slice(span.start, span.end)))
    }
    const texts = new Map<number, string>(), labels = new Map<number, string>(), summaries = new Map<number, string>()
    const visibleById = new Map([...visible].map(position => [records[position].semanticId, records[position]]))
    for (const position of visible) {
      const record = records[position], atom = atoms.get(position)!, title = recordLabel(record, atom, visibleById)
      labels.set(position, title)
      summaries.set(position, readableSummary(record, atom, recordProofs.get(position)!.map(id => spanText.get(id)).find(Boolean), visibleById))
      texts.set(position, [title, canonicalNarrativeJsonV1(atom), ...recordProofs.get(position)!.flatMap(id => spanText.has(id) ? [spanText.get(id)!] : [])].join('\n'))
    }
    const matchesEntity = (position: number) => !entities.length || entities.some(entity => entity === position || byEntity.get(entity)?.has(position))
    const matchesArc = (position: number) => !arcs.length || arcs.some(arc => threads.get(arc)?.has(position))
    const text = query.text === null ? null : lexical(query.text)
    // A separate disclosure atom reveals its subjects at that position. Only
    // already authorized disclosure records may contribute these positions.
    const disclosed = new Map<number, NarrativeTimeV1['disclosedAt']>()
    for (const disclosure of index.indexes.disclosures) if (visible.has(disclosure.recordIndex)) {
      for (const position of disclosure.subjectIndexes) if (visible.has(position)) {
        const values = disclosed.get(position) ?? []
        values.push(...disclosure.audienceKeys.map(audienceKey => ({ audienceKey,
          sequenceKey: disclosure.sequenceKey, unitKey: disclosure.unitKey })))
        disclosed.set(position, values)
      }
    }
    const timeMatch = (position: number) => matchWorldQueryTimeV1(records[position], query.time,
      projection.bundle.timeContext, disclosed.get(position))
    const matched = new Set<number>()
    for (const position of visible) {
      const record = records[position]
      if (exactReadSemanticId !== undefined && record.semanticId !== exactReadSemanticId
        || query.view === 'author-plan' && record.modality !== 'intent'
        || query.view === 'development' && record.kind !== 'development'
        || query.mode === 'events' && record.kind !== 'event'
        || query.mode === 'state-at' && record.kind !== 'assertion'
        || query.kinds.length && !query.kinds.includes(record.kind)
        || !matchesEntity(position) || !matchesArc(position) || text && !lexical(texts.get(position)!).includes(text)) continue
      const temporal = timeMatch(position)
      if (temporal === 'unknown-time' || temporal === 'incomparable-time') {
        if (!base.diagnostics.some(item => item.code === temporal)) base.diagnostics.push({ code: temporal, semanticIds: [], reason: `${query.time?.axis ?? 'time'}-not-comparable` })
        continue
      }
      if (temporal === 'matched') matched.add(position)
    }
    const allowedEdges = index.indexes.adjacency.filter(edge => visible.has(edge.fromIndex) && visible.has(edge.toIndex)
      && edge.evidenceIds.every(id => spanText.has(id))
      // A structural truth reference is never an epistemic view read grant.
      && !(grant.perspective.kind === 'character' && ['belief-proposition', 'revision', 'knowledge-source'].includes(edge.kind)))
    const expansionKinds = new Set(query.relationExpansion.kinds)
    const frontier = [...matched].map(position => ({ position, hop: 0, causal: 0 })), expanded = new Set(matched)
    let bounded = (query.relationExpansion.maxHops > 0 || query.causalDepth > 0) && expanded.size > query.relationExpansion.maxNodes
    for (let cursor = 0; cursor < frontier.length && !bounded; cursor++) {
      const current = frontier[cursor]
      for (const edge of allowedEdges) {
        if (edge.fromIndex !== current.position) continue
        const causal = edge.kind === 'causes' && edge.basis === 'declared' && edge.evidenceIds.length > 0 && current.causal < query.causalDepth
        const ordinary = expansionKinds.has(edge.kind) && current.hop < query.relationExpansion.maxHops
        if (!causal && !ordinary || expanded.has(edge.toIndex)) continue
        if (expanded.size >= query.relationExpansion.maxNodes) { bounded = true; break }
        expanded.add(edge.toIndex); frontier.push({ position: edge.toIndex, hop: current.hop + 1, causal: current.causal + Number(causal) })
      }
    }
    if (bounded) base.diagnostics.push({ code: 'budget', semanticIds: [], reason: 'graph-node-budget' })
    const candidates = [...expanded].sort((a, b) => {
      const score = (position: number) => text && lexical(labels.get(position)!).includes(text) ? 1 : 0
      return score(b) - score(a) || compareUnicodeCodePointsV1(records[a].semanticId, records[b].semanticId)
    })
    if (offset > candidates.length) worldQueryFailV1('cursor', '查询游标超出可读取集合')
    let nextOffset = offset
    for (let at = offset; at < candidates.length; at++) {
      const position = candidates[at], record = records[position], closure = new Set<number>([position]), pending = [position]
      while (pending.length && closure.size <= query.relationExpansion.maxNodes) for (const target of required.get(pending.pop()!) ?? []) if (!closure.has(target)) {
        closure.add(target); pending.push(target)
      }
      if (closure.size > query.relationExpansion.maxNodes) {
        if (!base.diagnostics.some(item => item.reason === 'atomic-closure-node-budget')) base.diagnostics.push({ code: 'insufficient', semanticIds: [], reason: 'atomic-closure-node-budget' })
        nextOffset = at + 1; continue
      }
      const proofIds = sorted([...closure].flatMap(target => recordProofs.get(target)!))
      const originalAvailable = [...closure].every(target => recordProofs.get(target)!.length > 0) && proofIds.every(id => spanText.has(id))
      const complete = query.evidenceDepth === 'original' ? originalAvailable : record.evidenceIds.every(id => spanText.has(id))
      const atom = atoms.get(position)!
      const item: WorldQueryItemV1 = { semanticId: record.semanticId, revisionHash: record.revisionHash, kind: record.kind,
        title: labels.get(position)!, summary: preview(summaries.get(position)!, 400), modality: record.modality,
        authority: record.authority, reviewState: record.reviewState, depth: query.evidenceDepth,
        atom: query.evidenceDepth === 'metadata' ? null : structuredClone(atom), supportingAtoms: query.evidenceDepth === 'metadata' ? []
          : [...closure].filter(target => target !== position).sort((a, b) => compareUnicodeCodePointsV1(records[a].semanticId, records[b].semanticId))
            .map(target => ({ semanticId: records[target].semanticId, revisionHash: records[target].revisionHash, kind: records[target].kind,
              modality: records[target].modality, payload: structuredClone(atoms.get(target)!) })),
        evidence: proofIds.flatMap(id => {
          const span = evidence.get(id)!
          return spanText.has(id) ? [{ evidenceId: id, sourceResourceKey: span.sourceResourceKey, fieldPointer: span.fieldPointer,
            sourceRevision: span.sourceRevision, sourceContentHash: span.sourceContentHash, spanHash: span.spanHash,
            start: span.start, end: span.end, text: query.evidenceDepth === 'original' ? spanText.get(id)! : null }] : []
        }),
        match: { selection: matched.has(position) ? 'direct' : 'expansion',
          entityKeys: entities.filter(entity => entity === position || byEntity.get(entity)?.has(position)).map(entity => records[entity].semanticId),
          storyArcKeys: arcs.filter(arc => threads.get(arc)?.has(position)).map(arc => records[arc].semanticId), time: query.time ? timeMatch(position) : 'not-requested' },
        closure: complete ? 'complete' : 'insufficient', originalAvailable }
      if (!complete && query.evidenceDepth !== 'metadata') {
        base.diagnostics.push({ code: 'insufficient', semanticIds: [record.semanticId], reason: 'evidence-closure-unavailable' })
        nextOffset = at + 1; continue
      }
      const evidenceBytes = item.evidence.reduce((sum, span) => sum + (span.text === null ? 0 : bytes(span.text)), 0)
      const tokens = estimateTokens(canonicalNarrativeJsonV1(item))
      if (base.items.length >= query.budget.maxRecords || base.budgetUsed.tokens + tokens > query.budget.maxTokens - 1024
        || base.budgetUsed.evidenceBytes + evidenceBytes > query.budget.maxEvidenceBytes) {
        base.diagnostics.push({ code: 'insufficient', semanticIds: [], reason: 'complete-item-exceeds-remaining-budget' })
        nextOffset = at; break
      }
      base.items.push(item); base.budgetUsed.tokens += tokens; base.budgetUsed.records++
      base.budgetUsed.evidenceBytes += evidenceBytes; nextOffset = at + 1
    }
    if (nextOffset < candidates.length && base.items.length) base.nextCursor = encodeURIComponent(JSON.stringify({ version: 1,
      offset: nextOffset, queryHash, viewFingerprint, sortVersion: WORLD_QUERY_SORT_VERSION_V1 } satisfies QueryCursorV1))
    const delivered = new Set(base.items.map(item => item.semanticId))
    for (const edge of allowedEdges) if (delivered.has(records[edge.fromIndex].semanticId) && delivered.has(records[edge.toIndex].semanticId)
      && (expansionKinds.has(edge.kind) || edge.kind === 'causes' && query.causalDepth > 0)) {
      const value = { from: records[edge.fromIndex].semanticId, to: records[edge.toIndex].semanticId, kind: edge.kind, evidenceIds: [...edge.evidenceIds] }
      const cost = estimateTokens(canonicalNarrativeJsonV1(value))
      if (base.budgetUsed.tokens + cost > query.budget.maxTokens - 1024) { base.diagnostics.push({ code: 'budget', semanticIds: [], reason: 'edge-output-budget' }); break }
      base.edges.push(value); base.budgetUsed.tokens += cost
    }
    const states = new Map<string, Array<Extract<NarrativeRecordV1, { kind: 'assertion' }>>>()
    if (query.mode === 'state-at') for (const position of matched) {
      const record = records[position]
      if (record.kind !== 'assertion') continue
      const key = canonicalNarrativeJsonV1([record.payload.subjectRef.semanticId, record.payload.predicateKey])
      const group = states.get(key) ?? []; group.push(record); states.set(key, group)
    }
    for (const group of states.values()) {
      const predicate = WORLD_SEMANTIC_REGISTRY_V4.predicates.find(item => item.key === group[0].payload.predicateKey)
      const positive = new Set(group.filter(record => record.payload.polarity === 'positive').map(record => canonicalNarrativeJsonV1(record.payload.value)))
      const negative = new Set(group.filter(record => record.payload.polarity === 'negative').map(record => canonicalNarrativeJsonV1(record.payload.value)))
      // Append/multi predicates may coexist. No conflict policy permits the
      // same accepted proposition to be both true and false at one instant.
      if (predicate?.cardinality === 'single' && positive.size > 1 || [...positive].some(value => negative.has(value))) {
        base.diagnostics.push({ code: 'conflict', semanticIds: group.filter(record => delivered.has(record.semanticId)).map(record => record.semanticId), reason: 'overlapping-accepted-states' })
      }
    }
    const status = base.diagnostics.some(diagnostic => diagnostic.code === 'conflict') ? 'conflict'
      : base.diagnostics.some(diagnostic => diagnostic.code === 'insufficient'
        || !base.items.length && ['unknown-time', 'incomparable-time', 'budget'].includes(diagnostic.code)) ? 'insufficient' : base.items.length ? 'matched' : 'missing'
    if (!base.items.length && !base.diagnostics.length) base.diagnostics.push({ code: 'missing', semanticIds: [], reason: 'no-match-within-authorized-compiled-scope' })
    return finish(status)
  }
}
