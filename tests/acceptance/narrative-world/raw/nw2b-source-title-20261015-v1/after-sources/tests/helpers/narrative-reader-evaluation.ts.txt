export const READER_CONDITIONS = ['v3-provider', 'lexical-original', 'ir-lexical', 'ir-no-time', 'ir-no-knowledge', 'ir-no-evidence'] as const
export type ReaderCondition = typeof READER_CONDITIONS[number]
export interface ReaderQuestion {
  id: string; operation: string; question: string; view: string; branch: string
  search: string | null; entityId: string | null; arcId: string | null; throughId: string | null; holderId: string | null
  sourceTitle: string | null
}
export interface ReaderDocument {
  id: string; sourceId: string | null; title: string; kind: string; text: string | null
  atom: unknown; evidence: Array<{ sourceId: string | null; text: string; sourceHash: string | null; spanHash: string | null }>
}
export interface ReaderContext {
  status: string; diagnostics: string[]; documents: ReaderDocument[]; provenance: unknown
}

/** The frozen corpus uses `search` for both literal text and a selected scene
 * name. Consume the latter only when a typed facet is present AND the actual
 * registered release description confirms that exact world name. Never infer
 * this distinction from gold, a mutable draft or a coincidentally empty query. */
export function readerIrTextRoute(question: ReaderQuestion, frozenWorldName: string): {
  version: 1; semanticText: string | null; kinds: WorldQueryV1['kinds']; consumedWorldLabel: string | null
} {
  const kinds: WorldQueryV1['kinds'] = question.operation === 'entity-filter' && question.entityId ? ['entity']
    : question.operation === 'arc-filter' && question.arcId ? ['development']
      : question.operation === 'chapter-cutoff' && question.throughId ? ['disclosure'] : []
  const consumedWorldLabel = question.view === 'author' && kinds.length && question.search === frozenWorldName
    ? question.search : null
  const semanticText = consumedWorldLabel || question.sourceTitle || question.view === 'character'
    || ['causal-query', 'development-query'].includes(question.operation) ? null : question.search ?? question.question
  return { version: 1, semanticText, kinds, consumedWorldLabel }
}

/** Explicit whitelist. Answer atoms, expected resource IDs, forbidden facts,
 * gold spans and edit annotations must never cross the retrieval boundary. */
export function readerQuestions(family: NarrativeFamily): ReaderQuestion[] {
  return family.queries.map(query => {
    const record = /《([^》]+)》记录(\d+)/u.exec(query.question)
    return { id: query.id, operation: query.operation, question: query.question, view: query.view, branch: query.branch,
      search: query.search ?? null, entityId: query.entityId ?? null, arcId: query.arcId ?? null,
      throughId: query.throughId ?? null, holderId: query.holderId ?? null,
      sourceTitle: query.operation === 'original-read' && record ? `${record[1]}·记录${record[2]}` : null }
  })
}

/** The seed contains explicit test-owned legacy records. This minimal adapter
 * adds only literal reading spans and annotations of those existing records.
 * No invented calendar, communication event, relation, causal edge or magic rule.
 * An empty disclosure subjects array makes no entity/event discovery claim. */
export async function seedReaderFamily(family: NarrativeFamily) {
  const source = await seedNarrativeFamily({ ...family, queries: [], edits: [] })
  const work = (await db.works.get(source.scope.workId))!
  const owner = { kind: 'work' as const, workCode: work.code }
  const input = { scope: source.scope, owner, branchKey: 'canon' }
  const privateMetadata = { narrativeModalityV1: 'established' as const,
    narrativeVisibilityV1: { class: 'author-only' as const, audienceKeys: [], policyVersion: 'reader-fixture-literal-v1' } }
  async function confirm(table: string, recordId: number, annotations: Partial<NarrativeDomainFieldsV1>) {
    const capture = await captureNarrativeDomainV1(input)
    const row = capture.rows.get(table)!.find(item => item.id === recordId)!
    await confirmNarrativeDomainAnnotationsV1({ scope: source.scope, table, recordId,
      expectedSourceHash: await hashNarrativeDomainSourceV1(capture, table, row), annotations })
  }
  async function evidence(text: string, field: string) {
    const hash = await hashNarrativeBytesV1(text)
    return { evidenceKey: 'literal', source: { self: true as const }, fieldPointer: `/${field}`,
      sourceContentHash: hash, normalizationVersion: 'raw-utf8-v1' as const, start: 0,
      end: new TextEncoder().encode(text).length, spanHash: hash, supportRole: 'supports' as const }
  }
  for (const item of family.sources.filter(item => item.kind === 'character')) {
    await confirm('characters', source.localIds.get(item.id)!, { ...privateMetadata,
      narrativeVisibilityV1: { class: 'public', audienceKeys: [], policyVersion: 'reader-fixture-literal-v1' },
      narrativeEvidenceV1: [await evidence(item.title, 'name')] })
  }
  // The corpus already declares B's knowledge. No historical-location or other
  // lifecycle control is promoted into semantic QA by this adapter.
  await confirm('temporalFacts', source.factIds.get('confirmed')!, { ...privateMetadata,
    narrativeAssertionV1: { polarity: 'positive' } })
  for (const suffix of ['a', 'b']) {
    const id = source.localIds.get(`${family.id}:knowledge-${suffix}`)!
    await db.knowledgeLedger.update(id, { factId: source.factIds.get('confirmed')! })
    await confirm('knowledgeLedger', id, { ...privateMetadata, narrativeModalityV1: 'belief',
      narrativeEpistemicV1: { attitude: suffix === 'a' ? 'believe' : 'know', transition: 'acquire',
        assertedValue: { kind: 'text', value: suffix === 'a' ? family.seed.belief : family.seed.truth },
        sourceEvent: null, revises: null } })
  }
  const uidToSource = new Map<string, string>()
  for (const item of family.sources.filter(item => item.kind !== 'hypothesis')) {
    const table = item.kind === 'chapter' ? db.chapters : item.kind === 'character' ? db.characters : db.storyArcs
    const row = (await table.get(source.localIds.get(item.id)!))!
    // Resource identity is added by the shared lifecycle stamper; some legacy
    // domain interfaces do not yet expose the registry-owned field.
    const sourceUid = (row as typeof row & { ragDocumentId?: string }).ragDocumentId
    if (!sourceUid) throw Error('Stamped source is missing its registered UID')
    uidToSource.set(sourceUid, item.id)
    if (item.kind === 'chapter') {
      await confirm('chapters', row.id!, { ...privateMetadata, narrativeDisclosuresV1: [{
        ...privateMetadata, semanticSubKey: 'sub:v1:00000000-0000-4000-8000-000000000101',
        subjects: [], audienceKeys: ['author'], evidenceKey: 'literal',
        narrativeTimeV1: { occurred: null, valid: null, acquiredAt: null, disclosedAt: [],
          narratedAt: { chapter: { chapterId: row.id!, sourceUid }, spanStart: 0 } },
        narrativeEvidenceV1: [await evidence(item.text, 'content')],
      }] })
    } else if (item.kind === 'story-arc') {
      await confirm('storyArcs', row.id!, { narrativeDevelopmentsV1: [{ ...privateMetadata, narrativeModalityV1: 'intent',
        semanticSubKey: 'sub:v1:00000000-0000-4000-8000-000000000102', statement: item.text,
        developmentType: 'thread', owners: [], members: [], state: 'open', transitions: [], resolutionEvidenceKeys: [],
        narrativeEvidenceV1: [await evidence(item.text, 'description')],
      }] })
    }
  }
  const revision = await createWorldRevision({ scope: source.scope, formatVersion: 4, includeWorkSources: true, label: '固定读者评测：原文跨度与既有声明' })
  const release = await publishWorldRevision(revision.id!)
  const verified = await verifyWorldReleaseRecordV1(release)
  if (verified.version !== 4) throw Error('reader fixture requires v4')
  const projection = verified.manifest.narrative
  const keyToLogical = (key: string) => [...uidToSource].find(([uid]) => key.endsWith(`:${uid}`))?.[1] ?? null
  const semanticFor = (id: string) => projection.bundle.records.find(record => keyToLogical(record.provenance.sourceResourceKey) === id)?.semanticId
  return { ...source, v4Release: release, projection, keyToLogical, semanticFor, uidToSource }
}
type ReaderFixture = Awaited<ReturnType<typeof seedReaderFamily>>

const empty = (status: string, reason: string): ReaderContext => ({ status, diagnostics: [reason], documents: [], provenance: null })
const lower = (text: string) => text.toLowerCase()
function lexicalScore(text: string, question: ReaderQuestion) {
  if (question.search) return lower(text).includes(lower(question.search)) ? 1 : 0
  const chars = Array.from(question.question.replace(/[\s\p{P}\p{S}]/gu, ''))
  const terms = new Set(chars.slice(0, -1).map((char, i) => char + chars[i + 1]))
  return [...terms].filter(term => lower(text).includes(lower(term))).length
}

export function lexicalReaderContext(family: NarrativeFamily, question: ReaderQuestion): ReaderContext {
  // This reference has no trusted character-knowledge projection. Refusal is
  // part of its measured boundary, rather than handing author secrets to it.
  if (question.view === 'character') return empty('unsupported', 'lexical-reference-has-no-character-policy')
  if (question.branch !== 'canon') return empty('unsupported', 'simulation-not-implemented')
  let sources = family.sources.filter(item => item.branch === 'canon')
  if (question.sourceTitle) sources = sources.filter(item => item.title === question.sourceTitle)
  if (question.entityId) sources = sources.filter(item => item.id === question.entityId)
  if (question.arcId) sources = sources.filter(item => item.id === question.arcId)
  // Literal baseline supports caller-supplied ID selection, but no chronology.
  const ranked = sources.map(item => ({ item, score: question.sourceTitle ? 1 : lexicalScore(`${item.title}\n${item.text}`, question) }))
    .filter(item => item.score > 0).sort((a, b) => b.score - a.score || a.item.id.localeCompare(b.item.id)).slice(0, 12)
  return { status: ranked.length ? 'matched' : 'missing', diagnostics: question.throughId ? ['lexical-reference-has-no-time-filter'] : [],
    documents: ranked.map(({ item }) => ({ id: item.id, sourceId: item.id, title: item.title, kind: item.kind,
      text: item.text, atom: null, evidence: [{ sourceId: item.id, text: item.text, sourceHash: null, spanHash: null }] })),
    provenance: { method: 'literal-search-or-question-bigrams-v1', maxRecords: 12 } }
}

export async function v3ReaderContext(fixture: ReaderFixture, question: ReaderQuestion): Promise<ReaderContext> {
  if (['causal-query', 'development-query', 'hypothesis-query'].includes(question.operation)) return empty('unsupported', 'v3-operation-not-implemented')
  try {
    let descriptors
    if (question.sourceTitle) descriptors = fixture.descriptors.filter(item => item.title === question.sourceTitle && item.worldSemantic?.resourceKind === 'chapter')
    else if (question.view === 'character') descriptors = (await listAllWorldReleaseResourceDescriptorsV1({ ...fixture.resourceScope,
      characterId: fixture.localIds.get(question.holderId!) })).filter(item => item.worldSemantic?.resourceKind === 'knowledge-event')
    else {
      const input: ResourceSearchInputV1 = { scope: fixture.resourceScope, query: question.search ?? question.question, limit: 12,
        kinds: question.entityId ? ['character'] : question.arcId ? ['story-arc'] : ['chapter'] }
      if (question.entityId) input.entityKeys = [fixture.descriptor(question.entityId).resourceKey]
      if (question.arcId) input.storyArcKeys = [fixture.descriptor(question.arcId).resourceKey]
      if (question.throughId) input.timeRange = { throughChapterId: fixture.localIds.get(question.throughId) }
      descriptors = (await searchWorldReleaseV1(input)).items
    }
    const documents: ReaderDocument[] = []
    for (const item of descriptors) {
      const original = await readWorldOriginalEvidenceV1({ scope: fixture.resourceScope, resourceKey: item.resourceKey,
        sourceRef: item.sourceRefs[0]!, maxTokens: 12000 })
      documents.push({ id: item.resourceKey, sourceId: fixture.byKey.get(item.resourceKey) ?? null, title: item.title,
        kind: item.worldSemantic!.resourceKind, text: original.content, atom: null,
        evidence: [{ sourceId: fixture.byKey.get(item.resourceKey) ?? null, text: original.content, sourceHash: null, spanHash: null }] })
    }
    return { status: documents.length ? 'matched' : 'missing', diagnostics: [], documents,
      provenance: { releaseHash: fixture.reference.releaseHash, provider: 'registered-v3', maxRecords: 12 } }
  } catch (error) {
    if (!error || typeof error !== 'object' || !('code' in error) || !['unsupported', 'access-unavailable', 'budget'].includes(String(error.code))) throw error
    return empty(String(error.code), String(error))
  }
}

export async function irReaderContext(fixture: ReaderFixture, question: ReaderQuestion, condition: ReaderCondition,
  declaredKnowledgeThrough: TimePointV1 | null = null): Promise<ReaderContext> {
  if (question.branch !== 'canon') return empty('unsupported', 'simulation-not-implemented')
  const characterKey = question.holderId ? fixture.semanticFor(question.holderId) : null
  const access = await openAuthorWorldQueryAccessV1({ scope: fixture.scope, localReleaseRecordId: fixture.v4Release.id!,
    view: question.view === 'character' ? 'character' : question.arcId ? 'author-plan' : question.operation === 'development-query' ? 'development' : 'canon',
    perspective: characterKey ? { kind: 'character', characterKey } : { kind: 'author' },
    // A supplemental fixture can supply an explicit, source-grounded cutoff.
    // The historical corpus path keeps unknown time; never derive a cutoff
    // from expected answers, chapter order or a desired retrieval result.
    knowledgeThrough: characterKey ? declaredKnowledgeThrough ?? { kind: 'unknown', label: null } : null, disclosedThrough: null })
  try {
    const description = await describeWorldQueryAccessV1(access)
    const route = readerIrTextRoute(question, description.identity.worldName)
    const query: WorldQueryV1 = { ...authorWorldQueryV1(access), evidenceDepth: 'original',
      text: route.semanticText, kinds: route.kinds,
      budget: { maxTokens: 100000, maxRecords: 12, maxEvidenceBytes: 32768 } }
    if (question.entityId) query.entityKeys = [fixture.semanticFor(question.entityId)!]
    if (question.arcId) query.storyArcKeys = [fixture.semanticFor(question.arcId)!]
    if (question.throughId && condition !== 'ir-no-time') {
      const unitKey = [...fixture.uidToSource].find(([, logical]) => logical === question.throughId)?.[0]
      const sequence = fixture.projection.bundle.timeContext.sequences.find(item => item.units.some(unit => unit.unitKey === unitKey))
      if (!sequence || !unitKey) throw Error('Explicit narrative cutoff has no frozen chapter')
      query.time = { axis: 'narrated', sequenceKey: sequence.sequenceKey, throughUnitKey: unitKey, audienceKey: null }
    }
    if (question.operation === 'causal-query') query.causalDepth = 2
    const result = await queryWorldReleaseV1(access, query)
    let items: WorldQueryItemV1[] = result.items
    if (question.sourceTitle) {
      const source = [...fixture.byLogical].find(([, descriptor]) => descriptor.title === question.sourceTitle && descriptor.worldSemantic?.resourceKind === 'chapter')?.[0]
      items = items.filter(item => item.evidence.some(span => fixture.keyToLogical(span.sourceResourceKey) === source))
    }
    if (condition === 'ir-no-knowledge') items = items.filter(item => item.kind !== 'epistemic-event')
    const noEvidence = condition === 'ir-no-evidence'
    return { status: result.status, diagnostics: result.diagnostics.map(item => `${item.code}:${item.reason}`),
      documents: items.map(item => ({ id: item.semanticId, sourceId: null, title: item.title, kind: item.kind,
        text: noEvidence ? null : item.summary,
        atom: { modality: item.modality, value: item.atom,
          supporting: noEvidence ? [] : item.supportingAtoms.filter(atom => condition !== 'ir-no-knowledge' || atom.kind !== 'epistemic-event') },
        evidence: noEvidence ? [] : item.evidence.flatMap(span => span.text === null ? [] : [{ sourceId: fixture.keyToLogical(span.sourceResourceKey),
          text: span.text, sourceHash: span.sourceContentHash, spanHash: span.spanHash }]) })),
      provenance: { releaseHash: access.reference.releaseHash, query, receipt: result.receiptRef, indexHash: access.reference.indexIdentity.indexHash,
        route, frozenWorldName: description.identity.worldName,
        returnedBeforeAblation: result.items.length, evidenceAblated: noEvidence, timeAblated: condition === 'ir-no-time', knowledgeAblated: condition === 'ir-no-knowledge' } }
  } finally { await closeWorldQueryAccessV1(access) }
}

export async function collectReaderFamily(family: NarrativeFamily) {
  const questions = readerQuestions(family)
  const fixture = await seedReaderFamily(family)
  const cases = []
  for (const question of questions) for (const condition of READER_CONDITIONS) {
    const context = condition === 'v3-provider' ? await v3ReaderContext(fixture, question)
      : condition === 'lexical-original' ? lexicalReaderContext(family, question)
        : await irReaderContext(fixture, question, condition)
    cases.push({ id: `${question.id}/${condition}`, familyId: family.id, split: family.split, condition, question, context })
  }
  return { familyId: family.id, annotationBoundary: 'Fixture-author literal spans and already-declared corpus inputs; no model extraction or human acceptance. Acquired times remain unknown. No invented events/causes/calendars.',
    coverage: fixture.projection.coverage, cases }
}
