import { NARRATIVE_KINDS_V1 } from '../types/narrative-ir'
import { NARRATIVE_INDEX_EDGE_KINDS_V1 } from '../types/narrative-index'
import type { WorldQueryGrantV1, WorldQueryPerspectiveV1, WorldQueryTimeV1, WorldQueryV1, WorldReferenceV2 } from '../types/world-query'
import type { WorldReferenceV1 } from '../types/world-product-contracts'
import { arrayV1, canonicalNarrativeJsonV1, enumV1, hashV1, keyV1, nullableV1, objectV1, textV1, uintV1, uniqueKeysV1,
  type DecoderV1 } from '../world-engine/narrative-json'
import { intervalDecoderV1, timePointDecoderV1 } from '../world-engine/narrative-time'

export const WORLD_QUERY_VERSION_V1 = 'world-query-v1' as const
export const WORLD_QUERY_SORT_VERSION_V1 = 'authorized-lexical-source-title-v2' as const
export const WORLD_QUERY_LIMITS_V1 = Object.freeze({ maxTokens: 100_000, maxRecords: 500, maxEvidenceBytes: 4 * 1024 * 1024,
  maxTextBytes: 4096, maxRequestBytes: 128 * 1024, maxHops: 3, maxNodes: 500, maxKeys: 100 })
export const WORLD_QUERY_VIEWS_V1 = ['canon', 'character', 'development', 'author-plan', 'hypothesis'] as const

export class WorldQueryErrorV1 extends Error {
  constructor(readonly code: string, message: string) { super(`[world-query:${code}] ${message}`); this.name = 'WorldQueryErrorV1' }
}
export function worldQueryFailV1(code: string, message: string): never { throw new WorldQueryErrorV1(code, message) }

const integer = (min: number, max: number): DecoderV1<number> => (value, path) => {
  const result = uintV1(value, path)
  if (result < min || result > max) worldQueryFailV1('range', `${path} 超出允许范围`)
  return result
}
const keys: DecoderV1<string[]> = (value, path) => {
  const result = uniqueKeysV1(value, path)
  if (result.length > WORLD_QUERY_LIMITS_V1.maxKeys || result.some(key => key.length > 1024)) worldQueryFailV1('range', `${path} 过大`)
  return result
}
const perspective: DecoderV1<WorldQueryPerspectiveV1> = (value, path) => {
  const kind = (value as WorldQueryPerspectiveV1 | null)?.kind
  return kind === 'character'
    ? objectV1<Extract<WorldQueryPerspectiveV1, { kind: 'character' }>>({ kind: enumV1(['character']), characterKey: keyV1 })(value, path)
    : objectV1<{ kind: 'author' | 'reader' }>({ kind: enumV1(['author', 'reader']) })(value, path)
}
const time: DecoderV1<WorldQueryTimeV1> = (value, path) => {
  const raw = value as WorldQueryTimeV1 | null
  if (raw?.axis === 'narrated' || raw?.axis === 'disclosed') return objectV1<Extract<WorldQueryTimeV1, { axis: 'narrated' | 'disclosed' }>>({ axis: enumV1(['narrated', 'disclosed']),
    sequenceKey: keyV1, throughUnitKey: keyV1, audienceKey: nullableV1(keyV1) })(value, path)
  return raw && 'mode' in raw && raw.mode === 'at'
    ? objectV1<Extract<WorldQueryTimeV1, { mode: 'at' }>>({ axis: enumV1(['occurred', 'valid', 'acquired']), mode: enumV1(['at']), point: timePointDecoderV1 })(value, path)
    : objectV1<Extract<WorldQueryTimeV1, { mode: 'overlaps' }>>({ axis: enumV1(['occurred', 'valid', 'acquired']), mode: enumV1(['overlaps']), interval: intervalDecoderV1 })(value, path)
}

/** Shape only. The provider separately validates portable identity hashes and
 * compares the entire request reference with the actual frozen release. */
export const worldQueryReferenceDecoderV1: DecoderV1<WorldReferenceV1 | WorldReferenceV2> = (value, path) => {
  const shared = { schema: enumV1(['storyforge.world-reference']), worldCode: keyV1, releaseUid: keyV1,
    releaseVersion: integer(1, Number.MAX_SAFE_INTEGER), releaseHash: hashV1,
    localReleaseRecordId: integer(0, Number.MAX_SAFE_INTEGER), referenceHash: hashV1 }
  return (value as WorldReferenceV2 | null)?.version === 2
    ? objectV1<WorldReferenceV2>({ ...shared, version: enumV1([2]),
      manifestIdentity: objectV1({ schema: enumV1(['storyforge.world-release']), version: enumV1([4]), semanticContract: enumV1([4]),
        registryVersion: keyV1, compilerVersion: keyV1 }),
      indexIdentity: objectV1({ inputRootHash: hashV1, recipeVersion: keyV1, indexHash: hashV1 }) })(value, path)
    : objectV1<WorldReferenceV1>({ ...shared, version: enumV1([1]),
      manifestIdentity: objectV1({ schema: enumV1(['storyforge.world-release']), version: enumV1([3]), semanticContract: enumV1([3]), schemaHash: hashV1 }),
      capabilityIdentity: objectV1({ catalogHash: hashV1, profileHash: hashV1 }) })(value, path)
}

const queryDecoder = objectV1<WorldQueryV1>({
  schema: enumV1(['storyforge.world-query']), version: enumV1([1]), worldReference: worldQueryReferenceDecoderV1,
  sourcePlanHash: nullableV1(hashV1), view: enumV1(WORLD_QUERY_VIEWS_V1), perspective,
  branch: (value, path) => (value as WorldQueryV1['branch'] | null)?.kind === 'canon'
    ? objectV1<Extract<WorldQueryV1['branch'], { kind: 'canon' }>>({ kind: enumV1(['canon']), key: keyV1 })(value, path)
    : objectV1<Extract<WorldQueryV1['branch'], { kind: 'hypothesis' }>>({ kind: enumV1(['hypothesis']), artifactRef: keyV1 })(value, path),
  mode: enumV1(['search', 'state-at', 'events', 'traverse']), text: nullableV1(textV1),
  kinds: arrayV1(enumV1(NARRATIVE_KINDS_V1)), entityKeys: keys, storyArcKeys: keys, time: nullableV1(time),
  evidenceDepth: enumV1(['metadata', 'atomic', 'original']),
  relationExpansion: objectV1({ kinds: arrayV1(enumV1(NARRATIVE_INDEX_EDGE_KINDS_V1)), maxHops: integer(0, 3), maxNodes: integer(1, 500) }),
  causalDepth: integer(0, 3), budget: objectV1({ maxTokens: integer(1, 100_000), maxRecords: integer(1, 500), maxEvidenceBytes: integer(0, 4 * 1024 * 1024) }),
  cursor: nullableV1(keyV1),
})

export function parseWorldQueryV1(value: unknown): WorldQueryV1 {
  if (new TextEncoder().encode(canonicalNarrativeJsonV1(value)).length > WORLD_QUERY_LIMITS_V1.maxRequestBytes) worldQueryFailV1('request-budget', '查询请求过大')
  const result = queryDecoder(value, 'query')
  if (result.text !== null && (!result.text.trim() || new TextEncoder().encode(result.text).length > WORLD_QUERY_LIMITS_V1.maxTextBytes)) worldQueryFailV1('text', '查询文本为空或过大')
  if (result.mode === 'state-at' && !(result.time?.axis === 'valid' && result.time.mode === 'at')) worldQueryFailV1('state-time', 'state-at 必须指定有效状态轴上的时点')
  if (result.time?.axis === 'disclosed' && !result.time.audienceKey) worldQueryFailV1('audience', '揭示查询必须指定受众')
  if (result.time?.axis === 'narrated' && result.time.audienceKey !== null) worldQueryFailV1('audience', '叙述顺序不接受揭示受众')
  if (result.view === 'character' && result.perspective.kind !== 'character') worldQueryFailV1('perspective', '角色视图必须指定角色')
  if (result.perspective.kind === 'character' && result.view !== 'character') worldQueryFailV1('perspective', '角色读取必须使用角色视图')
  for (const list of [result.kinds, result.relationExpansion.kinds]) if (new Set(list).size !== list.length) worldQueryFailV1('duplicate', '枚举筛选条件重复')
  result.kinds.sort(); result.relationExpansion.kinds.sort()
  return result
}

export function parseWorldQueryGrantV1(value: unknown): WorldQueryGrantV1 {
  canonicalNarrativeJsonV1(value)
  const grant = objectV1<WorldQueryGrantV1>({ version: enumV1([1]), releaseReferenceHash: hashV1, sourcePlanHash: nullableV1(hashV1),
    purpose: enumV1(['author-review', 'product-preparation', 'product-production', 'product-runtime']), views: arrayV1(enumV1(WORLD_QUERY_VIEWS_V1)), perspective,
    audienceKeys: keys, allowedSemanticIds: nullableV1(uniqueKeysV1), branchKey: keyV1, knowledgeThrough: nullableV1(timePointDecoderV1),
    disclosedThrough: nullableV1(objectV1({ sequenceKey: keyV1, unitKey: keyV1, audienceKey: keyV1 })),
    maxTokens: integer(1, 100_000), maxRecords: integer(1, 500), maxEvidenceBytes: integer(0, 4 * 1024 * 1024), policyRevision: keyV1 })(value, 'grant')
  if (!grant.views.length || new Set(grant.views).size !== grant.views.length) worldQueryFailV1('grant', '读取授权视图无效')
  if ((grant.sourcePlanHash === null) !== ['author-review', 'product-preparation'].includes(grant.purpose)) worldQueryFailV1('grant', '产品读取必须绑定冻结来源计划')
  if (grant.purpose === 'product-preparation' && (grant.perspective.kind !== 'author'
    || grant.views.length !== 1 || grant.views[0] !== 'canon')) worldQueryFailV1('grant', '产品定向只允许 Canon 目录')
  if (grant.perspective.kind === 'character' && !grant.knowledgeThrough) worldQueryFailV1('grant', '角色读取缺少获知截止时点')
  if (grant.perspective.kind === 'character' && grant.views.some(view => view !== 'character')) worldQueryFailV1('grant', '角色授权不能混入全知视图')
  if (grant.perspective.kind === 'reader' && !grant.disclosedThrough) worldQueryFailV1('grant', '读者读取缺少揭示截止位置')
  if (grant.perspective.kind !== 'author' && grant.views.some(view => !['canon', 'character'].includes(view))) worldQueryFailV1('grant', '该视角不能读取作者计划或假设')
  grant.views.sort()
  return grant
}
