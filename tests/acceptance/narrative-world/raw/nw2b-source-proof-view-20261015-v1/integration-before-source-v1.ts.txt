import { afterAll, afterEach, beforeEach, describe, expect, it, vi } from 'vitest'
import { createHash } from 'node:crypto'
import { readFileSync, writeFileSync } from 'node:fs'
import { db } from '../../src/lib/db/schema'
import { makeRagEntryKey } from '../../src/lib/retrieval/rag-library'
import { prepareNarrativeCandidateV1, compileNarrativeAnnotationProposalV1 } from '../../src/lib/world-engine/narrative-domain-candidate'
import { generateNarrativeAnnotationCandidateV1, adoptNarrativeAnnotationCandidateV1 } from '../../src/lib/agent/run/narrative-annotation-durable'
import { createWorldRevision, publishWorldRevision } from '../../src/lib/world-engine/releases'
import { verifyWorldReleaseRecordV1 } from '../../src/lib/world-engine/release-codec'
import { authorWorldQueryV1, openAuthorWorldQueryAccessV1, queryWorldReleaseV1, closeWorldQueryAccessV1 } from '../../src/lib/context-gateway/world-query-client'
import { irReaderContext, readerQuestions, seedReaderFamily } from '../helpers/narrative-reader-evaluation'
import type { ChatMessage } from '../../src/lib/types'
import corpus from '../acceptance/narrative-world/corpus-v1.json'

// This checks authored, test-owned proposals and the existing adoption boundary.
// It deliberately does not claim that a model discovered these annotations.
describe('NW-1C · 原文存在、候选与已确认语义的覆盖边界', () => {
  const family = corpus.families[0]
  const observations: unknown[] = []
  beforeEach(async () => {
    vi.useFakeTimers({ toFake: ['Date'] }); vi.setSystemTime(1_900_000_000_000)
    await db.delete(); await db.open()
  })
  afterEach(() => { db.close(); vi.useRealTimers() })
  afterAll(() => {
    if (!process.env.NW_CANDIDATE_COVERAGE_REPORT) return
    const files = ['tests/acceptance/narrative-world/corpus-v1.json',
      'tests/regression/R-NW-1C-source-coverage.test.ts', 'tests/helpers/narrative-reader-evaluation.ts',
      'src/lib/world-engine/narrative-domain-candidate.ts', 'src/lib/world-engine/narrative-domain-evidence.ts',
      'src/lib/world-engine/narrative-domain-projection.ts', 'src/lib/agent/run/narrative-annotation-durable.ts',
      'src/lib/world-engine/semantic-registry-v4.json']
    writeFileSync(process.env.NW_CANDIDATE_COVERAGE_REPORT, JSON.stringify({
      schema: 'storyforge.narrative-candidate-source-coverage-diagnostic', version: 1,
      boundary: 'Existing frozen synthetic corpus; deterministic fixture-authored candidates and simulated confirmation in an isolated database. No real model extraction, human review or repair of historical reader scores.',
      sourceHashes: Object.fromEntries(files.map(path => [path, createHash('sha256').update(readFileSync(path)).digest('hex')])),
      observations, modelCalls: 0, independentHumanReviews: 0, accepted: false,
    }, null, 2) + '\n', { flag: 'wx' })
  })

  async function setup() {
    const fixture = await seedReaderFamily(family)
    const actor = (await db.characters.get(fixture.localIds.get(`${family.id}:character-a`)!))!
    const peer = (await db.characters.get(fixture.localIds.get(`${family.id}:character-b`)!))!
    const owner = { kind: 'work' as const, workCode: (await db.works.get(fixture.scope.workId))!.code }
    return { fixture, actor, peer, owner }
  }
  const visibility = { class: 'author-only', audienceKeys: [], policyVersion: 'source-coverage-fixture-v1' }
  const quote = (source: Record<string, unknown>, fieldPointer: string, text: string, evidenceKey: string) => ({
    evidenceKey, source, fieldPointer, quote: text, occurrence: 0, supportRole: 'supports',
  })
  async function releaseAfter(fixture: Awaited<ReturnType<typeof seedReaderFamily>>) {
    const revision = await createWorldRevision({ scope: fixture.scope, formatVersion: 4, includeWorkSources: true,
      compilerVersion: 'narrative-domain-v2', label: '测试作者显式采纳后的独立版本' })
    return publishWorldRevision(revision.id!)
  }

  it('简介原字节已封存，但没有自动成为实体证据；选入上下文也不会扩大旧证据字段合同', async () => {
    const { fixture, actor, owner } = await setup()
    const verified = await verifyWorldReleaseRecordV1(fixture.v4Release)
    if (verified.version !== 4) throw Error('Expected v4')
    const frozen = verified.manifest.records.characters.find(row => row.ragDocumentId === actor.ragDocumentId)!
    expect(frozen.shortDescription).toBe(actor.shortDescription)
    const question = readerQuestions(family).find(item => item.operation === 'entity-filter')!
    const context = await irReaderContext(fixture, question, 'ir-lexical')
    expect(context.documents).toHaveLength(1)
    expect(JSON.stringify(context.documents)).not.toContain('青篙')
    const prepared = await prepareNarrativeCandidateV1({ scope: fixture.scope, selection: {
      table: 'characters', sourceUid: actor.ragDocumentId!, owner,
      entryKeys: [makeRagEntryKey(actor.ragDocumentId!, 'shortDescription')],
    } })
    expect(prepared.baseline.originalFields[0].text).toBe(actor.shortDescription)
    const proposal = { annotations: { narrativeAliasesV1: ['青篙'],
      narrativeEvidenceV1: [quote({ self: true }, '/shortDescription', '外号青篙的渡工', 'identity')] }, clearFields: [] }
    await expect(compileNarrativeAnnotationProposalV1({ ...prepared, raw: JSON.stringify(proposal) }))
      .rejects.toThrow('未登记为证据来源')
    expect(await db.characters.get(actor.id!)).toEqual(actor)
    observations.push({ case: 'stored-description-is-not-confirmed-identity', sourceText: actor.shortDescription,
      frozenRawPresent: true, context, selectedOriginals: prepared.baseline.originalFields, proposal,
      outcome: 'shortDescription can be selected as context but is not a quoted evidence field in the existing semantic registry; no source mutation' })
  }, 30_000)

  it('登记章节证据上的显式角色候选只在采纳及新封存后生效，同名另一人和旧版本保持', async () => {
    const { fixture, actor, peer, owner } = await setup()
    const chapter = (await db.chapters.get(fixture.localIds.get(`${family.id}:chapter-1`)!))!
    const selection = { table: 'characters', sourceUid: actor.ragDocumentId!, owner,
      entryKeys: [makeRagEntryKey(actor.ragDocumentId!, 'shortDescription'), makeRagEntryKey(chapter.ragDocumentId!, 'content')] }
    const proposal = { annotations: { narrativeAliasesV1: ['渡工'], narrativeVisibilityV1: visibility,
      narrativeEvidenceV1: [quote({ chapterId: null, sourceUid: chapter.ragDocumentId }, '/content', '渡工沈青', 'role')] }, clearFields: [] }
    const runAI = vi.fn(async () => ({ output: JSON.stringify(proposal), durationMs: 1 }))
    const generated = await generateNarrativeAnnotationCandidateV1({ scope: fixture.scope, selection,
      instruction: '测试作者指定的候选：只将该角色的渡工称呼与原文章节证据关联。',
      modelIdentity: { provider: 'fixture', model: 'test-only' }, runAI })
    expect(generated.candidate.artifact.status).toBe('ready')
    expect(await db.characters.get(actor.id!)).toEqual(actor)
    await adoptNarrativeAnnotationCandidateV1({ scope: fixture.scope, runId: generated.snapshot.run.id,
      candidateHash: generated.candidate.candidateHash })
    expect(await db.characters.get(peer.id!)).toEqual(peer)
    const release = await releaseAfter(fixture)
    const access = await openAuthorWorldQueryAccessV1({ scope: fixture.scope, localReleaseRecordId: release.id!,
      view: 'canon', perspective: { kind: 'author' }, knowledgeThrough: null, disclosedThrough: null })
    try {
      const result = await queryWorldReleaseV1(access, { ...authorWorldQueryV1(access), text: '渡工', kinds: ['entity'], evidenceDepth: 'original' })
      expect(result.items).toHaveLength(1)
      expect(result.items[0].authorSource?.sourceResourceKey).toContain(actor.ragDocumentId)
      expect(result.items[0].evidence.some(span => span.text === '渡工沈青')).toBe(true)
      expect((await db.worldReleases.get(fixture.v4Release.id!))!.manifestJson).toBe(fixture.v4Release.manifestJson)
      const old = await irReaderContext(fixture, readerQuestions(family).find(item => item.operation === 'entity-filter')!, 'ir-lexical')
      expect(JSON.stringify(old.documents)).not.toContain('渡工')
      observations.push({ case: 'explicit-role-candidate-adoption', proposal, result, oldContext: old,
        simulatedAuthorConfirmation: true, newReleaseHash: release.contentHash, oldReleaseHash: fixture.v4Release.contentHash })
    } finally { await closeWorldQueryAccessV1(access) }
    expect(runAI).toHaveBeenCalledTimes(1)
  }, 30_000)

  it('计划及无兑现证据的修改不能过关；带原文证据的作者候选保留子身份并在新版本可查', async () => {
    const { fixture, owner } = await setup()
    const arc = (await db.storyArcs.get(fixture.localIds.get(`${family.id}:arc-main`)!))!
    const chapter = (await db.chapters.get(fixture.localIds.get(`${family.id}:chapter-4`)!))!
    const question = readerQuestions(family).find(item => item.operation === 'development-query')!
    expect((await irReaderContext(fixture, question, 'ir-lexical')).documents).toEqual([])
    const selection = { table: 'storyArcs', sourceUid: arc.ragDocumentId!, owner,
      entryKeys: [makeRagEntryKey(arc.ragDocumentId!, 'description'), makeRagEntryKey(chapter.ragDocumentId!, 'content')] }
    const prepared = await prepareNarrativeCandidateV1({ scope: fixture.scope, selection })
    const existing = arc.narrativeDevelopmentsV1![0]
    const development = { ...existing, statement: '护送承诺', developmentType: 'commitment', state: 'fulfilled',
      narrativeModalityV1: 'established', narrativeVisibilityV1: visibility, resolutionEvidenceKeys: ['payoff'],
      narrativeEvidenceV1: [quote({ chapterId: null, sourceUid: chapter.ragDocumentId }, '/content', chapter.content, 'payoff')] }
    const raw = (item: unknown) => JSON.stringify({ annotations: { narrativeDevelopmentsV1: [item] }, clearFields: [] })
    for (const invalid of [{ ...development, narrativeModalityV1: 'intent' }, { ...development, resolutionEvidenceKeys: [] }]) {
      await expect(compileNarrativeAnnotationProposalV1({ ...prepared, raw: raw(invalid) })).rejects.toThrow('兑现必须有明确证据')
    }
    const runAI = vi.fn(async (_messages: ChatMessage[]) => ({ output: raw(development), durationMs: 1 }))
    const generated = await generateNarrativeAnnotationCandidateV1({ scope: fixture.scope, selection,
      instruction: '按选定正文审核这条既有发展的进展。', modelIdentity: { provider: 'fixture', model: 'test-only' }, runAI })
    expect(generated.candidate.artifact.status).toBe('ready')
    expect(await db.storyArcs.get(arc.id!)).toEqual(arc)
    await adoptNarrativeAnnotationCandidateV1({ scope: fixture.scope, runId: generated.snapshot.run.id,
      candidateHash: generated.candidate.candidateHash })
    expect((await db.storyArcs.get(arc.id!))!.narrativeDevelopmentsV1![0].semanticSubKey).toBe(existing.semanticSubKey)
    const release = await releaseAfter(fixture)
    const access = await openAuthorWorldQueryAccessV1({ scope: fixture.scope, localReleaseRecordId: release.id!,
      view: 'development', perspective: { kind: 'author' }, knowledgeThrough: null, disclosedThrough: null })
    try {
      const result = await queryWorldReleaseV1(access, { ...authorWorldQueryV1(access), text: null,
        kinds: ['development'], evidenceDepth: 'original' })
      expect(result.items).toHaveLength(1)
      expect(result.items[0].modality).toBe('established')
      expect(result.items[0].atom).toHaveProperty('state', 'fulfilled')
      expect(result.items[0].evidence.some(span => span.text === chapter.content)).toBe(true)
      expect((await irReaderContext(fixture, question, 'ir-lexical')).documents).toEqual([])
      expect((await db.worldReleases.get(fixture.v4Release.id!))!.manifestJson).toBe(fixture.v4Release.manifestJson)
      const targetSubKeyInPrompt = JSON.stringify(runAI.mock.calls[0][0]).includes(existing.semanticSubKey)
      observations.push({ case: 'explicit-development-candidate-adoption', sourceText: chapter.content,
        proposal: JSON.parse(raw(development)), result, simulatedAuthorConfirmation: true, targetSubKeyInPrompt,
        boundary: 'Fixture supplies the existing child identity. This is not proof that the current model prompt can update existing child annotations.',
        newReleaseHash: release.contentHash, oldReleaseHash: fixture.v4Release.contentHash })
    } finally { await closeWorldQueryAccessV1(access) }
    expect(runAI).toHaveBeenCalledTimes(1)
  }, 30_000)
})
