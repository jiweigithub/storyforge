from pathlib import Path
import datetime
import hashlib
import json
import subprocess
import sys

root = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
prefix = Path('/tmp/nw-retained-heap-20261017-v1')
pin_path = Path('/tmp/nw-retained-heap-20261017-v1-sources.json')
pin = json.loads(pin_path.read_text())
assert Path.cwd() == root
assert not Path(str(prefix) + '-terminal.json').exists()
record = {'schema': 'storyforge.narrative-retained-heap-run', 'version': 1,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'head': pin['head'], 'mergeHead': pin['mergeHead'],
          'pinSha256': hashlib.sha256(pin_path.read_bytes()).hexdigest(),
          'diagnosticOnly': True, 'accepted': False, 'phases': [], 'actualModelCalls': 0,
          'diagnosticInputScope': pin['diagnosticInputScope']}
for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
    assert subprocess.check_output(['git', 'rev-parse', ref], text=True).strip() == pin[key]
code = 2
try:
    for name, command in [
        ('probe', ['node', '/tmp/nw-retained-heap-probe-20261017-v1.mjs', str(prefix), str(pin_path)]),
        ('summary', ['python3', '/tmp/nw-retained-heap-summary-20261017-v1.py', str(prefix) + '-result.json']),
    ]:
        log = Path(str(prefix) + '-' + name + '.log')
        item = {'name': name, 'command': command, 'log': str(log),
                'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat()}
        record['phases'].append(item)
        print(json.dumps({'starting': name, 'log': str(log)}), flush=True)
        with log.open('x') as output:
            code = subprocess.call(command, stdout=output, stderr=subprocess.STDOUT)
        item.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat(),
                    logSha256=hashlib.sha256(log.read_bytes()).hexdigest())
        print(json.dumps({'completed': name, 'exitCode': code}), flush=True)
        if code:
            break
finally:
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    with Path(str(prefix) + '-terminal.json').open('x') as output:
        json.dump(record, output, indent=2)
        output.write('\n')
sys.exit(code)
