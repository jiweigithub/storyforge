// External, isolated diagnostics only. This script is never a performance acceptance run.
import { createRequire } from 'node:module'
import { spawn } from 'node:child_process'
import { createHash } from 'node:crypto'
import { createServer } from 'node:net'
import { once } from 'node:events'
import { createReadStream, createWriteStream, openSync, closeSync, writeSync, readFileSync, existsSync, statfsSync, unlinkSync, writeFileSync } from 'node:fs'
import { pipeline } from 'node:stream/promises'
import { createGzip, createGunzip } from 'node:zlib'
import { resolve } from 'node:path'

const root = process.cwd()
const prefix = resolve(process.argv[2] || '/tmp/nw-retained-heap-20261017-v1')
const pinPath = process.argv[3]
if (!pinPath) throw Error('An explicit frozen source pin is required')
const pinBytes = readFileSync(pinPath), pin = JSON.parse(pinBytes)
const hashes = pin.sourceHashes ?? pin.files
if (!hashes || typeof hashes !== 'object') throw Error('Missing source hashes')
const digest = bytes => createHash('sha256').update(bytes).digest('hex')
const verify = () => {
  if (digest(readFileSync(pinPath)) !== digest(pinBytes)) throw Error('Pin changed')
  const changed = Object.entries(hashes).filter(([path, sha]) => !existsSync(resolve(root, path)) || digest(readFileSync(resolve(root, path))) !== sha).map(([path]) => path)
  if (changed.length) throw Error('Pinned source changed: ' + JSON.stringify(changed))
}
if (existsSync(prefix + '-result.json')) throw Error('Probe result already exists; do not overwrite')
verify()
const disk = statfsSync(root)
if (disk.bavail * disk.bsize < 8 * 2 ** 30) throw Error('Probe requires 8 GiB free')
const require = createRequire(resolve(root, 'package.json'))
const { chromium } = require('@playwright/test')
const modulePath = '/storyforge/tests/helpers/narrative-scale-browser.ts'
const vectors = JSON.parse(readFileSync(resolve(root, 'tests/acceptance/narrative-world/scale-vectors-20261015-v2.json'), 'utf8')).results.find(row => row.size === 1000000)
if (!vectors) throw Error('Frozen million-character vector absent')
const record = { schema: 'storyforge.narrative-retained-heap-probe', version: 1,
  startedAt: new Date().toISOString(), pin: pinPath, pinSha256: digest(pinBytes), sourceCount: Object.keys(hashes).length,
  scriptSha256: digest(readFileSync(new URL(import.meta.url))), diagnosticOnly: true, accepted: false,
  instrumentation: 'CDP CPU profile and heap snapshots with forced GC; neither timings nor heap sizes are acceptance RSS measurements',
  actualModelCalls: 0, snapshots: [], profiles: [], sourceUnchanged: false }
let server, browserServer, browser, serverFd, phase = 'start'
const rawSnapshots = []
const log = value => console.log(JSON.stringify({ phase, at: new Date().toISOString(), ...value }))
async function snapshot(cdp, label) {
  const path = prefix + '-' + label + '.heapsnapshot'
  const fd = openSync(path, 'wx')
  let bytes = 0
  const hash = createHash('sha256')
  const consume = ({ chunk }) => { const data = Buffer.from(chunk); writeSync(fd, data); hash.update(data); bytes += data.length }
  cdp.on('HeapProfiler.addHeapSnapshotChunk', consume)
  try { await cdp.send('HeapProfiler.takeHeapSnapshot', { reportProgress: false, captureNumericValue: true }) }
  finally { cdp.off('HeapProfiler.addHeapSnapshotChunk', consume); closeSync(fd) }
  const item = { label, path, bytes, sha256: hash.digest('hex') }
  record.snapshots.push(item); rawSnapshots.push(item); log({ snapshot: label, bytes })
}
async function heap(cdp, label, capture = true) {
  const beforeGC = await cdp.send('Runtime.getHeapUsage')
  await cdp.send('HeapProfiler.collectGarbage')
  const afterGC = await cdp.send('Runtime.getHeapUsage')
  record[label + 'Heap'] = { beforeGC, afterGC }
  if (capture) await snapshot(cdp, label)
}
async function profiled(cdp, label, operation) {
  await cdp.send('Profiler.enable'); await cdp.send('Profiler.start')
  try { return await operation() }
  finally {
    const { profile } = await cdp.send('Profiler.stop')
    const path = prefix + '-' + label + '.cpuprofile', bytes = JSON.stringify(profile)
    writeFileSync(path, bytes, { flag: 'wx' })
    record.profiles.push({ label, path, bytes: Buffer.byteLength(bytes), sha256: digest(bytes) })
  }
}
try {
  const socket = createServer(); socket.listen(0, '127.0.0.1'); await once(socket, 'listening')
  const port = socket.address().port; await new Promise(resolveClose => socket.close(resolveClose))
  const baseURL = `http://127.0.0.1:${port}/storyforge/`
  serverFd = openSync(prefix + '-server.log', 'wx')
  server = spawn(process.execPath, ['scripts/serve-e2e-snapshot.mjs', '--port', String(port)], { cwd: root,
    stdio: ['ignore', serverFd, serverFd], env: { ...process.env, CI: '1', PLAYWRIGHT_FROZEN_WORKSPACE: '1' } })
  const deadline = Date.now() + 120000
  let ready = false
  while (Date.now() < deadline && server.exitCode === null) {
    try { const response = await fetch(baseURL, { signal: AbortSignal.timeout(2000) }); ready = response.ok; await response.arrayBuffer() } catch {}
    if (ready) break
    await new Promise(resolveWait => setTimeout(resolveWait, 500))
  }
  if (!ready) throw Error('Isolated snapshot server did not become ready')
  browserServer = await chromium.launchServer({ headless: true })
  browser = await chromium.connect(browserServer.wsEndpoint()); record.browserVersion = browser.version()
  const context = await browser.newContext(), page = await context.newPage()
  page.setDefaultTimeout(120000)
  await page.goto(baseURL)
  phase = 'prepare'; log({})
  record.prepared = await page.evaluate(async modulePath => (await new Function('p', 'return import(p)')(modulePath)).prepareScaleV2(1000000), modulePath)
  if (record.prepared.inputHash !== vectors.inputHash) throw Error('Frozen input vector differs')
  const cdp = await context.newCDPSession(page)
  await heap(cdp, 'prepared')
  phase = 'compile'; log({})
  record.compilation = await profiled(cdp, 'compile', () => page.evaluate(async modulePath => (await new Function('p', 'return import(p)')(modulePath)).compileScaleV1(), modulePath))
  if (record.compilation.contentHash !== vectors.contentHash || record.compilation.indexHash !== vectors.indexHash) throw Error('Frozen compilation vector differs')
  await heap(cdp, 'compiled')
  phase = 'export'; log({})
  const { bytes, ...metadata } = await page.evaluate(async modulePath => (await new Function('p', 'return import(p)')(modulePath)).exportScaleV1(), modulePath)
  record.export = metadata
  await context.close()
  phase = 'import'; log({})
  const restored = await browser.newContext(), imported = await restored.newPage()
  await imported.goto(baseURL)
  record.import = await imported.evaluate(async ({ modulePath, bytes }) => (await new Function('p', 'return import(p)')(modulePath)).importScaleV1(bytes), { modulePath, bytes })
  await imported.reload()
  await imported.evaluate(async modulePath => {
    const load = new Function('p', 'return import(p)')
    await Promise.all([load(modulePath), load('/storyforge/src/lib/context-gateway/world-release-provider.ts'), load('/storyforge/src/lib/context-gateway/world-release-query-provider.ts')])
  }, modulePath)
  await restored.setOffline(true)
  const coldCdp = await restored.newCDPSession(imported)
  await heap(coldCdp, 'beforeCold', false)
  phase = 'cold-open'; log({})
  record.cold = await profiled(coldCdp, 'cold', () => imported.evaluate(async modulePath => (await new Function('p', 'return import(p)')(modulePath)).openScaleV1(), modulePath))
  await heap(coldCdp, 'opened')
  record.queries = []
  for (const needle of record.prepared.needles) record.queries.push(await imported.evaluate(async ({ modulePath, needle }) => (await new Function('p', 'return import(p)')(modulePath)).queryScaleV1(needle), { modulePath, needle }))
  if (record.queries.some(row => !row.exactEvidenceFound)) throw Error('Exact needle evidence absent')
  await restored.close()
} catch (error) { record.error = { phase, message: String(error), stack: error.stack }; process.exitCode = 1 }
finally {
  if (browser) await browser.close()
  if (browserServer) await browserServer.close()
  if (server && server.exitCode === null) { const exited = once(server, 'exit'); server.kill('SIGTERM'); await exited }
  if (serverFd !== undefined) closeSync(serverFd)
  phase = 'compress-evidence'; log({})
  for (const item of rawSnapshots) {
    try {
      const compressed = item.path + '.gz'
      await pipeline(createReadStream(item.path), createGzip({ level: 6, mtime: 0 }), createWriteStream(compressed, { flags: 'wx' }))
      const hash = createHash('sha256')
      for await (const chunk of createReadStream(compressed).pipe(createGunzip())) hash.update(chunk)
      if (hash.digest('hex') !== item.sha256) throw Error('Compressed snapshot differs')
      unlinkSync(item.path)
      item.path = compressed; item.compressedSha256 = digest(readFileSync(compressed))
    } catch (error) { item.archiveError = String(error); process.exitCode = 1 }
  }
  try { verify(); record.sourceUnchanged = true } catch (error) { record.sourceError = String(error); process.exitCode = 1 }
  record.endedAt = new Date().toISOString()
  writeFileSync(prefix + '-result.json', JSON.stringify(record, null, 2) + '\n', { flag: 'wx' })
  log({ completed: true, error: record.error ?? null, sourceUnchanged: record.sourceUnchanged, result: prefix + '-result.json' })
}
