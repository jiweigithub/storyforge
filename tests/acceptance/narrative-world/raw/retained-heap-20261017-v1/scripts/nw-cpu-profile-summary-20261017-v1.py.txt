"""Summarize the current owned diagnostic profiles, including idle/native time."""
from collections import Counter
from pathlib import Path
import hashlib
import json
import sys

report_path = Path(sys.argv[1])
report = json.loads(report_path.read_text())
result = {'schema': 'storyforge.narrative-current-cpu-profile-summary', 'version': 1,
          'probe': str(report_path), 'probeSha256': hashlib.sha256(report_path.read_bytes()).hexdigest(),
          'diagnosticOnly': True, 'accepted': False, 'profiles': []}

def category(frame):
    url = frame.get('url', '')
    name = frame.get('functionName', '')
    if name in ['(idle)', '(garbage collector)', '(program)']:
        return name
    if 'noble' in url or ('sha' in url.lower() and 'node_modules' in url):
        return 'SHA implementation'
    if '/narrative-domain/json.ts' in url:
        return 'canonical JSON and hash sinks'
    if '/world-engine/narrative-index' in url or '/world-engine/narrative-lexical' in url:
        return 'index construction and cache'
    if '/world-engine/narrative-ir' in url or '/world-engine/narrative-time' in url or '/world-engine/narrative-identity' in url:
        return 'IR identity, evidence and validation'
    if '/world-engine/narrative-domain' in url or '/narrative-domain/' in url or '/world-engine/narrative-projection' in url:
        return 'source domain projection'
    if '/world-engine/narrative-release' in url or '/world-engine/release-' in url:
        return 'release verification and compilation'
    if '/context-gateway/world-' in url:
        return 'query authorization and preparation'
    if '/chapter-memory/text-normalization' in url:
        return 'native crypto text hashing'
    if 'dexie' in url:
        return 'IndexedDB'
    return None

for item in report['profiles']:
    path = Path(item['path'])
    raw = path.read_bytes()
    assert hashlib.sha256(raw).hexdigest() == item['sha256']
    profile = json.loads(raw)
    nodes = {node['id']: node for node in profile['nodes']}
    parents = {child: node['id'] for node in profile['nodes'] for child in node.get('children', [])}
    self_time, categories = Counter(), Counter()
    samples, deltas = profile.get('samples', []), profile.get('timeDeltas', [])
    assert len(samples) == len(deltas)
    for node_id, micros in zip(samples, deltas):
        frame = nodes[node_id]['callFrame']
        url = frame.get('url', '')
        if '/storyforge/' in url:
            url = url.split('/storyforge/', 1)[1]
        self_time[(frame.get('functionName', ''), url, frame.get('lineNumber', -1) + 1)] += micros
        current, found = node_id, None
        while current is not None and found is None:
            found = category(nodes[current]['callFrame'])
            current = parents.get(current)
        categories[found or 'other native/app work'] += micros
    total = sum(deltas)
    result['profiles'].append({'label': item['label'], 'path': str(path),
        'sourceUnchanged': report['sourceUnchanged'], 'wallMs': (profile['endTime'] - profile['startTime']) / 1000,
        'sampledMs': total / 1000, 'samples': len(samples),
        'categoryMs': {key: round(value / 1000, 3) for key, value in categories.most_common()},
        'topSelf': [{'function': key[0], 'url': key[1], 'line': key[2], 'ms': round(value / 1000, 3),
                     'samplePercent': round(value / total * 100, 3)} for key, value in self_time.most_common(30)]})

output = Path(str(report_path).replace('-result.json', '-cpu-summary.json'))
with output.open('x') as stream:
    json.dump(result, stream, indent=2)
    stream.write('\n')
print(str(output))
