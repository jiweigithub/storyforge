"""Summarize owned V8 heap snapshots without printing source text or config."""
import collections
import gzip
import hashlib
import json
from pathlib import Path
import sys

report_path = Path(sys.argv[1])
report = json.loads(report_path.read_text())
result = {'schema': 'storyforge.narrative-retained-heap-summary', 'version': 1,
          'probe': str(report_path), 'probeSha256': hashlib.sha256(report_path.read_bytes()).hexdigest(),
          'diagnosticOnly': True, 'accepted': False,
          'limits': 'Post-GC V8 self sizes are not peak browser RSS. Component reachability is not a dominator retained-size measurement; shared nodes can appear in multiple components.',
          'snapshots': []}

for item in report['snapshots']:
    path = Path(item['path'])
    with (gzip.open(path, 'rt') if path.suffix == '.gz' else path.open()) as stream:
        heap = json.load(stream)
    meta, nodes, edges, strings = heap['snapshot']['meta'], heap['nodes'], heap['edges'], heap['strings']
    fields, ef = meta['node_fields'], meta['edge_fields']
    width, ew = len(fields), len(ef)
    tp, name, size, edge_count = [fields.index(key) for key in ['type', 'name', 'self_size', 'edge_count']]
    et, en, to = [ef.index(key) for key in ['type', 'name_or_index', 'to_node']]
    types, edge_types = meta['node_types'][tp], meta['edge_types'][et]
    starts, totals, counts, compiler_objects, manifest_strings = [], collections.Counter(), collections.Counter(), [], []
    cursor = 0
    for pos in range(0, len(nodes), width):
        starts.append(cursor)
        kind = types[nodes[pos + tp]]
        totals[kind] += nodes[pos + size]; counts[kind] += 1
        props = {}
        for ep in range(cursor, cursor + nodes[pos + edge_count] * ew, ew):
            edge_kind = edge_types[edges[ep + et]]
            key = strings[edges[ep + en]] if edge_kind == 'property' else None
            if edge_kind == 'property' and key in ['manifestJson', 'manifest', 'contentHash', 'index']:
                props[key] = edges[ep + to]
            if edge_kind == 'property' and key == 'manifestJson':
                target = edges[ep + to]
                manifest_strings.append({'ownerNode': pos // width, 'targetNode': target // width,
                    'type': types[nodes[target + tp]], 'selfBytes': nodes[target + size]})
        if len(props) == 4:
            compiler_objects.append({'node': pos // width, 'properties': props})
        cursor += nodes[pos + edge_count] * ew
    assert cursor == len(edges)

    def component(start):
        seen, todo, sizes, amounts = set(), [start], collections.Counter(), collections.Counter()
        while todo:
            pos = todo.pop()
            if pos in seen:
                continue
            seen.add(pos)
            kind = types[nodes[pos + tp]]
            sizes[kind] += nodes[pos + size]; amounts[kind] += 1
            first = starts[pos // width]
            for ep in range(first, first + nodes[pos + edge_count] * ew, ew):
                edge_kind, target = edge_types[edges[ep + et]], edges[ep + to]
                target_kind = types[nodes[target + tp]]
                key = strings[edges[ep + en]] if edge_kind in ['property', 'internal'] else None
                if edge_kind == 'element' or (edge_kind == 'property' and key != '__proto__'):
                    todo.append(target)
                elif edge_kind == 'internal' and (key in ['elements', 'properties'] or ('string' in kind and 'string' in target_kind)):
                    todo.append(target)
        return {'reachableSelfBytes': sum(sizes.values()), 'nodes': len(seen),
                'bytesByType': dict(sizes.most_common()), 'countsByType': dict(amounts.most_common())}

    compiled = []
    for obj in compiler_objects:
        compiled.append({'node': obj['node'], 'components': {key: component(target) for key, target in obj['properties'].items()}})
    result['snapshots'].append({'label': item['label'], 'file': str(path),
        'totalSelfBytes': sum(totals.values()), 'bytesByType': dict(totals.most_common()),
        'countsByType': dict(counts.most_common()), 'manifestJsonReferences': manifest_strings,
        'compiledObjects': compiled, 'nodeCount': len(nodes) // width})
    print(json.dumps({'label': item['label'], 'MiB': round(sum(totals.values()) / 2**20, 3),
                      'compilerObjects': len(compiled), 'manifestJsonReferences': len(manifest_strings)}), flush=True)
    del heap, nodes, edges, strings, starts

output = Path(str(report_path).replace('-result.json', '-summary.json'))
with output.open('x') as stream:
    json.dump(result, stream, indent=2)
    stream.write('\n')
print(str(output))
