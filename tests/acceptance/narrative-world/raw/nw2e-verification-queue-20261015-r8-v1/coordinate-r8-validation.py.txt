from pathlib import Path
import datetime, json, subprocess, sys, time

root = '/Users/qinyingying/.codex/worktrees/nw2e-verification-r8/storyforge'
terminal = Path('/tmp/nw2e-r8-ci-terminal.json')
print('Waiting for exact r8 full CI before full native validation.', flush=True)
while not terminal.exists():
    time.sleep(10)
result = json.loads(terminal.read_text())
if result['exitCode'] != 0 or not result['sourceUnchanged']:
    print('r8 CI failed; full native validation not dispatched.', flush=True)
    sys.exit(2)
print('r8 CI passed; requesting shared slot for full native validation.', flush=True)
with Path('/tmp/nw2e-full-e2e-frozen-r8.log').open('x') as stream:
    status = subprocess.call(['python3', '/tmp/storyforge-serial-ci-20261013.py', '--',
                              'python3', '/tmp/nw2e-run-frozen-r8.py', 'e2e'],
                             cwd=root, stdout=stream, stderr=subprocess.STDOUT)
Path('/tmp/nw2e-r8-coordinator-terminal.json').write_text(json.dumps({
    'schema': 'storyforge.validation-coordinator-terminal', 'version': 1,
    'endedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'exitCode': status,
    'performanceDispatched': False, 'accepted': False}, indent=2) + '\n')
print('r8 native wrapper exited ' + str(status) + '; no performance or model calls dispatched.', flush=True)
sys.exit(status)
