from pathlib import Path
import datetime, hashlib, json, shutil

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
BASE = Path('/tmp/nw-index-keys-paired-validation-20261021-v1')
SNAPSHOTS = Path('/tmp/nw-index-keys-pair-snapshots-20261021-v1')
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2); stream.write('\n')

assert not SNAPSHOTS.exists()
probe_receipt = ROOT / 'tests/acceptance/narrative-world/index-keys-probe-result-20261021-v1.json'
assert sha(probe_receipt) == '94fba1a6eb0a877b4ba1109469ce3437bcd3d9fe2e6657135cf9bbc7804c20ac'
prior = json.loads(Path('/tmp/nw-buffered-text-paired-validation-20261021-v1-snapshot-sources.json').read_text())
support = json.loads(Path('/tmp/nw-buffered-text-paired-validation-20261021-v1-snapshot-support-sources.json').read_text())
paths = set(prior['originHashes']) | set(support['originHashes'])
paths.update(str(p.relative_to(ROOT)) for p in (ROOT / 'tests/acceptance/narrative-world').iterdir() if p.is_file())
paths.update(str(p.relative_to(ROOT)) for p in (ROOT / 'docs').rglob('*') if p.is_file())
paths.update(['AGENTS.md', 'README.md'])
assert not any(any(part.startswith('.env') for part in Path(name).parts) for name in paths)
origin = {name: sha(ROOT / name) for name in sorted(paths)}
index_path = 'src/lib/world-engine/narrative-index.ts'
test_path = 'tests/regression/R-NW-2E-index-keys.test.ts'
assert test_path not in paths and not (ROOT / test_path).exists()
draft = Path('/tmp/nw-index-keys-integration-regression-20261021-v1.ts')
helpers = Path('/tmp/nw-index-keys-probe-20261021-v1/keys.ts')
helper_text = helpers.read_text().split('// Experimental private-compiler writers.', 1)[1]
helper_text = '// Private writers for fresh compiler-owned tuple/edge keys.' + helper_text
helper_text = helper_text.replace('export const compilerEdgeKey', 'const compilerEdgeKey').replace('export function compilerTupleKey', 'function compilerTupleKey')
helper_text = helper_text.replace('export { canonicalNarrativeJsonV1, compareUnicodeCodePointsV1 }\n', '')
source = (ROOT / index_path).read_text()
anchor = 'const numberSet = '
assert source.count(anchor) == 1
candidate = source.replace(anchor, helper_text + '\n' + anchor)
replacements = {
    'canonicalNarrativeJsonV1([from, to, role])': 'compilerTupleKey(from, to, role)',
    'canonicalNarrativeJsonV1([source.sourceResourceKey, source.sourceRevision, source.fieldPointer])':
        'compilerTupleKey(source.sourceResourceKey, source.sourceRevision, source.fieldPointer)',
    'canonicalNarrativeJsonV1([source.sourceResourceKey, source.fieldPointer, source.sourceRevision])':
        'compilerTupleKey(source.sourceResourceKey, source.fieldPointer, source.sourceRevision)',
    'adjacency.map(item => [canonicalNarrativeJsonV1(item), item])': 'adjacency.map(item => [compilerEdgeKey(item), item])',
}
for old, new in replacements.items():
    assert candidate.count(old) == (2 if 'source.fieldPointer, source.sourceRevision' in old else 1), old
    candidate = candidate.replace(old, new)
scale = json.loads(Path('/tmp/nw-buffered-text-rollback-validation-20261021-v1-scale-sources.json').read_text())
variants = {}
for variant in ['baseline', 'candidate']:
    target = SNAPSHOTS / variant
    target.mkdir(parents=True)
    for name in sorted(paths):
        dest = target / name; dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(ROOT / name, dest)
    (target / test_path).write_bytes(draft.read_bytes())
    if variant == 'candidate': (target / index_path).write_text(candidate)
    (target / 'node_modules').symlink_to(ROOT / 'node_modules', target_is_directory=True)
    variants[variant] = {name: sha(target / name) for name in sorted(paths | {test_path})}
    save(str(BASE) + '-' + variant + '-scale-sources.json', {'sourceHashes': {name: variants[variant][name] for name in scale['sourceHashes']}})
assert [name for name in sorted(variants['baseline']) if variants['baseline'][name] != variants['candidate'][name]] == [index_path]
assert all(variants['baseline'][name] == digest for name, digest in origin.items())
runtime = dict(prior['runtimeHashes'])
probe_pin = json.loads(Path('/tmp/nw-index-keys-probe-20261021-v1/pin.json').read_text())
runtime.update({p: v for p, v in probe_pin['files'].items() if p.startswith(str(ROOT / 'node_modules') + '/')})
assert all(sha(path) == digest for path, digest in runtime.items())
inputs = {'schema': 'storyforge.narrative-paired-snapshot-inputs', 'version': 1, 'createdAt': now(),
    'sourceWorkspace': str(ROOT), 'snapshots': str(SNAPSHOTS), 'originHashes': origin,
    'variants': variants, 'runtimeHashes': runtime, 'nodeModulesTarget': str(ROOT / 'node_modules'),
    'onlyVariantDifference': index_path, 'sameNewTestInBothVariants': test_path,
    'draftInputs': {str(draft): sha(draft), str(helpers): sha(helpers), str(Path(__file__)): sha(__file__)}}
save(str(BASE) + '-snapshot-sources.json', inputs)
rollback_plan = json.loads((ROOT / 'tests/acceptance/narrative-world/buffered-text-rollback-plan-20261021-v1.json').read_text())
plan = {'schema': 'storyforge.narrative-index-keys-paired-validation-plan', 'version': 1, 'createdAt': now(),
    'product': 'World engine S1 product engineering',
    'boundary': 'Independent compiler experiment only; no owner/schema/registry, public decoder, recipe, gold, threshold, source text, release or author-data changes. Production index remains untouched.',
    'purpose': 'Decide retention using integrated regressions and complete compile/export/import/offline-open measurements, following the isolated key probe.',
    'probeReceipt': str(probe_receipt.relative_to(ROOT)), 'probeReceiptSha256': sha(probe_receipt),
    'snapshotInputs': str(BASE) + '-snapshot-sources.json', 'snapshotInputsSha256': sha(str(BASE) + '-snapshot-sources.json'),
    'sourcePathsPerVariant': len(variants['baseline']), 'runtimeFiles': len(runtime), 'onlyVariantDifference': index_path,
    'baselineSha256': variants['baseline'][index_path], 'candidateSha256': variants['candidate'][index_path],
    'newRegression': test_path, 'newRegressionSha256': sha(draft),
    'directedTests': rollback_plan['directedTests'] + [test_path],
    'checksBeforeTiming': ['main workspace diff and docs', 'new regression on baseline', '32 directed files on candidate', 'candidate changed-file lint with zero warnings', 'candidate TypeScript'],
    'sequence': [['baseline', 'candidate'], ['candidate', 'baseline'], ['baseline', 'candidate'], ['candidate', 'baseline'], ['baseline', 'candidate']],
    'size': 1000000, 'mode': 'diagnostic', 'harness': 'tests/e2e/narrative-scale.spec.ts',
    'policy': {'freshBrowserEachRun': True, 'profiling': False, 'retries': 0, 'allAttemptsRetained': True,
        'stopOnCommandOrIntegrityFailure': True, 'stopOnPerformanceGateFailure': False, 'goldOrThresholdChanges': False,
        'sharedSerialEntry': '/tmp/storyforge-serial-ci-20261013.py', 'acceptance': False, 'aggregateAsFormalAcceptance': False,
        'retention': 'All integrated conformance/identity checks must pass, every paired cold open must improve, and candidate median additional RSS must not exceed baseline median. Any mixed cold-open or median RSS regression rejects this candidate. This is an experiment disposition rule, not a change to any product acceptance threshold.'},
    'productionChanged': False, 'adopted': False, 'authorDataTouched': False, 'actualModelCalls': 0, 'independentHumanReviews': 0,
    'requiredAfterRetention': ['current-source baseline and governance gates', 'full build', 'final-source full CI and E2E', 'formal three-size cold/warm/RSS acceptance', 'independent quality and impact evidence']}
plan_path = ROOT / 'tests/acceptance/narrative-world/index-keys-paired-plan-20261021-v1.json'
save(plan_path, plan)
print(json.dumps({'plan': str(plan_path), 'filesPerVariant': len(variants['baseline']), 'runtimeFiles': len(runtime),
    'onlyVariantDifference': index_path, 'baselineSha256': plan['baselineSha256'], 'candidateSha256': plan['candidateSha256']}))
