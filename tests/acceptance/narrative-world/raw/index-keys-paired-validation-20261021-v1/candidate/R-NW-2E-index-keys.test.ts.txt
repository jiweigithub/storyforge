import { createHash } from 'node:crypto'
import { describe, expect, it } from 'vitest'
import { narrativeIRFixture, irSourceIdentity, NW_HASH } from '../helpers/narrative-ir-fixture'
import { createNarrativeSemanticIdV1 } from '../../src/lib/world-engine/narrative-identity'
import { compileNarrativeIRBundleV1 } from '../../src/lib/world-engine/narrative-ir'
import { compileNarrativeIndexV1, compileNarrativeIndexIncrementalV1, verifyNarrativeIndexV1 } from '../../src/lib/world-engine/narrative-index'
import { canonicalNarrativeJsonV1, compareUnicodeCodePointsV1, hashNarrativeValueV1 } from '../../src/lib/world-engine/narrative-json'
import type { NarrativeDomainProjectionV1 } from '../../src/lib/world-engine/narrative-domain-projection'

describe('NW-2E · 私有索引键保持完整编译的字节、排序和来源边界', () => {
  it('转义分隔符、Unicode、多位数位置和增量复用与严格规范键一致', async () => {
    const fixture = await narrativeIRFixture()
    const template = fixture.input.records.find(record => record.kind === 'entity')!
    if (template.kind !== 'entity') throw new Error('missing entity fixture')
    const event = fixture.input.records.find(record => record.kind === 'event')!
    if (event.kind !== 'event') throw new Error('missing event fixture')
    const roles = ['quoted"key', 'slash\\key', 'nul\0key', '\ue000', '𐀀', '😀', 'e\u0301', 'é', 'bracket[],key']
    for (let offset = 0; offset < 32; offset++) {
      const identity = irSourceIdentity('character', offset + 100)
      const record = structuredClone(template)
      record.semanticId = createNarrativeSemanticIdV1(fixture.input.scope.owner, 'entity', identity)
      record.payload.sourceIdentity = identity
      record.payload.canonicalLabel = `人物${offset}`
      record.provenance.sourceResourceKey = `character:${identity.sourceUid}`
      fixture.sources.push({ ...fixture.sources[0], sourceResourceKey: record.provenance.sourceResourceKey })
      fixture.input.records.push(record)
      event.payload.participants.push({ entityRef: { kind: 'entity', semanticId: record.semanticId, revisionHash: NW_HASH },
        role: roles[offset % roles.length] })
    }
    // Two fields of the same source, plus escaped revisions, exercise tuple
    // boundaries without weakening the public source decoder.
    const lastSource = fixture.sources.at(-1)!
    const extraSource = { ...lastSource, sourceRevision: 'rev\0with"quote', fieldPointer: '/field~0/~1😀' }
    fixture.sources.push(extraSource)
    const extraRecord = fixture.input.records.at(-1)!
    extraRecord.provenance.sourceRevision = extraSource.sourceRevision
    extraRecord.provenance.sourceField = extraSource.fieldPointer
    for (let position = 0; position < fixture.sources.length - 1; position++) {
      const source = fixture.sources[position]
      const oldRevision = source.sourceRevision, oldPointer = source.fieldPointer
      source.sourceRevision = `revision:${roles[position % roles.length]}`
      source.fieldPointer = `/field:${roles[(position + 3) % roles.length]}`
      for (const record of fixture.input.records) {
        if (record.provenance.sourceResourceKey === source.sourceResourceKey
          && record.provenance.sourceRevision === oldRevision && record.provenance.sourceField === oldPointer) {
          record.provenance.sourceRevision = source.sourceRevision
          record.provenance.sourceField = source.fieldPointer
        }
      }
      for (const span of fixture.input.evidence) {
        if (span.sourceResourceKey === source.sourceResourceKey && span.sourceRevision === oldRevision && span.fieldPointer === oldPointer) {
          span.sourceRevision = source.sourceRevision
          span.fieldPointer = source.fieldPointer
        }
      }
    }
    event.dependencies.push({ semanticId: template.semanticId, revisionHash: NW_HASH, role: 'supports' })
    const bundle = await compileNarrativeIRBundleV1(fixture.input, fixture.sources)
    const sourceKeys = [...new Set(bundle.records.map(record => record.provenance.sourceResourceKey))].sort(compareUnicodeCodePointsV1)
    const projection: NarrativeDomainProjectionV1 = { version: 1, bundle, sources: fixture.sources,
      sourceGroups: bundle.records.map(record => ({ semanticId: record.semanticId, worldGroupSourceUid: null })),
      dependencyEvidence: [{ fromSemanticId: event.semanticId, toSemanticId: template.semanticId, role: 'supports', evidenceIds: event.evidenceIds }],
      coverage: { version: 1, eligibleSelected: sourceKeys, projectedSourceKeys: sourceKeys, explicitlyUnsupportedSourceKeys: [],
        gaps: [], excluded: [], fields: [], declaredAtoms: bundle.records.length, projectedAtoms: bundle.records.length, sourceSpans: [] } }
    const input = { inputRootHash: await hashNarrativeValueV1(projection), projection }
    const retained = await compileNarrativeIndexV1(input)
    const direct = await compileNarrativeIndexV1({ ...input, retainIncrementalShards: false })
    expect(canonicalNarrativeJsonV1(direct)).toBe(canonicalNarrativeJsonV1(retained))
    const positions = new Map(direct.indexes.recordKeys.map((key, position) => [key, position]))
    const participantEdges = event.payload.participants.map(participant => ({
      fromIndex: positions.get(event.semanticId)!, toIndex: positions.get(participant.entityRef.semanticId)!,
      kind: 'participant', role: participant.role, evidenceIds: [...event.evidenceIds].sort(compareUnicodeCodePointsV1), basis: 'structural',
    }))
    const expectedEdges = participantEdges.map(canonicalNarrativeJsonV1).sort(compareUnicodeCodePointsV1)
    const actualEdges = direct.indexes.adjacency.filter(edge => edge.kind === 'participant').map(canonicalNarrativeJsonV1)
    expect(actualEdges).toEqual(expectedEdges)
    expect(participantEdges.some(edge => edge.toIndex >= 10)).toBe(true)
    const edgeKeys = direct.indexes.adjacency.map(canonicalNarrativeJsonV1)
    expect(edgeKeys).toEqual([...new Set(edgeKeys)].sort(compareUnicodeCodePointsV1))
    expect(direct.indexes.adjacency.filter(edge => edge.kind === 'supports')).toEqual([{
      fromIndex: positions.get(template.semanticId), toIndex: positions.get(event.semanticId),
      kind: 'supports', role: null, evidenceIds: [...event.evidenceIds].sort(compareUnicodeCodePointsV1), basis: 'declared',
    }])
    const sourceOrder = fixture.sources.map(source => canonicalNarrativeJsonV1([source.sourceResourceKey, source.fieldPointer, source.sourceRevision]))
      .sort(compareUnicodeCodePointsV1)
    const chunks = direct.indexes.chunks
    expect(chunks.map(chunk => canonicalNarrativeJsonV1([chunk.sourceResourceKey, chunk.fieldPointer, chunk.sourceRevision]))).toEqual(sourceOrder)
    for (const chunk of chunks) {
      const expected = bundle.records.filter(record => record.provenance.sourceResourceKey === chunk.sourceResourceKey
        && record.provenance.sourceRevision === chunk.sourceRevision && record.provenance.sourceField === chunk.fieldPointer
        || record.evidenceIds.some(id => bundle.evidence.some(span => span.evidenceId === id
          && span.sourceResourceKey === chunk.sourceResourceKey && span.sourceRevision === chunk.sourceRevision && span.fieldPointer === chunk.fieldPointer)))
        .map(record => positions.get(record.semanticId)!).sort((a, b) => a - b)
      expect(chunk.recordIndexes).toEqual(expected)
    }
    const incremental = await compileNarrativeIndexIncrementalV1({ ...input,
      projection: { ...projection, sources: [...projection.sources].reverse() } }, retained)
    expect(incremental.receipt.rebuiltSources).toEqual([])
    expect(incremental.receipt.reusedSources).toEqual(sourceOrder)
    expect(canonicalNarrativeJsonV1(incremental.index)).toBe(canonicalNarrativeJsonV1(direct))
    expect(await verifyNarrativeIndexV1(direct, input)).toEqual(direct)
    const { indexHash, ...body } = direct
    expect(indexHash).toBe(createHash('sha256').update(canonicalNarrativeJsonV1(body)).digest('hex'))
    const staleProof = structuredClone(projection)
    staleProof.dependencyEvidence[0].role = 'contradicts'
    await expect(compileNarrativeIndexV1({ ...input, projection: staleProof })).rejects.toThrow('index-dependency-evidence')
  })
})
