// Private writers for fresh compiler-owned tuple/edge keys. These accept only compiler-created
// dense rows and arrays. They are not an external-object validation API.
function compilerStringKey(value: string): string {
  if (typeof value !== 'string') narrativeFail('json', '索引键字符串字段无效')
  assertUnicodeV1(value)
  return JSON.stringify(value)
}
function compilerNumberKey(value: number): string {
  if (!Number.isSafeInteger(value) || value < 0 || Object.is(value, -0)) narrativeFail('json', '索引数字必须为安全非负整数')
  return String(value)
}
function compilerStringsKey(values: string[]): string {
  let encoded = '['
  for (let index = 0; index < values.length; index++) {
    if (index) encoded += ','
    encoded += compilerStringKey(values[index])
  }
  return encoded + ']'
}
type CompilerKeyFields<T> = { [K in keyof T]-?: (value: T[K]) => string }
function compilerObjectKey<T extends object>(fields: CompilerKeyFields<T>): (value: T) => string {
  const keys = (Object.keys(fields) as Array<keyof T & string>).sort(compareUnicodeCodePointsV1)
  const tokens = keys.map(key => `${JSON.stringify(key)}:`)
  return value => {
    let encoded = '{'
    for (let index = 0; index < keys.length; index++) {
      if (index) encoded += ','
      const key = keys[index]
      encoded += tokens[index] + fields[key](value[key])
    }
    return encoded + '}'
  }
}
const compilerEdgeKey = compilerObjectKey<NarrativeIndexV1['indexes']['adjacency'][number]>({
  fromIndex: compilerNumberKey,
  toIndex: compilerNumberKey,
  kind: compilerStringKey,
  role: value => value === null ? 'null' : compilerStringKey(value),
  evidenceIds: compilerStringsKey,
  basis: compilerStringKey,
})
function compilerTupleKey(a: string, b: string, c: string): string {
  return '[' + compilerStringKey(a) + ',' + compilerStringKey(b) + ',' + compilerStringKey(c) + ']'
}

const numberSet = (values: Iterable<number>): number[] => [...(values instanceof Set ? values : new Set(values))].sort((a, b) => a - b)
/** Compiler-owned posting membership is usually appended in source/record
 * order. Sorted arrays avoid retaining a hash-table Set and then allocating a
 * second Set/array for the same members. The fallback handles any out-of-order
 * overlap exactly; it never assumes that a smaller index was already present. */
function insertPostingIndex(values: number[], value: number): void {
  const last = values[values.length - 1]
  if (last === undefined || value > last) { values.push(value); return }
  if (last === value) return
  let low = 0, high = values.length
  while (low < high) {
    const middle = (low + high) >>> 1
    if (values[middle] < value) low = middle + 1
    else high = middle
  }
  if (values[low] !== value) values.splice(low, 0, value)
}
const stringSet = (values: Iterable<string>): string[] => [...new Set(values)].sort(compareUnicodeCodePointsV1)
const UNKNOWN_POLICY: VisibilityV1 = { class: 'unclassified', audienceKeys: [], policyVersion: 'narrative-index-evidence-v1' }
const PUBLIC_POLICY: VisibilityV1 = { class: 'public', audienceKeys: [], policyVersion: 'narrative-index-evidence-v1' }
const utf8Size = (code: number) => code <= 0x7f ? 1 : code <= 0x7ff ? 2 : code <= 0xffff ? 3 : 4

/** Fixed scalar rules, independent of locale, model, browser word segmenter or
 * Unicode normalization. Offsets always address the original UTF-8 bytes. */
function forEachNarrativeLexicalTermV1(text: string, emit: (term: string, start: number, end: number) => void): void {
  let offset = 0, ascii = '', asciiStart = 0
  let previous = '', previousStart = 0
  const flush = () => {
    if (ascii) emit(ascii, asciiStart, asciiStart + ascii.length)
    ascii = ''
  }
  for (const scalar of text) {
    const code = scalar.codePointAt(0)!, end = offset + utf8Size(code)
    const asciiWord = (code >= 48 && code <= 57) || (code >= 65 && code <= 90) || (code >= 97 && code <= 122) || code === 95
    if (asciiWord) {
      previous = ''
      if (!ascii) asciiStart = offset
      ascii += code >= 65 && code <= 90 ? String.fromCharCode(code + 32) : scalar
      if (ascii.length === 128) flush()
    } else {
      flush()
      const separator = code <= 0x7f || code === 0xa0 || (code >= 0x2000 && code <= 0x206f)
        || (code >= 0x2e00 && code <= 0x2e7f) || (code >= 0x3000 && code <= 0x303f)
        || (code >= 0xfe10 && code <= 0xfe1f) || (code >= 0xfe30 && code <= 0xfe6f)
        || (code >= 0xff01 && code <= 0xff0f) || (code >= 0xff1a && code <= 0xff20)
        || (code >= 0xff3b && code <= 0xff40) || (code >= 0xff5b && code <= 0xff65)
      if (separator) previous = ''
      else {
        if (previous) emit(previous + scalar, previousStart, end)
        emit(scalar, offset, end)
        previous = scalar; previousStart = offset
      }
    }
    offset = end
  }
  flush()
}

export function narrativeLexicalTermsV1(text: string): Array<{ term: string; start: number; end: number }> {
  const terms: Array<{ term: string; start: number; end: number }> = []
  forEachNarrativeLexicalTermV1(text, (term, start, end) => terms.push({ term, start, end }))
  return terms
}

function verifyCoverage(projection: NarrativeDomainProjectionV1): void {
  const coverage = projection.coverage
  if (projection.version !== 1 || coverage.version !== 1) narrativeFail('index-version', '叙事索引输入版本不受支持')
  const eligible = stringSet(coverage.eligibleSelected), projected = stringSet(coverage.projectedSourceKeys)
  const unsupported = stringSet(coverage.explicitlyUnsupportedSourceKeys)
  if (eligible.length !== coverage.eligibleSelected.length || projected.length !== coverage.projectedSourceKeys.length
    || unsupported.length !== coverage.explicitlyUnsupportedSourceKeys.length || projected.some(key => unsupported.includes(key))
    || canonicalNarrativeJsonV1(eligible) !== canonicalNarrativeJsonV1(stringSet([...projected, ...unsupported]))
    || canonicalNarrativeJsonV1(projected) !== canonicalNarrativeJsonV1(stringSet(projection.bundle.records.map(record => record.provenance.sourceResourceKey)))
    || coverage.projectedAtoms !== projection.bundle.records.length) narrativeFail('index-coverage', '索引输入的来源覆盖分区不完整')
}

function atomText(record: NarrativeRecordV1): string {
  switch (record.kind) {
    case 'entity': return [record.payload.canonicalLabel, ...record.payload.aliases].join('\n')
    case 'assertion': return canonicalNarrativeJsonV1({ predicate: record.payload.predicateKey, value: record.payload.value })
    case 'relation': return canonicalNarrativeJsonV1({ type: record.payload.relationType, value: record.payload.value })
    case 'rule': return record.payload.statement
    case 'epistemic-event': return canonicalNarrativeJsonV1({ attitude: record.payload.attitude, value: record.payload.assertedValue })
    // Original prose is indexed separately at its evidence/permission spans.
    // It must not inherit public visibility from a partial annotation.
    case 'development': return `${record.payload.developmentType}\n${record.payload.state}`
    case 'event': return record.payload.eventType
    case 'disclosure': return ''
  }
}


export interface NarrativeIndexReuseReceiptV1 {
  version: 1
  baseIndexHash: string | null
  rebuiltSources: string[]
  reusedSources: string[]
  rebuiltAtomPostings: string[]
  reusedAtomPostings: string[]
  /** IR verification and global ID remapping/graph assembly still visit all
   * records. Only original chunks/postings and atom lexical shards are reused. */
  normalization: 'full-verification'
  assembly: 'global-deterministic'
}
type SourceShard = { signature: string; chunks: NarrativeIndexV1['indexes']['chunks']; postings: NarrativeIndexV1['indexes']['sourcePostings'] }
type IndexShards = { indexHash: string; recordKeys: string[]; sources: Map<string, SourceShard>; atoms: Map<string, { text: string; terms: string[] }> }
// Compiler-created private snapshots only. A re-hashed external cache cannot
// become trusted incremental input. A cold restore first verifies the base.
const compiledShards = new WeakMap<NarrativeIndexV1, IndexShards>()
const copy = <T>(value: T): T => JSON.parse(JSON.stringify(value)) as T

export async function compileNarrativeIndexV1(input: {
  inputRootHash: string; projection: NarrativeDomainProjectionV1; predicates?: readonly FactPredicateSpec[]
  /** Runtime-only allocation policy; never part of recipe bytes. */
  retainIncrementalShards?: boolean
}): Promise<NarrativeIndexV1> {
  return (await compileWithShards(input)).index
}
export async function compileNarrativeIndexIncrementalV1(input: Parameters<typeof compileNarrativeIndexV1>[0], baseIndex: NarrativeIndexV1) {
  const prior = compiledShards.get(baseIndex)
  if (!prior) narrativeFail('index-incremental-base', '增量基线须先按冻结配方验证，不能信任外部缓存')
  return compileWithShards(input, prior)
}

/** Pure recipe used by the versioned release compiler. inputRootHash must come
 * from its verified semantic payload; this function neither creates a release
 * nor authorizes a read. The verifier recompiles against that same payload. */
async function compileWithShards(input: {
  inputRootHash: string
  projection: NarrativeDomainProjectionV1
  predicates?: readonly FactPredicateSpec[]
  retainIncrementalShards?: boolean
}, prior?: IndexShards): Promise<{ index: NarrativeIndexV1; receipt: NarrativeIndexReuseReceiptV1 }> {
  hashV1(input.inputRootHash, 'inputRootHash')
  const { projection } = input
  const retainShards = !!prior || input.retainIncrementalShards !== false
  const bundle = await verifyNarrativeIRBundleV1(projection.bundle, projection.sources, input.predicates)
  verifyCoverage(projection)
  const records = [...bundle.records].sort((a, b) => compareUnicodeCodePointsV1(a.semanticId, b.semanticId))
  const next: IndexShards | null = retainShards
    ? { indexHash: '', recordKeys: records.map(record => record.semanticId), sources: new Map(), atoms: new Map() } : null
  const receipt: NarrativeIndexReuseReceiptV1 = { version: 1, baseIndexHash: prior?.indexHash ?? null, rebuiltSources: [], reusedSources: [],
    rebuiltAtomPostings: [], reusedAtomPostings: [], normalization: 'full-verification', assembly: 'global-deterministic' }
  const positions = new Map(records.map((record, index) => [record.semanticId, index]))
  const at = (ref: Pick<SemanticRefV1, 'semanticId'>) => positions.get(ref.semanticId) ?? narrativeFail('index-reference', '索引引用未在同一输入包解析')
  const dependencyProofs = new Map<string, string[]>()
  const supportingEvidence = new Set(bundle.evidence.filter(span => span.supportRole === 'supports').map(span => span.evidenceId))
  const dependencyKey = (from: string, to: string, role: string) => compilerTupleKey(from, to, role)
  for (const proof of projection.dependencyEvidence) {
    const record = records[at({ semanticId: proof.fromSemanticId })]
    at({ semanticId: proof.toSemanticId })
    const key = dependencyKey(proof.fromSemanticId, proof.toSemanticId, proof.role)
    if (dependencyProofs.has(key) || !record.dependencies.some(item => item.semanticId === proof.toSemanticId && item.role === proof.role)
      || !proof.evidenceIds.length || proof.evidenceIds.some(id => !record.evidenceIds.includes(id)
        || !supportingEvidence.has(id))) {
      narrativeFail('index-dependency-evidence', '依赖边缺少同一原子中的有效支持证据')
    }
    dependencyProofs.set(key, stringSet(proof.evidenceIds))
  }
  const adjacency: NarrativeIndexV1['indexes']['adjacency'] = []
  const dependencies: NarrativeIndexV1['indexes']['dependencies'] = []
  const bySource = new Map<string, number[]>(), byEntity = new Map<number, Set<number>>()
  const atomPostings = new Map<string, number[]>()
  const edge = (fromIndex: number, toIndex: number, kind: NarrativeIndexEdgeKindV1, evidenceIds: string[], role: string | null = null,
    basis: 'declared' | 'structural' = 'structural') => {
    adjacency.push({ fromIndex, toIndex, kind, role, evidenceIds: stringSet(evidenceIds), basis })
  }
  const epistemic: NarrativeIndexV1['indexes']['epistemic'] = []
  const disclosures: NarrativeIndexV1['indexes']['disclosures'] = []
  const developments: NarrativeIndexV1['indexes']['developments'] = []
  const evidenceConsumers = new Map<string, Set<number>>()
  const sourceKey = (source: { sourceResourceKey: string; sourceRevision: string; fieldPointer: string }) =>
    compilerTupleKey(source.sourceResourceKey, source.sourceRevision, source.fieldPointer)
  const provenanceBySource = new Map<string, number[]>()
  const evidenceBySource = new Map<string, typeof bundle.evidence>()
  for (const span of bundle.evidence) {
    const key = sourceKey(span), spans = evidenceBySource.get(key) ?? []
    spans.push(span); evidenceBySource.set(key, spans)
  }
  records.forEach((record, index) => {
    const source = bySource.get(record.provenance.sourceResourceKey) ?? []
    source.push(index); bySource.set(record.provenance.sourceResourceKey, source)
    const provenanceKey = sourceKey({ ...record.provenance, fieldPointer: record.provenance.sourceField })
    const provenance = provenanceBySource.get(provenanceKey) ?? []
    provenance.push(index); provenanceBySource.set(provenanceKey, provenance)
    for (const evidenceId of record.evidenceIds) {
      const consumers = evidenceConsumers.get(evidenceId) ?? new Set<number>()
      consumers.add(index); evidenceConsumers.set(evidenceId, consumers)
    }
    const required = new Set<number>()
    forEachNarrativeReferenceV1(record, ref => {
      const target = at(ref)
      required.add(target)
      if (ref.kind === 'entity') {
        const related = byEntity.get(target) ?? new Set<number>()
        related.add(index); byEntity.set(target, related)
      }
    })
    for (const dependency of record.dependencies) {
      const target = at(dependency)
      required.add(target)
      const support = dependencyProofs.get(dependencyKey(record.semanticId, dependency.semanticId, dependency.role)) ?? []
      if (!support.length && dependency.role !== 'requires') narrativeFail('index-dependency-evidence', '声明依赖不能在索引中丢失支持证据')
      const inbound = ['supports', 'contradicts', 'causes'].includes(dependency.role)
      edge(inbound ? target : index, inbound ? index : target, dependency.role, support, null, support.length ? 'declared' : 'structural')
    }
    dependencies.push({ recordIndex: index, requiredIndexes: numberSet([...required].filter(target => target !== index)) })
    const text = atomText(record), previous = prior?.atoms.get(record.semanticId)
    const uniqueTerms = new Set<string>()
    if (previous?.text !== text) forEachNarrativeLexicalTermV1(text, term => uniqueTerms.add(term))
    const terms = previous?.text === text ? [...previous.terms] : [...uniqueTerms].sort(compareUnicodeCodePointsV1)
    receipt[previous?.text === text ? 'reusedAtomPostings' : 'rebuiltAtomPostings'].push(record.semanticId)
    next?.atoms.set(record.semanticId, { text, terms: [...terms] })
    for (const term of terms) {
      const posting = atomPostings.get(term) ?? []
      insertPostingIndex(posting, index); atomPostings.set(term, posting)
    }
    const link = (ref: SemanticRefV1, kind: NarrativeIndexEdgeKindV1, role: string | null = null) => edge(index, at(ref), kind, record.evidenceIds, role)
    switch (record.kind) {
      case 'entity': {
        const related = byEntity.get(index) ?? new Set<number>(); related.add(index); byEntity.set(index, related); break
      }
      case 'event':
        record.payload.participants.forEach(item => link(item.entityRef, 'participant', item.role))
        record.payload.locationRefs.forEach(ref => link(ref, 'location'))
        record.payload.propositionRefs.forEach(ref => link(ref, 'proposition'))
        record.payload.effectRefs.forEach(ref => link(ref, 'effect'))
        if (record.payload.communication) {
          link(record.payload.communication.senderRef, 'sender')
          record.payload.communication.recipientRefs.forEach(ref => link(ref, 'recipient'))
        }
        if (record.payload.observerRef) link(record.payload.observerRef, 'observer')
        break
      case 'assertion': link(record.payload.subjectRef, 'subject'); record.payload.supersedesRefs.forEach(ref => link(ref, 'supersedes')); break
      case 'relation':
        link(record.payload.fromRef, 'relation', 'from'); link(record.payload.toRef, 'relation', 'to')
        if (record.payload.causeEventRef) edge(at(record.payload.causeEventRef), index, 'causes', record.evidenceIds, null, 'declared')
        break
      case 'rule': record.payload.scopeRefs.forEach(ref => link(ref, 'rule-scope')); record.payload.preconditionRefs.forEach(ref => link(ref, 'precondition')); break
      case 'epistemic-event':
        link(record.payload.holderRef, 'belief-holder'); link(record.payload.propositionRef, 'belief-proposition')
        if (record.payload.sourceEventRef) link(record.payload.sourceEventRef, 'knowledge-source')
        if (record.payload.revisesRef) link(record.payload.revisesRef, 'revision')
        epistemic.push({ recordIndex: index, holderIndex: at(record.payload.holderRef), propositionIndex: at(record.payload.propositionRef),
          sourceEventIndex: record.payload.sourceEventRef ? at(record.payload.sourceEventRef) : null,
          attitude: record.payload.attitude, transition: record.payload.transition, acquiredAt: record.time.acquiredAt })
        break
      case 'development':
        record.payload.ownerRefs.forEach(ref => link(ref, 'owner')); record.payload.memberRefs.forEach(ref => link(ref, 'member'))
        record.payload.transitionEventRefs.forEach(ref => link(ref, 'transition'))
        developments.push({ recordIndex: index, type: record.payload.developmentType, state: record.payload.state,
          ownerIndexes: numberSet(record.payload.ownerRefs.map(at)), memberIndexes: numberSet(record.payload.memberRefs.map(at)),
          transitionEventIndexes: numberSet(record.payload.transitionEventRefs.map(at)), modality: record.modality })
        break
      case 'disclosure':
        record.payload.subjectRefs.forEach(ref => link(ref, 'discloses'))
        disclosures.push({ recordIndex: index, subjectIndexes: numberSet(record.payload.subjectRefs.map(at)), audienceKeys: stringSet(record.payload.audienceKeys),
          sequenceKey: record.payload.sequenceKey, unitKey: record.payload.unitKey, evidenceId: record.payload.spanRef })
    }
  })
  for (const before of bundle.timeContext.before) {
    edge(at({ semanticId: before.beforeEventKey }), at({ semanticId: before.afterEventKey }), 'before', [])
  }
  const allChunks: NarrativeIndexV1['indexes']['chunks'] = []
  const mergedSourcePostings = new Map<string, Map<string, { visibility: VisibilityV1; indexes: number[] }>>()
  const sources = projection.sources.map(source => ({ source,
    sortKey: compilerTupleKey(source.sourceResourceKey, source.fieldPointer, source.sourceRevision) }))
    .sort((a, b) => compareUnicodeCodePointsV1(a.sortKey, b.sortKey)).map(item => item.source)
  // Policies below are freshly constructed by this invocation (or copied
  // from a private shard), never mutable caller objects. A source range may
  // share one policy across thousands of lexical terms; encode it once.
  const policyKeys = new WeakMap<VisibilityV1, string>()
  const policyKey = (visibility: VisibilityV1) => {
    let value = policyKeys.get(visibility)
    if (value === undefined) { value = canonicalNarrativeJsonV1(visibility); policyKeys.set(visibility, value) }
    return value
  }
  const mergeShard = (shard: SourceShard) => {
    const offset = allChunks.length
    // Stored capabilities own a separate copy; these fresh/remapped chunks
    // can become the public result directly without another JSON round trip.
    allChunks.push(...shard.chunks)
    for (const item of shard.postings) {
      const posting = mergedSourcePostings.get(item.term) ?? new Map<string, { visibility: VisibilityV1; indexes: number[] }>()
      for (const original of item.groups) {
        const key = policyKey(original.visibility), group = posting.get(key) ?? { visibility: copy(original.visibility), indexes: [] as number[] }
        original.chunkIndexes.forEach(index => insertPostingIndex(group.indexes, index + offset)); posting.set(key, group)
      }
      mergedSourcePostings.set(item.term, posting)
    }
  }
  for (const source of sources) {
    // Source-local offsets make unchanged shards portable across insertions,
    // deletions and reorderings of the global sorted arrays.
    const chunks: NarrativeIndexV1['indexes']['chunks'] = []
    // A non-incremental read has no use for a second source-local posting
    // graph. Assemble its global offsets directly; retained/reused shards
    // still use local coordinates and their private copies below.
    const sourcePostings = retainShards
      ? new Map<string, Map<string, { visibility: VisibilityV1; indexes: number[] }>>() : mergedSourcePostings
    const postingOffset = retainShards ? 0 : allChunks.length
    const bytes = new TextEncoder().encode(source.text)
    const sourceContentHash = await hashNarrativeBytesV1(source.text)
    const spans = evidenceBySource.get(sourceKey(source)) ?? []
    const provenanceConsumers = provenanceBySource.get(sourceKey(source)) ?? []
    const shardKey = compilerTupleKey(source.sourceResourceKey, source.fieldPointer, source.sourceRevision)
    const consumers = numberSet([...provenanceConsumers, ...spans.flatMap(span => [...(evidenceConsumers.get(span.evidenceId) ?? [])])])
    const signature = retainShards ? await hashNarrativeValueV1({ sourceContentHash, spans,
      consumers: consumers.map(index => ({ semanticId: records[index].semanticId, visibility: records[index].visibility })),
      evidenceConsumers: spans.map(span => [span.evidenceId, stringSet([...(evidenceConsumers.get(span.evidenceId) ?? [])].map(index => records[index].semanticId))]),
      provenanceConsumers: provenanceConsumers.map(index => records[index].semanticId) }) : ''
    const previous = prior?.sources.get(shardKey)
    if (previous?.signature === signature) {
      const shard = copy(previous)
      shard.chunks.forEach(chunk => { chunk.recordIndexes = numberSet(chunk.recordIndexes.map(index =>
        positions.get(prior!.recordKeys[index]) ?? narrativeFail('index-incremental-reference', '复用分片的引用已丢失'))) })
      next!.sources.set(shardKey, copy(shard)); mergeShard(shard); receipt.reusedSources.push(shardKey)
      continue
    }
    receipt.rebuiltSources.push(shardKey)
    const boundaries = numberSet([0, bytes.length, ...spans.flatMap(span => [span.start, span.end])])
    const first = chunks.length
    for (let b = 0; b + 1 < boundaries.length; b++) {
      let start = boundaries[b]
      while (start < boundaries[b + 1]) {
        let end = Math.min(start + 1024, boundaries[b + 1])
        while (end < bytes.length && (bytes[end] & 0xc0) === 0x80) end--
        if (end <= start) narrativeFail('index-span', '索引分块不能拆开 UTF-8 字符')
        const covering = spans.filter(span => span.start <= start && span.end >= end)
        const proofConsumers = numberSet(covering.flatMap(span => [...(evidenceConsumers.get(span.evidenceId) ?? [])]))
        const recordIndexes = numberSet([...proofConsumers, ...provenanceConsumers])
        const visibility = intersectNarrativeVisibilityV1(proofConsumers.length ? PUBLIC_POLICY : UNKNOWN_POLICY,
          recordIndexes.map(index => records[index].visibility))
        chunks.push({ sourceResourceKey: source.sourceResourceKey, sourceRevision: source.sourceRevision, fieldPointer: source.fieldPointer,
          sourceContentHash, start, end, spanHash: await hashNarrativeBytesV1(new TextDecoder('utf-8', { fatal: true }).decode(bytes.slice(start, end))),
          recordIndexes, evidenceIds: stringSet(covering.map(span => span.evidenceId)), visibility })
        start = end
      }
    }
    // Postings record membership, not term frequency. Collect distinct terms
    // only while their complete overlapping chunk interval stays identical.
    // This bounds temporary terms to one chunk interval, while cross-boundary
    // terms still inherit every touched chunk's policy and global offset.
    let low = first, rangeStart = first, rangeEnd = first
    const rangeTerms = new Set<string>()
    const flushTerms = () => {
      if (!rangeTerms.size) return
      const policies: VisibilityV1[] = []
      for (let c = rangeStart; c < rangeEnd; c++) policies.push(chunks[c].visibility)
      const visibility = intersectNarrativeVisibilityV1(PUBLIC_POLICY, policies)
      const visibilityKey = policyKey(visibility)
      for (const term of rangeTerms) {
        let posting = sourcePostings.get(term)
        if (!posting) { posting = new Map(); sourcePostings.set(term, posting) }
        let group = posting.get(visibilityKey)
        if (!group) {
          group = { visibility: retainShards ? visibility : copy(visibility), indexes: [] as number[] }
          posting.set(visibilityKey, group)
        }
        for (let c = rangeStart; c < rangeEnd; c++) insertPostingIndex(group.indexes, c + postingOffset)
      }
      rangeTerms.clear()
    }
    forEachNarrativeLexicalTermV1(source.text, (term, start, end) => {
      while (low < chunks.length && chunks[low].end <= start) low++
      let after = low
      while (after < chunks.length && chunks[after].start < end) after++
      if (rangeStart !== low || rangeEnd !== after) {
        flushTerms(); rangeStart = low; rangeEnd = after
      }
      rangeTerms.add(term)
    })
    flushTerms()
    if (next) {
      const shard: SourceShard = { signature, chunks, postings: [...sourcePostings.entries()].map(([term, groups]) => ({ term,
        groups: [...groups.values()].map(group => ({ visibility: group.visibility, chunkIndexes: group.indexes })) })) }
      next.sources.set(shardKey, copy(shard))
      mergeShard(shard)
    } else allChunks.push(...chunks)
  }
  const unknownPoint = (point: NonNullable<NarrativeRecordV1['time']['acquiredAt']>) => point.kind === 'unknown'
    || (point.kind === 'event-anchor' && (point.side !== 'at' || bundle.timeContext.anchors.find(anchor => anchor.eventKey === point.eventKey)?.value == null))
  const unknown = (axis: 'occurred' | 'valid' | 'acquired' | 'narrated' | 'disclosed') => records.flatMap((record, index) => {
    const value = axis === 'acquired' ? record.time.acquiredAt : axis === 'narrated' ? record.time.narratedAt
      : axis === 'disclosed' ? record.time.disclosedAt : record.time[axis]
    const absent = value === null || (Array.isArray(value) && value.length === 0)
      || (typeof value === 'object' && value !== null && ('kind' in value ? unknownPoint(value)
        : 'start' in value ? unknownPoint(value.start) || value.endMeaning === 'unknown' || (value.end !== null && unknownPoint(value.end)) : false))
    return absent ? [index] : []
  })
  const uniqueEdges = [...new Map(adjacency.map(item => [compilerEdgeKey(item), item])).entries()]
    .sort(([a], [b]) => compareUnicodeCodePointsV1(a, b)).map(([, value]) => value)
  const body: Omit<NarrativeIndexV1, 'indexHash'> = {
    schema: 'storyforge.narrative-index', version: 1, inputRootHash: input.inputRootHash, recipeVersion: NARRATIVE_INDEX_RECIPE_V1,
    indexes: { recordKeys: records.map(record => record.semanticId),
      byKind: NARRATIVE_KINDS_V1.map(kind => ({ kind, recordIndexes: records.flatMap((record, index) => record.kind === kind ? [index] : []) })),
      bySource: [...bySource.entries()].sort(([a], [b]) => compareUnicodeCodePointsV1(a, b)).map(([sourceResourceKey, recordIndexes]) => ({ sourceResourceKey, recordIndexes })),
      byEntity: [...byEntity.entries()].sort(([a], [b]) => a - b).map(([entityIndex, indexes]) => ({ entityIndex, recordIndexes: numberSet(indexes) })),
      adjacency: uniqueEdges, dependencies, time: records.map((record, recordIndex) => ({ recordIndex, time: record.time })),
      epistemic, disclosures, developments, chunks: allChunks,
      sourcePostings: [...mergedSourcePostings.entries()].sort(([a], [b]) => compareUnicodeCodePointsV1(a, b)).map(([term, groups]) => ({ term,
        groups: [...groups.entries()].sort(([a], [b]) => compareUnicodeCodePointsV1(a, b)).map(([, group]) => ({ visibility: group.visibility, chunkIndexes: group.indexes })),
      })),
      atomPostings: [...atomPostings.entries()].sort(([a], [b]) => compareUnicodeCodePointsV1(a, b)).map(([term, values]) => ({ term, recordIndexes: values })),
      unknownTime: (['occurred', 'valid', 'acquired', 'narrated', 'disclosed'] as const).map(axis => ({ axis, recordIndexes: unknown(axis) })),
      intentIndexes: records.flatMap((record, index) => record.modality === 'intent' ? [index] : []),
      coverageHash: await hashNarrativeValueV1(projection.coverage),
    },
  }
  // The completed arrays own their contents. Drop temporary lookup/posting
  // containers before hashing the large output; async compiler closures must
  // not retain their Set/Map graphs alongside the final serialized index.
  // Private incremental shards in next own separate copies and remain intact.
  positions.clear(); dependencyProofs.clear(); supportingEvidence.clear()
  bySource.clear(); byEntity.clear(); atomPostings.clear(); evidenceConsumers.clear()
  provenanceBySource.clear(); evidenceBySource.clear(); mergedSourcePostings.clear()
  const index = { ...body, indexHash: await hashNarrativeTextPartsV1(compiledIndexParts(body, policyKey)) }
  if (next) { next.indexHash = index.indexHash; compiledShards.set(index, next) }
  return { index, receipt }

}

/** This writer is private to the assembler above, not an imported-index codec.
 * Every traversed row/list is newly assembled here or copied from our private
 * shard. Scalars are checked; nested IR time and policy use the strict codec.
 * It never visits caller objects as plain JSON and never freezes the returned
 * index. Mapped field writers require every schema field at compile time.
 * External verification below still recompiles and strictly compares all data. */
type IndexParts<T> = (value: T) => Iterable<string>
type IndexFields<T> = { [K in keyof T]-?: IndexParts<T[K]> }
type Indexes = NarrativeIndexV1['indexes']
type IndexRow<K extends keyof Indexes> = Indexes[K] extends Array<infer Row> ? Row : never
function compiledObjectParts<T extends object>(fields: IndexFields<T>): IndexParts<T> {
  const keys = (Object.keys(fields) as Array<keyof T & string>).sort(compareUnicodeCodePointsV1)
  const tokens = keys.map(key => `${JSON.stringify(key)}:`)
  return function* (value) {
    yield '{'
    for (let index = 0; index < keys.length; index++) {
      if (index) yield ','
      const key = keys[index]
      yield tokens[index]
      yield* fields[key](value[key])
    }
    yield '}'
  }
}
function compiledListParts<T>(item: IndexParts<T>): IndexParts<T[]> {
  return function* (values) {
    yield '['
    for (let index = 0; index < values.length; index++) {
      if (index) yield ','
      yield* item(values[index])
    }
    yield ']'
  }
}
function* compiledStringParts(value: string): Iterable<string> {
  if (typeof value !== 'string') narrativeFail('json', '索引字符串字段无效')
  assertUnicodeV1(value)
  yield JSON.stringify(value)
}
function indexNumber(value: number): string {
  if (!Number.isSafeInteger(value) || value < 0 || Object.is(value, -0)) narrativeFail('json', '索引数字必须为安全非负整数')
  return String(value)
}
function* compiledNumberParts(value: number): Iterable<string> { yield indexNumber(value) }
function* compiledNumberListParts(values: number[]): Iterable<string> {
  yield '['
  // Members come only from compiler positions/boundaries. A bounded scalar
  // copy avoids both descriptor objects per member and a full index encoding.
  for (let start = 0; start < values.length; start += 256) {
    if (start) yield ','
    const tokens: string[] = []
    for (let end = Math.min(start + 256, values.length), at = start; at < end; at++) tokens.push(indexNumber(values[at]))
    yield tokens.join(',')
  }
  yield ']'
}
function nullableIndexParts<T>(parts: IndexParts<T>): IndexParts<T | null> {
  return function* (value) { if (value === null) yield 'null'; else yield* parts(value) }
}
function* strictIndexParts(value: unknown): Iterable<string> { yield canonicalNarrativeJsonV1(value) }
function compiledIndexParts(body: Omit<NarrativeIndexV1, 'indexHash'>, policyKey: (value: VisibilityV1) => string): Iterable<string> {
  const strings = compiledListParts(compiledStringParts), numbers = compiledNumberListParts
  const text = compiledStringParts, number = compiledNumberParts, rows = compiledListParts
  const policy: IndexParts<VisibilityV1> = function* (value) { yield policyKey(value) }
  const indexes = compiledObjectParts<Indexes>({
    recordKeys: strings,
    byKind: rows(compiledObjectParts<IndexRow<'byKind'>>({ kind: text, recordIndexes: numbers })),
    bySource: rows(compiledObjectParts<IndexRow<'bySource'>>({ sourceResourceKey: text, recordIndexes: numbers })),
    byEntity: rows(compiledObjectParts<IndexRow<'byEntity'>>({ entityIndex: number, recordIndexes: numbers })),
    adjacency: rows(compiledObjectParts<IndexRow<'adjacency'>>({
      fromIndex: number, toIndex: number, kind: text, role: nullableIndexParts(text), evidenceIds: strings, basis: text,
    })),
    dependencies: rows(compiledObjectParts<IndexRow<'dependencies'>>({ recordIndex: number, requiredIndexes: numbers })),
    time: rows(compiledObjectParts<IndexRow<'time'>>({ recordIndex: number, time: strictIndexParts })),
    epistemic: rows(compiledObjectParts<IndexRow<'epistemic'>>({
      recordIndex: number, holderIndex: number, propositionIndex: number, sourceEventIndex: nullableIndexParts(number),
      attitude: text, transition: text, acquiredAt: strictIndexParts,
    })),
    disclosures: rows(compiledObjectParts<IndexRow<'disclosures'>>({
      recordIndex: number, subjectIndexes: numbers, audienceKeys: strings, sequenceKey: text, unitKey: text, evidenceId: text,
    })),
    developments: rows(compiledObjectParts<IndexRow<'developments'>>({
      recordIndex: number, type: text, state: text, ownerIndexes: numbers, memberIndexes: numbers, transitionEventIndexes: numbers, modality: text,
    })),
    chunks: rows(compiledObjectParts<IndexRow<'chunks'>>({
      sourceResourceKey: text, sourceRevision: text, fieldPointer: text, sourceContentHash: text, start: number, end: number,
      spanHash: text, recordIndexes: numbers, evidenceIds: strings, visibility: policy,
    })),
    sourcePostings: rows(compiledObjectParts<IndexRow<'sourcePostings'>>({
      term: text, groups: rows(compiledObjectParts<IndexRow<'sourcePostings'>['groups'][number]>({ visibility: policy, chunkIndexes: numbers })),
    })),
    atomPostings: rows(compiledObjectParts<IndexRow<'atomPostings'>>({ term: text, recordIndexes: numbers })),
    unknownTime: rows(compiledObjectParts<IndexRow<'unknownTime'>>({ axis: text, recordIndexes: numbers })),
    intentIndexes: numbers, coverageHash: text,
  })
  return compiledObjectParts<Omit<NarrativeIndexV1, 'indexHash'>>({
    schema: text, version: number, inputRootHash: text, recipeVersion: text, indexes,
  })(body)
}

/** Rebuilding is the exact codec: unknown/extra fields, wrong recipes, corrupt
 * postings and evidence all fail comparison instead of being silently fixed. */
export async function verifyNarrativeIndexV1(value: unknown, input: Parameters<typeof compileNarrativeIndexV1>[0]): Promise<NarrativeIndexV1> {
  const expected = await compileNarrativeIndexV1(input)
  if (!equalNarrativeJsonV1(value, expected)) narrativeFail('index-integrity', '索引与冻结输入或登记编译规则不一致')
  return expected
}

export function resolveNarrativeIndexClosureV1(index: NarrativeIndexV1, selectedIndexes: readonly number[], maxNodes: number): number[] {
  if (!Number.isSafeInteger(maxNodes) || maxNodes <= 0) narrativeFail('index-budget', '依赖闭包预算必须是正整数')
  const selected = numberSet(selectedIndexes)
  if (selected.some(item => !Number.isSafeInteger(item) || item < 0 || item >= index.indexes.recordKeys.length)) narrativeFail('index-reference', '依赖闭包起点不存在')
  const required = new Map(index.indexes.dependencies.map(item => [item.recordIndex, item.requiredIndexes]))
  if (required.size !== index.indexes.recordKeys.length || index.indexes.dependencies.length !== required.size
    || [...required.entries()].some(([key, values]) => !Number.isSafeInteger(key) || key < 0 || key >= required.size
      || values.some(value => !Number.isSafeInteger(value) || value < 0 || value >= required.size))) {
    narrativeFail('index-reference', '依赖闭包表不完整或含有越界引用')
  }
  if (selected.length > maxNodes) narrativeFail('index-budget', '依赖闭包起点超过预算')
  const seen = new Set<number>(), pending = [...selected]
  const queued = new Set(selected)
  for (let cursor = 0; cursor < pending.length; cursor++) {
    const current = pending[cursor]
    if (seen.has(current)) continue
    seen.add(current)
    if (seen.size > maxNodes) narrativeFail('index-budget', '依赖闭包超过预算，不能静默丢弃必需来源')
    for (const target of required.get(current) ?? []) if (!queued.has(target)) {
      queued.add(target)
      if (queued.size > maxNodes) narrativeFail('index-budget', '依赖闭包超过预算，不能静默丢弃必需来源')
      pending.push(target)
    }
  }
  return numberSet(seen)
}
