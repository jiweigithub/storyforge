def metric(items, nested=True):
 scores=[r['score'] if nested else r for r in items];tp=sum(r['truePositive'] for r in scores);fp=sum(r['falsePositive'] for r in scores);fn=sum(r['falseNegative'] for r in scores)
 out={'cases':len(scores),'truePositive':tp,'falsePositive':fp,'falseNegative':fn,'precision':tp/(tp+fp) if tp+fp else None,'recall':tp/(tp+fn) if tp+fn else None}
 if nested:out|={'uncertainRecommendations':sum(r['uncertainRecommendations'] for r in scores),'unscoredPredictions':sum(len(r['unscoredPredictions']) for r in scores)}
 return out
