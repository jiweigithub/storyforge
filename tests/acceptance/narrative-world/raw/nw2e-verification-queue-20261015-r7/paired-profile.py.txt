from pathlib import Path
import datetime, hashlib, json, os, socket, subprocess, sys

entries = [
    ('before', 'r5', '100000', True),
    ('after', 'r7', '100000', True),
    ('after-million', 'r7', '1000000', False),
]
results = []
for label, generation, size, profile in entries:
    root = Path('/Users/qinyingying/.codex/worktrees/nw2e-verification-' + generation + '/storyforge')
    frozen = json.loads(Path('/tmp/nw2e-frozen-files-' + generation + '.json').read_text())
    def verify():
        for path, expected in frozen['files'].items():
            assert hashlib.sha256((root / path).read_bytes()).hexdigest() == expected, path
    verify()
    report = Path('/tmp/nw2e-r7-profile-' + label + '.json')
    assert not report.exists()
    with socket.socket() as sock:
        sock.bind(('127.0.0.1', 0))
        port = str(sock.getsockname()[1])
    env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1', PLAYWRIGHT_PORT=port,
               PLAYWRIGHT_HTML_OUTPUT_DIR='/tmp/nw2e-r7-profile-' + label + '-html',
               NW_SCALE_BROWSER_MODE='diagnostic', NW_SCALE_BROWSER_SIZES=size,
               NW_SCALE_BROWSER_REPORT=str(report),
               NW_SCALE_BROWSER_SOURCES='/tmp/nw2e-r7-profile-' + label + '-sources.json')
    if profile:
        env['NW_SCALE_BROWSER_PROFILE_DIR'] = '/tmp/nw2e-r7-profile-' + label + '-cpu'
    print('Starting ' + label + ' diagnostic, ' + size + ' characters, isolated port ' + port, flush=True)
    code = subprocess.call(['npx', 'playwright', 'test', 'tests/e2e/narrative-scale.spec.ts', '--reporter=line',
                            '--retries=0', '--output=/tmp/nw2e-r7-profile-' + label + '-test-results'], cwd=root, env=env)
    verify()
    results.append({'label': label, 'generation': generation, 'size': int(size), 'profiled': profile,
                    'exitCode': code, 'report': str(report), 'sourceUnchanged': True})
    if code:
        break
terminal = Path('/tmp/nw2e-r7-paired-profile-terminal.json')
terminal.write_text(json.dumps({'schema': 'storyforge.paired-profile-terminal', 'version': 1,
    'endedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'runs': results,
    'accepted': False, 'boundary': 'Serialized single diagnostics; no formal repeated-performance acceptance.'}, indent=2) + '\n')
sys.exit(results[-1]['exitCode'])
