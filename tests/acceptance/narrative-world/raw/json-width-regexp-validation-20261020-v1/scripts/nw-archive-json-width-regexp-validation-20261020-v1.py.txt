from pathlib import Path
import datetime,gzip,hashlib,json,re,shutil,subprocess,sys
ROOT=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
PREFIX=Path('/tmp/nw-json-width-regexp-validation-20261020-v1')
BASE=ROOT/'tests/acceptance/narrative-world'
RAW=BASE/'raw/json-width-regexp-validation-20261020-v1'
RECEIPT=BASE/'json-width-regexp-validation-result-20261020-v1.json'
def digest(path):
    h=hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda:f.read(1024*1024),b''):h.update(chunk)
    return h.hexdigest()
def main():
    assert sys.argv[1:]==['--owned-session-closed']
    terminal=json.loads(Path(str(PREFIX)+'-terminal.json').read_text())
    assert terminal.get('endedAt') and isinstance(terminal.get('exitCode'),int) and terminal['sourceUnchanged']
    pin_path=Path(str(PREFIX)+'-sources.json');pin=json.loads(pin_path.read_text())
    assert digest(pin_path)==terminal['pinSha256']
    for ref,key in [('HEAD','head'),('MERGE_HEAD','mergeHead')]:
        assert subprocess.check_output(['git','-C',str(ROOT),'rev-parse',ref],text=True).strip()==pin[key]
    changed=[p for p,h in pin['files'].items() if not (ROOT/p).is_file() or digest(ROOT/p)!=h]
    assert not changed,changed
    assert not RECEIPT.exists()
    pairs=[]
    for p in sorted(PREFIX.parent.glob(PREFIX.name+'*')):
        if p.is_file() and p!=pin_path and p.suffix in ('.json','.log'):pairs.append((p,Path(p.name)))
        elif p.is_dir() and p.name.endswith(('-test-results','-cpu','-heap')):
            pairs.extend((c,Path(p.name)/c.relative_to(p)) for c in sorted(p.rglob('*')) if c.is_file())
    scripts=[Path(__file__),Path(str(PREFIX)+'.py')]
    supervisor=Path(str(PREFIX)+'-queue-supervisor-r1.py')
    if supervisor.exists():
        queue_terminal=json.loads(Path(str(PREFIX)+'-queue-r1-terminal.json').read_text())
        assert queue_terminal.get('endedAt') and queue_terminal['runnerTerminalPresent']
        assert queue_terminal['pinSha256']==terminal['pinSha256']
        assert queue_terminal['exitCode']==terminal['exitCode']
        scripts.append(supervisor)
    for script in scripts:pairs.append((script,Path('scripts')/(script.name+'.txt')))
    assert shutil.disk_usage(ROOT).free>sum(p.stat().st_size for p,_ in pairs)+5*2**30
    RAW.mkdir(parents=True,exist_ok=True);artifacts=[]
    for source,relative in pairs:
        target=RAW/relative;target.parent.mkdir(parents=True,exist_ok=True)
        if not target.exists():
            with source.open('rb') as src,target.open('xb') as dst:shutil.copyfileobj(src,dst)
        assert digest(source)==digest(target)
        artifacts.append({'path':str(target.relative_to(BASE)),'bytes':target.stat().st_size,'sha256':digest(target)})
    packed=RAW/(pin_path.name+'.gz')
    if not packed.exists():
        with packed.open('xb') as f:f.write(gzip.compress(pin_path.read_bytes(),mtime=0))
    assert hashlib.sha256(gzip.decompress(packed.read_bytes())).hexdigest()==digest(pin_path)
    artifacts.append({'path':str(packed.relative_to(BASE)),'bytes':packed.stat().st_size,'sha256':digest(packed),'decompressedSha256':digest(pin_path)})
    for p,h in terminal.get('generatedReports',{}).items():
        assert digest(ROOT/p)==h, 'Generated report changed: '+p
    phases=[]
    for phase in terminal['phases']:
        log=Path(phase['log']);assert digest(log)==phase['logSha256']
        text=re.sub(r'\x1b\[[0-9;]*m','',log.read_text(errors='replace'))
        phases.append({**phase,'log':str((RAW/log.name).relative_to(BASE)),
            'summaryLines':[line.strip() for line in text.splitlines() if re.search(r'^\s*(Test Files|Tests|Duration|\d+ (passed|failed|skipped))\b',line)]})
    receipt={**terminal,'schema':'storyforge.narrative-json-width-regexp-validation-result',
        'archivedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),
        'head':pin['head'],'mergeHead':pin['mergeHead'],'phases':phases,'artifacts':artifacts}
    diagnostic=Path(str(PREFIX)+'-diagnostic.json')
    if diagnostic.exists():
        d=json.loads(diagnostic.read_text())
        receipt['diagnostic']={'path':str((RAW/diagnostic.name).relative_to(BASE)),
            'summaries':d['summaries'],'failures':d['failures'],'deviceMatches':d['deviceMatches'],
            'runs':[{k:r[k] for k in ('size','run','prepared','compilation','memory','export','open','queries') if k in r} for r in d['runs']]}
    with RECEIPT.open('x') as f:json.dump(receipt,f,ensure_ascii=False,indent=2);f.write('\n')
    print(json.dumps({'receipt':str(RECEIPT),'artifacts':len(artifacts),'exitCode':receipt['exitCode']}))
if __name__=='__main__':main()
