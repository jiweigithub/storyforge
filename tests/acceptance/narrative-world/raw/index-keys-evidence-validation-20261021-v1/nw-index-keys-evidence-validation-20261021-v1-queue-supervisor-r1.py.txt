from pathlib import Path
from datetime import datetime, timezone
import fcntl, hashlib, json, os, signal, subprocess, sys

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
PREFIX = Path('/tmp/nw-index-keys-evidence-validation-20261021-v1')
STATE = Path(str(PREFIX) + '-queue-r1-state.json')
RECEIPT = Path(str(PREFIX) + '-queue-r1-terminal.json')
LOG = Path(str(PREFIX) + '-queue-r1.log')
now = lambda: datetime.now(timezone.utc).isoformat()
digest = lambda path: hashlib.sha256(Path(path).read_bytes()).hexdigest()
def save(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2); stream.write('\n')

def main():
    signal.signal(signal.SIGHUP, signal.SIG_IGN)
    with Path(str(PREFIX) + '-queue-r1.lock').open('a') as owned_lock:
        fcntl.flock(owned_lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        assert not STATE.exists() and not RECEIPT.exists() and not LOG.exists()
        assert not Path(str(PREFIX) + '-terminal.json').exists()
        assert not Path(str(PREFIX) + '-diff-check.log').exists()
        pin_path = Path(str(PREFIX) + '-sources.json')
        pin = json.loads(pin_path.read_text())
        assert digest(pin_path) == sys.argv[1]
        assert digest(str(PREFIX) + '.py') == pin['runnerSha256']
        for path, sha in pin['executionFiles'].items(): assert digest(path) == sha, path
        for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
            assert subprocess.check_output(['git', '-C', str(ROOT), 'rev-parse', ref], text=True).strip() == pin[key]
        assert all(digest(ROOT / path) == sha for path, sha in pin['files'].items())
        command = [sys.executable, '/tmp/storyforge-serial-ci-20261013.py', '--', sys.executable, str(PREFIX) + '.py']
        record = {'schema': 'storyforge.narrative-durable-queue-attempt', 'version': 1, 'attempt': 1,
            'startedAt': now(), 'supervisorPid': os.getpid(), 'root': str(ROOT), 'runner': str(PREFIX) + '.py',
            'pinSha256': digest(pin_path), 'log': str(LOG), 'command': command,
            'productionChanged': False, 'actualModelCalls': 0, 'accepted': False}
        with LOG.open('x') as log:
            child = subprocess.Popen(command, cwd=ROOT, stdin=subprocess.DEVNULL, stdout=log,
                stderr=subprocess.STDOUT, start_new_session=True)
            record['queuePid'] = child.pid
            save(STATE, record)
            def forward(signum, frame):
                if child.poll() is None:
                    try: os.killpg(child.pid, signum)
                    except ProcessLookupError: pass
            signal.signal(signal.SIGTERM, forward); signal.signal(signal.SIGINT, forward)
            code = child.wait()
        record.update(endedAt=now(), exitCode=code, logSha256=digest(LOG),
            runnerTerminalPresent=Path(str(PREFIX) + '-terminal.json').exists())
        save(RECEIPT, record)

if __name__ == '__main__': main()
