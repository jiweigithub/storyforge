from pathlib import Path
import datetime, hashlib, json, os, shutil, subprocess

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
BASE = Path('/tmp/nw-index-keys-evidence-validation-20261021-v1')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
original = HANDOFF.read_bytes()
h = json.loads(original)
q = h['durableQueue']
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()
def save(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

assert h['currentRunner'] == str(BASE) + '.py'
assert h['sourceFrozen'] is True
assert h['preparedCurrentArchive']['sha256'] == sha(__file__)
pin_path = Path(h['currentPin'])
pin = json.loads(pin_path.read_text())
terminal_path = Path(h['currentTerminal'])
terminal = json.loads(terminal_path.read_text())
queue_terminal = json.loads(Path(q['terminal']).read_text())
assert terminal['exitCode'] == queue_terminal['exitCode'] == 0
assert terminal['sourceUnchanged'] and queue_terminal['runnerTerminalPresent']
assert terminal['productionChanged'] is False and terminal['actualModelCalls'] == 0
assert terminal['independentHumanReviews'] == 0 and terminal['performanceAcceptance'] is False
assert [phase['name'] for phase in terminal['phases']] == ['diff-check', 'check-docs']
assert sha(pin_path) == h['currentPinSha256'] == terminal['pinSha256']
assert sha(h['currentRunner']) == pin['runnerSha256']
assert sha(q['log']) == queue_terminal['logSha256']
for phase in terminal['phases']:
    assert phase['exitCode'] == 0 and sha(phase['log']) == phase['logSha256']
def verify_inputs():
    for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
        assert subprocess.check_output(['git', 'rev-parse', ref], cwd=ROOT, text=True).strip() == pin[key]
    for path, digest in pin['files'].items():
        assert sha(ROOT / path) == digest, path
    for group in ['runtimeFiles', 'executionFiles']:
        for path, digest in pin[group].items():
            assert sha(path) == digest, path
verify_inputs()
listed = subprocess.check_output(['git', 'ls-files', '-z', '--cached', '--others', '--exclude-standard'], cwd=ROOT).decode().split('\0')
actual = {name for name in listed if name and (ROOT / name).is_file() and not any(part.startswith('.env') for part in Path(name).parts)}
assert actual == set(pin['files']), {'added': sorted(actual - set(pin['files'])), 'missing': sorted(set(pin['files']) - actual)}

pids = set(terminal['observedPids']) | {queue_terminal['supervisorPid'], queue_terminal['queuePid'], terminal['runnerPid']}
pids.update(phase['childPid'] for phase in terminal['phases'])
observations = Path(str(BASE) + '-process-observations.jsonl')
if observations.exists():
    for line in observations.read_text().splitlines():
        pids.update(json.loads(line)['pids'])
ps = subprocess.run(['ps', '-p', ','.join(map(str, sorted(pids))), '-o', 'pid=,ppid=,state=,command='], capture_output=True, text=True)
assert ps.returncode == 1 and not ps.stdout.strip(), ps.stdout
rows = {int(parts[0]): parts for line in subprocess.check_output(['ps', '-axo', 'pid=,ppid=,command='], text=True).splitlines() if len(parts := line.strip().split(None, 2)) == 3}
ancestors = {os.getpid()}
pid = os.getpid()
while pid in rows and int(rows[pid][1]) not in ancestors:
    pid = int(rows[pid][1]); ancestors.add(pid)
scan = subprocess.run(['lsof', '-nP', '-d', 'cwd', '-Fpn'], capture_output=True, text=True, timeout=20)
assert scan.returncode == 0
cwd_matches = []
pid = None
for line in scan.stdout.splitlines():
    if line.startswith('p'): pid = int(line[1:])
    elif line.startswith('n'):
        path = Path(line[1:])
        if path == ROOT.resolve() or ROOT.resolve() in path.parents:
            if pid in rows and pid not in ancestors: cwd_matches.append(rows[pid])
assert not cwd_matches, cwd_matches
command_matches = [parts for pid, parts in rows.items() if pid not in ancestors and str(BASE) in parts[2]]
assert not command_matches, command_matches
exit_proof = {
    'schema': 'storyforge.narrative-validation-owned-exit', 'version': 1, 'checkedAt': now(),
    'pids': sorted(pids), 'allAbsent': True, 'sourceUnchanged': True,
    'sourceFiles': len(pin['files']), 'runtimeUnchanged': True, 'runtimeFiles': len(pin['runtimeFiles']),
    'head': pin['head'], 'mergeHead': pin['mergeHead'], 'pinSha256': sha(pin_path),
    'terminalSha256': sha(terminal_path), 'queueTerminalSha256': sha(q['terminal']),
    'workspaceProcessesRemaining': [],
    'limitation': 'All recorded wrappers, runner, phase children and observed descendants are absent. Cwd and command scans exclude this evidence writer and its ancestors. Unobserved transient processes are not individually enumerated.'}
save(str(BASE) + '-owned-exit.json', exit_proof)

dest = ROOT / 'tests/acceptance/narrative-world/raw/index-keys-evidence-validation-20261021-v1'
assert not dest.exists()
dest.mkdir()
artifacts = []
def archive(source, relative):
    target = dest / relative
    if source.suffix in ['.ts', '.tsx', '.js', '.mjs', '.cjs', '.py']: target = Path(str(target) + '.txt')
    target.parent.mkdir(parents=True, exist_ok=True)
    assert not target.exists()
    shutil.copy2(source, target)
    digest = sha(source)
    assert digest == sha(target)
    artifacts.append({'path': str(target.relative_to(ROOT)), 'sha256': digest, 'bytes': target.stat().st_size})
for source in sorted(BASE.parent.glob(BASE.name + '*')):
    if source.name.endswith(('.lock', '.tmp')) or any(token in source.name for token in ['-archived-handoff', '-heartbeat-']): continue
    assert source.is_file(), source
    archive(source, Path(source.name))
archive(Path(__file__), Path('archive-script.py'))
receipt = dict(terminal)
receipt.update(
    schema='storyforge.narrative-index-keys-evidence-validation-receipt', archivedAt=now(),
    artifacts=artifacts, exitProof=exit_proof,
    disposition='documentation-and-evidence-checks-passed',
    checksPassed=['git diff --check', 'npm run check:docs'],
    limitations=['Documentation and evidence validation only; no new performance measurement, candidate adoption, full CI, full E2E, model call, or human review.'],
    productionChanged=False, adopted=False, performanceAcceptance=False, formalAcceptance=False,
    goldOrThresholdChanges=False,
    next='Keep the private-key candidate external and unadopted. Continue evidence-supported canonical JSON traversal investigation, then declare and validate any independent candidate through the shared serial entry before production adoption. Remaining performance, quality and cross-product gates are still open.')
receipt_path = ROOT / 'tests/acceptance/narrative-world/index-keys-evidence-validation-result-20261021-v1.json'
save(receipt_path, receipt)
assert all(sha(ROOT / item['path']) == item['sha256'] for item in artifacts)
verify_inputs()
verification = {
    'schema': 'storyforge.narrative-archive-verification', 'version': 1, 'verifiedAt': now(),
    'receipt': str(receipt_path.relative_to(ROOT)), 'receiptSha256': sha(receipt_path),
    'artifactsVerified': len(artifacts), 'allMatch': True, 'sourceFilesUnchanged': len(pin['files']),
    'runtimeFilesUnchanged': len(pin['runtimeFiles']), 'knownPidsAbsent': sorted(pids),
    'productionChanged': False, 'performanceAcceptance': False, 'actualModelCalls': 0}
verification_path = ROOT / 'tests/acceptance/narrative-world/index-keys-evidence-archive-verification-20261021-v1.json'
save(verification_path, verification)
assert HANDOFF.read_bytes() == original
h.update(status='documentation-validation-completed-archived', sourceFrozen=False, updatedAt=now(),
    latestCompletedReceipt=str(receipt_path), latestCompletedReceiptSha256=sha(receipt_path),
    latestArchiveVerification=str(verification_path), latestArchiveVerificationSha256=sha(verification_path),
    lastCompletedSourcePinSha256=h['currentPinSha256'])
h['durableQueue'].update(status='completed-archived', completed=True, exitCode=0, endedAt=queue_terminal['endedAt'])
h['preparedCurrentArchive'].update(executed=True, artifacts=len(artifacts))
h['pendingDocumentationVerification'].update(status='passed-and-archived', receipt=str(receipt_path))
h['currentValidationObservation'] = {'at': now(), 'status': 'completed-archived', 'receipt': str(receipt_path),
    'sourceFilesUnchanged': len(pin['files']), 'knownPidsAbsent': sorted(pids), 'performanceAcceptance': False, 'resubmitted': False}
h['next'] = receipt['next']
stage = Path(str(BASE) + '-archived-handoff.tmp')
stage.write_text(json.dumps(h, ensure_ascii=False, indent=2) + '\n'); os.replace(stage, HANDOFF)
print(json.dumps({key: value for key, value in verification.items() if key != 'knownPidsAbsent'}, ensure_ascii=False))
print('knownPidsAbsent', len(pids))
