"""Reconcile an actually generated baseline without editing its measurements."""
from pathlib import Path
import hashlib
import json
import shutil
import sys


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def verify(root, manifest_path, previous_path, generated_path, target_path):
    root, manifest_path, previous_path, generated_path, target_path = map(
        Path, (root, manifest_path, previous_path, generated_path, target_path))
    manifest = json.loads(manifest_path.read_bytes())
    previous = json.loads(previous_path.read_bytes())
    generated = json.loads(generated_path.read_bytes())
    relative_manifest = str(manifest_path.relative_to(root))
    relative_target = str(target_path.relative_to(root))
    if manifest['baseline']['path'] != relative_target:
        raise ValueError('Final baseline reference was not declared before measurement')
    if relative_manifest not in manifest['sourceFingerprintPaths']:
        raise ValueError('Manifest self-fingerprint must remain in scope')
    observed = lambda value: {k: v for k, v in value.items() if k not in ('input', 'run')}
    if observed(previous) != observed(generated):
        raise ValueError('Observed behavior changed; preserve result for explicit review')
    expected = sorted([*manifest['sourceFingerprintPaths'], manifest['corpus']['path']])
    hashes = generated['input']['hashes']
    if sorted(hashes) != expected:
        raise ValueError('Measured fingerprint scope differs from the complete declared scope')
    for name, digest in hashes.items():
        if sha(root / name) != digest:
            raise ValueError('Measured input no longer matches: ' + name)
    encoded = json.dumps(hashes, ensure_ascii=False, separators=(',', ':')).encode('utf-8')
    if hashlib.sha256(encoded).hexdigest() != generated['input']['sha256']:
        raise ValueError('Baseline aggregate input digest differs')
    if sha(root / manifest['corpus']['path']) != manifest['corpus']['sha256']:
        raise ValueError('Original corpus changed')
    if target_path.exists():
        raise FileExistsError('Target exists; inspect prior outcome without overwrite')
    return {'observationsEqualPrevious': True, 'fingerprintCount': len(hashes),
            'manifestSelfFingerprintRetained': True, 'generatedSha256': sha(generated_path)}


def promote(root, manifest_path, previous_path, generated_path, target_path):
    result = verify(root, manifest_path, previous_path, generated_path, target_path)
    # Copy the original bytes, including measured inputs and actual run identity.
    with Path(target_path).open('xb') as target, Path(generated_path).open('rb') as source:
        shutil.copyfileobj(source, target)
    if sha(target_path) != result['generatedSha256']:
        raise ValueError('Copied baseline bytes differ')
    return result


if __name__ == '__main__':
    print(json.dumps(promote(*sys.argv[1:]), ensure_ascii=False))
