from pathlib import Path
import datetime, hashlib, json, os, subprocess, sys

B=Path(__file__).parent
Q=B/'archive-regression'
Q.mkdir(exist_ok=True)
sha=lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
now=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(name,data):
    p=Q/name;temp=p.with_suffix(p.suffix+'.tmp');temp.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n');os.replace(temp,p)
assert not (Q/'input-pin.json').exists() and not (Q/'terminal.json').exists(), 'Never dispatch twice'
paths=[B/'archive-delivery-validation-original.py',B/'archive-delivery-validation-candidate.py',
       B/'test_archive_fix.py',Path(__file__),Path('/tmp/storyforge-serial-ci-20261013.py')]
pin={'schema':'storyforge.nw-archive-regression-inputs','version':1,'createdAt':now(),
 'files':{str(p):sha(p) for p in paths},'workspaceFrozen':False,
 'scope':'Independent archiver regression fixtures only; no product/model/performance acceptance.',
 'automaticRetries':0,'command':[sys.executable,'/tmp/storyforge-serial-ci-20261013.py','--',sys.executable,str(B/'test_archive_fix.py')]}
save('input-pin.json',pin)
state={'startedAt':now(),'supervisorPid':os.getpid(),'status':'submitting-to-existing-serial-entry',
 'inputPinSha256':sha(Q/'input-pin.json'),'automaticRetries':0,'workspaceFrozen':False}
save('state.json',state)
exit_code=None
try:
    with (Q/'queue.log').open('x') as output:
        child=subprocess.Popen(pin['command'],cwd=B,stdout=output,stderr=subprocess.STDOUT)
        state.update(queuePid=child.pid,status='queued-or-running');save('state.json',state)
        exit_code=child.wait()
    state.update(exitCode=exit_code,status='passed' if exit_code==0 else 'failed')
except BaseException as error:
    state.update(status='orchestration-failed-result-needs-review',error=type(error).__name__+': '+str(error))
finally:
    state.update(endedAt=now(),inputReadback={str(p):sha(p) for p in paths})
    state['inputsUnchanged']=state['inputReadback']==pin['files']
    if (Q/'queue.log').exists():state['logSha256']=sha(Q/'queue.log')
    save('terminal.json',state)
sys.exit(exit_code if exit_code is not None else 1)
