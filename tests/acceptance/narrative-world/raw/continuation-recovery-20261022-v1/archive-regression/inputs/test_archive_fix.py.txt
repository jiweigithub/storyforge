"""Archive regression fixtures only; no StoryForge product/model acceptance."""
from pathlib import Path
import hashlib, json, subprocess, sys, tempfile, unittest

B = Path(__file__).parent
ORIGINAL = B / 'archive-delivery-validation-original.py'
CANDIDATE = B / 'archive-delivery-validation-candidate.py'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()


class ArchiveIntegrityTest(unittest.TestCase):
    def fixture(self, variant, collision=None):
        temporary = tempfile.TemporaryDirectory(prefix='archive-fixture-', dir=B)
        self.addCleanup(temporary.cleanup)
        base = Path(temporary.name)
        root = base/'workspace'
        runs = base/'runs'
        attempt = runs/'fixture'
        attempt.mkdir(parents=True)
        sources = {
            'src/lib/consistency/world-impact.ts': 'export const implementation = 1;\n',
            'src/lib/types/world-impact.ts': 'export interface FixtureType { value: number }\n',
            'tests/e2e/narrative-impact.spec.ts': '// isolated archive input, not a product test\n',
            'tests/regression/R-NW-2D-confirmed-sequence.test.ts': '// isolated archive input, not a product test\n',
        }
        for name, content in sources.items():
            p=root/name; p.parent.mkdir(parents=True, exist_ok=True); p.write_text(content)
        (root/'tests/acceptance/narrative-world/raw').mkdir(parents=True)
        runner=attempt/'validation.py';runner.write_text('# fixture only, not executed\n')
        queue_log=attempt/'validation-queue-r1.log';queue_log.write_text('fixture archive inputs\n')
        execution = {}
        if collision == 'raw-renamed-path':
            (attempt/'same.py').write_text('first fixture bytes\n')
            (attempt/'same.py.txt').write_text('second fixture bytes\n')
        if collision == 'external-input-name':
            for directory in ['outside-a', 'outside-b']:
                p=base/directory/'input.py';p.parent.mkdir();p.write_text('# equal fixture bytes\n')
                execution[str(p)]=sha(p)
        pin = {'files': {name: sha(root/name) for name in sources}, 'runtimeFiles': {}, 'extraFiles': {},
               'executionFiles': execution, 'runnerSha256': sha(runner)}
        pin_path=attempt/'validation-sources.json';pin_path.write_text(json.dumps(pin))
        # Synthetic successful terminals exist solely to test the archiver.
        # They are never copied into project acceptance evidence.
        terminal={'sourceUnchanged':True,'exitCode':0,'phases':[],'observedPids':[],'productionChanged':False}
        queue={'exitCode':0,'pinSha256':sha(pin_path),'logSha256':sha(queue_log),
               'supervisorPid':2147483646,'queuePid':2147483647,'productionChanged':False}
        (attempt/'validation-terminal.json').write_text(json.dumps(terminal))
        (attempt/'validation-queue-r1-terminal.json').write_text(json.dumps(queue))
        (attempt/'plan.json').write_text(json.dumps({'productionChanged':False,'fixtureOnly':True}))
        handoff=base/'handoff.json';handoff.write_text(json.dumps({'currentPin':str(pin_path),'durableQueue':{}}))
        source=variant.read_text()
        replacements = {
            "R=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')":f'R=Path({str(root)!r})',
            "B=Path('/tmp/nw-impact-sequence-20261022-v1')/sys.argv[1]":f'B=Path({str(runs)!r})/sys.argv[1]',
            "hp=Path('/tmp/nw-handoff-current-20261017.json')":f'hp=Path({str(handoff)!r})',
        }
        for old, new in replacements.items():
            self.assertEqual(source.count(old),1)
            source=source.replace(old,new)
        program=base/'archive-under-test.py';program.write_text(source)
        result=subprocess.run([sys.executable,str(program),'fixture','SYNTHETIC ARCHIVER FIXTURE ONLY'],
                              text=True,capture_output=True,timeout=20)
        archive=root/'tests/acceptance/narrative-world/raw/impact-sequence-fixture-20261022'
        receipt=root/'tests/acceptance/narrative-world/impact-sequence-fixture-20261022.json'
        return result, root, archive, receipt, sources

    def test_same_basename_reproduction_and_repaired_complete_archive(self):
        original,root,archive,receipt,sources=self.fixture(ORIGINAL)
        self.assertEqual(original.returncode,0,original.stderr)
        old=json.loads(receipt.read_text())
        self.assertLess(len({x['path'] for x in old['files']}),len(old['files']))
        self.assertEqual(sum(sha(root/x['path'])!=x['sha256'] for x in old['files']),1)
        corrected,root,archive,receipt,sources=self.fixture(CANDIDATE)
        self.assertEqual(corrected.returncode,0,corrected.stderr)
        new=json.loads(receipt.read_text())
        self.assertEqual(len({x['path'] for x in new['files']}),len(new['files']))
        for entry in new['files']:
            self.assertEqual(sha(root/entry['path']),entry['sha256'])
        for path,content in sources.items():
            self.assertEqual((archive/'source-inputs'/(path+'.txt')).read_text(),content)
            self.assertEqual((root/path).read_text(),content)

    def test_raw_extension_collision_rejected_without_overwrite_or_receipt(self):
        result,root,archive,receipt,sources=self.fixture(CANDIDATE,'raw-renamed-path')
        self.assertNotEqual(result.returncode,0)
        self.assertIn('Archive destination already exists',result.stderr)
        self.assertFalse(receipt.exists())
        self.assertEqual((archive/'same.py.txt').read_text(),'first fixture bytes\n')

    def test_identical_external_input_destination_rejected_without_success_receipt(self):
        result,root,archive,receipt,sources=self.fixture(CANDIDATE,'external-input-name')
        self.assertNotEqual(result.returncode,0)
        self.assertIn('Archive destination already exists',result.stderr)
        self.assertFalse(receipt.exists())
        outputs=list((archive/'execution-inputs').iterdir())
        self.assertEqual(len(outputs),1)
        self.assertEqual(outputs[0].read_text(),'# equal fixture bytes\n')


if __name__ == '__main__':
    unittest.main(verbosity=2)
