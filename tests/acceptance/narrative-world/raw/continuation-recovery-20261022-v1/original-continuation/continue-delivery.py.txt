from pathlib import Path
import datetime, fcntl, hashlib, json, os, re, shutil, subprocess, sys, time

BASE = Path('/tmp/nw-impact-sequence-20261022-v1')
ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
STATE = BASE / 'delivery-continuation-state.json'
TERMINAL = BASE / 'delivery-continuation-terminal.json'

def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def read(path): return json.loads(Path(path).read_text())
def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def save(path, data):
    tmp = Path(str(path) + '.tmp')
    tmp.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    os.replace(tmp, path)
def alive(pid):
    try: os.kill(pid, 0); return True
    except ProcessLookupError: return False

record = {'schema':'storyforge.nw-authorized-delivery-continuation', 'version':1,
          'startedAt':now(), 'pid':os.getpid(), 'modelCalls':0, 'automaticRetries':0,
          'thresholdChanges':False, 'goldChanges':False, 'steps':[]}

def stage(name, **data):
    record.update(stage=name, updatedAt=now(), **data)
    save(STATE, record)
    h = read(HANDOFF)
    h['deliveryContinuation'] = {'pid':os.getpid(), 'script':__file__, 'state':str(STATE),
                                 'terminal':str(TERMINAL), 'stage':name, 'updatedAt':now()}
    save(HANDOFF, h)
    print(json.dumps({'stage':name, **data}, ensure_ascii=False), flush=True)

def wait_validation(label):
    directory = BASE / label
    terminal = directory / 'validation-queue-r1-terminal.json'
    missing = 0
    while not terminal.exists():
        h = read(HANDOFF)
        pid = h['durableQueue']['supervisorPid']
        missing = 0 if alive(pid) else missing + 1
        if missing >= 3:
            raise RuntimeError('Validation supervisor disappeared without a terminal; preserve unknown outcome, do not resubmit.')
        time.sleep(10)
    queue = read(terminal)
    result_path = directory / 'validation-terminal.json'
    if not result_path.exists():
        raise RuntimeError('Queue ended without a runner terminal; inspect dispatch evidence before any retry.')
    result = read(result_path)
    pids = set(result['observedPids']) | {queue['queuePid'], queue['supervisorPid']}
    deadline = time.monotonic() + 60
    while any(alive(pid) for pid in pids):
        if time.monotonic() >= deadline:
            raise RuntimeError('Terminal exists but an observed PID still exists; preserve evidence for identity inspection.')
        time.sleep(2)
    assessment = ('All declared phases completed. This is engineering evidence; performance, model, semantic and human acceptance remain separate.'
                  if result['exitCode'] == 0 else
                  'Validation failed; preserve the original failure and stop this continuation. No automatic rerun or standard changes.')
    subprocess.run([sys.executable, str(BASE/'archive-delivery-validation.py'), label, assessment], check=True)
    record['steps'].append({'label':label, 'exitCode':result['exitCode'], 'terminal':str(result_path)})
    if result['exitCode'] != 0:
        raise RuntimeError('Validation failed in '+label+'; archived before further source edits.')
    return result

def launch(label, plan):
    assert not read(HANDOFF)['sourceFrozen']
    assert not (BASE/label).exists(), 'This attempt already exists; never dispatch twice.'
    subprocess.run([sys.executable, str(BASE/'prepare-delivery-validation.py'), label, str(BASE/plan)], check=True)

def main():
    with (BASE/'delivery-continuation.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        assert not STATE.exists() and not TERMINAL.exists()
        stage('waiting-for-existing-metrics-job')
        metrics_path = BASE/'project-metrics-terminal.json'
        while not metrics_path.exists():
            if not alive(52840) and not alive(52842):
                raise RuntimeError('Metrics job disappeared without a terminal; do not repeat unknown work.')
            time.sleep(10)
        metrics = read(metrics_path)
        assert metrics['exitCode'] == 0 and sha(BASE/'project-metrics.log') == metrics['logSha256']
        before = (BASE/'before/docs/ARCHITECTURE.md').read_text()
        after = (ROOT/'docs/ARCHITECTURE.md').read_text()
        pattern = r'<!-- project-metrics:start -->[\s\S]*?<!-- project-metrics:end -->'
        assert re.sub(pattern, '', before) == re.sub(pattern, '', after), 'Metrics generator changed unrelated content.'
        assert sha(ROOT/'docs/ARCHITECTURE.md') == metrics['afterSha256']
        while alive(52840) or alive(52842): time.sleep(1)
        stage('baseline-v77')
        launch('baseline-v77', 'baseline-v77-plan.json')
        wait_validation('baseline-v77')
        stage('reviewing-actual-baseline')
        manifest_path = ROOT/'tests/acceptance/narrative-world/manifest-v1.json'
        manifest = read(manifest_path)
        old_rel = manifest['baseline']['path']
        assert old_rel == 'tests/acceptance/narrative-world/baseline-20261021-nw2e-v76.json'
        assert str(manifest_path.relative_to(ROOT)) not in manifest['sourceFingerprintPaths']
        old = read(ROOT/old_rel)
        new_path = BASE/'baseline-v77/baseline-v77.json'
        new = read(new_path)
        observed = lambda value: {k:v for k,v in value.items() if k not in ['input','run']}
        assert observed(old) == observed(new), 'Baseline observations changed; needs review, do not promote automatically.'
        expected = sorted([*manifest['sourceFingerprintPaths'], manifest['corpus']['path']])
        assert sorted(new['input']['hashes']) == expected
        for name,digest in new['input']['hashes'].items(): assert sha(ROOT/name) == digest, name
        assert sha(ROOT/manifest['corpus']['path']) == manifest['corpus']['sha256']
        target = ROOT/'tests/acceptance/narrative-world/baseline-20261022-nw2d-v77.json'
        assert not target.exists()
        shutil.copy2(new_path, target)
        assert sha(target) == sha(new_path)
        assert not read(HANDOFF)['sourceFrozen']
        manifest['baseline']['history'].append({'path':old_rel, 'workPackage':'NW-2E',
            'note':'Preserved v76 observations. v77 rebinds the two-file confirmed-sequence impact repair and its regressions; original gold, all observed outcomes and acceptance thresholds remain unchanged.'})
        manifest['baseline']['path'] = str(target.relative_to(ROOT))
        save(manifest_path, manifest)
        h = read(HANDOFF)
        h['latestBaseline'] = {'path':str(target), 'sha256':sha(target), 'observationsEqualPrevious':True,
                              'fingerprintCount':len(new['input']['hashes'])}
        h['pendingProjectMetrics']['completed'] = True
        h['pendingProjectMetrics']['terminalSha256'] = sha(metrics_path)
        h['currentPhase'] = 'sequence-repair-delivery-validation'
        save(HANDOFF, h)
        stage('full-ci-and-full-isolated-e2e')
        launch('delivery-v1', 'delivery-v1-plan.json')
        wait_validation('delivery-v1')
        stage('delivery-validation-archived')

try:
    main()
    record['exitCode'] = 0
except BaseException as error:
    record.update(exitCode=1, error=type(error).__name__+': '+str(error))
    stage('needs-evidence-review', error=record['error'])
finally:
    record['endedAt'] = now()
    if not TERMINAL.exists(): save(TERMINAL, record)
    print(json.dumps({'exitCode':record.get('exitCode',1), 'terminal':str(TERMINAL)}), flush=True)
sys.exit(record.get('exitCode',1))
