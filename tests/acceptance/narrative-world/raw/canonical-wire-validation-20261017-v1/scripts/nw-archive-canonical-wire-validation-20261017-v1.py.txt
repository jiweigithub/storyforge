from pathlib import Path
import datetime,gzip,hashlib,json,re,shutil,subprocess,sys
ROOT=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
PREFIX=Path('/tmp/nw-canonical-wire-validation-20261017-v1')
BASE=ROOT/'tests/acceptance/narrative-world'
RAW=BASE/'raw/canonical-wire-validation-20261017-v1'
RECEIPT=BASE/'canonical-wire-validation-result-20261017-v1.json'
def digest(path):
 h=hashlib.sha256()
 with path.open('rb') as f:
  for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
 return h.hexdigest()
def main():
 assert sys.argv[1:]==['--owned-session-closed']
 terminal_path=Path(str(PREFIX)+'-terminal.json');terminal=json.loads(terminal_path.read_text())
 assert terminal.get('endedAt') and isinstance(terminal.get('exitCode'),int)
 pin_path=Path(str(PREFIX)+'-sources.json');pin=json.loads(pin_path.read_text())
 assert digest(pin_path)==terminal['pinSha256']
 for ref,key in [('HEAD','head'),('MERGE_HEAD','mergeHead')]:
  assert subprocess.check_output(['git','-C',str(ROOT),'rev-parse',ref],text=True).strip()==pin[key]
 assert terminal['sourceUnchanged']
 changed=[p for p,h in pin['files'].items() if not (ROOT/p).is_file() or digest(ROOT/p)!=h]
 assert not changed,changed
 for p,h in terminal.get('generatedReports',{}).items():assert (ROOT/p).is_file() and digest(ROOT/p)==h
 if RECEIPT.exists():
  d=json.loads(RECEIPT.read_text());assert d['pinSha256']==digest(pin_path)
  for a in d['artifacts']:assert digest(BASE/a['path'])==a['sha256']
  print(json.dumps({'existingVerified':str(RECEIPT)}));return
 pairs=[]
 for p in sorted(PREFIX.parent.glob(PREFIX.name+'*')):
  if p.is_file() and p!=pin_path and p.suffix in ('.json','.log'):pairs.append((p,Path(p.name)))
  elif p.is_dir() and p.name.endswith('-test-results'):
   pairs.extend((c,Path(p.name)/c.relative_to(p)) for c in sorted(p.rglob('*')) if c.is_file())
 for script in [Path(__file__),Path('/tmp/nw-canonical-wire-validation-20261017-v1.py')]:
  pairs.append((script,Path('scripts')/(script.name+'.txt')))
 assert shutil.disk_usage(ROOT).free>sum(p.stat().st_size for p,_ in pairs)+5*2**30
 RAW.mkdir(parents=True,exist_ok=True);artifacts=[]
 for source,relative in pairs:
  target=RAW/relative;target.parent.mkdir(parents=True,exist_ok=True)
  if not target.exists():
   with source.open('rb') as src,target.open('xb') as dst:shutil.copyfileobj(src,dst)
  assert digest(source)==digest(target)
  artifacts.append({'path':str(target.relative_to(BASE)),'bytes':target.stat().st_size,'sha256':digest(target)})
 packed=RAW/(pin_path.name+'.gz')
 if not packed.exists():
  with packed.open('xb') as f:f.write(gzip.compress(pin_path.read_bytes(),mtime=0))
 assert hashlib.sha256(gzip.decompress(packed.read_bytes())).hexdigest()==digest(pin_path)
 artifacts.append({'path':str(packed.relative_to(BASE)),'bytes':packed.stat().st_size,'sha256':digest(packed),'decompressedSha256':digest(pin_path)})
 phases=[]
 for row in terminal['phases']:
  log=Path(row['log']);assert digest(log)==row['logSha256']
  plain=re.sub(r'\x1b\[[0-9;]*m','',log.read_text(errors='replace'))
  phases.append({**row,'log':str((RAW/log.name).relative_to(BASE)),'summaryLines':[l.strip() for l in plain.splitlines() if re.search(r'^\s*(Test Files|Tests|Duration|\d+ (passed|failed|skipped))\b',l)]})
 receipt={'schema':'storyforge.narrative-canonical-wire-validation-result','version':1,'recordedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'head':pin['head'],'mergeHead':pin['mergeHead'],'pinSha256':digest(pin_path),'sourceFiles':len(pin['files']),'sourceUnchanged':True,'exitCode':terminal['exitCode'],'error':terminal.get('error'),'phases':phases,'artifacts':artifacts,'nativeTwoProductsPassed':terminal.get('nativeTwoProductsPassed',False),'fullCI':False,'fullE2E':False,'performanceAcceptance':False,'humanAcceptance':False,'actualModelCalls':0,'accepted':False}
 receipt['generatedReports']=terminal.get('generatedReports',{})
 receipt['observationComparisons']=terminal.get('observationComparisons',{})
 report_path=Path(str(PREFIX)+'-diagnostic.json')
 if report_path.exists():
  report=json.loads(report_path.read_text());assert not report['accepted'] and not report['cpuProfiling'] and not report['heapProfiling']
  assert report['sourceUnchanged'] and all(pin['files'][p]==h for p,h in report['sourceHashes'].items())
  receipt['diagnostic']={'path':str((RAW/report_path.name).relative_to(BASE)),'summaries':report['summaries'],'failures':report['failures'],'deviceMatches':report['deviceMatches'],'runs':[{k:r[k] for k in ('size','run','prepared','compilation','memory','export','open','queries') if k in r} for r in report['runs']]}
 with RECEIPT.open('x') as f:json.dump(receipt,f,ensure_ascii=False,indent=2);f.write('\n')
 print(json.dumps({'receipt':str(RECEIPT),'artifactCount':len(artifacts),'bytes':sum(a['bytes'] for a in artifacts),'exitCode':receipt['exitCode'],'nativeTwoProductsPassed':receipt['nativeTwoProductsPassed']}))
if __name__=='__main__':main()
