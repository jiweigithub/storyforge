from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, os, shutil, signal, subprocess, time

ROOT=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
BASE=Path('/tmp/nw-frozen-scalar-operations-20261022-v2')
OUT=Path('/tmp/nw-low-load-diagnosis-20261022')
HANDOFF=Path('/tmp/nw-handoff-current-20261017.json')
now=lambda:datetime.now(timezone.utc).isoformat()
def sha(path):
    with Path(path).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def save(path,value):
    with path.open('x') as f:json.dump(value,f,ensure_ascii=False,indent=2);f.write('\n')
def processes():
    result={}
    for line in subprocess.check_output(['ps','-axo','pid=,ppid=,stat=,lstart=,command='],text=True).splitlines():
        p=line.strip().split(None,8)
        if len(p)==9:result[int(p[0])]={'pid':int(p[0]),'ppid':int(p[1]),'state':p[2],'started':' '.join(p[3:8]),'command':p[8]}
    return result
def verify(h,pin):
    assert sha(BASE/'validation-sources.json')==h['currentPinSha256']
    assert sha(BASE/'validation.py')==pin['runnerSha256']
    for p,digest in pin['files'].items():assert sha(ROOT/p)==digest,p
    for group in ['runtimeFiles','extraFiles','executionFiles']:
        for p,digest in pin[group].items():assert sha(p)==digest,p
    for ref,key in [('HEAD','head'),('MERGE_HEAD','mergeHead')]:
        assert subprocess.check_output(['git','rev-parse',ref],cwd=ROOT,text=True).strip()==pin[key]
    return {k:len(pin[k]) for k in ['files','runtimeFiles','extraFiles','executionFiles']}
