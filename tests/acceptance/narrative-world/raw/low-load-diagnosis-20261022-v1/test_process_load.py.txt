import unittest
from process_load import classify,is_test_process

def row(command,pid=1,ppid=0,cpu=0,state='S'):
    return dict(command=command,pid=pid,ppid=ppid,cpu=cpu,state=state)

class ProcessLoadTests(unittest.TestCase):
    def test_actual_tests_are_detected(self):
        for c in ['npm run ci','npm run ci:e2e','npm run test:coverage','node (vitest)','node (vitest 2)','/usr/local/bin/node /r/node_modules/vitest/vitest.mjs run','node /r/node_modules/playwright/cli.js test']:
            with self.subTest(command=c):self.assertTrue(is_test_process(row(c)))
    def test_queued_wrappers_are_not_running_tests(self):
        for c in ['python3 /tmp/storyforge-serial-ci-20261013.py -- npm run ci','python3 guard.py npm run ci','caffeinate -i npm run ci']:
            with self.subTest(command=c):self.assertFalse(is_test_process(row(c)))
    def test_quoted_command_text_does_not_count(self):
        self.assertFalse(is_test_process(row("sh -c 'echo npm run ci'")))
        self.assertFalse(is_test_process(row('python3 -c "print(\'node (vitest)\')"')))
        self.assertFalse(is_test_process(row('node -e "console.log(\'npm run ci\')"')))
    def test_idle_dev_server_is_not_test(self):
        self.assertFalse(is_test_process(row('node /r/node_modules/vite/bin/vite.js --host')))
    def test_dead_worker_is_not_active(self):
        self.assertTrue(classify([row('node (vitest 1)',cpu=90,state='Z')])['quiet'])
    def test_wrapper_child_is_still_detected(self):
        data=classify([row('python3 guard.py npm run ci',pid=10),row('npm run ci',pid=11,ppid=10)])
        self.assertFalse(data['quiet']);self.assertEqual([r['pid'] for r in data['activeTestRoots']],[11])
    def test_counts_roots_without_losing_children(self):
        data=classify([row('npm run ci',pid=10),row('sh -c vitest',pid=11,ppid=10),row('node (vitest)',pid=12,ppid=11),row('node (vitest 1)',pid=13,ppid=12)])
        self.assertEqual(len(data['activeTests']),3);self.assertEqual([r['pid'] for r in data['activeTestRoots']],[10])
    def test_cpu_threshold_and_own_exclusion_are_unchanged(self):
        self.assertTrue(classify([row('WindowServer',cpu=19.9)])['quiet'])
        self.assertFalse(classify([row('WindowServer',cpu=20)])['quiet'])
        self.assertTrue(classify([row('node (vitest)',pid=10,cpu=100)],excluded={10})['quiet'])

if __name__=='__main__':unittest.main(verbosity=2)
