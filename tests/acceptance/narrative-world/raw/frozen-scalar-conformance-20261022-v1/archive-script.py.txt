from pathlib import Path
import datetime, hashlib, json, os, shutil, subprocess

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
DRAFT = Path('/tmp/nw-frozen-scalar-draft-20261021-v1')
BASE = Path('/tmp/nw-frozen-scalar-conformance-20261022-v1')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
original = HANDOFF.read_bytes()
h = json.loads(original)
q = h['durableQueue']
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()
def save(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2); stream.write('\n')

assert h['currentRunner'] == str(BASE) + '.py' and h['sourceFrozen'] is True
assert h['preparedCurrentArchive']['sha256'] == sha(__file__)
pin = json.loads(Path(h['currentPin']).read_text())
plan = json.loads(Path(str(BASE) + '-plan.json').read_text())
terminal = json.loads(Path(h['currentTerminal']).read_text())
queue_terminal = json.loads(Path(q['terminal']).read_text())
assert terminal['exitCode'] == queue_terminal['exitCode']
assert terminal['sourceUnchanged'] and queue_terminal['runnerTerminalPresent']
assert terminal['productionChanged'] is False and terminal['actualModelCalls'] == 0
assert terminal['independentHumanReviews'] == 0 and terminal['performanceAcceptance'] is False
assert sha(h['currentPin']) == h['currentPinSha256'] == terminal['pinSha256'] == queue_terminal['pinSha256']
assert sha(h['currentRunner']) == pin['runnerSha256']
assert sha(q['log']) == queue_terminal['logSha256']
for expected, actual in zip(plan['phases'], terminal['phases']):
    for key in ['name', 'command', 'cwd']: assert expected[key] == actual[key], key
    assert sha(actual['log']) == actual['logSha256']
passed = terminal['exitCode'] == 0
if passed:
    assert len(terminal['phases']) == len(plan['phases']) == 6
    assert all(phase['exitCode'] == 0 for phase in terminal['phases'])
    assert set(terminal['testSummaries']) == {'baseline', 'candidate'}
for variant, summary in terminal.get('testSummaries', {}).items():
    assert sha(summary['path']) == summary['sha256']
    report = json.loads(Path(summary['path']).read_text())
    assert len(report['testResults']) == 4
    if passed:
        assert summary['success'] and summary['numTotalTests'] == summary['numPassedTests'] > 0
        assert summary['numFailedTests'] == summary['numPendingTests'] == summary['numTodoTests'] == 0
if passed:
    assert terminal['testSummaries']['baseline']['numTotalTests'] == terminal['testSummaries']['candidate']['numTotalTests']

def verify_inputs():
    for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
        assert subprocess.check_output(['git', 'rev-parse', ref], cwd=ROOT, text=True).strip() == pin[key]
    for path, digest in pin['files'].items(): assert sha(ROOT / path) == digest, path
    for group in ['runtimeFiles', 'executionFiles', 'extraFiles']:
        for path, digest in pin[group].items(): assert sha(path) == digest, path
    for variant in ['baseline', 'candidate']:
        assert (DRAFT / variant / 'node_modules').resolve() == (ROOT / 'node_modules').resolve()
verify_inputs()
listed = subprocess.check_output(['git', 'ls-files', '-z', '--cached', '--others', '--exclude-standard'], cwd=ROOT).decode().split('\0')
actual = {name for name in listed if name and (ROOT / name).is_file() and not any(part.startswith('.env') for part in Path(name).parts)}
assert actual == set(pin['files']), {'added': sorted(actual - set(pin['files'])), 'missing': sorted(set(pin['files']) - actual)}

pids = set(terminal['observedPids']) | {queue_terminal['supervisorPid'], queue_terminal['queuePid'], terminal['runnerPid']}
pids.update(phase['childPid'] for phase in terminal['phases'])
observations = Path(str(BASE) + '-process-observations.jsonl')
if observations.exists():
    for line in observations.read_text().splitlines(): pids.update(json.loads(line)['pids'])
ps = subprocess.run(['ps', '-p', ','.join(map(str, sorted(pids))), '-o', 'pid=,ppid=,state=,command='], capture_output=True, text=True)
assert ps.returncode == 1 and not ps.stdout.strip(), ps.stdout
rows = {int(parts[0]): parts for line in subprocess.check_output(['ps', '-axo', 'pid=,ppid=,command='], text=True).splitlines() if len(parts := line.strip().split(None, 2)) == 3}
ancestors = {os.getpid()}
pid = os.getpid()
while pid in rows and int(rows[pid][1]) not in ancestors:
    pid = int(rows[pid][1]); ancestors.add(pid)
scan = subprocess.run(['lsof', '-nP', '-d', 'cwd', '-Fpn'], capture_output=True, text=True, timeout=20)
assert scan.returncode == 0
cwd_matches = []
pid = None
for line in scan.stdout.splitlines():
    if line.startswith('p'): pid = int(line[1:])
    elif line.startswith('n'):
        path = Path(line[1:])
        if any(path == root.resolve() or root.resolve() in path.parents for root in [ROOT, DRAFT]):
            if pid in rows and pid not in ancestors: cwd_matches.append(rows[pid])
assert not cwd_matches, cwd_matches
assert not [parts for pid, parts in rows.items() if pid not in ancestors and str(BASE) in parts[2]]
exit_proof = {'schema': 'storyforge.narrative-validation-owned-exit', 'version': 1, 'checkedAt': now(),
    'pids': sorted(pids), 'allAbsent': True, 'sourceUnchanged': True,
    'sourceFiles': len(pin['files']), 'runtimeFiles': len(pin['runtimeFiles']), 'extraFiles': len(pin['extraFiles']),
    'head': pin['head'], 'mergeHead': pin['mergeHead'], 'pinSha256': sha(h['currentPin']),
    'terminalSha256': sha(h['currentTerminal']), 'queueTerminalSha256': sha(q['terminal']),
    'workspaceProcessesRemaining': [],
    'limitation': 'All recorded processes are absent; cwd/command scans exclude this evidence writer and its ancestors. Unobserved transient descendants are not individually enumerated.'}
save(str(BASE) + '-owned-exit.json', exit_proof)
dest = ROOT / 'tests/acceptance/narrative-world/raw/frozen-scalar-conformance-20261022-v1'
assert not dest.exists()
dest.mkdir()
artifacts = []
def archive(source, relative):
    target = dest / relative
    if source.suffix in ['.ts', '.tsx', '.js', '.mjs', '.cjs', '.py']: target = Path(str(target) + '.txt')
    target.parent.mkdir(parents=True, exist_ok=True)
    assert not target.exists()
    shutil.copy2(source, target)
    digest = sha(source)
    assert digest == sha(target)
    artifacts.append({'path': str(target.relative_to(ROOT)), 'sha256': digest, 'bytes': target.stat().st_size})
for source in sorted(BASE.parent.glob(BASE.name + '*')):
    if source.name.endswith(('.lock', '.tmp')) or any(token in source.name for token in ['-archived-handoff', '-heartbeat-']): continue
    assert source.is_file(), source
    archive(source, Path(source.name))
for path in sorted(pin['extraFiles']):
    source = Path(path)
    if DRAFT in source.parents: archive(source, Path('draft') / source.relative_to(DRAFT))
archive(Path(__file__), Path('archive-script.py'))
receipt = dict(terminal)
receipt.update(schema='storyforge.narrative-frozen-scalar-conformance-receipt', archivedAt=now(),
    artifacts=artifacts, exitProof=exit_proof, conformancePassed=passed,
    disposition='conformance-passed-no-adoption' if passed else 'conformance-failed-no-adoption',
    failedPhases=[p['name'] for p in terminal['phases'] if p['exitCode'] != 0],
    limitations=['External pure-codec variants only. No performance measurement, full product integration, full CI/E2E, candidate adoption, model call or human review.'],
    productionChanged=False, adopted=False, performanceAcceptance=False, formalAcceptance=False, goldOrThresholdChanges=False,
    next=plan['nextIfPassed'] if passed else 'Inspect the archived actual failure; preserve this attempt and make only a supported repair before a separately declared validation.')
receipt_path = ROOT / 'tests/acceptance/narrative-world/frozen-scalar-conformance-result-20261022-v1.json'
save(receipt_path, receipt)
assert all(sha(ROOT / item['path']) == item['sha256'] for item in artifacts)
verify_inputs()
verification = {'schema': 'storyforge.narrative-archive-verification', 'version': 1, 'verifiedAt': now(),
    'receipt': str(receipt_path.relative_to(ROOT)), 'receiptSha256': sha(receipt_path),
    'artifactsVerified': len(artifacts), 'allMatch': True, 'sourceFilesUnchanged': len(pin['files']),
    'runtimeFilesUnchanged': len(pin['runtimeFiles']), 'draftAndPlanFilesUnchanged': len(pin['extraFiles']),
    'knownPidsAbsent': sorted(pids), 'productionChanged': False, 'conformancePassed': passed, 'performanceAcceptance': False}
verification_path = ROOT / 'tests/acceptance/narrative-world/frozen-scalar-conformance-archive-verification-20261022-v1.json'
save(verification_path, verification)
assert HANDOFF.read_bytes() == original

    latestCompletedReceipt=str(receipt_path), latestCompletedReceiptSha256=sha(receipt_path),

    lastCompletedSourcePinSha256=h['currentPinSha256'])

    testsExecuted=True, sourceFrozen=False, receipt=str(receipt_path), candidateAdopted=False)
h['activeIndependentProbe'].update(status='completed-archived', testsExecuted=True, conformancePassed=passed,
    receipt=str(receipt_path), adopted=False)

    'sourceFilesUnchanged': len(pin['files']), 'knownPidsAbsent': sorted(pids), 'performanceAcceptance': False, 'resubmitted': False}
h['next'] = receipt['next']
stage = Path(str(BASE) + '-archived-handoff.tmp')

print('knownPidsAbsent', len(pids))
