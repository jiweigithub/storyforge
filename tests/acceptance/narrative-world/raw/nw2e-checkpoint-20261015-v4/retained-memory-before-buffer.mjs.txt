import { createRequire } from 'node:module';
import { readFileSync, writeFileSync, openSync, closeSync } from 'node:fs';
import { spawn, execFileSync } from 'node:child_process';
import { createServer } from 'node:net';
import { createHash } from 'node:crypto';
import { join } from 'node:path';

const root = '/Users/qinyingying/.codex/worktrees/narrative-world-design/storyforge';
const require = createRequire(join(root, 'package.json'));
const { chromium } = require('playwright');
const paths = Object.keys(JSON.parse(readFileSync(join(root, 'tests/acceptance/narrative-world/scale-sources-20261015-v10.json'), 'utf8')).sourceHashes);
const fingerprint = () => Object.fromEntries(paths.map(p => [p, createHash('sha256').update(readFileSync(join(root, p))).digest('hex')]));
const sourceHashes = fingerprint();
const listener = createServer();
await new Promise(resolve => listener.listen(0, '127.0.0.1', resolve));
const port = listener.address().port;
await new Promise(resolve => listener.close(resolve));
const log = openSync('/tmp/nw2e-retained-memory-server.log', 'w');
const vite = spawn(process.execPath, [join(root, 'node_modules/vite/bin/vite.js'), '--host', '127.0.0.1', '--port', String(port), '--strictPort'], { cwd: root, stdio: ['ignore', log, log] });
const url = `http://127.0.0.1:${port}/storyforge/`;
let server;
try {
  for (let attempt = 0; ; attempt++) {
    try { if ((await fetch(url)).ok) break; } catch {}
    if (attempt === 100 || vite.exitCode !== null) throw Error('Isolated development server unavailable');
    await new Promise(resolve => setTimeout(resolve, 100));
  }
  server = await chromium.launchServer({ headless: true });
  const browser = await chromium.connect(server.wsEndpoint());
  const context = await browser.newContext(), page = await context.newPage();
  await page.goto(url);
  const modulePath = '/storyforge/tests/helpers/narrative-scale-browser.ts';
  const prepared = await page.evaluate(async p => (await new Function('p', 'return import(p)')(p)).prepareScaleV1(1000000), modulePath);
  const cdp = await context.newCDPSession(page);
  const rss = () => {
    const rows = execFileSync('ps', ['-axo', 'pid=,ppid=,rss='], { encoding: 'utf8' }).trim().split('\n').map(line => line.trim().split(/\s+/).map(Number));
    const ids = new Set([server.process().pid]);
    for (let before = -1; before !== ids.size;) { before = ids.size; for (const [id, parent] of rows) if (ids.has(parent)) ids.add(id); }
    return rows.filter(([id]) => ids.has(id)).reduce((sum, row) => sum + row[2] * 1024, 0);
  };
  await cdp.send('HeapProfiler.collectGarbage');
  const baseline = { heap: await cdp.send('Runtime.getHeapUsage'), rssBytes: rss() };
  const compilation = await page.evaluate(async p => (await new Function('p', 'return import(p)')(p)).compileScaleV1(), modulePath);
  const afterCompile = { heap: await cdp.send('Runtime.getHeapUsage'), rssBytes: rss() };
  await cdp.send('HeapProfiler.collectGarbage');
  const afterCollection = { heap: await cdp.send('Runtime.getHeapUsage'), rssBytes: rss() };
  const report = { schema: 'storyforge.narrative-retained-memory-diagnostic', version: 1, createdAt: new Date().toISOString(),
    evidenceBoundary: 'One isolated browser diagnostic of retained heap and process RSS after explicit GC. Not the frozen performance protocol, not peak-memory acceptance, and not a mathematical lower bound. No threshold changes.',
    browserVersion: browser.version(), sourceHashes, sourceUnchanged: JSON.stringify(sourceHashes) === JSON.stringify(fingerprint()), prepared, compilation,
    baseline, afterCompile, afterCollection, accepted: false, modelCalls: 0, independentHumanReviews: 0 };
  writeFileSync(join(root, 'tests/acceptance/narrative-world/scale-retained-memory-20261015-v1.json'), JSON.stringify(report, null, 2) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ compilation, baseline, afterCompile, afterCollection, sourceUnchanged: report.sourceUnchanged }));
  await context.close(); await browser.close();
} finally { await server?.close(); vite.kill('SIGTERM'); closeSync(log); }
