from pathlib import Path
from datetime import datetime, timezone
import fcntl
import hashlib
import json
import os
import signal
import subprocess
import sys

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
PREFIX = Path('/tmp/nw-utf8-token-bytes-validation-20261019-v1')
STATE = Path(str(PREFIX) + '-queue-r1-state.json')
RECEIPT = Path(str(PREFIX) + '-queue-r1-terminal.json')
LOG = Path(str(PREFIX) + '-queue-r1.log')

def now():
    return datetime.now(timezone.utc).isoformat()

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

def main():
    signal.signal(signal.SIGHUP, signal.SIG_IGN)
    with Path(str(PREFIX) + '-queue-r1.lock').open('a') as owned_lock:
        fcntl.flock(owned_lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        assert not STATE.exists() and not RECEIPT.exists() and not LOG.exists()
        assert not Path(str(PREFIX) + '-terminal.json').exists()
        assert not Path(str(PREFIX) + '-diff-check.log').exists()
        pin_path = Path(str(PREFIX) + '-sources.json')
        pin = json.loads(pin_path.read_text())
        assert digest(pin_path) == '430e8fe05c9faa4695b48b13f1af3f73cf1b710e2be60c8518f3fb8ccfc0ac89'
        assert digest(Path(str(PREFIX) + '.py')) == pin['runnerSha256']
        assert digest(Path(str(PREFIX) + '-scale-sources.json')) == pin['scaleInputSha256']
        for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
            assert subprocess.check_output(['git', '-C', str(ROOT), 'rev-parse', ref], text=True).strip() == pin[key]
        changed = [p for p, sha in pin['files'].items() if not (ROOT/p).is_file() or digest(ROOT/p) != sha]
        assert not changed, changed
        command = [sys.executable, '/tmp/storyforge-serial-ci-20261013.py', '--', sys.executable, str(PREFIX) + '.py']
        record = {'schema': 'storyforge.narrative-durable-queue-attempt', 'version': 1, 'attempt': 1,
                  'startedAt': now(), 'supervisorPid': os.getpid(), 'root': str(ROOT),
                  'runner': str(PREFIX) + '.py', 'pinSha256': digest(pin_path),
                  'log': str(LOG), 'actualModelCalls': 0, 'accepted': False}
        with LOG.open('x') as log:
            child = subprocess.Popen(command, cwd=ROOT, stdin=subprocess.DEVNULL,
                                     stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
            record['queuePid'] = child.pid
            save(STATE, record)
            def forward(signum, frame):
                if child.poll() is None:
                    try:
                        os.killpg(child.pid, signum)
                    except ProcessLookupError:
                        pass
            signal.signal(signal.SIGTERM, forward)
            signal.signal(signal.SIGINT, forward)
            code = child.wait()
        record.update(endedAt=now(), exitCode=code, logSha256=digest(LOG),
                      runnerTerminalPresent=Path(str(PREFIX) + '-terminal.json').exists())
        save(RECEIPT, record)

if __name__ == '__main__':
    main()
