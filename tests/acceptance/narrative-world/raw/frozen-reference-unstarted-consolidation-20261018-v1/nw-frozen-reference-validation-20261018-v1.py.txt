from pathlib import Path
import datetime, hashlib, json, os, shutil, socket, subprocess, sys

ROOT=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
PREFIX=Path('/tmp/nw-frozen-reference-validation-20261018-v1')
PIN=Path(str(PREFIX)+'-sources.json')
SCALE=Path(str(PREFIX)+'-scale-sources.json')
def digest(path):
    h=hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda:f.read(1024*1024),b''):h.update(chunk)
    return h.hexdigest()
def now():return datetime.datetime.now(datetime.timezone.utc).isoformat()
pin=json.loads(PIN.read_text())
pin_hash=digest(PIN)
def verify():
    assert Path.cwd()==ROOT and digest(PIN)==pin_hash
    assert digest(Path(__file__))==pin['runnerSha256'] and digest(SCALE)==pin['scalePinSha256']
    for ref,key in [('HEAD','head'),('MERGE_HEAD','mergeHead')]:
        assert subprocess.check_output(['git','rev-parse',ref],text=True).strip()==pin[key]
    changed=[p for p,h in pin['files'].items() if not (ROOT/p).is_file() or digest(ROOT/p)!=h]
    assert not changed,changed

record={'schema':'storyforge.narrative-frozen-reference-validation','version':1,'startedAt':now(),
    'pinSha256':pin_hash,'sourceFiles':len(pin['files']),'phases':[],
    'actualModelCalls':0,'fullCI':False,'fullE2E':False,'performanceAcceptance':False,'accepted':False}
env=dict(os.environ,CI='1',PLAYWRIGHT_FROZEN_WORKSPACE='1')
for key in list(env):
    if key.startswith('NW_'):env.pop(key)
def phase(name,command,extra=None):
    verify()
    free=shutil.disk_usage(ROOT).free/2**30
    assert free>=5
    phase_env=dict(env)
    with socket.socket() as s:s.bind(('127.0.0.1',0));phase_env['PLAYWRIGHT_PORT']=str(s.getsockname()[1])
    phase_env['PLAYWRIGHT_HTML_OUTPUT_DIR']=str(PREFIX)+'-'+name+'-html'
    if extra:phase_env.update(extra)
    log=Path(str(PREFIX)+'-'+name+'.log')
    item={'name':name,'command':command,'startedAt':now(),'freeGiB':free,'log':str(log)}
    record['phases'].append(item)
    print(json.dumps({'starting':name,'at':item['startedAt']}),flush=True)
    with log.open('x') as out:code=subprocess.call(command,env=phase_env,stdout=out,stderr=subprocess.STDOUT)
    item.update(exitCode=code,endedAt=now(),logSha256=digest(log))
    verify()
    print(json.dumps({'completed':name,'exitCode':code}),flush=True)
    if code:raise RuntimeError(name+' failed with exit '+str(code))

code=2
try:
    phase('targeted',['node','node_modules/vitest/vitest.mjs','run',
        'tests/regression/R-NW-2E-reference-index.test.ts','tests/regression/R-NW-1B-projection.test.ts',
        'tests/regression/R-NW-2E-domain-allocations.test.ts','tests/regression/R-NW-2A-semantic-export.test.ts',
        'tests/regression/R-NW-2A-index-recipe.test.ts','tests/regression/R-NW-2A-index-cache.test.ts',
        'tests/regression/R-NW-2A-release-content.test.ts','tests/regression/R-NW-2A-package.test.ts',
        'tests/regression/R-NW-2B-query.test.ts','tests/regression/R-NW-2B-provider.test.ts',
        'tests/regression/R-NW-2D-impact.test.ts','tests/regression/R-NW-2D-recovery.test.ts',
        'tests/regression/R-NW-2E-packing.test.ts'])
    phase('lint',['node','node_modules/eslint/bin/eslint.js','src/lib/world-engine/narrative-domain-evidence.ts',
        'src/lib/world-engine/narrative-release-v4.ts','tests/regression/R-NW-2E-reference-index.test.ts'])
    phase('types',['node','node_modules/typescript/bin/tsc','--noEmit'])
    phase('diagnostic',['node','node_modules/@playwright/test/cli.js','test','tests/e2e/narrative-scale.spec.ts',
        '--reporter=line','--retries=0','--output='+str(PREFIX)+'-test-results'],{
        'NW_SCALE_BROWSER_MODE':'diagnostic','NW_SCALE_PROFILE':'compiler-v2',
        'NW_SCALE_BROWSER_REPORT':str(PREFIX)+'-diagnostic.json',
        'NW_SCALE_BROWSER_VECTORS':'tests/acceptance/narrative-world/scale-vectors-20261015-v2.json',
        'NW_SCALE_BROWSER_SOURCES':str(SCALE),'NW_SCALE_BROWSER_SIZES':'1000000'})
    report=json.loads(Path(str(PREFIX)+'-diagnostic.json').read_text())
    assert report['sourceUnchanged'] and not report['accepted']
    assert not report['cpuProfiling'] and not report['heapProfiling']
    assert all(pin['files'][p]==h for p,h in report['sourceHashes'].items())
    record['diagnosticSummary']=report['summaries']
    record['diagnosticFailures']=report['failures']
    code=0
except BaseException as e:
    record['error']=type(e).__name__+': '+str(e)
finally:
    try:verify();record['sourceUnchanged']=True
    except BaseException as e:record['sourceUnchanged']=False;record['sourceCheckError']=str(e);code=2
    record.update(exitCode=code,endedAt=now())
    with Path(str(PREFIX)+'-terminal.json').open('x') as f:json.dump(record,f,ensure_ascii=False,indent=2);f.write('\n')
    print(json.dumps({'terminal':str(PREFIX)+'-terminal.json','exitCode':code}),flush=True)
sys.exit(code)
