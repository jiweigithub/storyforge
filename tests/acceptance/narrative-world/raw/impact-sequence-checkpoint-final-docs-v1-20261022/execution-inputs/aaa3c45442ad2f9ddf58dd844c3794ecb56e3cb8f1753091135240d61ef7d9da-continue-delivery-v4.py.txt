from pathlib import Path
import datetime, fcntl, hashlib, json, os, re, shutil, subprocess, sys, time, traceback

BASE = Path('/tmp/nw-impact-sequence-20261022-v1')
ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
REPAIR = Path('/tmp/nw-workspace-load-repair-20261022-v1')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
STATE = BASE / 'delivery-continuation-v4-state.json'
TERMINAL = BASE / 'delivery-continuation-v4-terminal.json'

def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def read(path): return json.loads(Path(path).read_text())
def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def save(path, data):
    tmp = Path(str(path) + '.tmp')
    tmp.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    os.replace(tmp, path)
def alive(pid):
    try: os.kill(pid, 0); return True
    except ProcessLookupError: return False

record = {'schema':'storyforge.nw-authorized-delivery-continuation', 'version':1,
          'startedAt':now(), 'pid':os.getpid(), 'modelCalls':0, 'automaticRetries':0,
          'thresholdChanges':False, 'goldChanges':False, 'steps':[]}

def stage(name, **data):
    record.update(stage=name, updatedAt=now(), **data)
    save(STATE, record)
    h = read(HANDOFF)
    h['deliveryContinuation'] = {'pid':os.getpid(), 'script':__file__, 'state':str(STATE),
                                 'terminal':str(TERMINAL), 'stage':name, 'updatedAt':now()}
    save(HANDOFF, h)
    print(json.dumps({'stage':name, **data}, ensure_ascii=False), flush=True)

def wait_validation(label, expected_failure=False):
    directory = BASE / label
    terminal = directory / 'validation-queue-r1-terminal.json'
    missing = 0
    while not terminal.exists():
        h = read(HANDOFF)
        assert h['currentPin'] == str(directory/'validation-sources.json')
        pid = h['durableQueue']['supervisorPid']
        missing = 0 if alive(pid) else missing + 1
        if missing >= 3:
            raise RuntimeError('Supervisor disappeared without a terminal; preserve unknown outcome, do not resubmit.')
        time.sleep(10)
    queue = read(terminal)
    result_path = directory / 'validation-terminal.json'
    if not result_path.exists():
        raise RuntimeError('Queue ended without a runner terminal; inspect dispatch evidence before retry.')
    result = read(result_path)
    pids = set(result['observedPids']) | {queue['queuePid'], queue['supervisorPid']}
    deadline = time.monotonic() + 60
    while any(alive(pid) for pid in pids):
        if time.monotonic() >= deadline:
            raise RuntimeError('An observed PID still exists after terminal; preserve evidence for identity inspection.')
        time.sleep(2)
    assessment = ('Expected deterministic reproduction of the preserved delivery-v2 unhandled workspace-load rejection. A failing reproduction is not a fix.'
                  if expected_failure else
                  'All declared phases completed; engineering evidence only.' if result['exitCode'] == 0 else
                  'Validation failed; preserve the original result and stop. No automatic rerun or standard changes.')
    subprocess.run([sys.executable, str(BASE/'archive-delivery-validation-v2.py'), label, assessment], check=True)
    record['steps'].append({'label':label, 'exitCode':result['exitCode'], 'terminal':str(result_path)})
    if expected_failure:
        log = (directory/'validation-workspace-load-red.log').read_text()
        assert result['exitCode'] == 1 and result['phases'][0]['exitCode'] == 1
        assert 'WorldReferenceErrorV1' in log and '[world-reference:release-missing]' in log
        assert 'Unhandled Rejection' in log and 'shows the current frozen-release load failure' in log
        assert re.search(r'Tests\s+1 failed\s*\|\s*2 skipped', log), 'Unexpected reproduction result; inspect before editing'
    elif result['exitCode'] != 0:
        raise RuntimeError('Validation failed in '+label+'; archived before further edits.')
    return result

def launch(label, plan):
    assert not read(HANDOFF)['sourceFrozen']
    assert not (BASE/label).exists(), 'Attempt already exists; never dispatch twice.'
    subprocess.run([sys.executable, str(BASE/'prepare-delivery-validation-v4.py'), label, str(BASE/plan)], check=True)

def apply_repair():
    assert not read(HANDOFF)['sourceFrozen']
    assert not (REPAIR/'applied.json').exists()
    changes = read(REPAIR/'changes.json')['changes']
    for item in changes:
        assert sha(ROOT/item['path']) == item['beforeSha256'], item['path']
        assert sha(item['after']) == item['afterSha256'], item['after']
    for item in changes:
        (ROOT/item['path']).write_bytes(Path(item['after']).read_bytes())
        assert sha(ROOT/item['path']) == item['afterSha256']
    save(REPAIR/'applied.json', {'at':now(), 'changes':changes, 'originalFailurePreserved':True,
                               'sourceFrozenBeforeEdits':False, 'goldChanged':False, 'thresholdsChanged':False})
    h = read(HANDOFF)
    h['workspaceLoadRecovery']['repairApplied'] = True
    h['pendingDocumentationDraft']['supersededBy'] = str(REPAIR/'changes.json')
    h['pendingDocumentationDraft']['note'] = 'The prior pending delivery-v2 sentence was reconciled to its actual failure; final delivery-v3 documentation remains pending.'
    save(HANDOFF, h)

def refresh_metrics():
    assert not read(HANDOFF)['sourceFrozen']
    log_path = REPAIR/'project-metrics.log'
    result_path = REPAIR/'project-metrics-terminal.json'
    assert not log_path.exists() and not result_path.exists(), 'Metrics action already started; inspect unknown outcome'
    paths = subprocess.check_output(['git','ls-files','-z','--cached','--others','--exclude-standard'], cwd=ROOT).decode().split('\0')
    inputs = {p:sha(ROOT/p) for p in sorted(set(paths)) if p and (ROOT/p).is_file()}
    output = 'docs/ARCHITECTURE.md'
    (REPAIR/'architecture-before-metrics.md').write_bytes((ROOT/output).read_bytes())
    save(REPAIR/'project-metrics-inputs.json', inputs)
    command = [sys.executable, '/tmp/storyforge-serial-ci-20261013.py', '--',
               '/usr/local/bin/node', 'scripts/generate-project-metrics.mjs']
    state = {'startedAt':now(), 'command':command, 'allowedOutput':output, 'sourceInputs':str(REPAIR/'project-metrics-inputs.json')}
    with log_path.open('x') as log:
        process = subprocess.Popen(command, cwd=ROOT, stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT)
        state['queuePid'] = process.pid
        save(REPAIR/'project-metrics-state.json', state)
        code = process.wait()
    mismatches = [p for p,digest in inputs.items() if p != output and sha(ROOT/p) != digest]
    state.update(endedAt=now(), exitCode=code, logSha256=sha(log_path), otherInputMismatches=mismatches,
                 outputBeforeSha256=inputs[output], outputAfterSha256=sha(ROOT/output), queueProcessExited=True)
    save(result_path, state)
    assert code == 0 and not mismatches, 'Metrics failed or changed an undeclared input; preserve evidence'
    (REPAIR/'architecture-after-metrics.md').write_bytes((ROOT/output).read_bytes())

def preserve_repair_receipt(promotion):
    assert not read(HANDOFF)['sourceFrozen']
    archive = ROOT/'tests/acceptance/narrative-world/raw/workspace-load-repair-20261022-v1'
    archive.mkdir()
    files = []
    for source in sorted(REPAIR.rglob('*')):
        if not source.is_file(): continue
        target = archive/source.relative_to(REPAIR)
        if source.suffix in ('.py','.ts','.tsx','.mjs'):
            target = target.with_name(target.name+'.txt')
        target.parent.mkdir(parents=True, exist_ok=True)
        assert not target.exists()
        shutil.copy2(source, target)
        assert sha(source) == sha(target)
        files.append({'path':str(target.relative_to(ROOT)), 'sha256':sha(target)})
    receipt = {'schema':'storyforge.nw-workspace-load-repair', 'version':1, 'at':now(),
               'failureReceipt':'tests/acceptance/narrative-world/impact-sequence-delivery-v2-20261022.json',
               'reproductionReceipt':'tests/acceptance/narrative-world/impact-sequence-workspace-load-red-v1-20261022.json',
               'targetedReceipt':'tests/acceptance/narrative-world/impact-sequence-workspace-load-green-v1-20261022.json',
               'baselinePromotion':promotion, 'files':files, 'goldChanged':False, 'thresholdsChanged':False,
               'newModelCalls':0, 'engineeringDelivery':'full CI and isolated E2E pending',
               'semanticQualityAcceptance':False, 'performanceAcceptance':False}
    target = ROOT/'tests/acceptance/narrative-world/workspace-load-repair-20261022-v1.json'
    assert not target.exists()
    save(target, receipt)
    h = read(HANDOFF)
    h['workspaceLoadRecovery'].update(receipt=str(target), receiptSha256=sha(target), targetedPassed=True)
    h['remainingGates'][0] = 'delivery-v2 preserved 4949 assertions passed with 1 unhandled workspace load rejection, so full CI failed and E2E never started. The failure was reproduced, narrowly repaired and targeted regressions passed; actual v79 baseline was promoted with unchanged observations. Finish exactly one delivery-v3 full CI and isolated E2E, terminal/archive review, final documentation and local integration commit. No push, PR or main merge.'
    save(HANDOFF, h)

def main():
    with (BASE/'delivery-continuation-v4.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        assert not STATE.exists() and not TERMINAL.exists(), 'This continuation already started'
        assert not (BASE/'workspace-load-green-v1').exists() and not (BASE/'delivery-v3').exists()
        assert read(HANDOFF)['currentPin'] == str(BASE/'workspace-load-red-v1/validation-sources.json')
        stage('awaiting-existing-workspace-load-reproduction')
        wait_validation('workspace-load-red-v1', expected_failure=True)
        stage('applying-reviewed-workspace-load-repair')
        apply_repair()
        stage('refreshing-documentation-metrics-through-shared-queue')
        refresh_metrics()
        stage('targeted-workspace-lifecycle-and-baseline-v79')
        launch('workspace-load-green-v1', 'workspace-load-green-v1-plan.json')
        wait_validation('workspace-load-green-v1')
        stage('reviewing-actual-baseline-v79')
        manifest = ROOT/'tests/acceptance/narrative-world/manifest-v1.json'
        target = ROOT/'tests/acceptance/narrative-world/baseline-20261022-nw2d-v79.json'
        command = [sys.executable, '/tmp/nw-continuation-recovery-20261022-v1/baseline_promotion.py', str(ROOT),
                   str(manifest), str(ROOT/'tests/acceptance/narrative-world/baseline-20261022-nw2d-v78.json'),
                   str(BASE/'workspace-load-green-v1/baseline-v79.json'), str(target)]
        promotion = json.loads(subprocess.check_output(command, text=True))
        save(BASE/'baseline-v79-promotion.json', {'at':now(), 'command':command, **promotion})
        h = read(HANDOFF)
        h['latestBaseline'] = {'path':str(target), 'sha256':sha(target), **promotion}
        save(HANDOFF, h)
        preserve_repair_receipt(promotion)
        stage('full-ci-and-full-isolated-e2e')
        launch('delivery-v3', 'delivery-v3-plan.json')
        wait_validation('delivery-v3')
        stage('delivery-validation-archived')

try:
    main()
    record['exitCode'] = 0
except BaseException as error:
    record.update(exitCode=1, error=type(error).__name__+': '+str(error), traceback=traceback.format_exc())
    stage('needs-evidence-review', error=record['error'])
finally:
    record['endedAt'] = now()
    if not TERMINAL.exists(): save(TERMINAL, record)
    print(json.dumps({'exitCode':record.get('exitCode',1), 'terminal':str(TERMINAL)}), flush=True)
sys.exit(record.get('exitCode',1))
