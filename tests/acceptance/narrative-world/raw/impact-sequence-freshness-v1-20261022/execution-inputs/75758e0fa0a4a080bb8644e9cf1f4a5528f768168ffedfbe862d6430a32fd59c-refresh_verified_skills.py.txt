"""Refresh only reviewed dates after exact frozen contract regressions passed."""
from pathlib import Path
import datetime
import hashlib
import json
import re
import shutil

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
BASE = Path('/tmp/nw-impact-sequence-20261022-v1')
RECOVERY = Path('/tmp/nw-continuation-recovery-20261022-v1')
sha = lambda path: hashlib.sha256(Path(path).read_bytes()).hexdigest()
read = lambda path: json.loads(Path(path).read_text())
handoff = read('/tmp/nw-handoff-current-20261017.json')
assert not handoff['sourceFrozen']
result = read(BASE / 'freshness-v1/validation-terminal.json')
receipt_path = ROOT / 'tests/acceptance/narrative-world/impact-sequence-freshness-v1-20261022.json'
receipt = read(receipt_path)
assert result['exitCode'] == receipt['exitCode'] == 0
assert result['sourceUnchanged'] and receipt['allInputsUnchanged']
assert len(result['phases']) == 4 and all(p['exitCode'] == 0 for p in result['phases'])
pin = read(BASE / 'freshness-v1/validation-sources.json')
name = 'src/lib/agent/skill-registry.ts'
registry = ROOT / name
assert sha(registry) == pin['files'][name]
assert sha(ROOT / 'scripts/check-agent-freshness.mjs') == pin['files']['scripts/check-agent-freshness.mjs']
closure = read(RECOVERY / 'stale-skill-closure.json')
expected_ids = {s['id'] for s in closure['skills']}
failure_log = (BASE / 'delivery-v1/validation-full-ci.log').read_text()
failed_ids = set(re.findall(r'^- (\S+) verification is stale:', failure_log, flags=re.M))
assert expected_ids == failed_ids and len(expected_ids) == 36
old = registry.read_text()
parts = old.split('\n  {')
date = datetime.datetime.now(datetime.timezone.utc).date().isoformat()
changed = set()
for i, part in enumerate(parts[1:], 1):
    identity = re.search(r"\bid: '([^']+)'", part)
    if identity and identity[1] in expected_ids:
        assert part.count("lastVerifiedAt: '2026-09-06'") == 1
        parts[i] = part.replace("lastVerifiedAt: '2026-09-06'", "lastVerifiedAt: '" + date + "'")
        changed.add(identity[1])
assert changed == expected_ids
new = '\n  {'.join(parts)
assert old.count("lastVerifiedAt: '2026-09-06'") == len(changed)
assert new == old.replace("lastVerifiedAt: '2026-09-06'", "lastVerifiedAt: '" + date + "'")
manifest = read(ROOT / 'tests/acceptance/narrative-world/manifest-v1.json')
assert name not in manifest['sourceFingerprintPaths'], 'Review baseline binding before changing an actual measured input'
destination = ROOT / 'tests/acceptance/narrative-world/raw/skill-freshness-review-20261022-v1'
destination.mkdir()
items = []
def copy_file(source, relative):
    target = destination / relative
    assert not target.exists()
    shutil.copy2(source, target)
    assert sha(source) == sha(target)
    items.append({'path':str(target.relative_to(ROOT)), 'sha256':sha(target)})
copy_file(registry, 'skill-registry-before.ts.txt')
copy_file(RECOVERY / 'stale-skill-closure.json', 'stale-skill-closure.json')
copy_file(__file__, 'refresh_verified_skills.py.txt')
copy_file(BASE / 'delivery-continuation-v2-terminal.json', 'previous-continuation-terminal.json')
copy_file(BASE / 'delivery-continuation-v2-state.json', 'previous-continuation-state.json')
copy_file(BASE / 'delivery-continuation-v2.log', 'previous-continuation.log')
registry.write_text(new)
copy_file(registry, 'skill-registry-after.ts.txt')
report = {'schema':'storyforge.nw-skill-freshness-review', 'version':1,
          'reviewedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'scope':'Reverify the 36 expired Skill contracts after the full CI stopped at the unchanged 45-day gate.',
          'owner':'Existing product-specific Skill owners; no product, runtime, author-data or release ownership change.',
          'method':'Read registered read/write boundaries and their regression references; execute all nine referenced suites plus shared read-tool, budget and provider-contract regressions; retain raw logs before changing only verification dates.',
          'skills':sorted(changed), 'dateBefore':'2026-09-06', 'dateAfter':date,
          'sourceBeforeSha256':pin['files'][name], 'sourceAfterSha256':sha(registry),
          'targetedReceipt':str(receipt_path.relative_to(ROOT)), 'targetedReceiptSha256':sha(receipt_path),
          'originalFailureReceipt':'tests/acceptance/narrative-world/impact-sequence-delivery-v1-20261022.json',
          'maxVerificationAgeDays':45, 'freshnessCheckerUnchanged':True,
          'registeredReadsWritesOwnersAndPromptsUnchanged':True,
          'baselinePointerUnchanged':manifest['baseline']['path'],
          'baselineRegenerationRequired':False, 'reason':'The changed metadata file is outside the existing baseline input scope; no measured fingerprint is patched.',
          'actualModelCalls':0, 'humanEvaluations':0, 'semanticQualityAcceptance':False,
          'performanceAcceptance':False, 'fullDeliveryAcceptance':False,
          'files':items}
report_path = ROOT / 'tests/acceptance/narrative-world/skill-freshness-review-20261022-v1.json'
with report_path.open('x') as f:
    json.dump(report, f, ensure_ascii=False, indent=2)
    f.write('\n')
handoff = read('/tmp/nw-handoff-current-20261017.json')
assert not handoff['sourceFrozen']
handoff['freshnessReview'] = {'receipt':str(report_path), 'sha256':sha(report_path), 'skillCount':36,
                            'verifiedAt':date, 'status':'targeted-passed-metadata-refreshed-awaiting-delivery'}
Path('/tmp/nw-handoff-current-20261017.json').write_text(json.dumps(handoff, ensure_ascii=False, indent=2)+'\n')
print(json.dumps({'receipt':str(report_path), 'skillCount':36, 'sourceSha256':sha(registry)}))
