import type { NarrativeRecordV1, TypedValueV1, VisibilityV1 } from '../types/narrative-ir'
import type { NarrativeIndexV1 } from '../types/narrative-index'
import type { NarrativeTimeContextV1 } from '../types/narrative-time'
import type { WorldQueryGrantV1, WorldQueryItemV1, WorldQueryV1 } from '../types/world-query'
import { canonicalNarrativeJsonV1 } from '../world-engine/narrative-json'
import { sequenceOrdinalV1, timeAtOrBeforeV1 } from './world-query-time'

export function queryVisibilityAllowsV1(visibility: VisibilityV1, grant: WorldQueryGrantV1): boolean {
  if (grant.perspective.kind === 'author') return grant.purpose === 'author-review' || visibility.class !== 'unclassified'
  return visibility.class === 'public' || visibility.class === 'restricted' && visibility.audienceKeys.some(key => grant.audienceKeys.includes(key))
}
export function queryValueEntityKeysV1(value: TypedValueV1): string[] {
  return value.kind === 'entity-ref' ? [value.value.semanticId] : value.kind === 'ordered-set' ? value.values.flatMap(queryValueEntityKeysV1) : []
}

/** Only canonical, already verified records enter this pure projection.
 * Epistemic references are not read authority for the proposition's truth. */
export function buildWorldQueryVisibilityV1(input: {
  records: NarrativeRecordV1[]; index: NarrativeIndexV1; timeContext: NarrativeTimeContextV1
  query: WorldQueryV1; grant: WorldQueryGrantV1
  knowledgeProjectionVersion?: 1 | 2
}): { visible: Set<number>; required: Map<number, number[]>; atoms: Map<number, NonNullable<WorldQueryItemV1['atom']>> } {
  const { records, index, timeContext, query, grant } = input
  const positions = new Map(records.map((record, position) => [record.semanticId, position]))
  const allow = grant.allowedSemanticIds === null ? null : new Set(grant.allowedSemanticIds)
  const eligible = new Set(records.flatMap((record, position) => record.scope.branchKey === grant.branchKey
    && ['accepted', 'historical'].includes(record.reviewState) && record.modality !== 'hypothesis'
    && (allow === null || allow.has(record.semanticId)) && queryVisibilityAllowsV1(record.visibility, grant) ? [position] : []))
  const visible = new Set<number>(), required = new Map(index.indexes.dependencies.map(item => [item.recordIndex, [...item.requiredIndexes]]))
  const atoms = new Map<number, NonNullable<WorldQueryItemV1['atom']>>()
  const character = grant.perspective.kind === 'character' ? grant.perspective.characterKey : null
  const add = (position: number) => { if (eligible.has(position)) visible.add(position) }
  if (character) {
    for (const position of eligible) if (records[position].kind === 'entity' && records[position].modality === 'established') add(position)
    const known = index.indexes.epistemic.filter(item => eligible.has(item.recordIndex) && records[item.holderIndex].semanticId === character
      && item.acquiredAt !== null && timeAtOrBeforeV1(item.acquiredAt, grant.knowledgeThrough!, timeContext) === 'matched')
    const revised = new Set(known.flatMap(item => {
      const record = records[item.recordIndex]
      return record.kind === 'epistemic-event' && record.payload.revisesRef ? [record.payload.revisesRef.semanticId] : []
    }))
    for (const item of known) {
      const record = records[item.recordIndex]
      if (record.kind !== 'epistemic-event' || revised.has(record.semanticId)
        || !['acquire', 'revise'].includes(record.payload.transition) || record.payload.attitude === 'unaware') continue
      const proposition = records[item.propositionIndex]
      if (proposition.kind !== 'assertion') continue
      const independent = input.knowledgeProjectionVersion === 2
      const refs = [record.payload.holderRef.semanticId, ...(independent ? [] : [proposition.payload.subjectRef.semanticId]),
        ...queryValueEntityKeysV1(record.payload.assertedValue), ...record.dependencies.map(ref => ref.semanticId)]
      const dependencies = refs.map(key => positions.get(key)!)
      if (dependencies.some(position => !eligible.has(position))) continue
      required.set(item.recordIndex, [...new Set(dependencies)])
      const knowledge = { holderKey: character, assertedValue: structuredClone(record.payload.assertedValue),
        attitude: record.payload.attitude, transition: record.payload.transition }
      atoms.set(item.recordIndex, independent ? { ...knowledge, projectionVersion: 2 } : {
        ...knowledge, subjectKey: proposition.payload.subjectRef.semanticId, predicateKey: proposition.payload.predicateKey })
      add(item.recordIndex)
      if (eligible.has(item.propositionIndex) && record.payload.attitude === 'know'
        && canonicalNarrativeJsonV1(record.payload.assertedValue) === canonicalNarrativeJsonV1(proposition.payload.value)) add(item.propositionIndex)
      if (item.sourceEventIndex !== null) add(item.sourceEventIndex)
    }
  } else {
    for (const position of eligible) {
      const record = records[position]
      const mode = query.view === 'author-plan' ? record.modality === 'intent'
        : query.view === 'development' ? record.kind === 'development' && record.modality === 'established'
          : record.modality === 'established'
      if (mode) add(position)
    }
    // Dependencies of an explicitly requested author plan remain readable as
    // support, but are not themselves author-plan search hits.
    if (grant.perspective.kind === 'author' && ['author-plan', 'development'].includes(query.view)) {
      const pending = [...visible]
      while (pending.length) for (const target of required.get(pending.pop()!) ?? []) {
        if (eligible.has(target) && !visible.has(target) && records[target].modality === 'established') { visible.add(target); pending.push(target) }
      }
    }
    if (grant.perspective.kind === 'reader') {
      const cutoff = grant.disclosedThrough!, through = sequenceOrdinalV1(timeContext, cutoff.sequenceKey, cutoff.unitKey)
      const revealed = new Set<number>()
      for (const disclosure of index.indexes.disclosures) if (eligible.has(disclosure.recordIndex)
        && disclosure.audienceKeys.includes(cutoff.audienceKey) && disclosure.sequenceKey === cutoff.sequenceKey
        && sequenceOrdinalV1(timeContext, disclosure.sequenceKey, disclosure.unitKey) <= through) {
        disclosure.subjectIndexes.forEach(position => revealed.add(position)); revealed.add(disclosure.recordIndex)
      }
      for (const position of visible) {
        const record = records[position]
        const disclosed = record.time.disclosedAt.some(at => at.audienceKey === cutoff.audienceKey && at.sequenceKey === cutoff.sequenceKey
          && sequenceOrdinalV1(timeContext, at.sequenceKey, at.unitKey) <= through)
        if (record.kind !== 'entity' && !revealed.has(position) && !disclosed) visible.delete(position)
      }
    }
  }
  // Fixed point: a reference never exposes a hidden endpoint. This also
  // removes otherwise-public edges whose closure contains a private record.
  let changed = true
  while (changed) {
    changed = false
    for (const position of visible) if ((required.get(position) ?? []).some(target => !visible.has(target))) {
      visible.delete(position); changed = true
    }
  }
  for (const position of visible) if (!atoms.has(position)) atoms.set(position, structuredClone(records[position].payload))
  return { visible, required, atoms }
}
