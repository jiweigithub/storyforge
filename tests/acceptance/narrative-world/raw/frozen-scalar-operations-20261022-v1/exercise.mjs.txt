const clone = value => JSON.parse(JSON.stringify(value))
const ensure = (condition, message) => { if (!condition) throw new Error(message) }
const median = values => [...values].sort((a, b) => a - b)[Math.floor(values.length / 2)]

// Synthetic operation input only. These strings are not author material or a
// quality gold. Fixed counts and all operation orders are declared in plan.json.
export function makeFixture() {
  return {
    schema: 'storyforge.frozen-scalar-operation-fixture', version: 1,
    stamp: { sequence: 17, source: 'diagnostic-only' },
    rows: Array.from({ length: 512 }, (_, index) => ({
      id: `record-${index}`, owner: `owner-${index % 19}`, enabled: index % 3 !== 0,
      text: `${index}:` + '山路上的灯仍亮着。Astral 🧭 and 𐀀; quote " slash \\ newline\n'.repeat(24),
      confidence: (index % 101) / 101, tick: index * 256,
      stamp: { keepNested: true, source: `source-${index % 23}` },
      values: [null, true, false, index, -index - 1, 0.125, '中', '🧭', '\u0000', '\ue000'],
      spans: Array.from({ length: 17 }, (_, offset) => index * 31 + offset),
    })),
    numericRuns: Array.from({ length: 16 }, (_, row) =>
      Array.from({ length: 259 }, (_, col) => row * 8192 + col)),
    longProse: Array.from({ length: 4 }, (_, row) =>
      `${row}:` + '远处的山路🧭𐀀保留原样，正文与顺序都不改变。'.repeat(1537)),
    keyOrder: { '10': 10, '2': 2, '\ue000': 'private-use', '𐀀': 'astral', a: null },
  }
}

export async function exercise(variants, fixture, expected, plan, checkpoint) {
  const report = { startedAt: new Date().toISOString(), backend: {}, conformance: {}, samples: [], summaries: [] }
  try {
    const prepared = {}
    for (const variant of ['baseline', 'candidate']) {
      const api = variants[variant]
      const backend = await api.initializeNarrativeSha256V1()
      report.backend[variant] = api.narrativeSha256StatusV1()
      ensure(backend === 'wasm' && report.backend[variant].initialization === 'ready', `${variant}: expected explicit WASM initialization`)
      const mutable = clone(fixture), frozen = clone(fixture)
      const firstHash = await api.freezeAndHashNarrativeJsonV1(frozen)
      ensure(firstHash === expected.hash && Object.isFrozen(frozen) && Object.isFrozen(frozen.rows[0]), `${variant}: freeze/hash conformance`)
      const full = api.canonicalNarrativeJsonWithHashV1(frozen)
      ensure(full.json === expected.wire && full.hash === expected.hash, `${variant}: canonical conformance`)
      const inspected = api.inspectCanonicalNarrativeObjectV1(frozen, expected.wire, plan.omittedRootKeys)
      ensure(inspected.matches && inspected.projectionHash === expected.projectionHash, `${variant}: root projection conformance`)
      ensure(!api.inspectCanonicalNarrativeObjectV1(frozen, expected.wire + ' ', plan.omittedRootKeys).matches, `${variant}: mismatched wire must fail`)
      ensure(!Object.isFrozen(mutable) && !Object.isFrozen(mutable.rows[0]), `${variant}: mutable control`)
      prepared[variant] = { mutable, frozen }
      report.conformance[variant] = { matched: true, fullHash: firstHash, projectionHash: inspected.projectionHash, backend }
    }
    await checkpoint('conformance', { backend: report.backend, conformance: report.conformance })
    const check = (operation, value, variant) => {
      if (operation === 'frozen-wire-hash') {
        ensure(value.json === expected.wire && value.hash === expected.hash, `${variant}/${operation}: output mismatch`)
      } else if (operation === 'frozen-inspect') {
        ensure(value.matches && value.projectionHash === expected.projectionHash, `${variant}/${operation}: output mismatch`)
      } else ensure(value === expected.hash, `${variant}/${operation}: hash mismatch`)
    }
    const measure = async (operation, variant, pair, warmup) => {
      const api = variants[variant]
      const { frozen, mutable } = prepared[variant]
      // Allocation and input cloning precede the timer in every variant.
      const inputs = operation === 'first-freeze-hash'
        ? Array.from({ length: plan.repetitions }, () => clone(fixture)) : undefined
      const outputs = new Array(plan.repetitions)
      await checkpoint('before-sample', { operation, variant, pair, warmup })
      const startedAt = new Date().toISOString(), start = performance.now()
      for (let iteration = 0; iteration < plan.repetitions; iteration++) {
        switch (operation) {
          case 'frozen-hash': outputs[iteration] = await api.hashNarrativeValueV1(frozen); break
          case 'frozen-wire-hash': outputs[iteration] = api.canonicalNarrativeJsonWithHashV1(frozen); break
          case 'frozen-inspect': outputs[iteration] = api.inspectCanonicalNarrativeObjectV1(frozen, expected.wire, plan.omittedRootKeys); break
          case 'mutable-hash': outputs[iteration] = await api.hashNarrativeValueV1(mutable); break
          case 'first-freeze-hash': outputs[iteration] = await api.freezeAndHashNarrativeJsonV1(inputs[iteration]); break
          default: throw new Error(`Unknown declared operation: ${operation}`)
        }
      }
      const durationMs = performance.now() - start, endedAt = new Date().toISOString()
      for (const output of outputs) check(operation, output, variant)
      if (inputs) for (const input of inputs) ensure(Object.isFrozen(input) && Object.isFrozen(input.rows[0]), `${variant}: first freeze receipt missing`)
      ensure(!Object.isFrozen(mutable) && !Object.isFrozen(mutable.rows[0]), `${variant}: mutable control was frozen`)
      const sample = { operation, variant, pair, warmup, startedAt, endedAt, repetitions: plan.repetitions,
        durationMs, perOperationMs: durationMs / plan.repetitions, outputMatched: true }
      report.samples.push(sample)
      await checkpoint('sample', sample)
    }
    for (const operation of plan.operations) {
      for (const variant of ['baseline', 'candidate']) await measure(operation, variant, 0, true)
      for (let pair = 1; pair <= plan.pairs; pair++) {
        const order = pair % 2 === 1 ? ['baseline', 'candidate'] : ['candidate', 'baseline']
        for (const variant of order) await measure(operation, variant, pair, false)
      }
      const rows = report.samples.filter(row => row.operation === operation && !row.warmup)
      const baselineMedianMs = median(rows.filter(row => row.variant === 'baseline').map(row => row.perOperationMs))
      const candidateMedianMs = median(rows.filter(row => row.variant === 'candidate').map(row => row.perOperationMs))
      const pairedRatios = Array.from({ length: plan.pairs }, (_, index) => {
        const pair = index + 1
        return rows.find(row => row.pair === pair && row.variant === 'candidate').perOperationMs
          / rows.find(row => row.pair === pair && row.variant === 'baseline').perOperationMs
      })
      report.summaries.push({ operation, baselineMedianMs, candidateMedianMs,
        candidateToBaselineRatio: candidateMedianMs / baselineMedianMs,
        pairedRatios, fasterCandidatePairs: pairedRatios.filter(ratio => ratio < 1).length })
    }
    report.exitCode = 0
  } catch (error) { report.error = String(error?.stack ?? error); report.exitCode = 1 }
  report.endedAt = new Date().toISOString()
  return report
}
