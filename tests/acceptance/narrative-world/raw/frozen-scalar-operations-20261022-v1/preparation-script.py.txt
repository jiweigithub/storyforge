    for parent in [issuer, *issuer.parents]:

def collect_tree(path):
    for directory, dirs, files in os.walk(path):
        dirs[:] = [name for name in dirs if name not in ['node_modules', '.cache', '.vite']]
        for name in files:
            p = Path(directory) / name

def collect_package(path):
    path = path.resolve()
