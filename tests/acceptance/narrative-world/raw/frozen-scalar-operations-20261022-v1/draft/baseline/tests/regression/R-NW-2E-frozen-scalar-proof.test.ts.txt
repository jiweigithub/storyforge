import { createHash } from 'node:crypto'
import { describe, expect, it, vi } from 'vitest'
import {
  assertNarrativeJsonV1, boundedCanonicalNarrativeJsonV1, canonicalNarrativeJsonV1,
  canonicalNarrativeJsonWithHashV1, freezeAndHashNarrativeJsonV1, freezeNarrativeJsonV1,
  hashNarrativeValueV1, hashNarrativeValueWithoutKeysV1, inspectCanonicalNarrativeObjectV1,
} from '../../src/lib/world-engine/narrative-json'

const digest = (text: string) => createHash('sha256').update(text).digest('hex')
function reference(value: unknown, omitted: ReadonlySet<string> = new Set()): string {
  if (value === null || typeof value !== 'object') return JSON.stringify(value)
  if (Array.isArray(value)) return `[${value.map(item => reference(item, omitted)).join(',')}]`
  const compare = (left: string, right: string) => {
    const a = Array.from(left, char => char.codePointAt(0)!), b = Array.from(right, char => char.codePointAt(0)!)
    for (let index = 0; index < Math.min(a.length, b.length); index++) if (a[index] !== b[index]) return a[index] - b[index]
    return a.length - b.length
  }
  return `{${Object.keys(value).filter(key => !omitted.has(key)).sort(compare)
    .map(key => `${JSON.stringify(key)}:${reference((value as Record<string, unknown>)[key], omitted)}`).join(',')}}`
}

describe('NW independent draft · completed frozen JSON scalar proof', () => {
  it('retains complete long text, mixed scalar arrays and distinct recursive/root omission bytes', async () => {
    const value = {
      revisionHash: 'root',
      rows: Array.from({ length: 35 }, (_, row) => ({
        revisionHash: `nested-${row}`,
        text: `桥😀𐀀e\u0301\r\n"\\\u0000${row}`.repeat(140),
        mixed: [...Array.from({ length: 259 }, (_, index) => index), false, null, 1e-7, -2.5, Number.MAX_SAFE_INTEGER],
      })),
      keys: Object.fromEntries(['𐀀', '\ue000', '__proto__', '10', '2', 'a\u0000b'].map(key => [key, key])),
    }
    const wire = reference(value)
    freezeNarrativeJsonV1(value)
    expect(canonicalNarrativeJsonWithHashV1(value)).toEqual({ json: wire, hash: digest(wire) })
    expect(await hashNarrativeValueV1(value)).toBe(digest(wire))
    expect(await hashNarrativeValueWithoutKeysV1(value, ['revisionHash']))
      .toBe(digest(reference(value, new Set(['revisionHash']))))
    expect(inspectCanonicalNarrativeObjectV1(value, wire, ['revisionHash'])).toEqual({
      matches: true, projectionHash: digest(reference({ rows: value.rows, keys: value.keys })),
    })
    expect(boundedCanonicalNarrativeJsonV1(value, Buffer.byteLength(wire))).toBe(wire)
    expect(boundedCanonicalNarrativeJsonV1(value, Buffer.byteLength(wire) - 1)).toBeNull()
    expect(Reflect.set(value.rows[0].mixed, '0', -0)).toBe(false)
    expect(await hashNarrativeValueV1(value)).toBe(digest(wire))
  })

  it('caller freezing and array batching never authorize invalid scalars or omitted values', () => {
    const invalid = [undefined, NaN, Infinity, -Infinity, -0, Number.MAX_SAFE_INTEGER + 1, '\ud800', '\udfff', 1n, Symbol('invalid')]
    const checks = [
      (value: unknown) => assertNarrativeJsonV1(value),
      (value: unknown) => canonicalNarrativeJsonV1(value),
      (value: unknown) => hashNarrativeValueV1(value),
      (value: unknown) => hashNarrativeValueWithoutKeysV1(value, ['hidden']),
      (value: unknown) => freezeAndHashNarrativeJsonV1(value, ['hidden']),
    ]
    for (const scalar of invalid) for (const check of checks) {
      const fields = Object.freeze({ keep: 'valid', hidden: Object.freeze({ value: scalar }) })
      const array = Object.freeze([...Array.from({ length: 35 }, (_, index) => index), scalar])
      expect(() => check(fields)).toThrow()
      expect(() => check(Object.freeze({ keep: 'valid', hidden: array }))).toThrow()
    }
  })

  it('a completed child cannot authorize its failed parent, a clone or a mutable sibling', () => {
    const child = { text: '已校验子节点', values: [1, 2, 3] }
    const failed = { a: child, z: { value: '\ud800' } }
    expect(() => freezeNarrativeJsonV1(failed)).toThrow('unicode')
    expect(canonicalNarrativeJsonV1(child)).toBe(reference(child))
    expect(() => hashNarrativeValueWithoutKeysV1(failed, ['z'])).toThrow('unicode')
    const clone = structuredClone(child)
    clone.values[0] = -0
    expect(() => hashNarrativeValueV1(clone)).toThrow('json')
    const sibling = { a: child, z: { value: 'valid' } }
    expect(canonicalNarrativeJsonV1(sibling)).toBe(reference(sibling))
    sibling.z.value = '\ud800'
    expect(() => canonicalNarrativeJsonV1(sibling)).toThrow('unicode')
  })

  it('never executes accessors or accepts holes, symbols or cycles beside a receipted subtree', () => {
    const good = freezeNarrativeJsonV1({ text: 'valid', values: [1, 2, 3] })
    let getterCalls = 0
    const accessor = Object.defineProperty({}, 'value', { enumerable: true, get() { getterCalls++; return 'bad' } })
    const holes = new Array(40)
    const symbols = { [Symbol('private')]: 1 }
    const cycle: { self?: unknown } = {}; cycle.self = cycle
    for (const bad of [accessor, holes, symbols, cycle, new Date()]) {
      const value = Object.freeze({ a: good, hidden: bad })
      expect(() => canonicalNarrativeJsonV1(value)).toThrow()
      expect(() => hashNarrativeValueWithoutKeysV1(value, ['hidden'])).toThrow()
    }
    expect(getterCalls).toBe(0)
  })

  it('reentrant hashing keeps immutable descendant scalars and active sinks independent', async () => {
    const outer = freezeNarrativeJsonV1({ text: '桥😀'.repeat(20000), rows: Array.from({ length: 260 }, (_, index) => index) })
    const inner = freezeNarrativeJsonV1({ text: '内层𐀀', rows: [1, 2, 3] })
    const expectedOuter = digest(reference(outer)), expectedInner = digest(reference(inner))
    const encodeInto = TextEncoder.prototype.encodeInto
    let entered = false, nested: Promise<string> | undefined
    const spy = vi.spyOn(TextEncoder.prototype, 'encodeInto').mockImplementation(function (this: TextEncoder, text, destination) {
      if (!entered) {
        entered = true
        expect(Reflect.set(outer.rows, '0', NaN)).toBe(false)
        nested = hashNarrativeValueV1(inner)
      }
      return encodeInto.call(this, text, destination)
    })
    try {
      expect(await hashNarrativeValueV1(outer)).toBe(expectedOuter)
      expect(entered).toBe(true)
      expect(await nested).toBe(expectedInner)
    } finally { spy.mockRestore() }
  })
})
