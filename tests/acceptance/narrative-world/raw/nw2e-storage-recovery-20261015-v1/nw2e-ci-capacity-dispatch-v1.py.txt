from pathlib import Path
import datetime, json, shutil, subprocess, sys

generation = sys.argv[1]
assert generation in ('r7', 'r8')
root = Path('/Users/qinyingying/.codex/worktrees/nw2e-verification-' + generation + '/storyforge')
assert Path.cwd() == root
free = shutil.disk_usage(root).free
record = {'schema': 'storyforge.ci-storage-preflight', 'version': 1, 'generation': generation,
          'createdAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'availableBytes': free, 'minimumBytes': 5 * 1024**3,
          'modelCalls': 0, 'testRunnerStarted': False, 'accepted': False}
if free >= record['minimumBytes']:
    record['testRunnerStarted'] = True
    code = subprocess.call(['python3', '/tmp/nw2e-run-frozen-' + generation + '.py', 'ci'])
else:
    print('Insufficient free space for full CI; frozen runner not dispatched.', flush=True)
    code = 2
record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
Path('/tmp/nw2e-' + generation + '-ci-capacity-r2-terminal.json').write_text(json.dumps(record, indent=2) + '\n')
sys.exit(code)
