from pathlib import Path
import datetime, json, subprocess, sys

generation = sys.argv[1]
assert generation in ('r7', 'r8')
root = Path('/Users/qinyingying/.codex/worktrees/nw2e-verification-' + generation + '/storyforge')
log = Path('/tmp/nw2e-full-ci-frozen-' + generation + '-launch-r2.log')
record = {'schema': 'storyforge.ci-launch-terminal', 'version': 1, 'generation': generation,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'log': str(log),
          'reason': 'Prior shared-gate launch failed before any test with ENOSPC. Same frozen source; no model retry.'}
with log.open('x') as stream:
    code = subprocess.call(['python3', '/tmp/storyforge-serial-ci-20261013.py', '--',
                            'python3', '/tmp/nw2e-ci-capacity-dispatch-v1.py', generation],
                           cwd=root, stdout=stream, stderr=subprocess.STDOUT)
record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat(),
              frozenCITerminalExists=Path('/tmp/nw2e-' + generation + '-ci-terminal.json').exists())
Path('/tmp/nw2e-' + generation + '-ci-launch-r2-terminal.json').write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record), flush=True)
sys.exit(code)
