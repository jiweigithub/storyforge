from pathlib import Path
import datetime, hashlib, json, os, shutil, socket, subprocess, sys

root = Path('/Users/qinyingying/.codex/worktrees/nw2e-verification-r9/storyforge')
manifest_path = Path('/tmp/nw2e-frozen-files-r9.json')
manifest = json.loads(manifest_path.read_text())
mode = sys.argv[1]
assert mode in ('ci', 'e2e')
terminal = Path('/tmp/nw2e-r9-' + mode + '-terminal.json')
assert not terminal.exists()

def verify():
    assert Path.cwd() == root
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip() == manifest['baseCommit']
    changed = [p for p, h in manifest['files'].items() if not (root / p).is_file()
               or hashlib.sha256((root / p).read_bytes()).hexdigest() != h]
    assert not changed, 'Frozen source drift: ' + json.dumps(changed)
    paths = set(subprocess.check_output(['git', 'diff', '--name-only', '-z']).decode().split('\0'))
    paths.update(subprocess.check_output(['git', 'ls-files', '--others', '--exclude-standard', '-z']).decode().split('\0'))
    paths.discard('')
    assert paths == set(manifest['files']), 'Frozen changed path set drift'

code = 2
record = {'schema': 'storyforge.frozen-run-terminal', 'version': 1, 'mode': mode,
          'root': str(root), 'manifest': str(manifest_path),
          'manifestSha256': hashlib.sha256(manifest_path.read_bytes()).hexdigest(),
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'sourceCount': len(manifest['files']), 'sourceUnchanged': False}
try:
    verify()
    record['freeBytesAtAdmission'] = shutil.disk_usage(root).free
    if record['freeBytesAtAdmission'] < 5 * 1024**3:
        raise RuntimeError('Insufficient free space before validation; no test process dispatched')
    env = dict(os.environ, CI='1')
    if mode == 'e2e':
        with socket.socket() as sock:
            sock.bind(('127.0.0.1', 0))
            port = sock.getsockname()[1]
        env.update(PLAYWRIGHT_FROZEN_WORKSPACE='1', PLAYWRIGHT_PORT=str(port),
                   PLAYWRIGHT_HTML_OUTPUT_DIR='/tmp/nw2e-r9-full-e2e-html')
        record['isolatedPort'] = port
    print('[nw2e-r9] Frozen ' + str(len(manifest['files'])) + ' files verified; starting ' + mode, flush=True)
    code = subprocess.call(['npm', 'run', 'ci:e2e' if mode == 'e2e' else 'ci'], env=env)
    verify()
    record['sourceUnchanged'] = True
except BaseException as error:
    record['error'] = type(error).__name__ + ': ' + str(error)
    code = 2
finally:
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    terminal.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
    print('[nw2e-r9] ' + mode + ' exit ' + str(code) + '; frozen sources unchanged: ' + str(record['sourceUnchanged']), flush=True)
sys.exit(code)
