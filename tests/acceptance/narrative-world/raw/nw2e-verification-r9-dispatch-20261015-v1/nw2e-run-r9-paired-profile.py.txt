from pathlib import Path
import datetime, hashlib, json, os, shutil, socket, subprocess, sys

entries = [('before', 'r8', 100000, True), ('after', 'r9', 100000, True), ('after-million', 'r9', 1000000, False)]
results = []
record = {'schema': 'storyforge.paired-profile-terminal', 'version': 1,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'runs': results,
          'accepted': False, 'modelCalls': 0,
          'boundary': 'Serialized single diagnostics; historical compiler-v1 scale fixture; not formal repeated-performance or compiler-v2 scale acceptance.'}
terminal = Path('/tmp/nw2e-r9-paired-profile-terminal.json')
assert not terminal.exists()
code = 2
try:
    for label, generation, size, profile in entries:
        root = Path('/Users/qinyingying/.codex/worktrees/nw2e-verification-' + generation + '/storyforge')
        manifest = Path('/tmp/nw2e-frozen-files-' + generation + '.json')
        frozen = json.loads(manifest.read_text())
        def verify():
            assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip() == frozen['baseCommit']
            for path, expected in frozen['files'].items():
                assert hashlib.sha256((root / path).read_bytes()).hexdigest() == expected, path
            paths = set(subprocess.check_output(['git', 'diff', '--name-only', '-z'], cwd=root).decode().split('\0'))
            paths.update(subprocess.check_output(['git', 'ls-files', '--others', '--exclude-standard', '-z'], cwd=root).decode().split('\0'))
            paths.discard('')
            assert paths == set(frozen['files']), 'Frozen changed path set drift'
        verify()
        free_bytes = shutil.disk_usage(root).free
        if free_bytes < 5 * 1024**3:
            raise RuntimeError('Insufficient space before performance diagnostic; no browser dispatched')
        report = Path('/tmp/nw2e-r9-profile-' + label + '.json')
        assert not report.exists()
        with socket.socket() as sock:
            sock.bind(('127.0.0.1', 0))
            port = str(sock.getsockname()[1])
        env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1', PLAYWRIGHT_PORT=port,
                   PLAYWRIGHT_HTML_OUTPUT_DIR='/tmp/nw2e-r9-profile-' + label + '-html',
                   NW_SCALE_BROWSER_MODE='diagnostic', NW_SCALE_BROWSER_SIZES=str(size),
                   NW_SCALE_BROWSER_REPORT=str(report),
                   NW_SCALE_BROWSER_SOURCES='/tmp/nw2e-r9-profile-' + label + '-sources.json')
        if profile:
            env['NW_SCALE_BROWSER_PROFILE_DIR'] = '/tmp/nw2e-r9-profile-' + label + '-cpu'
        print('Starting ' + label + ' diagnostic on ' + generation + ', characters=' + str(size), flush=True)
        code = subprocess.call(['npx', 'playwright', 'test', 'tests/e2e/narrative-scale.spec.ts', '--reporter=line',
                                '--retries=0', '--output=/tmp/nw2e-r9-profile-' + label + '-test-results'], cwd=root, env=env)
        verify()
        result = {'label': label, 'generation': generation, 'size': size, 'profiled': profile,
                  'exitCode': code, 'report': str(report), 'sourceUnchanged': True,
                  'manifestSha256': hashlib.sha256(manifest.read_bytes()).hexdigest(), 'freeBytesAtAdmission': free_bytes}
        results.append(result)
        if code:
            break
        assert report.exists(), 'Browser command returned without its measurement receipt'
        measured = json.loads(report.read_text())
        assert measured['sourceUnchanged'] is True
        assert measured['mode'] == 'diagnostic' and measured['modelCalls'] == 0
        run = measured['runs'][0]
        result['identity'] = {'inputHash': run['prepared']['inputHash'],
                              'contentHash': run['compilation']['contentHash'], 'indexHash': run['compilation']['indexHash'],
                              'packageHash': run['export']['packageHash']}
        if label == 'after':
            assert result['identity'] == results[0]['identity'], 'Before/after frozen input or output byte identity differs'
            record['beforeAfterIdentitiesEqual'] = True
except BaseException as error:
    record['error'] = type(error).__name__ + ': ' + str(error)
    code = 2
finally:
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    with terminal.open('x') as stream:
        json.dump(record, stream, ensure_ascii=False, indent=2)
        stream.write('\n')
sys.exit(code)
