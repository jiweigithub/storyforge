import { afterEach, expect, it } from 'vitest'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { db } from '../../src/lib/db/schema'
import { createWorkspace } from '../../src/lib/workspace/create-workspace'
import { verifyWorldReleaseRecordV1 } from '../../src/lib/world-engine/release-codec'
import { irReaderContext } from '../../tests/helpers/narrative-reader-evaluation'
import { readerMessages } from '../../scripts/narrative-reader-contract.mjs'

const base = 'tests/acceptance/narrative-world/'
const inputPath = base + 'reader-inputs-20261015-v6.json'
const judgeBase = base + 'raw/reader-incomplete-judge-20261016-v2/calls/345f88c9bb4471175cacd08e25e8394a437781cf73f3f088a7421d37a246ccac'
const protocolPath = base + 'reader-protocol-20261015-v4.json'
const hash = (path: string) => createHash('sha256').update(readFileSync(path)).digest('hex')
const read = (path: string) => JSON.parse(readFileSync(path, 'utf8'))
const checkpoint = read(base + 'engineering-checkpoint-20261016-v32.json')
const evidenceHashes = Object.fromEntries([inputPath, protocolPath, judgeBase + '.request.json', judgeBase + '.result.json'].map(path => [path, hash(path)]))
const report: Record<string, any> = { schema: 'storyforge.frozen-reader-title-replay', version: 1,
  boundary: 'One historical frozen v4 release replayed through the current registered provider and existing reader adapter, in isolated fake IndexedDB. No new model call, no original response editing, no human or complete semantic acceptance.',
  inputPath, sourceHashes: checkpoint.sourceHashes, evidenceHashes, currentProductionVersion: 31,
  completed: false, modelCalls: 0, accepted: false }
afterEach(() => db.close())

it('the actual theater snapshot now supplies its original title without rewriting frozen release or historical model bytes', async () => {
  const verifyPins = () => {
    for (const [path, expected] of Object.entries({ ...checkpoint.sourceHashes, ...evidenceHashes })) expect(hash(path), path).toBe(expected)
  }
  try {
    verifyPins()
    const inputs = read(inputPath), family = inputs.families.find((row: any) => row.familyId === 'theater-promise')
    const original = family.cases.find((row: any) => row.id === 'theater-promise:q01/ir-lexical')
    const saved = family.releaseSnapshots, release = saved.v4, judge = read(judgeBase + '.result.json')
    expect(judge.status).toBe('completed'); expect(judge.judgment.assessment).toBe('partly-correct')
    expect(original.context.documents.map((row: any) => row.title)).toEqual(['已确认揭示'])
    expect(original.question.holderId).toBeNull(); expect(original.question.throughId).toBeNull(); expect(original.question.sourceTitle).toBeNull()
    await db.delete(); await db.open()
    const created = await createWorkspace({ name: '冻结读者标题复核', genres: [], description: '隔离诊断', status: 'drafting', targetWordCount: 1000, enableMultiWorld: false },
      { purpose: 'world-engine', kind: 'novel', novelProfile: 'long' })
    expect(created.scope).toMatchObject(saved.scope)
    await db.worlds.update(created.scope.worldId, { code: release.sourceWorldCode })
    await db.worldRevisions.add({ id: release.revisionId, projectId: release.projectId, worldId: release.worldId,
      revision: release.version, parentRevisionId: null, label: release.label, manifestJson: release.manifestJson,
      contentHash: release.contentHash, createdAt: release.createdAt, updatedAt: release.createdAt })
    await db.worldReleases.add(release)
    const verified = await verifyWorldReleaseRecordV1(release)
    if (verified.version !== 4 || !verified.index) throw Error('Historical release did not verify as v4')
    const uidToSource = new Map<string, string>(Object.entries(saved.logicalSourcesByUid))
    const keyToLogical = (key: string) => [...uidToSource].find(([uid]) => key.endsWith(`:${uid}`))?.[1] ?? null
    const fixture = { scope: saved.scope, v4Release: release, projection: verified.manifest.narrative, uidToSource, keyToLogical,
      semanticFor: (id: string) => verified.manifest.narrative.bundle.records.find(record => keyToLogical(record.provenance.sourceResourceKey) === id)?.semanticId,
      byLogical: new Map() } as Parameters<typeof irReaderContext>[0]
    const current = await irReaderContext(fixture, original.question, 'ir-lexical')
    expect(current.status).toBe('matched'); expect(current.documents).toHaveLength(1)
    const expectedTitle = '已确认揭示｜封存来源：空弦剧场·记录1'
    expect(current.documents[0].title).toBe(expectedTitle)
    expect(current.documents[0].authorSource?.title).toBe(original.question.search)
    expect(current.documents[0].id).toBe(original.context.documents[0].id)
    expect(current.documents[0].evidence).toEqual(original.context.documents[0].evidence)
    const protocol = read(protocolPath), before = readerMessages(original, protocol)
    const after = readerMessages({ ...original, context: current }, protocol)
    const beforePayload = JSON.parse(before.messages[1].content), afterPayload = JSON.parse(after.messages[1].content)
    expect(afterPayload.documents[0].title).toBe(expectedTitle)
    const normalized = structuredClone(afterPayload); normalized.documents[0].title = beforePayload.documents[0].title
    expect(normalized).toEqual(beforePayload)
    expect(after.promptHash).not.toBe(before.promptHash)
    expect(await db.worldReleases.get(release.id)).toEqual(release)
    verifyPins()
    Object.assign(report, { completed: true, caseId: original.id, sourceRelease: { releaseUid: release.releaseUid, contentHash: release.contentHash,
      manifestSha256: createHash('sha256').update(release.manifestJson).digest('hex'), indexHash: verified.index.indexHash },
      originalTitle: beforePayload.documents[0].title, currentTitle: afterPayload.documents[0].title,
      originalReaderPromptHash: before.promptHash, currentReaderPromptHash: after.promptHash,
      modelPayloadChangedOnlyInTitle: true, originalEvidenceAndSemanticIdPreserved: true, frozenReleaseBytesPreserved: true,
      judge: { assessment: judge.judgment.assessment, score: judge.judgment.answerCorrectness, reason: judge.judgment.reason }, currentContext: current })
  } catch (error) { report.error = String(error); throw error }
  finally { writeFileSync('/tmp/nw-reader-title-replay-20261016-v1.json', JSON.stringify(report, null, 2) + '\n', { flag: 'wx' }) }
}, 30_000)
