from pathlib import Path
import datetime, fcntl, hashlib, json, os, re, shutil, subprocess, sys, time, traceback

BASE = Path('/tmp/nw-impact-sequence-20261022-v1')
ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
STATE = BASE / 'delivery-continuation-v2-state.json'
TERMINAL = BASE / 'delivery-continuation-v2-terminal.json'

def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def read(path): return json.loads(Path(path).read_text())
def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def save(path, data):
    tmp = Path(str(path) + '.tmp')
    tmp.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    os.replace(tmp, path)
def alive(pid):
    try: os.kill(pid, 0); return True
    except ProcessLookupError: return False

record = {'schema':'storyforge.nw-authorized-delivery-continuation', 'version':1,
          'startedAt':now(), 'pid':os.getpid(), 'modelCalls':0, 'automaticRetries':0,
          'thresholdChanges':False, 'goldChanges':False, 'steps':[]}

def stage(name, **data):
    record.update(stage=name, updatedAt=now(), **data)
    save(STATE, record)
    h = read(HANDOFF)
    h['deliveryContinuation'] = {'pid':os.getpid(), 'script':__file__, 'state':str(STATE),
                                 'terminal':str(TERMINAL), 'stage':name, 'updatedAt':now()}
    save(HANDOFF, h)
    print(json.dumps({'stage':name, **data}, ensure_ascii=False), flush=True)

def wait_validation(label):
    directory = BASE / label
    terminal = directory / 'validation-queue-r1-terminal.json'
    missing = 0
    while not terminal.exists():
        h = read(HANDOFF)
        pid = h['durableQueue']['supervisorPid']
        missing = 0 if alive(pid) else missing + 1
        if missing >= 3:
            raise RuntimeError('Validation supervisor disappeared without a terminal; preserve unknown outcome, do not resubmit.')
        time.sleep(10)
    queue = read(terminal)
    result_path = directory / 'validation-terminal.json'
    if not result_path.exists():
        raise RuntimeError('Queue ended without a runner terminal; inspect dispatch evidence before any retry.')
    result = read(result_path)
    pids = set(result['observedPids']) | {queue['queuePid'], queue['supervisorPid']}
    deadline = time.monotonic() + 60
    while any(alive(pid) for pid in pids):
        if time.monotonic() >= deadline:
            raise RuntimeError('Terminal exists but an observed PID still exists; preserve evidence for identity inspection.')
        time.sleep(2)
    assessment = ('All declared phases completed. This is engineering evidence; performance, model, semantic and human acceptance remain separate.'
                  if result['exitCode'] == 0 else
                  'Validation failed; preserve the original failure and stop this continuation. No automatic rerun or standard changes.')
    subprocess.run([sys.executable, str(BASE/'archive-delivery-validation-v2.py'), label, assessment], check=True)
    record['steps'].append({'label':label, 'exitCode':result['exitCode'], 'terminal':str(result_path)})
    if result['exitCode'] != 0:
        raise RuntimeError('Validation failed in '+label+'; archived before further source edits.')
    return result

def launch(label, plan):
    assert not read(HANDOFF)['sourceFrozen']
    assert not (BASE/label).exists(), 'This attempt already exists; never dispatch twice.'
    subprocess.run([sys.executable, str(BASE/'prepare-delivery-validation-v2.py'), label, str(BASE/plan)], check=True)

def main():
    with (BASE/'delivery-continuation-v2.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        assert not STATE.exists() and not TERMINAL.exists(), 'Recovery already started; inspect its actual outcome'
        assert not (BASE/'baseline-v78').exists() and not (BASE/'delivery-v1').exists(), 'Validation already exists'
        manifest_path = ROOT/'tests/acceptance/narrative-world/manifest-v1.json'
        target = ROOT/'tests/acceptance/narrative-world/baseline-20261022-nw2d-v78.json'
        assert read(manifest_path)['baseline']['path'] == str(target.relative_to(ROOT)), 'Final pointer must precede measurement'
        stage('baseline-v78-self-fingerprint')
        launch('baseline-v78', 'baseline-v78-plan.json')
        wait_validation('baseline-v78')
        stage('reviewing-actual-baseline-v78')
        assert not read(HANDOFF)['sourceFrozen'], 'Archive must release pin before exact-byte promotion'
        generated = BASE/'baseline-v78/baseline-v78.json'
        command = [sys.executable, '/tmp/nw-continuation-recovery-20261022-v1/baseline_promotion.py', str(ROOT),
                   str(manifest_path), str(ROOT/'tests/acceptance/narrative-world/baseline-20261021-nw2e-v76.json'),
                   str(generated), str(target)]
        promotion = json.loads(subprocess.check_output(command, text=True))
        save(BASE/'baseline-v78-promotion.json', {'at':now(), 'command':command, **promotion})
        h = read(HANDOFF)
        h['latestBaseline'] = {'path':str(target), 'sha256':sha(target), **promotion}
        h['currentPhase'] = 'sequence-repair-delivery-validation'
        save(HANDOFF, h)
        stage('full-ci-and-full-isolated-e2e')
        launch('delivery-v1', 'delivery-v1-plan.json')
        wait_validation('delivery-v1')
        stage('delivery-validation-archived')

try:
    main()
    record['exitCode'] = 0
except BaseException as error:
    record.update(exitCode=1, error=type(error).__name__+': '+str(error), traceback=traceback.format_exc())
    stage('needs-evidence-review', error=record['error'])
finally:
    record['endedAt'] = now()
    if not TERMINAL.exists(): save(TERMINAL, record)
    print(json.dumps({'exitCode':record.get('exitCode',1), 'terminal':str(TERMINAL)}), flush=True)
sys.exit(record.get('exitCode',1))
