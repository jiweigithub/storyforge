def observe():
    rows={int(parts[0]):parts for line in subprocess.check_output(['ps','-axo','pid=,ppid=,command='],text=True).splitlines() if len(parts:=line.strip().split(None,2))==3}
    family={os.getpid()}
    while True:
        children={pid for pid,parts in rows.items() if int(parts[1]) in family}
        if children<=family:break
        family|=children
    if family-owned:
        with Path(str(BASE)+'-process-observations.jsonl').open('a') as stream:
            stream.write(json.dumps({'at':now(),'pids':sorted(family),'processes':[rows[pid] for pid in sorted(family) if pid in rows]})+'\n')
        owned.update(family)
