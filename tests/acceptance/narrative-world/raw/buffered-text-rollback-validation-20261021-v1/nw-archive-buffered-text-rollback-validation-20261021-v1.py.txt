from pathlib import Path
import datetime, hashlib, json, os, shutil, subprocess

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
BASE = Path('/tmp/nw-buffered-text-rollback-validation-20261021-v1')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
original = HANDOFF.read_bytes()
h = json.loads(original)
q = h['durableQueue']
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()
def write_new(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

pin_path = Path(str(BASE) + '-sources.json')
pin = json.loads(pin_path.read_text())
terminal_path = Path(str(BASE) + '-terminal.json')
terminal = json.loads(terminal_path.read_text())
queue_terminal = json.loads(Path(q['terminal']).read_text())
runtime_path = Path(str(BASE) + '-runtime-sources.json')
runtime = json.loads(runtime_path.read_text())
plan_path = ROOT / h['validationPlan']
plan = json.loads(plan_path.read_text())
assert h['currentRunner'] == str(BASE) + '.py'
assert terminal['exitCode'] == queue_terminal['exitCode'] == 0
assert terminal['sourceUnchanged'] and terminal['runtimeUnchanged']
assert queue_terminal['runnerTerminalPresent']
assert sha(pin_path) == h['currentPinSha256'] == terminal['pinSha256']
assert sha(h['currentRunner']) == pin['runnerSha256']
assert sha(str(BASE) + '-scale-sources.json') == pin['scaleInputSha256']
assert sha(runtime_path) == pin['runtimeInputSha256']
assert sha(q['log']) == queue_terminal['logSha256']
assert all(sha(ROOT / path) == digest for path, digest in pin['files'].items())
assert all(sha(path) == digest for path, digest in runtime['files'].items())
for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
    assert subprocess.check_output(['git', 'rev-parse', ref], cwd=ROOT, text=True).strip() == pin[key]
for phase in terminal['phases']:
    assert phase['exitCode'] == 0 and sha(phase['log']) == phase['logSha256']
assert len(terminal['phases']) == 12
assert all(sha(path) == digest for path, digest in terminal['generatedReports'].items())
baseline = terminal['observationComparisons']['baseline']
assert sha(baseline['previous']) == baseline['previousSha256']
assert sha(baseline['current']) == baseline['currentSha256']
current = json.loads(Path(baseline['current']).read_text())
previous = json.loads(Path(baseline['previous']).read_text())
semantic = lambda obj: {key: value for key, value in obj.items() if key not in {'input', 'run'}}
assert semantic(current) == semantic(previous)
assert all(pin['files'][path] == digest for path, digest in current['input']['hashes'].items())
manifest = json.loads((ROOT / 'tests/acceptance/narrative-world/manifest-v1.json').read_text())
assert manifest['baseline']['path'] == plan['baselinePlanned']
assert sha(ROOT / 'src/lib/narrative-domain/json.ts') == plan['restoredSourceSha256']
assert sha(ROOT / 'tests/regression/R-NW-2E-byte-sink.test.ts') == plan['retainedTestSha256']
directed_log = Path(str(BASE) + '-directed-regression.log').read_text()
assert '31 passed (31)' in directed_log and '314 passed | 1 skipped (315)' in directed_log
for name in ['baseline-generate', 'baseline-default-readback']:
    assert '25 passed (25)' in Path(str(BASE) + '-' + name + '.log').read_text()

pids = set(terminal['observedPids']) | {queue_terminal['supervisorPid'], queue_terminal['queuePid'], terminal['runnerPid']}
pids.update(phase['childPid'] for phase in terminal['phases'])
for line in Path(str(BASE) + '-process-observations.jsonl').read_text().splitlines():
    pids.update(json.loads(line)['pids'])
ps = subprocess.run(['ps', '-p', ','.join(map(str, sorted(pids))), '-o', 'pid=,ppid=,state=,command='], capture_output=True, text=True)
assert ps.returncode == 1 and not ps.stdout.strip(), ps.stdout
rows = {int(parts[0]): parts for line in subprocess.check_output(['ps', '-axo', 'pid=,ppid=,command='], text=True).splitlines()
        if len(parts := line.strip().split(None, 2)) == 3}
ancestors = {os.getpid()}
pid = os.getpid()
while pid in rows and int(rows[pid][1]) not in ancestors:
    pid = int(rows[pid][1])
    ancestors.add(pid)
cwd_scan = subprocess.run(['lsof', '-nP', '-d', 'cwd', '-Fpn'], capture_output=True, text=True, timeout=15)
cwd_pids = []
pid = None
for line in cwd_scan.stdout.splitlines():
    if line.startswith('p'):
        pid = int(line[1:])
    elif line.startswith('n') and line[1:] == str(ROOT):
        cwd_pids.append(pid)
unexpected = [pid for pid in cwd_pids if pid in rows and pid not in ancestors]
command_matches = [parts for pid, parts in rows.items() if pid not in ancestors
                   and (str(BASE) in parts[2] or str(ROOT) in parts[2])]
assert not unexpected, [rows[pid] for pid in unexpected]
assert not command_matches, command_matches
proof = {'schema': 'storyforge.narrative-validation-owned-exit', 'version': 1, 'checkedAt': now(),
    'pids': sorted(pids), 'allAbsent': True, 'sourceUnchanged': True, 'runtimeUnchanged': True,
    'sourceFiles': len(pin['files']), 'runtimeFiles': len(runtime['files']), 'pinSha256': sha(pin_path),
    'head': pin['head'], 'mergeHead': pin['mergeHead'], 'terminalSha256': sha(terminal_path),
    'queueTerminalSha256': sha(q['terminal']), 'workspaceProcessesRemaining': [], 'commandMatchesRemaining': [],
    'workspaceScanExcludes': 'This evidence writer and its ancestors; no foreign processes were signaled.',
    'limitation': 'PID evidence covers recorded wrappers, runner, phase children and observed descendants; independent cwd and command scans found no remaining owned test processes.'}
write_new(str(BASE) + '-owned-exit.json', proof)

dest = ROOT / 'tests/acceptance/narrative-world/raw/buffered-text-rollback-validation-20261021-v1'
assert not dest.exists()
dest.mkdir()
artifacts = []
files = [p for p in sorted(BASE.parent.glob(BASE.name + '*')) if p.is_file()
         and not any(part in p.name for part in ['-heartbeat-', '-queue-r1.lock', '-archived-handoff'])]
files.extend([Path(__file__), plan_path, Path(baseline['current'])])
for source in files:
    target = dest / source.name
    if source.suffix in {'.py', '.js', '.mjs', '.cjs', '.ts', '.tsx'}:
        target = Path(str(target) + '.txt')
    assert not target.exists()
    shutil.copy2(source, target)
    assert sha(source) == sha(target)
    artifacts.append({'path': str(target.relative_to(ROOT)), 'sha256': sha(target), 'bytes': target.stat().st_size})
receipt = dict(terminal)
receipt.update(schema='storyforge.narrative-buffered-text-rollback-validation-receipt', archivedAt=now(),
    artifacts=artifacts, exitProof=proof,
    validationSummary={'directedFilesPassed': 31, 'directedTestsPassed': 314, 'directedTestsSkipped': 1,
        'baselineGenerationTestsPassed': 25, 'defaultBaselineReadbackTestsPassed': 25, 'baselineSemanticObservationsUnchanged': True,
        'changedFileLint': True, 'types': True, 'governanceChecks': ['narrative-world', 'architecture', 'required-tables', 'ai-manual', 'docs'],
        'build': True, 'performanceRun': False, 'performanceAcceptance': False, 'fullCI': False, 'fullE2E': False},
    rawEvidenceExclusions='Foreign queue heartbeat observations and lock files remain preserved in /tmp; not copied into product evidence.',
    next='Keep short-text buffering reverted. Continue evidence-supported performance work; formal three-size performance, quality and final same-source full CI/E2E remain open.')
receipt_path = ROOT / 'tests/acceptance/narrative-world/buffered-text-rollback-validation-result-20261021-v1.json'
write_new(receipt_path, receipt)
assert all(sha(ROOT / item['path']) == item['sha256'] for item in artifacts)
verification = {'schema': 'storyforge.narrative-archive-verification', 'version': 1, 'verifiedAt': now(),
    'receipt': str(receipt_path.relative_to(ROOT)), 'receiptSha256': sha(receipt_path),
    'artifactsVerified': len(artifacts), 'allMatch': True, 'sourceFilesUnchanged': len(pin['files']),
    'runtimeFilesUnchanged': len(runtime['files']), 'knownPidsAbsent': sorted(pids),
    'generatedBaselineSha256': baseline['currentSha256'], 'performanceAcceptance': False, 'actualModelCalls': 0}
verification_path = ROOT / 'tests/acceptance/narrative-world/buffered-text-rollback-archive-verification-20261021-v1.json'
write_new(verification_path, verification)
assert HANDOFF.read_bytes() == original
h.update(status='validation-completed-archived', sourceFrozen=False, updatedAt=now(),
    latestCompletedReceipt=str(receipt_path), latestCompletedReceiptSha256=sha(receipt_path),
    latestArchiveVerification=str(verification_path), latestArchiveVerificationSha256=sha(verification_path),
    lastCompletedSourcePinSha256=sha(pin_path),
    queue='The single rollback validation completed and released the shared slot; both terminal receipts and process exits verified.',
    currentTurnProgress='Rollback verification passed: 314 directed tests plus 25 baseline tests, lint/types/five governance gates/build; v76 semantic observations equal v75. Raw evidence archived and source freeze released.',
    next='Update current evidence documentation and prepare the next performance experiment based on existing profiles; all further tests use the shared serial entry.')
h['durableQueue'].update(completed=True, exitCode=0, endedAt=queue_terminal['endedAt'], runnerPid=terminal['runnerPid'])
h['preparedCurrentArchive'].update(executed=True, path=__file__, receipt=str(receipt_path))
h['candidate']['rollbackValidation'] = 'completed-archived'
h['currentValidationObservation'] = {'at': now(), 'status': 'completed-archived', 'receipt': str(receipt_path),
    'sourceFilesUnchanged': len(pin['files']), 'runtimeFilesUnchanged': len(runtime['files']),
    'knownPidsAbsent': sorted(pids), 'performanceAcceptance': False, 'resubmitted': False}
h['currentGoalTurnClassification'] = {'kind': 'rollback-validation-archived', 'goalStatus': h.get('goalStatus'), 'sourceFrozen': False}
stage = Path(str(BASE) + '-archived-handoff.tmp')
stage.write_text(json.dumps(h, ensure_ascii=False, indent=2) + '\n')
os.replace(stage, HANDOFF)
print(json.dumps(verification, ensure_ascii=False))
