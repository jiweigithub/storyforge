import datetime, hashlib, json, os, shutil, signal, subprocess, sys, time
ROOT=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')

sha=lambda path:hashlib.sha256(Path(path).read_bytes()).hexdigest()
now=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
pin=json.loads(PIN.read_text());pin_sha=sha(PIN)
assert not TERMINAL.exists()
owned={os.getpid()};active=None

def verify():
    assert sha(PIN)==pin_sha and sha(__file__)==pin['runnerSha256']
    for path,digest in pin['executionFiles'].items():assert sha(path)==digest,path
    for ref,key in [('HEAD','head'),('MERGE_HEAD','mergeHead')]:assert subprocess.check_output(['git','rev-parse',ref],cwd=ROOT,text=True).strip()==pin[key]
    for path,digest in pin['files'].items():assert sha(ROOT/path)==digest,path
    for group in ['runtimeFiles', 'extraFiles']:
        for path,digest in pin[group].items():assert sha(path)==digest,path
    for p in record['phases']:
        if 'logSha256' in p:assert sha(p['log'])==p['logSha256']
def observe():
    rows={int(v[0]):v for line in subprocess.check_output(['ps','-axo','pid=,ppid=,command='],text=True).splitlines() if len(v:=line.split(None,2))==3}
    descendants={os.getpid()}
    while True:
        children={pid for pid,v in rows.items() if int(v[1]) in descendants}
        if children<=descendants:break
        descendants|=children
    if descendants-owned:
        with Path(str(BASE)+'-process-observations.jsonl').open('a') as f:f.write(json.dumps({'at':now(),'pids':sorted(descendants),'processes':[rows[p] for p in sorted(descendants) if p in rows]})+'\n')
    owned.update(descendants)
def phase(name,command,cwd):
    global active
    verify();log=Path(str(BASE)+'-'+name+'.log')
    p={'name':name,'command':command,'cwd':cwd,'startedAt':now(),'log':str(log)};record['phases'].append(p)
    print(json.dumps({'starting':name,'at':p['startedAt']}),flush=True)
    with log.open('x') as f:
        active=subprocess.Popen(command,cwd=cwd,env=dict(os.environ,CI='1'),stdin=subprocess.DEVNULL,stdout=f,stderr=subprocess.STDOUT)
        p['childPid']=active.pid;owned.add(active.pid);observe()
        while active.poll() is None:time.sleep(1)
        observe()
        code=active.returncode;active=None
    p.update(exitCode=code,endedAt=now(),logSha256=sha(log));verify()
    print(json.dumps({'completed':name,'exitCode':code}),flush=True)
    return code
