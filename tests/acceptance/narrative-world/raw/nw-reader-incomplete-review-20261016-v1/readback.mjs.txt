import { readFile, writeFile, readdir } from 'node:fs/promises'
import { resolve, relative, dirname } from 'node:path'
import { pathToFileURL } from 'node:url'
import assert from 'node:assert/strict'

// Offline readback only: no model transport is imported or called.
const [rootArg, summaryArg, outputArg] = process.argv.slice(2)
assert(rootArg && summaryArg && outputArg, 'root, terminal summary and new output are required')
const root = resolve(rootArg)
const local = path => {
  const full = resolve(root, path)
  assert(!relative(root, full).startsWith('..'), 'Source must remain in this worktree')
  return full
}
const json = async path => JSON.parse(await readFile(local(path), 'utf8'))
const { sha256, validateReaderAnswer } = await import(pathToFileURL(local('scripts/narrative-reader-contract.mjs')))
const { judgeRequest, validatePriorJudgeCall, summarizeJudge, judgeExecutionCounts } = await import(pathToFileURL(local('scripts/narrative-reader-judge-contract.mjs')))
const base = 'tests/acceptance/narrative-world'
const run = dirname(summaryArg)
assert(run.startsWith(`${base}/raw/reader-incomplete-judge-20261016-`))
const summary = await json(summaryArg), manifest = await json(`${run}/manifest.json`)
assert.equal(summary.schema, 'storyforge.narrative-reader-incomplete-judge-results')
assert.equal(typeof summary.eligibleJudgmentsComplete, 'boolean')
assert.equal(typeof summary.eligibleJudgmentExecutionComplete, 'boolean')
assert.equal(summary.sourceUnchanged, true)
assert.equal(summary.dryRun, false)
assert.equal(summary.runError, null)
assert.equal(summary.complete, false)
assert.equal(summary.executionComplete, false)
assert.equal(summary.accepted, false)
assert.equal(summary.manifestSha256, sha256(manifest))
assert.equal(manifest.totalObservations, 720)
assert.equal(manifest.eligibleObservations, 718)
assert.equal(manifest.distinctPrompts, 459)
assert.equal(manifest.cases.length, 718)
assert.equal(new Set(manifest.cases.map(row => row.id)).size, 718)
const pinned = { ...manifest.sourceHashes, ...manifest.runtimeHashes }
async function verifyPins() {
  for (const [path, expected] of Object.entries(pinned)) assert.equal(sha256(await readFile(local(path))), expected, path)
}
await verifyPins()
const diagnostic = await json(`${base}/reader-incomplete-diagnostic-20261015-v1.json`)
const protocol = await json(`${base}/reader-judge-protocol-20261015-v2.json`)
const readerProtocol = await json(`${base}/reader-protocol-20261015-v3.json`)
const corpus = await json(`${base}/corpus-v1.json`)
const observations = new Map(diagnostic.observations.map(row => [row.id, row]))
const rawHashes = {}, results = new Map()
const localFiles = new Set(await readdir(local(`${run}/calls`)))
const inherited = new Set(manifest.continuation?.inheritedPromptHashes ?? [])
for (const hash of manifest.dispatchOrder) {
  assert(!results.has(hash), 'Duplicate dispatched prompt')
  const present = ['request', 'started', 'result'].some(kind => localFiles.has(`${hash}.${kind}.json`))
  assert(!(present && inherited.has(hash)), 'An inherited input was redispatched')
  if (!present && !inherited.has(hash)) continue
  const prefix = `${present ? run : manifest.continuation.run}/calls/${hash}`, prior = {}
  for (const [key, suffix] of [['request', 'request'], ['started', 'started'], ['result', 'result']]) {
    const path = `${prefix}.${suffix}.json`, bytes = await readFile(local(path))
    rawHashes[path] = sha256(bytes); prior[key] = JSON.parse(bytes)
  }
  const row = manifest.cases.find(item => item.judgePromptHash === hash), observation = observations.get(row?.id)
  assert.equal(observation?.status, 'completed')
  const readerRequest = await json(observation.resultPath.replace('.result.json', '.request.json'))
  const readerResult = await json(observation.resultPath)
  const family = corpus.families.find(item => item.id === row.familyId)
  const question = family.queries.find(item => item.id === row.questionId)
  const readerAnswer = validateReaderAnswer(readerResult.rawText, readerProtocol)
  const request = judgeRequest({ readerRequest, readerAnswer, family, question }, protocol)
  assert.equal(request.promptHash, hash)
  assert.equal(sha256(readerRequest.messages), row.readerPromptHash)
  assert.deepEqual(prior.request.messages, request.messages)
  const checked = validatePriorJudgeCall(prior, request, protocol)
  if (checked.status === 'completed') assert.deepEqual(checked.judgment, prior.result.judgment)
  else assert.equal(checked.judgment, undefined, 'A failed or unknown call must not be assigned a judgment')
  results.set(hash, checked)
}
assert.equal(results.size, summary.attemptedPrompts)
const actualFiles = (await readdir(local(`${run}/calls`))).sort()
assert.deepEqual(actualFiles, Object.keys(rawHashes).filter(path => path.startsWith(`${run}/calls/`)).map(path => path.slice(path.lastIndexOf('/') + 1)).sort())
const rows = manifest.cases.filter(row => results.get(row.judgePromptHash)?.status === 'completed')
  .map(({ inputBytes, estimatedInputTokens, ...row }) => ({ ...row, judgment: results.get(row.judgePromptHash).judgment }))
assert.deepEqual(rows, summary.rows)
assert.deepEqual(summarizeJudge(rows), summary.metrics)
const counts = judgeExecutionCounts([...results.values()], manifest.distinctPrompts)
for (const [key, value] of Object.entries(counts)) assert.equal(summary[key === 'executionComplete' ? 'eligibleJudgmentExecutionComplete' : key], value)
assert.equal(summary.eligibleJudgmentsComplete, rows.length === manifest.eligibleObservations && [...results.values()].every(row => row.status === 'completed'))
const usage = { inputTokens: [...results.values()].reduce((n, row) => n + (row.usage?.inputTokens ?? 0), 0),
  outputTokens: [...results.values()].reduce((n, row) => n + (row.usage?.outputTokens ?? 0), 0),
  unknownDispatchedCalls: [...results.values()].filter(row => row.dispatched && (!Number.isSafeInteger(row.usage?.inputTokens) || !Number.isSafeInteger(row.usage?.outputTokens))).length }
assert.deepEqual(usage, summary.usage)
assert.equal(summary.readerCoverage.unavailable.length, 2)
for (const unavailable of summary.readerCoverage.unavailable) assert(!rows.some(row => row.id === unavailable.id))
for (const [condition, coverage] of Object.entries(summary.coverageByCondition)) {
  const selected = rows.filter(row => row.condition === condition)
  const cases = manifest.cases.filter(row => row.condition === condition)
  assert.equal(cases.length, coverage.eligibleObservations)
  assert.equal(coverage.completed, selected.length)
  for (const status of ['failed', 'invalid-judgment', 'unknown-or-interrupted', 'not-attempted']) {
    assert.equal(coverage[status], cases.filter(row => (results.get(row.judgePromptHash)?.status ?? 'not-attempted') === status).length)
  }
  assert.equal(coverage.totalObservations, 120)
  assert.equal(coverage.totalObservations, coverage.eligibleObservations + coverage.readerFailed + coverage.readerUnknown)
}
const checkpoint = await json(`${base}/engineering-checkpoint-20261016-v32.json`)
for (const [path, expected] of Object.entries(checkpoint.sourceHashes)) assert.equal(sha256(await readFile(local(path))), expected, path)
await verifyPins()
for (const [path, expected] of Object.entries(rawHashes)) assert.equal(sha256(await readFile(local(path))), expected, path)
const brief = row => ({ id: row.id, condition: row.condition, operation: row.operation, judgePromptHash: row.judgePromptHash,
  assessment: row.judgment.assessment, answerCorrectness: row.judgment.answerCorrectness,
  materialFaithfulness: row.judgment.materialFaithfulness, scopeCompliance: row.judgment.scopeCompliance, reason: row.judgment.reason })
const record = { schema: 'storyforge.reader-incomplete-review-audit', version: 1, createdAt: new Date().toISOString(),
  summaryPath: summaryArg, summarySha256: sha256(await readFile(local(summaryArg))),
  manifestSha256: sha256(await readFile(local(`${run}/manifest.json`))),
  plannedObservations: 720, eligibleObservations: manifest.eligibleObservations, gradedObservations: rows.length,
  distinctValidatedJudgments: [...results.values()].filter(row => row.status === 'completed').length, executionCounts: counts,
  originalReaderMissing: summary.readerCoverage.unavailable, completeOriginalMatrix: false,
  eligibleJudgmentsComplete: summary.eligibleJudgmentsComplete, eligibleJudgmentExecutionComplete: summary.eligibleJudgmentExecutionComplete, sourceUnchanged: true,
  pinnedEvidenceFiles: Object.keys(manifest.sourceHashes).length, pinnedRuntimeFiles: Object.keys(manifest.runtimeHashes).length,
  sourceCheckpoint: 32, productionVersion: 31, checkpointFingerprintsUnchanged: Object.keys(checkpoint.sourceHashes).length,
  validation: ['Frozen evidence/runtime hashes matched before and after readback', 'Every raw request regenerated from its original successful reader call',
    'All started/request/result identities checked; successful generations include valid usage and exact quotations',
    'All reported rows and metrics reproduced from valid raw judgments; failed and unknown judgments remain ungraded', 'Raw calls directory exactly matches expected request/start/result files',
    'Failed and unknown original reader observations retain planned denominators without judgments'],
  usage, metrics: summary.metrics, coverageByCondition: summary.coverageByCondition,
  unresolvedJudgeInputs: manifest.dispatchOrder.filter(hash => results.get(hash)?.status !== 'completed').map(hash => ({
    promptHash: hash, status: results.get(hash)?.status ?? 'not-attempted',
    caseIds: manifest.cases.filter(row => row.judgePromptHash === hash).map(row => row.id),
    diagnostic: results.get(hash)?.codexDiagnostic ?? null })),
  scopeFlaggedCases: rows.filter(row => row.judgment.scopeCompliance !== 'compliant').map(brief),
  partialOrIncorrectCases: rows.filter(row => ['partly-correct', 'incorrect'].includes(row.judgment.assessment)).map(brief),
  unjustifiedAbstentions: rows.filter(row => row.judgment.assessment === 'unjustified-abstention').map(brief),
  rawHashes, newModelCallsDuringReadback: 0, independentHumanReviews: 0, costUsd: null, accepted: false,
  evidenceBoundary: 'Historical frozen incomplete reader cohort only. One judgment per deduplicated prompt within this run; existing successful overlaps are disclosed separately and not independent repetitions. This audit neither fills missing readers nor proves current production, literary, human, product, full CI or performance acceptance.' }
await writeFile(resolve(outputArg), JSON.stringify(record, null, 2) + '\n', { flag: 'wx' })
console.log(JSON.stringify({ output: resolve(outputArg), attempted: results.size, judgments: record.distinctValidatedJudgments, observations: rows.length,
  verifiedSourceFiles: Object.keys(pinned).length, rawFiles: Object.keys(rawHashes).length, sourceUnchanged: true,
  scopeFlags: record.scopeFlaggedCases.length, partialOrIncorrect: record.partialOrIncorrectCases.length, accepted: false }))
