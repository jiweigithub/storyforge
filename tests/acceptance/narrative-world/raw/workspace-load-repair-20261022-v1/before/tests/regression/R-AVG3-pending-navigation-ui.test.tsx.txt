import { act } from 'react'
import { createRoot } from 'react-dom/client'
import { MemoryRouter, Route, Routes, useLocation } from 'react-router'
import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest'
import { db } from '../../src/lib/db/schema'
import { createAvgDraftV1, readAvgDraftV1, selectAvgWorldV1 } from '../../src/lib/avg/draft-service'
import { seedCurrentProductWorld } from '../helpers/current-product-world'
import AvgPage from '../../src/pages/AvgPage'

const pending = vi.hoisted(() => ({ selection: null as Promise<void> | null }))

vi.mock('../../src/lib/avg/draft-service', async importOriginal => {
  const actual = await importOriginal<typeof import('../../src/lib/avg/draft-service')>()
  return {
    ...actual,
    selectAvgWorldV1: vi.fn(async (...args: Parameters<typeof actual.selectAvgWorldV1>) => {
      await pending.selection
      return actual.selectAvgWorldV1(...args)
    }),
  }
})
vi.mock('../../src/hooks/useAutoBackup', () => ({ useAutoBackup: () => undefined }))
vi.mock('../../src/hooks/useGistAutoBackup', () => ({ useGistAutoBackup: () => undefined }))
vi.mock('../../src/components/examples/ExampleLibrary', () => ({ default: () => null }))
vi.mock('../../src/components/world-engine/WorldResourceBrowser', () => ({ default: () => null }))

globalThis.IS_REACT_ACT_ENVIRONMENT = true

let host: HTMLDivElement
let root: ReturnType<typeof createRoot>
let finishSelection: () => void
let rejectSelection: (error: Error) => void

function LocationProbe() {
  const location = useLocation()
  return <output data-testid="location">{location.pathname}{location.search}</output>
}

function button(label: string, selector = '') {
  const result = Array.from(host.querySelectorAll<HTMLButtonElement>(`${selector} button`))
    .find(element => element.textContent?.trim() === label)
  if (!result) throw new Error(`Missing button: ${label}`)
  return result
}

function location() {
  return host.querySelector('[data-testid="location"]')?.textContent
}

async function settle(check: () => void) {
  const started = Date.now()
  let failure: unknown
  while (Date.now() - started < 1000) {
    await act(async () => { await new Promise(resolve => setTimeout(resolve, 10)) })
    try { check(); return }
    catch (error) { failure = error }
  }
  throw failure
}

beforeEach(async () => {
  await db.delete()
  await db.open()
  vi.mocked(selectAvgWorldV1).mockClear()
  pending.selection = new Promise<void>((resolve, reject) => {
    finishSelection = resolve
    rejectSelection = reject
  })
  host = document.createElement('div')
  document.body.append(host)
  root = createRoot(host)
})

afterEach(async () => {
  finishSelection()
  await act(async () => {
    await Promise.allSettled(vi.mocked(selectAvgWorldV1).mock.results.map(result => result.value))
    root.unmount()
  })
  host.remove()
  pending.selection = null
  db.close()
})

describe('AVG source selection navigation', () => {
  it.each(['saved', 'rejected'] as const)(
    'makes pending navigation visibly unavailable and restores it when selection is %s',
    async outcome => {
      const source = await seedCurrentProductWorld('导航回归来源')
      const avg = await createAvgDraftV1('导航回归 AVG')
      const query = `?project=${avg.scope.projectId}&work=${avg.scope.workId}`
      await act(async () => root.render(
        <MemoryRouter initialEntries={[`/avg/source${query}`]}>
          <LocationProbe />
          <Routes><Route path="/avg/:pageId" element={<AvgPage />} /></Routes>
        </MemoryRouter>,
      ))
      await settle(() => expect(button('选择此版本').disabled).toBe(false))
      await act(async () => button('选择此版本').click())
      await settle(() => expect(selectAvgWorldV1).toHaveBeenCalledTimes(1))

      const navigation = '[aria-label="AVG 页面导航"]'
      const controls = host.querySelectorAll<HTMLButtonElement>(
        '[aria-label="AVG 页面导航"] button, [aria-label="制作阶段"] button, .lf-current',
      )
      expect(controls.length).toBeGreaterThan(3)
      for (const control of controls) expect(control.disabled).toBe(true)
      expect(button('返回制作方案').disabled).toBe(true)
      await act(async () => button('制作台', navigation).click())
      expect(location()).toBe(`/avg/source${query}`)
      expect((await readAvgDraftV1(avg.scope)).worldReleaseId).toBeNull()

      await act(async () => {
        if (outcome === 'saved') finishSelection()
        else rejectSelection(new Error('本次世界选择失败'))
        await Promise.allSettled(vi.mocked(selectAvgWorldV1).mock.results.map(result => result.value))
      })
      await settle(() => {
        expect(button('制作台', navigation).disabled).toBe(false)
        expect(host.textContent).toContain(outcome === 'saved' ? '世界版本已引用' : '本次世界选择失败')
      })
      const draft = await readAvgDraftV1(avg.scope)
      if (outcome === 'saved') {
        const localReference = await db.worldReleases.get(draft.worldReleaseId!)
        expect(localReference?.projectId).toBe(avg.scope.projectId)
        expect(localReference?.releaseUid).toBe(source.release.releaseUid)
        expect(localReference?.contentHash).toBe(source.release.contentHash)
      } else {
        expect(draft.worldReleaseId).toBeNull()
        expect(draft.revision).toBe(0)
        expect(host.querySelector('[role="alert"]')?.textContent).toBe('本次世界选择失败')
      }
      expect(await db.worldReleases.get(source.release.id!)).toEqual(source.release)
      expect(await db.productProductions.count()).toBe(0)
      expect(selectAvgWorldV1).toHaveBeenCalledTimes(1)
      await act(async () => button('制作台', navigation).click())
      await settle(() => {
        expect(location()).toBe(`/avg/vision${query}`)
        expect(button('确认制作方案', '[aria-label="AVG 内容导航"]').disabled).toBe(false)
      })
      await act(async () => button('确认制作方案', '[aria-label="AVG 内容导航"]').click())
      // The saved branch loads the real Studio lazily before the route can commit.
      await act(async () => { await vi.dynamicImportSettled() })
      await settle(() => expect(location()).toBe(`/avg/confirm${query}`))
      expect(await db.productProductions.count()).toBe(0)
      expect(await db.worldReleases.get(source.release.id!)).toEqual(source.release)
    },
  )
})
