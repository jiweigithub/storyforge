import assert from 'node:assert/strict'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { performance } from 'node:perf_hooks'
import * as esbuild from '/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge/node_modules/esbuild/lib/main.js'

const base = '/tmp/nw-buffered-text-probe-20261021-v1'
const root = '/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge'
const hash = value => createHash('sha256').update(value).digest('hex')
const pin = JSON.parse(readFileSync(`${base}/pin.json`, 'utf8'))
const verify = () => { for (const [path, digest] of Object.entries(pin.files)) assert.equal(hash(readFileSync(path)), digest, path) }
verify()
const result = { schema: 'storyforge.narrative-buffered-text-probe', version: 1, startedAt: new Date().toISOString(),
  evidenceLevel: 'isolated Node codec experiment; not browser or end-to-end performance acceptance',
  runnerPid: process.pid, pinSha256: hash(readFileSync(`${base}/pin.json`)), environment: { node: process.version, arch: process.arch, platform: process.platform },
  conformance: {}, samples: [], productionChanged: false, actualModelCalls: 0, accepted: false }
try {
  // Initialization is deliberately excluded from the paired sink experiment.
  // The same real SHA implementation is bundled independently for each variant.
  // Export initialization from the same module graph, without touching product sources.
  const modules = {}
  for (const variant of ['baseline', 'candidate']) {
    await esbuild.build({ stdin: { contents: `export * from './json.ts'; export { initializeNarrativeSha256V1 } from './sha256.ts'`,
      resolveDir: `${base}/${variant}`, loader: 'ts' }, outfile: `${base}/${variant}-ready.mjs`,
      bundle: true, platform: 'node', format: 'esm', target: 'node22', nodePaths: [`${root}/node_modules`] })
    modules[variant] = await import(`${base}/${variant}-ready.mjs`)
    assert.equal(await modules[variant].initializeNarrativeSha256V1(), 'wasm')
  }
  const fixture = JSON.parse(readFileSync(`${base}/worldrelease-golden.json`, 'utf8')).compiled
  const reference = modules.baseline
  const values = [null, false, true, 0, 1.25, Number.MAX_SAFE_INTEGER, '', '汉字😀\r\n"\\\u0000',
    { '\uE000': 'BMP', '😀': 'astral', '2': 2, '10': 10, '0': 'zero' },
    fixture.manifest.narrative.bundle, fixture.manifest.narrative.sources, fixture.index, fixture.manifest]
  const rawTexts = ['', 'a'.repeat(127), 'é'.repeat(64), '中'.repeat(129), '😀'.repeat(65),
    ...[4095, 4096, 4097, 32767, 32768, 32769].flatMap(n => [ 'x'.repeat(n), '汉字😀'.repeat(n).slice(0, n - n % 4) ])]
  const partitions = [[], [''], ['a', 'é', '中', '😀', '\r\n'],
    Array.from({ length: 1000 }, (_, i) => ['a', 'é', '汉字', '😀', '"\\\n'][i % 5]),
    ['a'.repeat(128), '中'.repeat(129), 'b'.repeat(4096), '😀'.repeat(16385), 'tail'],
    Array.from({ length: 1400 }, (_, i) => 'x'.repeat(i % 128 + 1)).concat('😀'.repeat(30000), '终')]
  const omit = value => Array.isArray(value) ? value.map(omit) : value && typeof value === 'object'
    ? Object.fromEntries(Object.entries(value).filter(([key]) => !['revisionHash', 'revisionVectorHash'].includes(key)).map(([key, child]) => [key, omit(child)])) : value
  for (const [name, api] of Object.entries(modules)) {
    for (const value of values) {
      const canonical = reference.canonicalNarrativeJsonV1(value), expected = hash(canonical)
      assert.equal(await api.hashNarrativeValueV1(value), expected)
      assert.deepEqual(api.canonicalNarrativeJsonWithHashV1(value), { json: canonical, hash: expected })
      const owned = structuredClone(value)
      assert.equal(await api.freezeAndHashNarrativeJsonV1(owned, ['revisionHash', 'revisionVectorHash']), hash(reference.canonicalNarrativeJsonV1(omit(value))))
    }
    for (const text of rawTexts) assert.equal(await api.hashNarrativeBytesV1(text), hash(text))
    for (const parts of partitions) assert.equal(await api.hashNarrativeTextPartsV1(parts), hash(parts.join('')))
    for (const invalid of ['\ud800', '\udc00', 'a\ud800b']) {
      await assert.rejects(api.hashNarrativeBytesV1(invalid), /unicode/)
      await assert.rejects(api.hashNarrativeTextPartsV1(['prefix', invalid]), /unicode/)
    }
    let getterCalls = 0
    const accessor = { prior: 'x'.repeat(1000) }
    Object.defineProperty(accessor, 'z', { enumerable: true, get() { getterCalls++; return 1 } })
    assert.throws(() => api.hashNarrativeValueV1(accessor), /accessor/)
    assert.equal(getterCalls, 0)
    const sentinel = new Error('producer stopped after buffered text')
    function* broken() { yield 'a'.repeat(100); yield '中'; throw sentinel }
    await assert.rejects(api.hashNarrativeTextPartsV1(broken()), error => error === sentinel)
    assert.equal(await api.hashNarrativeBytesV1('after-failure'), hash('after-failure'))
    const nested = []
    function* reentrant() {
      yield 'outer'
      nested.push(api.hashNarrativeTextPartsV1((function* () { yield 'inner'; yield '😀' })()))
      yield 'tail'
    }
    assert.equal(await api.hashNarrativeTextPartsV1(reentrant()), hash('outertail'))
    assert.deepEqual(await Promise.all(nested), [hash('inner😀')])
    result.conformance[name] = { canonicalValues: values.length, rawTexts: rawTexts.length, partitionCases: partitions.length,
      invalidUnicodeCases: 3, accessorInvocations: getterCalls, failureRecovery: true, reentrantSinkIsolation: true, passed: true }
  }
  const records = fixture.manifest.narrative.bundle.records
  const expectedRecords = records.map(record => hash(reference.canonicalNarrativeJsonV1(omit(record))))
  const partList = partitions.at(-1), partHash = hash(partList.join(''))
  const expectedBundle = hash(reference.canonicalNarrativeJsonV1(fixture.manifest.narrative.bundle))
  const operations = [
    { name: 'fresh-record-freeze-and-hash', iterations: 1200, prepare: () => Array.from({ length: 1200 }, (_, i) => structuredClone(records[i % records.length])),
      run: async (api, inputs) => { for (let i = 0; i < inputs.length; i++) assert.equal(await api.freezeAndHashNarrativeJsonV1(inputs[i], ['revisionHash', 'revisionVectorHash']), expectedRecords[i % records.length]) } },
    { name: 'mixed-short-and-long-raw-parts', iterations: 80, prepare: () => null,
      run: async api => { for (let i = 0; i < 80; i++) assert.equal(await api.hashNarrativeTextPartsV1(partList), partHash) } },
    { name: 'bundle-value-hash', iterations: 120, prepare: () => fixture.manifest.narrative.bundle,
      run: async (api, input) => { for (let i = 0; i < 120; i++) assert.equal(await api.hashNarrativeValueV1(input), expectedBundle) } },
  ]
  for (const op of operations) {
    for (const api of Object.values(modules)) await op.run(api, op.prepare())
    for (let repetition = 0; repetition < 7; repetition++) {
      const order = repetition % 2 ? ['candidate', 'baseline'] : ['baseline', 'candidate']
      for (const variant of order) {
        const input = op.prepare(), started = performance.now()
        await op.run(modules[variant], input)
        result.samples.push({ operation: op.name, repetition: repetition + 1, variant, iterations: op.iterations, elapsedMs: performance.now() - started })
      }
    }
  }
  const median = xs => [...xs].sort((a, b) => a - b)[Math.floor(xs.length / 2)]
  result.summaries = operations.map(op => {
    const timings = Object.fromEntries(['baseline', 'candidate'].map(variant => [variant, result.samples.filter(s => s.operation === op.name && s.variant === variant).map(s => s.elapsedMs)]))
    return { operation: op.name, repetitionsPerVariant: 7, baselineMedianMs: median(timings.baseline), candidateMedianMs: median(timings.candidate),
      candidateOverBaseline: median(timings.candidate) / median(timings.baseline), pairedRatios: timings.baseline.map((value, i) => timings.candidate[i] / value) }
  })
  result.limitations = ['These are copied-code Node operation measurements, not Chromium end-to-end timings or formal scale samples.',
    'Preparation and SHA initialization are outside measured operation loops; each fresh-record sample gets separate mutable clones.',
    'No heap/RSS improvement is measured or claimed. A browser comparison is required before adopting any production change.',
    'All raw paired timings are retained; the experiment does not change acceptance thresholds or decide product readiness.']
  verify(); result.sourceUnchanged = true; result.exitCode = 0
} catch (error) {
  result.error = String(error?.stack ?? error); result.exitCode = 1; process.exitCode = 1
} finally {
  result.endedAt = new Date().toISOString()
  writeFileSync(`${base}/result.json`, `${JSON.stringify(result, null, 2)}\n`, { flag: 'wx' })
  console.log(JSON.stringify({ exitCode: result.exitCode, conformance: result.conformance, summaries: result.summaries, result: `${base}/result.json` }))
}
