from pathlib import Path
import datetime, hashlib, json, os, shutil, statistics, subprocess

ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
BASE = Path('/tmp/nw-index-keys-probe-20261021-v1')
HANDOFF = Path('/tmp/nw-handoff-current-20261017.json')
original = HANDOFF.read_bytes()
h = json.loads(original)
active = h['activeIndependentProbe']
assert active['directory'] == str(BASE) and not active['completed']
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()
def write_new(path, obj):
    with Path(path).open('x') as stream:
        json.dump(obj, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

pin = json.loads((BASE / 'pin.json').read_text())
terminal = json.loads((BASE / 'terminal.json').read_text())
result = json.loads((BASE / 'result.json').read_text())
assert sha(BASE / 'pin.json') == active['pinSha256'] == terminal['pinSha256'] == result['pinSha256']
assert terminal['exitCode'] == result['exitCode'] == 0
assert terminal['sourceUnchanged'] and result['sourceUnchanged']
assert not result['adopted'] and not result['productionChanged'] and not result['accepted']
assert len(pin['files']) == pin['fileCount'] == 658
assert all(sha(path) == digest for path, digest in pin['files'].items())
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip() == pin['head']
assert sha(BASE / 'queue.log') == terminal['logSha256']
for phase in result['phases']:
    assert phase['exitCode'] == 0 and sha(phase['log']) == phase['logSha256']
assert all(sha(path) == digest for path, digest in result['generatedBundles'].items())
for runtime in ['node', 'chromium']:
    data = result[runtime]
    assert data['conformance']['passed'] and data['conformance']['exactBytes']
    assert data['conformance']['canonicalOrderingAndLastDuplicateIdentity']
    assert data['conformance']['originalGoldenEdges'] == 15
    assert data['conformance']['tupleCases'] == 2744 and data['conformance']['edgeCases'] == 1415
    assert len(data['samples']) == 28
    for summary in data['summaries']:
        rows = [row for row in data['samples'] if row['operation'] == summary['operation']]
        assert len(rows) == 14
        for variant in ['baseline', 'candidate']:
            assert summary[variant + 'MedianMs'] == statistics.median(row['elapsedMs'] for row in rows if row['variant'] == variant)
        for pair in range(1, 8):
            pair_rows = {row['variant']: row for row in rows if row['pair'] == pair}
            assert set(pair_rows) == {'baseline', 'candidate'}
            assert pair_rows['candidate']['elapsedMs'] < pair_rows['baseline']['elapsedMs']

pids = {row['pid'] for row in result['ownedProcesses']} | {terminal['supervisorPid'], terminal['queuePid'], result['runnerPid']}
ps = subprocess.run(['ps', '-p', ','.join(map(str, sorted(pids))), '-o', 'pid=,ppid=,state=,command='], capture_output=True, text=True)
assert ps.returncode == 1 and not ps.stdout.strip(), ps.stdout
rows = {int(parts[0]): parts for line in subprocess.check_output(['ps', '-axo', 'pid=,ppid=,command='], text=True).splitlines()
        if len(parts := line.strip().split(None, 2)) == 3}
ancestors = {os.getpid()}
pid = os.getpid()
while pid in rows and int(rows[pid][1]) not in ancestors:
    pid = int(rows[pid][1]); ancestors.add(pid)
cwd_scan = subprocess.run(['lsof', '-nP', '-d', 'cwd', '-Fpn'], capture_output=True, text=True, timeout=15)
cwd_pids = []
for line in cwd_scan.stdout.splitlines():
    if line.startswith('p'): pid = int(line[1:])
    elif line.startswith('n') and line[1:] in {str(ROOT), str(BASE)}: cwd_pids.append(pid)
remaining = [pid for pid in cwd_pids if pid in rows and pid not in ancestors]
matches = [parts for pid, parts in rows.items() if pid not in ancestors and (str(ROOT) in parts[2] or str(BASE) in parts[2])]
assert not remaining, [rows[pid] for pid in remaining]
assert not matches, matches
proof = {'schema': 'storyforge.narrative-validation-owned-exit', 'version': 1, 'checkedAt': now(),
    'pids': sorted(pids), 'allAbsent': True, 'sourceUnchanged': True, 'pinnedFiles': len(pin['files']),
    'head': pin['head'], 'pinSha256': sha(BASE / 'pin.json'), 'terminalSha256': sha(BASE / 'terminal.json'),
    'resultSha256': sha(BASE / 'result.json'), 'workspaceProcessesRemaining': [], 'commandMatchesRemaining': [],
    'limitation': 'Recorded wrappers and observed descendants are absent; short synchronous precheck descendants were not individually recorded. Independent cwd and command scans found no remaining owned processes. This evidence writer and its ancestors excluded; no foreign process signaled.'}
write_new(BASE / 'owned-exit.json', proof)

dest = ROOT / 'tests/acceptance/narrative-world/raw/index-keys-probe-20261021-v1'
assert not dest.exists()
dest.mkdir()
artifacts = []
sources = [(path, path.relative_to(BASE)) for path in sorted(BASE.rglob('*')) if path.is_file()
           and not path.is_symlink() and not path.name.startswith('heartbeat-') and path.name != 'supervisor.lock']
sources += [(Path('/tmp/nw-prepare-index-keys-probe-20261021-v1.py'), Path('prepare.py')), (Path(__file__), Path('archive.py'))]
for source, name in sources:
    target = dest / name
    if target.suffix in {'.py', '.js', '.mjs', '.cjs', '.ts', '.tsx'}: target = Path(str(target) + '.txt')
    target.parent.mkdir(parents=True, exist_ok=True)
    assert not target.exists()
    shutil.copy2(source, target)
    assert sha(source) == sha(target)
    artifacts.append({'path': str(target.relative_to(ROOT)), 'sha256': sha(target), 'bytes': target.stat().st_size})
receipt = dict(result)
receipt.update(schema='storyforge.narrative-index-private-key-probe-receipt', archivedAt=now(),
    artifacts=artifacts, exitProof=proof, formalPerformance=False, performanceAcceptance=False,
    retentionForIntegrationMet=True, productionAdoption=False,
    rawEvidenceExclusions='Foreign queue heartbeat observations and lock files remain in /tmp; not copied into product evidence.',
    next='Create baseline and candidate external snapshots; validate exact integrated behavior and five interleaved million-character cold-open/RSS diagnostic pairs before production adoption.')
receipt_path = ROOT / 'tests/acceptance/narrative-world/index-keys-probe-result-20261021-v1.json'
write_new(receipt_path, receipt)
assert all(sha(ROOT / item['path']) == item['sha256'] for item in artifacts)
verification = {'schema': 'storyforge.narrative-archive-verification', 'version': 1, 'verifiedAt': now(),
    'receipt': str(receipt_path.relative_to(ROOT)), 'receiptSha256': sha(receipt_path),
    'artifactsVerified': len(artifacts), 'allMatch': True, 'pinnedFilesUnchanged': len(pin['files']),
    'knownPidsAbsent': sorted(pids), 'productionAdoption': False, 'performanceAcceptance': False, 'actualModelCalls': 0}
verification_path = ROOT / 'tests/acceptance/narrative-world/index-keys-probe-archive-verification-20261021-v1.json'
write_new(verification_path, verification)
assert HANDOFF.read_bytes() == original
h.setdefault('independentProbeHistory', []).append(h.get('latestIndependentProbe'))
active.update(status='completed-archived', completed=True, exitCode=0, endedAt=result['endedAt'],
    runnerPid=result['runnerPid'], sourceUnchanged=True, ownedPidsAbsent=True,
    receipt=str(receipt_path), receiptSha256=sha(receipt_path), artifactCount=len(artifacts),
    verification=str(verification_path), verificationSha256=sha(verification_path),
    next=receipt['next'], sourceFrozenScope=None)
h.update(activeIndependentProbe=None, latestIndependentProbe=active,
    status='independent-probe-completed-archived', updatedAt=now(), sourceFrozen=False,
    currentTurnProgress='Private index-key probe passed all conformance and 28 paired operation comparisons. Raw evidence, 658-file integrity and 11 known process exits archived. Production index unchanged.',
    next=receipt['next'])
stage = Path('/tmp/nw-index-keys-probe-20261021-v1-archived-handoff.tmp')
stage.write_text(json.dumps(h, ensure_ascii=False, indent=2) + '\n')
os.replace(stage, HANDOFF)
print(json.dumps(verification, ensure_ascii=False))
