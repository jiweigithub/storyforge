import assert from 'node:assert/strict'
import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'
import { execFileSync } from 'node:child_process'
import * as esbuild from '/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge/node_modules/esbuild/lib/main.js'
import { chromium } from '/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge/node_modules/playwright/index.mjs'

const base = '/tmp/nw-index-keys-probe-20261021-v1'
const root = '/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge'
const hash = value => createHash('sha256').update(value).digest('hex')
const pin = JSON.parse(readFileSync(`${base}/pin.json`, 'utf8'))
const verify = () => { for (const [path, digest] of Object.entries(pin.files)) assert.equal(hash(readFileSync(path)), digest, path) }
const result = { schema: 'storyforge.narrative-index-private-key-probe', version: 1, startedAt: new Date().toISOString(), runnerPid: process.pid,
  pinSha256: hash(readFileSync(`${base}/pin.json`)), productionChanged: false, adopted: false, actualModelCalls: 0, accepted: false,
  evidenceLevel: 'Copied private-key operations in Node and isolated Chromium; no product, cold-open or RSS acceptance', phases: [], ownedProcesses: [] }
let server, browser
const capture = () => {
  const rows = execFileSync('ps', ['-axo', 'pid=,ppid=,stat=,command='], { encoding: 'utf8' }).trim().split('\n')
    .map(line => line.trim().match(/^(\d+)\s+(\d+)\s+(\S+)\s+(.*)$/)).filter(Boolean)
    .map(([, pid, ppid, state, command]) => ({ pid: Number(pid), ppid: Number(ppid), state, command }))
  const family = new Set([process.pid])
  for (;;) { const size = family.size; for (const row of rows) if (family.has(row.ppid)) family.add(row.pid); if (family.size === size) break }
  for (const row of rows) if (family.has(row.pid) && !result.ownedProcesses.some(old => old.pid === row.pid)) result.ownedProcesses.push(row)
}
try {
  verify()
  for (const [name, command, args] of [['diff-check', 'git', ['diff', '--check']], ['docs', 'npm', ['run', 'check:docs']],
    ['draft-types', pin.node, [`${root}/node_modules/typescript/bin/tsc`, '--project', `${base}/tsconfig.json`]]]) {
    let output, failure
    try { output = execFileSync(command, args, { cwd: root, encoding: 'utf8', timeout: 120000, env: { ...process.env, CI: '1' } }) }
    catch (error) { output = String(error.stdout ?? '') + String(error.stderr ?? ''); failure = error }
    const path = `${base}/${name}.log`; writeFileSync(path, output, { flag: 'wx' }); result.phases.push({ name, command: [command, ...args], exitCode: failure?.status ?? (failure ? 1 : 0), log: path, logSha256: hash(output) })
    if (failure) throw failure
  }
  const entry = `export * from './keys.ts'; export { exercise } from './exercise.mjs'; export { NARRATIVE_INDEX_EDGE_KINDS_V1 as edgeKinds } from './types/narrative-index.ts'`
  for (const [platform, format, target, suffix] of [['node', 'esm', 'node22', 'mjs'], ['browser', 'iife', 'chrome151', 'js']]) {
    await esbuild.build({ stdin: { contents: entry, resolveDir: base, loader: 'ts' }, outfile: `${base}/${platform}-ready.${suffix}`,
      bundle: true, platform, format, target, ...(platform === 'browser' ? { globalName: 'keyProbe' } : {}), nodePaths: [`${root}/node_modules`] })
  }
  result.generatedBundles = Object.fromEntries(['node-ready.mjs', 'browser-ready.js'].map(file => [`${base}/${file}`, hash(readFileSync(`${base}/${file}`))]))
  const fixture = JSON.parse(readFileSync(`${base}/index-golden.json`, 'utf8'))
  const api = await import(`${base}/node-ready.mjs`)
  result.node = { version: process.version, ...api.exercise(api, fixture, api.edgeKinds) }
  capture(); verify()
  server = await chromium.launchServer({ executablePath: pin.browser, headless: true })
  browser = await chromium.connect(server.wsEndpoint())
  result.browserPid = server.process().pid; result.browserVersion = browser.version()
  const context = await browser.newContext()
  await context.route('**/*', route => route.abort())
  const page = await context.newPage()
  await page.addScriptTag({ path: `${base}/browser-ready.js` })
  capture()
  result.chromium = await page.evaluate(fixture => {
    const api = globalThis.keyProbe
    return { userAgent: navigator.userAgent, ...api.exercise(api, fixture, api.edgeKinds) }
  }, fixture)
  capture()
  await context.close(); await browser.close(); browser = undefined
  await server.close(); server = undefined
  verify(); result.sourceUnchanged = true; result.exitCode = 0
  result.limitations = ['Only fresh compiler-owned rows and dense arrays are in scope; these helpers cannot validate external objects.',
    'Exact bytes and ordering are checked against the unchanged strict canonical codec and original fixed golden rows.',
    'Generated timing rows exercise index-key encoding only; they are not a million-character product run or author quality evidence.',
    'No RSS reduction is inferred. A retained production candidate still requires integration regression and paired complete-chain measurements.',
    'All seven alternating pairs are retained, including unfavorable pairs; no retries.']
} catch (error) { result.error = String(error?.stack ?? error); result.exitCode = 1; process.exitCode = 1 }
finally {
  if (browser) await browser.close().catch(() => {})
  if (server) await server.close().catch(() => {})
  result.endedAt = new Date().toISOString()
  writeFileSync(`${base}/result.json`, `${JSON.stringify(result, null, 2)}\n`, { flag: 'wx' })
  console.log(JSON.stringify({ exitCode: result.exitCode, result: `${base}/result.json`, node: result.node?.summaries, chromium: result.chromium?.summaries, error: result.error }))
}
