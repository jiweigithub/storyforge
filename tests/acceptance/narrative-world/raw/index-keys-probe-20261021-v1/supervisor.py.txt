from pathlib import Path
import datetime, fcntl, hashlib, json, os, signal, subprocess, sys

BASE = Path('/tmp/nw-index-keys-probe-20261021-v1')
ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')

def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def save(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

signal.signal(signal.SIGHUP, signal.SIG_IGN)
with (BASE/'supervisor.lock').open('a') as lock:
    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    assert not (BASE/'state.json').exists() and not (BASE/'terminal.json').exists()
    assert not (BASE/'queue.log').exists() and not (BASE/'result.json').exists()
    pin_path = BASE/'pin.json'
    pin_bytes = pin_path.read_bytes()
    pin = json.loads(pin_bytes)
    def verify():
        assert pin_path.read_bytes() == pin_bytes
        changed = [path for path, expected in pin['files'].items()
                   if not Path(path).is_file() or digest(Path(path)) != expected]
        assert not changed, changed
    verify()
    record = {'schema': 'storyforge.narrative-isolated-probe-queue', 'version': 1,
              'submittedAt': now(), 'supervisorPid': os.getpid(), 'pinSha256': digest(pin_path),
              'root': str(ROOT), 'frozenScope': 'external draft, fixed fixture and dependency inputs only',
              'productionMutation': False, 'actualModelCalls': 0, 'accepted': False}
    code = 2
    try:
        with (BASE/'queue.log').open('x') as log:
            command = [sys.executable, '/tmp/storyforge-serial-ci-20261013.py', '--', pin['node'], str(BASE/'probe.mjs')]
            record['command'] = command
            child = subprocess.Popen(command, cwd=ROOT, stdin=subprocess.DEVNULL, stdout=log,
                                     stderr=subprocess.STDOUT, start_new_session=True)
            record['queuePid'] = child.pid
            save(BASE/'state.json', record)
            code = child.wait()
        verify()
        record['sourceUnchanged'] = True
    except BaseException as error:
        record['error'] = type(error).__name__ + ': ' + str(error)
        code = 2
    finally:
        record.update(exitCode=code, endedAt=now(), logSha256=digest(BASE/'queue.log'),
                      resultPresent=(BASE/'result.json').exists())
        save(BASE/'terminal.json', record)
sys.exit(code)
