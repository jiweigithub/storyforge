export function exercise(api, fixture, edgeKinds) {
  const equal = (a, b, label) => { if (a !== b) throw Error(`${label}: mismatch`) }
  const rejected = (fn, label) => { let threw = false; try { fn() } catch { threw = true } if (!threw) throw Error(`${label}: expected rejection`) }
  const strings = ['', 'a', 'a\u0000b', '汉字', '😀', '\uE000', '"\\\r\n\t', '\u0000\b\f', '\u2028\u2029', '\ufeff', 'e\u0301', 'é', 'x'.repeat(1025), '__proto__']
  const tuples = []
  for (const a of strings) for (const b of strings) for (const c of strings) tuples.push([a, b, c])
  for (const tuple of tuples) equal(api.compilerTupleKey(...tuple), api.canonicalNarrativeJsonV1(tuple), 'tuple bytes')
  const tupleKeys = tuples.map(tuple => api.compilerTupleKey(...tuple))
  equal(new Set(tupleKeys).size, tuples.length, 'tuple injectivity')
  // Embedded delimiters, quote escapes, raw scalars and prefix numbers remain
  // byte-for-byte canonical; existing scalar-order sorting is unchanged.
  const numbers = [0, 1, 2, 9, 10, 11, 99, 100, 101, 999, 1000, Number.MAX_SAFE_INTEGER]
  const edges = fixture.adjacency.map(row => ({ ...row, evidenceIds: [...row.evidenceIds] }))
  for (let i = 0; i < 1200; i++) edges.push({ fromIndex: numbers[i % numbers.length], toIndex: numbers[Math.floor(i / numbers.length) % numbers.length],
    kind: edgeKinds[i % edgeKinds.length], role: i % 5 ? strings[i % strings.length] : null,
    evidenceIds: Array.from({ length: i % 4 }, (_, n) => strings[(i + n) % strings.length]).sort(api.compareUnicodeCodePointsV1),
    basis: i % 2 ? 'declared' : 'structural' })
  edges.push(...edges.slice(0, 200).map(row => ({ ...row, evidenceIds: [...row.evidenceIds] })))
  for (const row of edges) equal(api.compilerEdgeKey(row), api.canonicalNarrativeJsonV1(row), 'edge bytes')
  const dedupe = (rows, encode) => [...new Map(rows.map(row => [encode(row), row])).entries()]
    .sort(([a], [b]) => api.compareUnicodeCodePointsV1(a, b)).map(([, row]) => row)
  const assertRows = (actual, expected) => { equal(actual.length, expected.length, 'dedupe count'); actual.forEach((row, i) => equal(row, expected[i], 'canonical ordering and last duplicate identity')) }
  assertRows(dedupe(edges, api.compilerEdgeKey), dedupe(edges, api.canonicalNarrativeJsonV1))
  for (const bad of ['\ud800', '\udc00', 'a\ud800b']) {
    rejected(() => api.compilerTupleKey(bad, '', ''), 'tuple unicode')
    rejected(() => api.canonicalNarrativeJsonV1([bad, '', '']), 'reference unicode')
    for (const field of ['kind', 'role', 'basis']) rejected(() => api.compilerEdgeKey({ ...edges[0], [field]: bad }), 'edge unicode')
    rejected(() => api.compilerEdgeKey({ ...edges[0], evidenceIds: [bad] }), 'evidence unicode')
  }
  for (const bad of [undefined, null, 1, {}, []]) rejected(() => api.compilerTupleKey(bad, '', ''), 'tuple scalar domain')
  for (const bad of [-1, -0, 1.5, NaN, Infinity, Number.MAX_SAFE_INTEGER + 1]) {
    rejected(() => api.compilerEdgeKey({ ...edges[0], fromIndex: bad }), 'nonnegative safe index domain')
  }
  const mutable = { ...edges[0], evidenceIds: ['original'] }
  const before = api.compilerEdgeKey(mutable)
  mutable.evidenceIds.push('changed😀'); mutable.role = 'changed-role'
  equal(api.compilerEdgeKey(mutable), api.canonicalNarrativeJsonV1(mutable), 'mutation readback')
  if (api.compilerEdgeKey(mutable) === before) throw Error('stale memoized key')
  const descriptor = Object.getOwnPropertyDescriptor(Object.prototype, 'toJSON')
  if (descriptor) throw Error('unexpected inherited toJSON before isolated test')
  let inheritedCalls = 0
  try {
    Object.defineProperty(Object.prototype, 'toJSON', { configurable: true, value() { inheritedCalls++; return 'wrong' } })
    equal(api.compilerEdgeKey(mutable), api.canonicalNarrativeJsonV1(mutable), 'inherited hook ignored')
    equal(api.compilerTupleKey('a', 'b', 'c'), '["a","b","c"]', 'tuple hook ignored')
  } finally { delete Object.prototype.toJSON }
  equal(inheritedCalls, 0, 'inherited hook calls')

  const timingTuples = Array.from({ length: 20000 }, (_, i) => ['source:' + i, 'revision:' + (i % 31), '/field/' + strings[i % 12]])
  const timingEdges = Array.from({ length: 40000 }, (_, i) => ({ fromIndex: i % 10845, toIndex: (i * 7) % 10845,
    kind: edgeKinds[i % edgeKinds.length], role: i % 3 ? 'role:' + (i % 17) : null,
    evidenceIds: ['evidence:' + (i % 10845), '证据😀:' + (i % 37)], basis: i % 2 ? 'declared' : 'structural' }))
  timingEdges.push(...timingEdges.slice(0, 5000).map(row => ({ ...row, evidenceIds: [...row.evidenceIds] })))
  const expectedTuples = timingTuples.map(tuple => api.canonicalNarrativeJsonV1(tuple))
  const expectedEdges = dedupe(timingEdges, api.canonicalNarrativeJsonV1)
  const operations = [
    { name: 'tuple-keys', rows: timingTuples.length, run: variant => timingTuples.map(tuple => variant === 'baseline' ? api.canonicalNarrativeJsonV1(tuple) : api.compilerTupleKey(...tuple)),
      verify: actual => { equal(actual.length, expectedTuples.length, 'tuple sample length'); actual.forEach((key, i) => equal(key, expectedTuples[i], 'tuple sample bytes')) } },
    { name: 'edge-deduplicate-sort', rows: timingEdges.length, run: variant => dedupe(timingEdges, variant === 'baseline' ? api.canonicalNarrativeJsonV1 : api.compilerEdgeKey), verify: actual => assertRows(actual, expectedEdges) },
  ]
  const samples = []
  for (const operation of operations) {
    for (const variant of ['baseline', 'candidate']) operation.verify(operation.run(variant))
    for (let pair = 1; pair <= 7; pair++) for (const variant of pair % 2 ? ['baseline', 'candidate'] : ['candidate', 'baseline']) {
      const started = performance.now(), output = operation.run(variant), elapsedMs = performance.now() - started
      operation.verify(output)
      samples.push({ operation: operation.name, pair, variant, rows: operation.rows, elapsedMs })
    }
  }
  const median = xs => [...xs].sort((a, b) => a - b)[Math.floor(xs.length / 2)]
  const summaries = operations.map(op => {
    const groups = Object.fromEntries(['baseline', 'candidate'].map(variant => [variant, samples.filter(row => row.operation === op.name && row.variant === variant).map(row => row.elapsedMs)]))
    return { operation: op.name, pairs: 7, baselineMedianMs: median(groups.baseline), candidateMedianMs: median(groups.candidate),
      pairedRatios: groups.baseline.map((value, i) => groups.candidate[i] / value) }
  })
  return { conformance: { tupleCases: tuples.length, edgeCases: edges.length, originalGoldenEdges: fixture.adjacency.length,
    exactBytes: true, injectiveTuples: true, canonicalOrderingAndLastDuplicateIdentity: true, mutableReadback: true,
    malformedUnicodeRejected: true, scalarDomainRejected: true, inheritedToJSONCalls: inheritedCalls, passed: true }, samples, summaries }
}
