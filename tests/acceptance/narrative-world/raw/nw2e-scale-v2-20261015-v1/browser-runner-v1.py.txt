from pathlib import Path
import datetime, hashlib, json, os, shutil, socket, subprocess, sys

root = Path('/Users/qinyingying/.codex/worktrees/narrative-world-design/storyforge')
fingerprint = Path('/tmp/nw2e-scale-v2-frozen-sources-v1.json')
frozen = json.loads(fingerprint.read_text())
terminal = Path('/tmp/nw2e-scale-v2-browser-v1-terminal.json')
report = Path('/tmp/nw2e-scale-v2-browser-v1.json')
assert not terminal.exists() and not report.exists()
record = {'schema': 'storyforge.scale-v2-browser-terminal', 'version': 1,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'fingerprintSha256': hashlib.sha256(fingerprint.read_bytes()).hexdigest(),
          'boundary': 'Two single browser diagnostics on separately frozen compiler-v2 vectors. Not formal repeated performance or author acceptance.',
          'modelCalls': 0, 'accepted': False}
code = 2

def verify():
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip() == frozen['baseCommit']
    for path, expected in frozen['sourceHashes'].items():
        assert hashlib.sha256((root / path).read_bytes()).hexdigest() == expected, 'Frozen diagnostic source changed: ' + path

try:
    verify()
    record['sourceVerifiedBefore'] = True
    record['freeBytesAtAdmission'] = shutil.disk_usage(root).free
    if record['freeBytesAtAdmission'] < 5 * 1024**3:
        raise RuntimeError('Insufficient storage before browser dispatch')
    with socket.socket() as sock:
        sock.bind(('127.0.0.1', 0))
        port = str(sock.getsockname()[1])
    env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1', PLAYWRIGHT_PORT=port,
               PLAYWRIGHT_HTML_OUTPUT_DIR='/tmp/nw2e-scale-v2-browser-v1-html',
               NW_SCALE_BROWSER_MODE='diagnostic', NW_SCALE_PROFILE='compiler-v2',
               NW_SCALE_BROWSER_SIZES='100000,1000000', NW_SCALE_BROWSER_REPORT=str(report),
               NW_SCALE_BROWSER_VECTORS='tests/acceptance/narrative-world/scale-vectors-20261015-v2.json',
               NW_SCALE_BROWSER_SOURCES=str(fingerprint))
    env.pop('NW_SCALE_BROWSER_PROFILE_DIR', None)
    command = ['npx', 'playwright', 'test', 'tests/e2e/narrative-scale.spec.ts', '--reporter=line', '--retries=0',
               '--output=/tmp/nw2e-scale-v2-browser-v1-test-results']
    record['command'] = command
    print('Starting compiler-v2 scale diagnostics in isolated browser data.', flush=True)
    code = subprocess.call(command, cwd=root, env=env)
    verify()
    record['sourceUnchanged'] = True
    if code == 0:
        measured = json.loads(report.read_text())
        assert measured['mode'] == 'diagnostic' and measured['profile'] == 'compiler-v2'
        assert measured['sourceUnchanged'] is True and measured['accepted'] is False and measured['modelCalls'] == 0
        record['report'] = str(report)
        record['reportSha256'] = hashlib.sha256(report.read_bytes()).hexdigest()
        record['performanceFailures'] = measured['failures']
except BaseException as error:
    record['error'] = type(error).__name__ + ': ' + str(error)
    code = 2
finally:
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    with terminal.open('x') as stream:
        json.dump(record, stream, ensure_ascii=False, indent=2)
        stream.write('\n')
print('Compiler-v2 diagnostic terminal exit=' + str(code) + '; acceptance remains separate.', flush=True)
sys.exit(code)
