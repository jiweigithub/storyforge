const sha = value => createHash('sha256').update(typeof value === 'string' || Buffer.isBuffer(value) ? value : JSON.stringify(value)).digest('hex')
const bytes = path => readFile(resolve(root, path))
const json = async path => JSON.parse(await bytes(path))
const leaves = value => typeof value === 'string' ? [value] : Array.isArray(value) ? value.flatMap(leaves)
  : value && typeof value === 'object' ? Object.values(value).flatMap(leaves) : []
const { validateJudgeProtocol, validatePriorJudgeCall, judgeExecutionCounts, summarizeJudge } = await import(pathToFileURL(resolve(root, 'scripts/narrative-reader-judge-contract.mjs')))

assert.deepEqual(manifest.dispatchOrder, parentManifest.dispatchOrder)
const verifyHashes = async map => {
  for (const [path, expected] of Object.entries(map)) assert.equal(sha(await bytes(path)), expected, `Changed evidence: ${path}`)
}
await verifyHashes(manifest.sourceHashes)
await verifyHashes(manifest.runtimeHashes)
let runtimeArchivedFiles = 0
for (const run of [parent, child]) {
  const snapshot = await json(`${run}/runtime-sources.json`)
  assert.deepEqual(Object.keys(snapshot.files).sort(), Object.keys(manifest.runtimeHashes).sort())
  for (const [path, value] of Object.entries(snapshot.files)) {
    assert.equal(sha(value.source), value.sha256)
    assert.equal(value.sha256, manifest.runtimeHashes[path])
    runtimeArchivedFiles++
  }
}

const calls = new Map(), rawHashes = {}, byRun = {}, callLocations = new Map()
for (const run of [parent, child]) {
  const files = (await readdir(resolve(root, run, 'calls'))).sort()
  const resultNames = files.filter(name => name.endsWith('.result.json'))
  assert.equal(files.length, resultNames.length * 3, 'Every attempt requires request, start and result; no orphan files')
  const runCounts = { attempts: 0, completed: 0, failed: 0 }
  for (const name of resultNames) {
    const hash = name.slice(0, -'.result.json'.length)
    assert(!calls.has(hash), 'A prior prompt was repeated in the continuation')
    const prefix = `${run}/calls/${hash}`
    const sent = await json(`${prefix}.request.json`)
    const started = await json(`${prefix}.started.json`)
    const result = await json(`${prefix}.result.json`)
    const payload = JSON.parse(sent.messages.find(message => message.role === 'user').content)
    assert(!Object.hasOwn(payload.response, 'status'), 'Reader status must be hidden from v2 judge')
    const prepared = { promptHash: hash, messages: sent.messages,
      referenceTexts: Object.fromEntries(payload.referenceSources.map(source => [source.sourceId, leaves(source.value)])) }
    const checked = validatePriorJudgeCall({ started, request: sent, result }, prepared, protocol)
    assert.deepEqual(checked, result)
    assert.equal(sent.messages[0].content, protocol.systemPrompt)
    assert(manifest.cases.some(row => row.judgePromptHash === hash))
    calls.set(hash, result); callLocations.set(hash, prefix)
    runCounts.attempts++; runCounts[result.status]++
    for (const suffix of ['request', 'started', 'result']) rawHashes[`${prefix}.${suffix}.json`] = sha(await bytes(`${prefix}.${suffix}.json`))
  }
  for (const name of (await readdir(resolve(root, run))).sort()) {
    if (name.endsWith('.json')) rawHashes[`${run}/${name}`] = sha(await bytes(`${run}/${name}`))
  }
  byRun[run] = runCounts
}

const count = selected => {
  const values = selected.map(row => calls.get(row.judgePromptHash))
  const assessments = Object.fromEntries(['correct', 'partly-correct', 'incorrect', 'justified-abstention', 'unjustified-abstention', 'cannot-grade'].map(label => [label, values.filter(call => call.status === 'completed' && call.judgment.assessment === label).length]))
  return { planned: selected.length, graded: values.filter(call => call.status === 'completed').length,
    ungraded: values.filter(call => call.status !== 'completed').length, assessments,
    substantive: values.filter(call => call.status === 'completed' && call.judgment.answerDisposition === 'substantive-answer').length,
    abstention: values.filter(call => call.status === 'completed' && call.judgment.answerDisposition === 'abstention').length }
}

const byConditionAndOperation = Object.fromEntries(conditions.map(condition => [condition, Object.fromEntries(operations.map(operation => [operation,

const bySplitAndCondition = Object.fromEntries(Object.keys(splitFamilies).map(split => [split, Object.fromEntries(conditions.map(condition => [condition,
  count(manifest.cases.filter(row => row.condition === condition && row.split === split))]))]))
const byFamilyAndCondition = Object.fromEntries(familyIds.map(id => [id, Object.fromEntries(conditions.map(condition => [condition,
  count(manifest.cases.filter(row => row.condition === condition && row.familyId === id))]))]))

// Exploratory resampling of this finite synthetic corpus, not independent model reruns.
// Resample families, with the same draw across all conditions; retain 8/4 split strata.
let randomState = 20261015
const random = () => { randomState ^= randomState << 13; randomState ^= randomState >>> 17; randomState ^= randomState << 5; return (randomState >>> 0) / 4294967296 }
const reps = 10000
const quantile = (values, probability) => {
  const a = [...values].sort((x, y) => x - y), position = (a.length - 1) * probability
  return a[Math.floor(position)] + (a[Math.ceil(position)] - a[Math.floor(position)]) * (position % 1)
}
const describe = values => ({ lower: quantile(values, .025), upper: quantile(values, .975) })
const pairRows = condition => manifest.cases.filter(row => row.condition === 'ir-lexical').flatMap(ir => {
  const other = manifest.cases.find(row => row.questionId === ir.questionId && row.condition === condition)
  const a = calls.get(ir.judgePromptHash), b = calls.get(other.judgePromptHash)
  return a.status !== 'completed' || b.status !== 'completed' ? [] : [{ familyId: ir.familyId,
    difference: Number(a.judgment.assessment === 'correct') - Number(b.judgment.assessment === 'correct') }]
})
const comparators = conditions.filter(condition => condition !== 'ir-lexical')
const paired = Object.fromEntries(comparators.map(condition => [condition, pairRows(condition)]))
const distribution = Object.fromEntries(conditions.map(condition => [condition, []]))
const differences = Object.fromEntries(comparators.map(condition => [condition, []]))
for (let iteration = 0; iteration < reps; iteration++) {
  const selected = Object.values(splitFamilies).flatMap(ids => ids.map(() => ids[Math.floor(random() * ids.length)]))
  for (const condition of conditions) {
    const records = selected.map(id => byFamilyAndCondition[id][condition])
    distribution[condition].push(records.reduce((sum, value) => sum + value.assessments.correct, 0) / records.reduce((sum, value) => sum + value.planned, 0))
  }
  for (const condition of comparators) {
    const selectedPairs = selected.flatMap(id => paired[condition].filter(row => row.familyId === id))
    differences[condition].push(selectedPairs.reduce((sum, value) => sum + value.difference, 0) / selectedPairs.length)
  }
}
const bootstrap = { schema: 'storyforge.descriptive-family-bootstrap', version: 1, replicates: reps,

  unit: 'original synthetic family; not case, condition, shared prompt or model call', strata: splitFamilies,
  pairedAcrossConditions: true, preregisteredInference: false,
  metric: 'Confirmed-correct answers / planned questions; failed judgment remains unknown, not classified incorrect. Pair differences use only questions graded in both conditions.',
  limitation: 'Post hoc descriptive intervals for 12 fixed synthetic families. Shared prompts and one judge draw do not become independent replications. Not a population, repeated-model, human or product quality acceptance claim.',
  confirmedCorrectFraction: Object.fromEntries(conditions.map(condition => [condition, {
    estimate: byCondition[condition].assessments.correct / byCondition[condition].planned,
    numerator: byCondition[condition].assessments.correct, denominator: byCondition[condition].planned,
    ungraded: byCondition[condition].ungraded,
    unknownJudgmentRange: [byCondition[condition].assessments.correct / byCondition[condition].planned,
      (byCondition[condition].assessments.correct + byCondition[condition].ungraded) / byCondition[condition].planned],
    exploratory95PercentileInterval: describe(distribution[condition]),
  }])),
  irMinusComparatorCompletePairs: Object.fromEntries(comparators.map(condition => [condition, {

      ir.readerPromptHash !== manifest.cases.find(row => row.questionId === ir.questionId && row.condition === condition).readerPromptHash).length,
    featureBenefitEstablished: false,
  }])),
}

  const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2) + '\n'

}
