def observe():
    rows=subprocess.check_output(['ps','-axo','pid=,ppid=,command='],text=True)
    parsed={int(f[0]):f for line in rows.splitlines() if len(f:=line.strip().split(None,2))==3}
    descendants={os.getpid()}
    while True:
        more={p for p,f in parsed.items() if int(f[1]) in descendants}
        if more<=descendants:break
        descendants|=more
    fresh=descendants-owned_pids
    owned_pids.update(descendants)
    if fresh:
        with Path(str(BASE)+'-process-observations.jsonl').open('a') as f:
            f.write(json.dumps({'at':now(),'pids':sorted(descendants),'processes':[parsed[p] for p in sorted(descendants) if p in parsed]})+'\n')
