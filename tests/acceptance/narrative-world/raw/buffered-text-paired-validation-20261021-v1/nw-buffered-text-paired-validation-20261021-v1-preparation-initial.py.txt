from pathlib import Path
import datetime, hashlib, json, shutil

root=Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
base=Path('/tmp/nw-buffered-text-paired-validation-20261021-v1')
snapshots=Path('/tmp/nw-buffered-text-pair-snapshots-20261021-v1')
baseline_source=Path('/tmp/nw-buffered-text-probe-20261021-v1/baseline/json.ts')
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
assert sha(baseline_source)=='fc1fe9db86daaf85808ecc1d1754a0dc524a9d57e1c63625edc87d8101bf52b8'
assert sha(root/'src/lib/narrative-domain/json.ts')=='5da9d19ed5aff738e1dee9c42afb1bd67257e2bcca38d52f6adf64213923b23c'
assert not snapshots.exists()
scale=json.loads(Path('/tmp/nw-buffered-text-validation-20261021-v2-scale-sources.json').read_text())
entries=['src','public','data','showcase','tests/helpers','scripts','ui-preview',
    'index.html','package.json','package-lock.json','postcss.config.js','tailwind.config.ts',
    'tsconfig.json','vite.config.ts','playwright.config.ts','tests/e2e/narrative-scale.spec.ts']
source_paths=set(scale['sourceHashes'])
for name in entries:
    path=root/name
    if path.is_dir():source_paths.update(str(p.relative_to(root)) for p in path.rglob('*') if p.is_file() and not any(part.startswith('.env') for part in p.relative_to(root).parts))
    else:source_paths.add(name)
origin_hashes={name:sha(root/name) for name in sorted(source_paths)}
variant_hashes={}
for variant in ['baseline','candidate']:
    target=snapshots/variant
    target.mkdir(parents=True)
    for name in sorted(source_paths):
        dest=target/name;dest.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(root/name,dest)
    if variant=='baseline':shutil.copy2(baseline_source,target/'src/lib/narrative-domain/json.ts')
    (target/'node_modules').symlink_to(root/'node_modules',target_is_directory=True)
    hashes={name:sha(target/name) for name in sorted(source_paths)}
    differences=[name for name in source_paths if hashes[name]!=origin_hashes[name]]
    assert differences==([] if variant=='candidate' else ['src/lib/narrative-domain/json.ts'])
    variant_hashes[variant]=hashes
    variant_scale={'sourceHashes':{p:hashes[p] for p in scale['sourceHashes']}}
    with Path(str(base)+'-'+variant+'-scale-sources.json').open('x') as f:json.dump(variant_scale,f,indent=2);f.write('\n')
runtime_prior=json.loads(Path('/tmp/nw-buffered-text-browser-probe-20261021-v1/pin.json').read_text())
runtime={p:v for p,v in runtime_prior['files'].items() if not p.startswith('/tmp/')}
for folder in ['vite','@vitejs/plugin-react','vite-plugin-pwa']:
    for p in (root/'node_modules'/folder).rglob('*'):
        if p.is_file():runtime[str(p)]=sha(p)
for p,v in runtime.items():assert sha(p)==v, p
