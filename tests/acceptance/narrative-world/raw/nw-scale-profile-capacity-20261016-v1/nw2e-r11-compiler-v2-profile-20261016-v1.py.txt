from pathlib import Path
import datetime, hashlib, json, os, shutil, socket, subprocess, sys

root = Path('/Users/qinyingying/.codex/worktrees/nw2e-verification-r11/storyforge')
manifest_path = Path('/tmp/nw2e-frozen-files-r11.json')
manifest = json.loads(manifest_path.read_text())
prefix = Path('/tmp/nw2e-r11-compiler-v2-profile-20261016-v1')
terminal = Path(str(prefix) + '-terminal.json')
report = Path(str(prefix) + '.json')
fingerprint = Path('/tmp/nw2e-r11-compiler-v2-profile-sources-20261016-v1.json')
assert not terminal.exists() and not report.exists()

def verify():
    assert Path.cwd() == root
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip() == manifest['baseCommit']
    bad = [p for p, h in manifest['files'].items()
           if hashlib.sha256((root / p).read_bytes()).hexdigest() != h]
    assert not bad, 'Frozen source drift: ' + json.dumps(bad)

record = {'schema': 'storyforge.scale-profile-terminal', 'version': 1,
          'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'sourceGeneration': 'r11', 'sourceTree': manifest['sourceTree'],
          'sourceCheckpoint': 29, 'productionVersion': 28,
          'boundary': 'Instrumented compiler-v2 million-character CPU profile. Diagnostic only; not an uninstrumented performance sample or formal acceptance.',
          'accepted': False, 'modelCalls': 0, 'sourceUnchanged': False}
code = 2
try:
    verify()
    assert shutil.disk_usage(root).free >= 5 * 1024**3, 'Insufficient free space; no browser dispatched'
    with socket.socket() as sock:
        sock.bind(('127.0.0.1', 0)); port = sock.getsockname()[1]
    env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1', PLAYWRIGHT_PORT=str(port),
               PLAYWRIGHT_HTML_OUTPUT_DIR=str(prefix) + '-html',
               NW_SCALE_BROWSER_MODE='diagnostic', NW_SCALE_PROFILE='compiler-v2',
               NW_SCALE_BROWSER_SIZES='1000000', NW_SCALE_BROWSER_REPORT=str(report),
               NW_SCALE_BROWSER_VECTORS='tests/acceptance/narrative-world/scale-vectors-20261015-v2.json',
               NW_SCALE_BROWSER_SOURCES=str(fingerprint), NW_SCALE_BROWSER_PROFILE_DIR=str(prefix))
    print('Starting one instrumented compiler-v2 million-character diagnostic.', flush=True)
    code = subprocess.call(['npx', 'playwright', 'test', 'tests/e2e/narrative-scale.spec.ts',
                            '--reporter=line', '--retries=0', '--output=' + str(prefix) + '-test-results'], env=env)
    verify(); record['sourceUnchanged'] = True
    if report.exists():
        result = json.loads(report.read_text())
        record.update(report=str(report), reportSha256=hashlib.sha256(report.read_bytes()).hexdigest(),
                      failures=result['failures'], cpuProfiling=result['cpuProfiling'])
        assert result['profile'] == 'compiler-v2' and result['mode'] == 'diagnostic'
        assert result['sourceUnchanged'] and result['accepted'] is False
    elif code == 0:
        raise RuntimeError('Missing diagnostic report')
except BaseException as error:
    record['error'] = type(error).__name__ + ': ' + str(error)
    code = 2
finally:
    record.update(exitCode=code, endedAt=datetime.datetime.now(datetime.timezone.utc).isoformat())
    with terminal.open('x') as stream:
        json.dump(record, stream, ensure_ascii=False, indent=2); stream.write('\n')
print('Profile terminal exit=' + str(code) + '; formal performance remains separate.', flush=True)
sys.exit(code)
