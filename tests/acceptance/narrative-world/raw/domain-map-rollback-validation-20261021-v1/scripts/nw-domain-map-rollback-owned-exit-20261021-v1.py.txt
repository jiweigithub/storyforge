from pathlib import Path
import datetime
import hashlib
import json
import os
import re
import subprocess

PREFIX = Path('/tmp/nw-domain-map-rollback-validation-20261021-v1')
ROOT = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
OUT = Path(str(PREFIX) + '-owned-exit.json')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    assert not OUT.exists()
    runner_path = Path(str(PREFIX) + '-terminal.json')
    queue_path = Path(str(PREFIX) + '-queue-r1-terminal.json')
    runner = json.loads(runner_path.read_text())
    queue = json.loads(queue_path.read_text())
    assert runner.get('endedAt') and queue.get('endedAt')
    assert runner['sourceUnchanged'] and queue['runnerTerminalPresent']
    assert runner['exitCode'] == queue['exitCode']
    pids = {runner['runnerPid'], queue['queuePid'], queue['supervisorPid']}
    pids.update(phase['childPid'] for phase in runner['phases'])
    for attempt in PREFIX.parent.glob(PREFIX.name + '-queue-r*-terminal.json'):
        prior = json.loads(attempt.read_text())
        pids.update([prior['queuePid'], prior['supervisorPid']])
    observations = []
    for file in PREFIX.parent.glob('nw-domain-map-rollback-validation-20261021-v1*observation*.json'):
        observations.append(file)
    observations.extend(PREFIX.parent.glob('nw-domain-map-rollback-validation-20261021-v1*heartbeat*.json'))
    for file in observations:
        data = json.loads(file.read_text())
        family = data.get('sharedQueue', {}).get('recordedOwnerFamily', [])
        if any(item['pid'] == queue['queuePid'] for item in family):
            pids.update(item['pid'] for item in family)
    process_text = subprocess.check_output(
        ['ps', '-axo', 'pid=,ppid=,stat=,command='], text=True
    )
    processes = {}
    for line in process_text.splitlines():
        match = re.match(r'\s*(\d+)\s+(\d+)\s+(\S+)\s+(.*)', line)
        if match:
            pid, parent, state, command = match.groups()
            processes[int(pid)] = {
                'pid': int(pid), 'ppid': int(parent),
                'state': state, 'command': command,
            }
    assert not pids.intersection(processes), [
        processes[pid] for pid in pids.intersection(processes)
    ]
    ancestors = {os.getpid()}
    current = os.getpid()
    while current in processes and processes[current]['ppid'] not in ancestors:
        current = processes[current]['ppid']
        ancestors.add(current)
    matching = [
        item for pid, item in processes.items()
        if pid not in ancestors and str(PREFIX) in item['command']
    ]
    assert not matching, matching
    cwd_query = subprocess.run(
        ['lsof', '-nP', '-d', 'cwd', '-Fpcn'], capture_output=True, text=True
    )
    assert cwd_query.returncode in (0, 1), cwd_query.stderr
    current_pid = None
    worktree_processes = []
    for line in cwd_query.stdout.splitlines():
        if line.startswith('p'):
            current_pid = int(line[1:])
        elif line.startswith('n') and current_pid not in ancestors:
            cwd = Path(line[1:])
            if cwd == ROOT or ROOT in cwd.parents:
                worktree_processes.append({'pid': current_pid, 'cwd': str(cwd)})
    assert not worktree_processes, worktree_processes
    result = {
        'schema': 'storyforge.narrative-owned-process-exit-verification',
        'version': 1,
        'observedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
        'runnerTerminal': {'path': str(runner_path), 'sha256': sha(runner_path)},
        'queueTerminal': {'path': str(queue_path), 'sha256': sha(queue_path)},
        'recordedPids': sorted(pids),
        'allRecordedPidsAbsent': True,
        'liveAttemptMatches': matching,
        'liveWorktreeCwds': worktree_processes,
        'observations': [{'path': str(p), 'sha256': sha(p)} for p in sorted(set(observations))],
        'limitation': 'Recorded descendant observations are sampled, not a continuous inventory. Exact attempt command and worktree cwd scans are also empty.',
        'signalsSent': [],
        'verifier': {'path': __file__, 'sha256': sha(Path(__file__))},
    }
    with OUT.open('x') as stream:
        json.dump(result, stream, ensure_ascii=False, indent=2)
        stream.write('\n')
    print(json.dumps({'receipt': str(OUT), 'recordedPids': len(pids)}))


if __name__ == '__main__':
    main()
