import { afterEach, beforeEach, describe, expect, it } from 'vitest'
import { db } from '../../src/lib/db/schema'
import { stampNewRecord } from '../../src/lib/workspace/scope'
import { projectNarrativeDomainsV1 } from '../../src/lib/world-engine/narrative-domain-projection'
import { hashNarrativeValueV1 as hash } from '../../src/lib/world-engine/narrative-json'
import { buildWorldImpactPlanV1 } from '../../src/lib/consistency/world-impact'
import { seedConfirmedNarrativeDomainV1 } from '../helpers/narrative-domain-fixture'

/** These explicit test-author declarations exercise the production confirmation
 * and projection boundaries. They are not additions to the semantic edit gold. */
async function orderedEvents(explicitRequirement = false) {
  const source = await seedConfirmedNarrativeDomainV1()
  const laterId = await db.storyTimelineEvents.add(stampNewRecord(source.scope, 'storyTimelineEvents', {
    projectId: source.scope.projectId, title: '后来开门', description: '后来，守门人打开了驿站的大门。',
    importance: 1, order: 1, createdAt: 1_800_000_000_000,
  } as never, { owner: 'work' })) as number
  const later = (await db.storyTimelineEvents.get(laterId))!
  const laterRef = { kind: 'event' as const, source: { storyEventId: laterId, sourceUid: later.ragDocumentId! }, subKey: 'root' }
  await source.confirm('storyTimelineEvents', laterId, {
    narrativeVisibilityV1: { class: 'public', audienceKeys: [], policyVersion: 'fixture-author-v1' },
    narrativeEventV1: { eventType: 'action', participants: [], locations: [], propositions: [], effects: [], communication: null, observer: null, before: [] },
    narrativeTimeV1: { occurred: { start: { kind: 'tick', timelineKey: 'postal-days', value: '3' }, end: null, endMeaning: 'open' },
      valid: null, acquiredAt: null, narratedAt: null, disclosedAt: [] },
    narrativeEvidenceV1: [await source.selfEvidence('storyTimelineEvents', laterId, 'description')],
    ...(explicitRequirement ? { narrativeDependenciesV1: [{ target: source.eventRef, role: 'requires' as const, evidenceKeys: ['self'] }] } : {}),
  })
  const earlier = (await db.storyTimelineEvents.get(source.event.id))!
  await source.confirm('storyTimelineEvents', source.event.id, {
    narrativeEventV1: { ...earlier.narrativeEventV1!, before: [laterRef] },
  })
  const projection = await projectNarrativeDomainsV1(source.input)
  const earlierAtom = projection.bundle.records.find(record => record.provenance.sourceResourceKey.endsWith(source.event.row.ragDocumentId))!
  const laterAtom = projection.bundle.records.find(record => record.provenance.sourceResourceKey.endsWith(later.ragDocumentId!))!
  expect(projection.bundle.timeContext.before).toContainEqual({ beforeEventKey: earlierAtom.semanticId, afterEventKey: laterAtom.semanticId })
  // The earlier event really does carry the later reference for closure. Impact
  // must distinguish that bookkeeping edge from an evidenced requirement.
  expect(earlierAtom.dependencies).toContainEqual(expect.objectContaining({ semanticId: laterAtom.semanticId, role: 'requires' }))
  const base = { projection, semanticRootHash: await hash(projection), lockedSemanticIds: [] }
  const oldRelease = await db.worldReleases.get(source.release.id!)
  const foreignChapter = await db.chapters.get(source.foreignChapterId)
  async function changeDescription(id: number) {
    const row = (await db.storyTimelineEvents.get(id))!
    await db.storyTimelineEvents.update(id, { description: row.description + ' 作者确认了新的现场细节。' })
    await source.confirm('storyTimelineEvents', id, {
      narrativeEvidenceV1: [await source.selfEvidence('storyTimelineEvents', id, 'description')],
    })
    const nextProjection = await projectNarrativeDomainsV1(source.input)
    const plan = await buildWorldImpactPlanV1({ base, next: {
      projection: nextProjection, semanticRootHash: await hash(nextProjection), lockedSemanticIds: [],
    } })
    expect(plan.writes).toEqual([])
    expect(await db.worldReleases.get(source.release.id!)).toEqual(oldRelease)
    expect(await db.chapters.get(source.foreignChapterId)).toEqual(foreignChapter)
    return plan
  }
  return { source, laterId, earlierAtom, laterAtom, base, changeDescription }
}

describe('NW-2D · 已确认先后顺序与有证据依赖的实际投影', () => {
  beforeEach(async () => { await db.delete(); await db.open() })
  afterEach(() => db.close())

  it('只声明 A 先于 B 时，修改 B 不把 A 的闭包引用当成反向硬依赖', async () => {
    const fixture = await orderedEvents()
    const plan = await fixture.changeDescription(fixture.laterId)
    expect(plan.items.find(item => item.key === fixture.earlierAtom.semanticId)).toMatchObject({
      disposition: 'review', confidence: 'unknown-impact', unknownEdges: ['incomplete-dependency-coverage'],
    })
    expect(plan.edges.filter(edge => edge.from === fixture.laterAtom.semanticId && edge.to === fixture.earlierAtom.semanticId
      && edge.strength === 'hard')).toEqual([])
    expect(plan.staleSemanticIds).not.toContain(fixture.earlierAtom.semanticId)
  })

  it('修改 A 时，后续 B 保留实际先后路径和 review，不伪造因果', async () => {
    const fixture = await orderedEvents()
    const plan = await fixture.changeDescription(fixture.source.event.id)
    expect(plan.items.find(item => item.key === fixture.laterAtom.semanticId)).toMatchObject({
      disposition: 'review', confidence: 'unknown-impact', pathRootKey: fixture.earlierAtom.semanticId,
      path: [expect.objectContaining({ from: fixture.earlierAtom.semanticId, to: fixture.laterAtom.semanticId, kind: 'sequence', strength: 'review' })],
    })
    expect(plan.staleSemanticIds).not.toContain(fixture.laterAtom.semanticId)
  })

  it('同一对事件另有已确认支持证据时，真实 requires 不会被时间顺序降级', async () => {
    const fixture = await orderedEvents(true)
    const proof = fixture.base.projection.dependencyEvidence.find(edge => edge.fromSemanticId === fixture.laterAtom.semanticId)!
    expect(proof).toMatchObject({ toSemanticId: fixture.earlierAtom.semanticId, role: 'requires' })
    expect(proof.evidenceIds).toHaveLength(1)
    const plan = await fixture.changeDescription(fixture.source.event.id)
    expect(plan.items.find(item => item.key === fixture.laterAtom.semanticId)).toMatchObject({
      disposition: 'stale', confidence: 'deterministic',
      path: [expect.objectContaining({ from: fixture.earlierAtom.semanticId, to: fixture.laterAtom.semanticId,
        kind: 'dependency', strength: 'hard', role: 'requires', evidenceIds: proof.evidenceIds })],
    })
    expect(plan.staleSemanticIds).toContain(fixture.laterAtom.semanticId)
    expect(plan.unknownCoverage.length).toBeGreaterThan(0)
  })

  it('跨原子伪造的支持证据仍拒绝，不能用顺序边遮蔽', async () => {
    const fixture = await orderedEvents(true)
    const corrupted = structuredClone(fixture.base)
    const proof = corrupted.projection.dependencyEvidence.find(edge => edge.fromSemanticId === fixture.laterAtom.semanticId)!
    proof.evidenceIds = [...fixture.earlierAtom.evidenceIds]
    await expect(buildWorldImpactPlanV1({ base: fixture.base, next: corrupted })).rejects.toThrow('impact-dependency-proof')
  })
})
