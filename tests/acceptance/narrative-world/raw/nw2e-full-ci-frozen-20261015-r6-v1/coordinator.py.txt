from pathlib import Path
import json, subprocess, sys, time

root = '/Users/qinyingying/.codex/worktrees/nw2e-verification-r6/storyforge'

def wait_success(mode):
    path = Path('/tmp/nw2e-r6-' + mode + '-terminal.json')
    while not path.exists():
        time.sleep(10)
    result = json.loads(path.read_text())
    if result['exitCode'] != 0 or not result['sourceUnchanged']:
        print('r6 ' + mode + ' failed; dependent checks not dispatched.', flush=True)
        sys.exit(2)

print('Waiting for exact r6 full CI before full native validation.', flush=True)
wait_success('ci')
print('r6 CI passed; requesting shared slot for full native validation.', flush=True)
log = Path('/tmp/nw2e-full-e2e-frozen-r6.log')
with log.open('x') as stream:
    status = subprocess.call(['python3', '/tmp/storyforge-serial-ci-20261013.py', '--',
                              'python3', '/tmp/nw2e-run-frozen-r6.py', 'e2e'], cwd=root, stdout=stream, stderr=subprocess.STDOUT)
if status:
    print('r6 native wrapper exited ' + str(status) + '; no diagnostic profile dispatched.', flush=True)
    sys.exit(status)
wait_success('e2e')
print('r6 full validation passed; requesting paired diagnostic slot, not formal performance acceptance.', flush=True)
log = Path('/tmp/nw2e-r6-paired-profile.log')
with log.open('x') as stream:
    status = subprocess.call(['python3', '/tmp/storyforge-serial-ci-20261013.py', '--',
                              'python3', '/tmp/nw2e-run-r6-paired-profile.py'], cwd=root, stdout=stream, stderr=subprocess.STDOUT)
print('Paired diagnostic wrapper exited ' + str(status), flush=True)
sys.exit(status)
