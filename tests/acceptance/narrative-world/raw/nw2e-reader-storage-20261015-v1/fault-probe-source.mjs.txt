import * as fs from 'node:fs'
import { resolve } from 'node:path'
import { createReaderReceiptStore, inspectReaderReceiptJournal, readerStorageDiagnostic } from '/Users/qinyingying/.codex/worktrees/narrative-world-design/storyforge/scripts/narrative-reader-storage.mjs'
import { sha256 } from '/Users/qinyingying/.codex/worktrees/narrative-world-design/storyforge/scripts/narrative-reader-contract.mjs'
const output = resolve(process.argv[2])
fs.mkdirSync(output)
const cases = []
for (const kind of ['low-space-before-dispatch', 'result-publication-ENOSPC', 'torn-journal-suffix']) {
  const directory = resolve(output, kind); fs.mkdirSync(directory); fs.mkdirSync(resolve(directory, 'calls'))
  const promptHash = sha256(`synthetic-storage-probe-only:${kind}`)
  const started = resolve(directory, 'calls', `${promptHash}.started.json`), result = resolve(directory, 'calls', `${promptHash}.result.json`)
  let deny = false
  const io = { ...fs, ...(kind === 'low-space-before-dispatch' ? { statfsSync: () => ({ bavail: 0n, bsize: 4096n }) } : {}),
    linkSync(...args) { if (deny) throw Object.assign(new Error('injected ENOSPC'), { code: 'ENOSPC' }); return fs.linkSync(...args) } }
  const store = createReaderReceiptStore(directory, io)
  let journal, failure = null, audit = null
  try {
    journal = store.begin({ promptHash, startedAt: '2026-10-15T00:00:00Z', model: 'synthetic-no-provider' }, started)
    journal.append('submission-intent', { fixture: true, providerInvoked: false })
    if (kind === 'result-publication-ENOSPC') {
      const value = { fixture: true, modelCalls: 0, status: 'completed', rawText: 'SYNTHETIC STORAGE FIXTURE; NOT A MODEL RESPONSE', usage: null }
      journal.append('generation-completed', value); journal.append('terminal-receipt', value)
      deny = true; store.writeJson(result, value)
    }
  } catch (error) { failure = readerStorageDiagnostic(error) }
  finally { journal?.close() }
  if (kind === 'torn-journal-suffix') fs.appendFileSync(journal.path, '{"body":{"kind":"terminal-receipt"')
  if (journal) audit = inspectReaderReceiptJournal(fs.readFileSync(journal.path), promptHash)
  const entry = { kind, promptHash, fixtureOnly: true, modelCalls: 0, failure, startExists: fs.existsSync(started),
    canonicalResultExists: fs.existsSync(result), forensicAudit: audit }
  fs.writeFileSync(resolve(directory, 'probe-outcome.json'), JSON.stringify(entry, null, 2) + '\n', { flag: 'wx' }); cases.push(entry)
}
if (cases[0].startExists || cases[0].failure?.code !== 'READER_STORAGE_LOW_SPACE'
  || cases[1].canonicalResultExists || !cases[1].forensicAudit?.generationCompletedEvidence
  || cases[1].forensicAudit.automaticRecoveryAllowed || cases[2].forensicAudit.corruption !== 'incomplete-final-line'
  || cases[2].forensicAudit.terminalEvidence) throw Error('Storage counterexample failed')
fs.writeFileSync(resolve(output, 'summary.json'), JSON.stringify({ schema: 'storyforge.reader-storage-fault-probe', version: 1,
  boundary: 'Real filesystem operations with injected failures and synthetic payloads; no model, no author data, no recovery of the historical unknown call.',
  cases, passed: 3, modelCalls: 0, accepted: false }, null, 2) + '\n', { flag: 'wx' })
console.log(JSON.stringify({ passed: 3, modelCalls: 0, accepted: false }))
