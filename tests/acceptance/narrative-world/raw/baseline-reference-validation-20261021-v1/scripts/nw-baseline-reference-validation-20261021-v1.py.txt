from pathlib import Path
import datetime
import hashlib
import json
import os
import shutil
import signal
import subprocess
import sys

root = Path('/Users/qinyingying/.codex/worktrees/nw-mainline-refresh-20261016/storyforge')
prefix = Path('/tmp/nw-baseline-reference-validation-20261021-v1')
pin_path = Path(str(prefix) + '-sources.json')
scale_path = Path(str(prefix) + '-scale-sources.json')
terminal = Path(str(prefix) + '-terminal.json')
pin = json.loads(pin_path.read_text())
assert not terminal.exists()


def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


pin_digest = digest(pin_path)
generated = {}


def verify():
    assert Path.cwd() == root
    assert digest(pin_path) == pin_digest
    assert digest(Path(__file__)) == pin['runnerSha256']
    assert digest(scale_path) == pin['scaleInputSha256']
    for ref, key in [('HEAD', 'head'), ('MERGE_HEAD', 'mergeHead')]:
        assert subprocess.check_output(['git', 'rev-parse', ref], text=True).strip() == pin[key]
    changed = [
        p for p, h in pin['files'].items()
        if not (root / p).is_file() or digest(root / p) != h
    ]
    assert not changed, 'Pinned sources changed: ' + json.dumps(changed)
    assert all(
        (root / p).is_file() and digest(root / p) == h for p, h in generated.items()
    ), 'Generated receipt changed after capture'


def stop(signum, frame):
    raise SystemExit(128 + signum)


signal.signal(signal.SIGTERM, stop)
signal.signal(signal.SIGINT, stop)
record = {
    'schema': 'storyforge.narrative-baseline-reference-validation',
    'version': 1, 'startedAt': utc(), 'runnerPid': os.getpid(),
    'root': str(root), 'head': pin['head'], 'mergeHead': pin['mergeHead'],
    'pinSha256': pin_digest, 'sourceFiles': len(pin['files']), 'phases': [],
    'sourceUnchanged': False, 'fullCI': False, 'fullE2E': False,
    'performanceAcceptance': False, 'humanQualityAcceptance': False,
    'actualModelCalls': 0, 'accepted': False,
}
env = dict(os.environ, CI='1', PLAYWRIGHT_FROZEN_WORKSPACE='1')
for key in list(env):
    if key.startswith('NW_'):
        env.pop(key, None)


def phase(name, command, extra=None):
    verify()
    free = shutil.disk_usage(root).free / 2**30
    assert free >= 5, 'Less than 5 GiB free; no new phase started'
    phase_env = dict(env)
    if extra:
        phase_env.update(extra)
    log = Path(str(prefix) + '-' + name + '.log')
    item = {
        'name': name, 'command': command, 'startedAt': utc(),
        'runnerPid': os.getpid(), 'log': str(log), 'freeGiB': free,
    }
    record['phases'].append(item)
    print(json.dumps({'starting': name, 'at': item['startedAt'], 'log': str(log)}), flush=True)
    with log.open('x') as output:
        child = subprocess.Popen(command, env=phase_env, stdout=output, stderr=subprocess.STDOUT)
        item['childPid'] = child.pid
        print(json.dumps({'phaseChild': name, 'pid': child.pid}), flush=True)
        result = child.wait()
    item.update(exitCode=result, endedAt=utc(), logSha256=digest(log))
    verify()
    print(json.dumps({'completed': name, 'exitCode': result}), flush=True)
    return result


class DirectedFailure(Exception):
    pass


code = 2

except BaseException as error:
