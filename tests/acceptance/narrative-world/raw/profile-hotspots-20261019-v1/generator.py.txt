from pathlib import Path
import collections,hashlib,json
prefix=Path('/tmp/nw-profile-ci-validation-20261019-v1')
result=Path('/tmp/nw-profile-ci-hotspots-20261019-v1.json')
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def frame(n):
 c=n['callFrame'];return (c.get('functionName') or '(anonymous)',c.get('url','').split('?')[0],c.get('lineNumber',-1)+1)
def cpu(path):
 x=json.loads(path.read_text());nodes={n['id']:n for n in x['nodes']};parents={c:n['id'] for n in x['nodes'] for c in n.get('children',[])}
 own=collections.Counter();tree=collections.Counter()
 for sample,delta in zip(x.get('samples',[]),x.get('timeDeltas',[])):
  own[frame(nodes[sample])]+=delta
  seen=set();cursor=sample
  while cursor in nodes:
   f=frame(nodes[cursor])
   if f not in seen:tree[f]+=delta;seen.add(f)
   if cursor not in parents:break
   cursor=parents[cursor]
 rows=lambda c:[{'function':f[0],'url':f[1],'line':f[2],'ms':round(t/1000,3)} for f,t in c.most_common(45)]
 return {'path':str(path),'sha256':sha(path),'elapsedMs':(x['endTime']-x['startTime'])/1000,'samples':len(x.get('samples',[])),'self':rows(own),'inclusive':rows(tree)}
def heap(path):
 x=json.loads(path.read_text());own=collections.Counter();tree=collections.Counter()
 def visit(n,ancestors):
  f=frame(n);own[f]+=n.get('selfSize',0)
  for a in set(ancestors+[f]):tree[a]+=n.get('selfSize',0)
  for child in n.get('children',[]):visit(child,ancestors+[f])
 visit(x['head'],[])
 rows=lambda c:[{'function':f[0],'url':f[1],'line':f[2],'MiB':round(n/2**20,3)} for f,n in c.most_common(45)]
 return {'path':str(path),'sha256':sha(path),'self':rows(own),'inclusive':rows(tree)}
terminal=json.loads(Path(str(prefix)+'-terminal.json').read_text());assert terminal['sourceUnchanged'] and terminal.get('endedAt')
assert terminal.get('profile')
for p,h in terminal['profile']['artifacts'].items():assert sha(Path(p))==h
out={'schema':'storyforge.narrative-profile-hotspots','version':1,'evidence':'Postprocessed actual CPU and heap allocation profiles; sampled allocations include collected objects, not peak RSS','pinSha256':terminal['pinSha256'],'accepted':False,
'cpu':[cpu(p) for p in sorted(Path(str(prefix)+'-cpu').glob('*.cpuprofile'))],
'heap':[heap(p) for p in sorted(Path(str(prefix)+'-heap').glob('*.heapprofile'))]}
assert len(out['cpu'])==2 and len(out['heap'])==2
with result.open('x') as f:json.dump(out,f,ensure_ascii=False,indent=2);f.write('\n')
for kind in ['cpu','heap']:
 for item in out[kind]:print(json.dumps({'kind':kind,'path':item['path'],'topSelf':item['self'][:12]},ensure_ascii=False))
